`וַיְהִי בַּעֲלוֹת הַכֹּהֲנִים נֹשְׂאֵי אֲרוֹן בְּרִית יְהוָה מִתּוֹךְ הַיַּרְדֵּן נִתְּקוּ כַּפּוֹת רַגְלֵי הַכֹּהֲנִים אֶל *הֶחָרָבָה* וַיָּשֻׁבוּ מֵי הַיַּרְדֵּן לִמְקוֹמָם וַיֵּלְכוּ כִתְמוֹל שִׁלְשׁוֹם עַל כָּל גְּדוֹתָיו` (יהושע ד:יח)
`וַיִּקַּח אֵלִיָּהוּ אֶת אַדַּרְתּוֹ וַיִּגְלֹם וַיַּכֶּה אֶת הַמַּיִם וַיֵּחָצוּ הֵנָּה וָהֵנָּה וַיַּעַבְרוּ שְׁנֵיהֶם *בֶּחָרָבָה*` (מלכים-ב ב:ח)
`וְנָתַתִּי יְאֹרִים *חָרָבָה* וּמָכַרְתִּי אֶת הָאָרֶץ בְּיַד רָעִים וַהֲשִׁמֹּתִי אֶרֶץ וּמְלֹאָהּ בְּיַד זָרִים אֲנִי יְהוָה דִּבַּרְתִּי` (יחזקאל ל:יב)
