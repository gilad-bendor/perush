`וַיֹּאמֶר אֱלֹהִים יִשְׁרְצוּ הַמַּיִם שֶׁרֶץ נֶפֶשׁ חַיָּה וְעוֹף יְעוֹפֵף עַל הָאָרֶץ עַל פְּנֵי רְקִיעַ הַשָּׁמָיִם` (בראשית א:כ)
`וַיִּבְרָא אֱלֹהִים אֶת הַתַּנִּינִם הַגְּדֹלִים וְאֵת כָּל נֶפֶשׁ הַחַיָּה הָרֹמֶשֶׂת אֲשֶׁר שָׁרְצוּ הַמַּיִם לְמִינֵהֶם וְאֵת כָּל עוֹף כָּנָף לְמִינֵהוּ וַיַּרְא אֱלֹהִים כִּי טוֹב` (בראשית א:כא)
`וַיֹּאמֶר אֱלֹהִים תּוֹצֵא הָאָרֶץ נֶפֶשׁ חַיָּה לְמִינָהּ בְּהֵמָה וָרֶמֶשׂ וְחַיְתוֹ אֶרֶץ לְמִינָהּ וַיְהִי כֵן` (בראשית א:כד)
`וַיַּעַשׂ אֱלֹהִים אֶת חַיַּת הָאָרֶץ לְמִינָהּ וְאֶת הַבְּהֵמָה לְמִינָהּ וְאֵת כָּל רֶמֶשׂ הָאֲדָמָה לְמִינֵהוּ וַיַּרְא אֱלֹהִים כִּי טוֹב` (בראשית א:כה)
`וַיְבָרֶךְ אֹתָם אֱלֹהִים וַיֹּאמֶר לָהֶם אֱלֹהִים פְּרוּ וּרְבוּ וּמִלְאוּ אֶת הָאָרֶץ וְכִבְשֻׁהָ וּרְדוּ בִּדְגַת הַיָּם וּבְעוֹף הַשָּׁמַיִם וּבְכָל חַיָּה הָרֹמֶשֶׂת עַל הָאָרֶץ` (בראשית א:כח)
`וּלְכָל חַיַּת הָאָרֶץ וּלְכָל עוֹף הַשָּׁמַיִם וּלְכֹל רוֹמֵשׂ עַל הָאָרֶץ אֲשֶׁר בּוֹ נֶפֶשׁ חַיָּה אֶת כָּל יֶרֶק עֵשֶׂב לְאָכְלָה וַיְהִי כֵן` (בראשית א:ל)
`וַיִּיצֶר יְהוָה אֱלֹהִים אֶת הָאָדָם עָפָר מִן הָאֲדָמָה וַיִּפַּח בְּאַפָּיו נִשְׁמַת חַיִּים וַיְהִי הָאָדָם לְנֶפֶשׁ חַיָּה` (בראשית ב:ז)
`וַיַּצְמַח יְהוָה אֱלֹהִים מִן הָאֲדָמָה כָּל עֵץ נֶחְמָד לְמַרְאֶה וְטוֹב לְמַאֲכָל וְעֵץ הַחַיִּים בְּתוֹךְ הַגָּן וְעֵץ הַדַּעַת טוֹב וָרָע` (בראשית ב:ט)
`וַיִּצֶר יְהוָה אֱלֹהִים מִן הָאֲדָמָה כָּל חַיַּת הַשָּׂדֶה וְאֵת כָּל עוֹף הַשָּׁמַיִם וַיָּבֵא אֶל הָאָדָם לִרְאוֹת מַה יִּקְרָא לוֹ וְכֹל אֲשֶׁר יִקְרָא לוֹ הָאָדָם נֶפֶשׁ חַיָּה הוּא שְׁמוֹ` (בראשית ב:יט)
`וַיִּקְרָא הָאָדָם שֵׁמוֹת לְכָל הַבְּהֵמָה וּלְעוֹף הַשָּׁמַיִם וּלְכֹל חַיַּת הַשָּׂדֶה וּלְאָדָם לֹא מָצָא עֵזֶר כְּנֶגְדּוֹ` (בראשית ב:כ)
`וְהַנָּחָשׁ הָיָה עָרוּם מִכֹּל חַיַּת הַשָּׂדֶה אֲשֶׁר עָשָׂה יְהוָה אֱלֹהִים וַיֹּאמֶר אֶל הָאִשָּׁה אַף כִּי אָמַר אֱלֹהִים לֹא תֹאכְלוּ מִכֹּל עֵץ הַגָּן` (בראשית ג:א)
`וַיֹּאמֶר יְהֹוָה אֱלֹהִים אֶל הַנָּחָשׁ כִּי עָשִׂיתָ זֹּאת אָרוּר אַתָּה מִכָּל הַבְּהֵמָה וּמִכֹּל חַיַּת הַשָּׂדֶה עַל גְּחֹנְךָ תֵלֵךְ וְעָפָר תֹּאכַל כָּל יְמֵי חַיֶּיךָ` (בראשית ג:יד)
`וּלְאָדָם אָמַר כִּי שָׁמַעְתָּ לְקוֹל אִשְׁתֶּךָ וַתֹּאכַל מִן הָעֵץ אֲשֶׁר צִוִּיתִיךָ לֵאמֹר לֹא תֹאכַל מִמֶּנּוּ אֲרוּרָה הָאֲדָמָה בַּעֲבוּרֶךָ בְּעִצָּבוֹן תֹּאכֲלֶנָּה כֹּל יְמֵי חַיֶּיךָ` (בראשית ג:יז)
`וַיִּקְרָא הָאָדָם שֵׁם אִשְׁתּוֹ חַוָּה כִּי הִוא הָיְתָה אֵם כָּל חָי` (בראשית ג:כ)
`וַיֹּאמֶר יְהוָה אֱלֹהִים הֵן הָאָדָם הָיָה כְּאַחַד מִמֶּנּוּ לָדַעַת טוֹב וָרָע וְעַתָּה פֶּן יִשְׁלַח יָדוֹ וְלָקַח גַּם מֵעֵץ הַחַיִּים וְאָכַל וָחַי לְעֹלָם` (בראשית ג:כב)
`וַיְגָרֶשׁ אֶת הָאָדָם וַיַּשְׁכֵּן מִקֶּדֶם לְגַן עֵדֶן אֶת הַכְּרֻבִים וְאֵת לַהַט הַחֶרֶב הַמִּתְהַפֶּכֶת לִשְׁמֹר אֶת דֶּרֶךְ עֵץ הַחַיִּים` (בראשית ג:כד)
`וַאֲנִי הִנְנִי מֵבִיא אֶת הַמַּבּוּל מַיִם עַל הָאָרֶץ לְשַׁחֵת כָּל בָּשָׂר אֲשֶׁר בּוֹ רוּחַ חַיִּים מִתַּחַת הַשָּׁמָיִם כֹּל אֲשֶׁר בָּאָרֶץ יִגְוָע` (בראשית ו:יז)
`וּמִכָּל הָחַי מִכָּל בָּשָׂר שְׁנַיִם מִכֹּל תָּבִיא אֶל הַתֵּבָה לְהַחֲיֹת אִתָּךְ זָכָר וּנְקֵבָה יִהְיוּ` (בראשית ו:יט)
`בִּשְׁנַת שֵׁשׁ מֵאוֹת שָׁנָה לְחַיֵּי נֹחַ בַּחֹדֶשׁ הַשֵּׁנִי בְּשִׁבְעָה עָשָׂר יוֹם לַחֹדֶשׁ בַּיּוֹם הַזֶּה נִבְקְעוּ כָּל מַעְיְנֹת תְּהוֹם רַבָּה וַאֲרֻבֹּת הַשָּׁמַיִם נִפְתָּחוּ` (בראשית ז:יא)
`הֵמָּה וְכָל הַחַיָּה לְמִינָהּ וְכָל הַבְּהֵמָה לְמִינָהּ וְכָל הָרֶמֶשׂ הָרֹמֵשׂ עַל הָאָרֶץ לְמִינֵהוּ וְכָל הָעוֹף לְמִינֵהוּ כֹּל צִפּוֹר כָּל כָּנָף` (בראשית ז:יד)
`וַיָּבֹאוּ אֶל נֹחַ אֶל הַתֵּבָה שְׁנַיִם שְׁנַיִם מִכָּל הַבָּשָׂר אֲשֶׁר בּוֹ רוּחַ חַיִּים` (בראשית ז:טו)
`וַיִּגְוַע כָּל בָּשָׂר הָרֹמֵשׂ עַל הָאָרֶץ בָּעוֹף וּבַבְּהֵמָה וּבַחַיָּה וּבְכָל הַשֶּׁרֶץ הַשֹּׁרֵץ עַל הָאָרֶץ וְכֹל הָאָדָם` (בראשית ז:כא)
`כֹּל אֲשֶׁר נִשְׁמַת רוּחַ חַיִּים בְּאַפָּיו מִכֹּל אֲשֶׁר בֶּחָרָבָה מֵתוּ` (בראשית ז:כב)
`וַיִּזְכֹּר אֱלֹהִים אֶת נֹחַ וְאֵת כָּל הַחַיָּה וְאֶת כָּל הַבְּהֵמָה אֲשֶׁר אִתּוֹ בַּתֵּבָה וַיַּעֲבֵר אֱלֹהִים רוּחַ עַל הָאָרֶץ וַיָּשֹׁכּוּ הַמָּיִם` (בראשית ח:א)
`כָּל הַחַיָּה אֲשֶׁר אִתְּךָ מִכָּל בָּשָׂר בָּעוֹף וּבַבְּהֵמָה וּבְכָל הָרֶמֶשׂ הָרֹמֵשׂ עַל הָאָרֶץ הוֹצֵא אִתָּךְ וְשָׁרְצוּ בָאָרֶץ וּפָרוּ וְרָבוּ עַל הָאָרֶץ` (בראשית ח:יז)
`כָּל הַחַיָּה כָּל הָרֶמֶשׂ וְכָל הָעוֹף כֹּל רוֹמֵשׂ עַל הָאָרֶץ לְמִשְׁפְּחֹתֵיהֶם יָצְאוּ מִן הַתֵּבָה` (בראשית ח:יט)
`וַיָּרַח יְהוָה אֶת רֵיחַ הַנִּיחֹחַ וַיֹּאמֶר יְהוָה אֶל לִבּוֹ לֹא אֹסִף לְקַלֵּל עוֹד אֶת הָאֲדָמָה בַּעֲבוּר הָאָדָם כִּי יֵצֶר לֵב הָאָדָם רַע מִנְּעֻרָיו וְלֹא אֹסִף עוֹד לְהַכּוֹת אֶת כָּל חַי כַּאֲשֶׁר עָשִׂיתִי` (בראשית ח:כא)
`וּמוֹרַאֲכֶם וְחִתְּכֶם יִהְיֶה עַל כָּל חַיַּת הָאָרֶץ וְעַל כָּל עוֹף הַשָּׁמָיִם בְּכֹל אֲשֶׁר תִּרְמֹשׂ הָאֲדָמָה וּבְכָל דְּגֵי הַיָּם בְּיֶדְכֶם נִתָּנוּ` (בראשית ט:ב)
`כָּל רֶמֶשׂ אֲשֶׁר הוּא חַי לָכֶם יִהְיֶה לְאָכְלָה כְּיֶרֶק עֵשֶׂב נָתַתִּי לָכֶם אֶת כֹּל` (בראשית ט:ג)
`וְאַךְ אֶת דִּמְכֶם לְנַפְשֹׁתֵיכֶם אֶדְרֹשׁ מִיַּד כָּל חַיָּה אֶדְרְשֶׁנּוּ וּמִיַּד הָאָדָם מִיַּד אִישׁ אָחִיו אֶדְרֹשׁ אֶת נֶפֶשׁ הָאָדָם` (בראשית ט:ה)
`וְאֵת כָּל נֶפֶשׁ הַחַיָּה אֲשֶׁר אִתְּכֶם בָּעוֹף בַּבְּהֵמָה וּבְכָל חַיַּת הָאָרֶץ אִתְּכֶם מִכֹּל יֹצְאֵי הַתֵּבָה לְכֹל חַיַּת הָאָרֶץ` (בראשית ט:י)
`וַיֹּאמֶר אֱלֹהִים זֹאת אוֹת הַבְּרִית אֲשֶׁר אֲנִי נֹתֵן בֵּינִי וּבֵינֵיכֶם וּבֵין כָּל נֶפֶשׁ חַיָּה אֲשֶׁר אִתְּכֶם לְדֹרֹת עוֹלָם` (בראשית ט:יב)
`וְזָכַרְתִּי אֶת בְּרִיתִי אֲשֶׁר בֵּינִי וּבֵינֵיכֶם וּבֵין כָּל נֶפֶשׁ חַיָּה בְּכָל בָּשָׂר וְלֹא יִהְיֶה עוֹד הַמַּיִם לְמַבּוּל לְשַׁחֵת כָּל בָּשָׂר` (בראשית ט:טו)
`וְהָיְתָה הַקֶּשֶׁת בֶּעָנָן וּרְאִיתִיהָ לִזְכֹּר בְּרִית עוֹלָם בֵּין אֱלֹהִים וּבֵין כָּל נֶפֶשׁ חַיָּה בְּכָל בָּשָׂר אֲשֶׁר עַל הָאָרֶץ` (בראשית ט:טז)
`וַיֹּאמֶר שׁוֹב אָשׁוּב אֵלֶיךָ כָּעֵת חַיָּה וְהִנֵּה בֵן לְשָׂרָה אִשְׁתֶּךָ וְשָׂרָה שֹׁמַעַת פֶּתַח הָאֹהֶל וְהוּא אַחֲרָיו` (בראשית יח:י)
`הֲיִפָּלֵא מֵיְהוָה דָּבָר לַמּוֹעֵד אָשׁוּב אֵלֶיךָ כָּעֵת חַיָּה וּלְשָׂרָה בֵן` (בראשית יח:יד)
`וַיִּהְיוּ חַיֵּי שָׂרָה מֵאָה שָׁנָה וְעֶשְׂרִים שָׁנָה וְשֶׁבַע שָׁנִים שְׁנֵי חַיֵּי שָׂרָה` (בראשית כג:א)
`וְלִבְנֵי הַפִּילַגְשִׁים אֲשֶׁר לְאַבְרָהָם נָתַן אַבְרָהָם מַתָּנֹת וַיְשַׁלְּחֵם מֵעַל יִצְחָק בְּנוֹ בְּעוֹדֶנּוּ חַי קֵדְמָה אֶל אֶרֶץ קֶדֶם` (בראשית כה:ו)
`וְאֵלֶּה יְמֵי שְׁנֵי חַיֵּי אַבְרָהָם אֲשֶׁר חָי מְאַת שָׁנָה וְשִׁבְעִים שָׁנָה וְחָמֵשׁ שָׁנִים` (בראשית כה:ז)
`וְאֵלֶּה שְׁנֵי חַיֵּי יִשְׁמָעֵאל מְאַת שָׁנָה וּשְׁלֹשִׁים שָׁנָה וְשֶׁבַע שָׁנִים וַיִּגְוַע וַיָּמָת וַיֵּאָסֶף אֶל עַמָּיו` (בראשית כה:יז)
`וַיַּחְפְּרוּ עַבְדֵי יִצְחָק בַּנָּחַל וַיִּמְצְאוּ שָׁם בְּאֵר מַיִם חַיִּים` (בראשית כו:יט)
`וַתֹּאמֶר רִבְקָה אֶל יִצְחָק קַצְתִּי בְחַיַּי מִפְּנֵי בְּנוֹת חֵת אִם לֹקֵחַ יַעֲקֹב אִשָּׁה מִבְּנוֹת חֵת כָּאֵלֶּה מִבְּנוֹת הָאָרֶץ לָמָּה לִּי חַיִּים` (בראשית כז:מו)
`וְעַתָּה לְכוּ וְנַהַרְגֵהוּ וְנַשְׁלִכֵהוּ בְּאַחַד הַבֹּרוֹת וְאָמַרְנוּ חַיָּה רָעָה אֲכָלָתְהוּ וְנִרְאֶה מַה יִּהְיוּ חֲלֹמֹתָיו` (בראשית לז:כ)
`וַיַּכִּירָהּ וַיֹּאמֶר כְּתֹנֶת בְּנִי חַיָּה רָעָה אֲכָלָתְהוּ טָרֹף טֹרַף יוֹסֵף` (בראשית לז:לג)
`בְּזֹאת תִּבָּחֵנוּ חֵי פַרְעֹה אִם תֵּצְאוּ מִזֶּה כִּי אִם בְּבוֹא אֲחִיכֶם הַקָּטֹן הֵנָּה` (בראשית מב:טו)
`שִׁלְחוּ מִכֶּם אֶחָד וְיִקַּח אֶת אֲחִיכֶם וְאַתֶּם הֵאָסְרוּ וְיִבָּחֲנוּ דִּבְרֵיכֶם הַאֱמֶת אִתְּכֶם וְאִם לֹא חֵי פַרְעֹה כִּי מְרַגְּלִים אַתֶּם` (בראשית מב:טז)
`וַיֹּאמְרוּ שָׁאוֹל שָׁאַל הָאִישׁ לָנוּ וּלְמוֹלַדְתֵּנוּ לֵאמֹר הַעוֹד אֲבִיכֶם חַי הֲיֵשׁ לָכֶם אָח וַנַגֶּד לוֹ עַל פִּי הַדְּבָרִים הָאֵלֶּה הֲיָדוֹעַ נֵדַע כִּי יֹאמַר הוֹרִידוּ אֶת אֲחִיכֶם` (בראשית מג:ז)
`וַיִּשְׁאַל לָהֶם לְשָׁלוֹם וַיֹּאמֶר הֲשָׁלוֹם אֲבִיכֶם הַזָּקֵן אֲשֶׁר אֲמַרְתֶּם הַעוֹדֶנּוּ חָי` (בראשית מג:כז)
`וַיֹּאמְרוּ שָׁלוֹם לְעַבְדְּךָ לְאָבִינוּ עוֹדֶנּוּ חָי וַיִּקְּדוּ וַיִּשְׁתַּחוּ` (בראשית מג:כח)
`וַיֹּאמֶר יוֹסֵף אֶל אֶחָיו אֲנִי יוֹסֵף הַעוֹד אָבִי חָי וְלֹא יָכְלוּ אֶחָיו לַעֲנוֹת אֹתוֹ כִּי נִבְהֲלוּ מִפָּנָיו` (בראשית מה:ג)
`וַיַּגִּדוּ לוֹ לֵאמֹר עוֹד יוֹסֵף חַי וְכִי הוּא מֹשֵׁל בְּכָל אֶרֶץ מִצְרָיִם וַיָּפָג לִבּוֹ כִּי לֹא הֶאֱמִין לָהֶם` (בראשית מה:כו)
`וַיֹּאמֶר יִשְׂרָאֵל רַב עוֹד יוֹסֵף בְּנִי חָי אֵלְכָה וְאֶרְאֶנּוּ בְּטֶרֶם אָמוּת` (בראשית מה:כח)
`וַיֹּאמֶר יִשְׂרָאֵל אֶל יוֹסֵף אָמוּתָה הַפָּעַם אַחֲרֵי רְאוֹתִי אֶת פָּנֶיךָ כִּי עוֹדְךָ חָי` (בראשית מו:ל)
`וַיֹּאמֶר פַּרְעֹה אֶל יַעֲקֹב כַּמָּה יְמֵי שְׁנֵי חַיֶּיךָ` (בראשית מז:ח)
`וַיֹּאמֶר יַעֲקֹב אֶל פַּרְעֹה יְמֵי שְׁנֵי מְגוּרַי שְׁלֹשִׁים וּמְאַת שָׁנָה מְעַט וְרָעִים הָיוּ יְמֵי שְׁנֵי חַיַּי וְלֹא הִשִּׂיגוּ אֶת יְמֵי שְׁנֵי חַיֵּי אֲבֹתַי בִּימֵי מְגוּרֵיהֶם` (בראשית מז:ט)
`וַיְחִי יַעֲקֹב בְּאֶרֶץ מִצְרַיִם שְׁבַע עֶשְׂרֵה שָׁנָה וַיְהִי יְמֵי יַעֲקֹב שְׁנֵי חַיָּיו שֶׁבַע שָׁנִים וְאַרְבָּעִים וּמְאַת שָׁנָה` (בראשית מז:כח)
`וַיְמָרְרוּ אֶת חַיֵּיהֶם בַּעֲבֹדָה קָשָׁה בְּחֹמֶר וּבִלְבֵנִים וּבְכָל עֲבֹדָה בַּשָּׂדֶה אֵת כָּל עֲבֹדָתָם אֲשֶׁר עָבְדוּ בָהֶם בְּפָרֶךְ` (שמות א:יד)
`וַיֵּלֶךְ מֹשֶׁה וַיָּשָׁב אֶל יֶתֶר חֹתְנוֹ וַיֹּאמֶר לוֹ אֵלֲכָה נָּא וְאָשׁוּבָה אֶל אַחַי אֲשֶׁר בְּמִצְרַיִם וְאֶרְאֶה הַעוֹדָם חַיִּים וַיֹּאמֶר יִתְרוֹ לְמֹשֶׁה לֵךְ לְשָׁלוֹם` (שמות ד:יח)
`וְאֵלֶּה שְׁמוֹת בְּנֵי לֵוִי לְתֹלְדֹתָם גֵּרְשׁוֹן וּקְהָת וּמְרָרִי וּשְׁנֵי חַיֵּי לֵוִי שֶׁבַע וּשְׁלֹשִׁים וּמְאַת שָׁנָה` (שמות ו:טז)
`וּבְנֵי קְהָת עַמְרָם וְיִצְהָר וְחֶבְרוֹן וְעֻזִּיאֵל וּשְׁנֵי חַיֵּי קְהָת שָׁלֹשׁ וּשְׁלֹשִׁים וּמְאַת שָׁנָה` (שמות ו:יח)
`וַיִּקַּח עַמְרָם אֶת יוֹכֶבֶד דֹּדָתוֹ לוֹ לְאִשָּׁה וַתֵּלֶד לוֹ אֶת אַהֲרֹן וְאֶת מֹשֶׁה וּשְׁנֵי חַיֵּי עַמְרָם שֶׁבַע וּשְׁלֹשִׁים וּמְאַת שָׁנָה` (שמות ו:כ)
`וְכִי יִגֹּף שׁוֹר אִישׁ אֶת שׁוֹר רֵעֵהוּ וָמֵת וּמָכְרוּ אֶת הַשּׁוֹר הַחַי וְחָצוּ אֶת כַּסְפּוֹ וְגַם אֶת הַמֵּת יֶחֱצוּן` (שמות כא:לה)
`אִם הִמָּצֵא תִמָּצֵא בְיָדוֹ הַגְּנֵבָה מִשּׁוֹר עַד חֲמוֹר עַד שֶׂה חַיִּים שְׁנַיִם יְשַׁלֵּם` (שמות כב:ד)
`וְהַשְּׁבִיעִת תִּשְׁמְטֶנָּה וּנְטַשְׁתָּהּ וְאָכְלוּ אֶבְיֹנֵי עַמֶּךָ וְיִתְרָם תֹּאכַל חַיַּת הַשָּׂדֶה כֵּן תַּעֲשֶׂה לְכַרְמְךָ לְזֵיתֶךָ` (שמות כג:יא)
`לֹא אֲגָרְשֶׁנּוּ מִפָּנֶיךָ בְּשָׁנָה אֶחָת פֶּן תִּהְיֶה הָאָרֶץ שְׁמָמָה וְרַבָּה עָלֶיךָ חַיַּת הַשָּׂדֶה` (שמות כג:כט)
`אוֹ נֶפֶשׁ אֲשֶׁר תִּגַּע בְּכָל דָּבָר טָמֵא אוֹ בְנִבְלַת חַיָּה טְמֵאָה אוֹ בְּנִבְלַת בְּהֵמָה טְמֵאָה אוֹ בְּנִבְלַת שֶׁרֶץ טָמֵא וְנֶעְלַם מִמֶּנּוּ וְהוּא טָמֵא וְאָשֵׁם` (ויקרא ה:ב)
`דַּבְּרוּ אֶל בְּנֵי יִשְׂרָאֵל לֵאמֹר זֹאת הַחַיָּה אֲשֶׁר תֹּאכְלוּ מִכָּל הַבְּהֵמָה אֲשֶׁר עַל הָאָרֶץ` (ויקרא יא:ב)
`וְכֹל אֲשֶׁר אֵין לוֹ סְנַפִּיר וְקַשְׂקֶשֶׂת בַּיַּמִּים וּבַנְּחָלִים מִכֹּל שֶׁרֶץ הַמַּיִם וּמִכֹּל נֶפֶשׁ הַחַיָּה אֲשֶׁר בַּמָּיִם שֶׁקֶץ הֵם לָכֶם` (ויקרא יא:י)
`וְכֹל הוֹלֵךְ עַל כַּפָּיו בְּכָל הַחַיָּה הַהֹלֶכֶת עַל אַרְבַּע טְמֵאִים הֵם לָכֶם כָּל הַנֹּגֵעַ בְּנִבְלָתָם יִטְמָא עַד הָעָרֶב` (ויקרא יא:כז)
`זֹאת תּוֹרַת הַבְּהֵמָה וְהָעוֹף וְכֹל נֶפֶשׁ הַחַיָּה הָרֹמֶשֶׂת בַּמָּיִם וּלְכָל נֶפֶשׁ הַשֹּׁרֶצֶת עַל הָאָרֶץ` (ויקרא יא:מו)
`לְהַבְדִּיל בֵּין הַטָּמֵא וּבֵין הַטָּהֹר וּבֵין הַחַיָּה הַנֶּאֱכֶלֶת וּבֵין הַחַיָּה אֲשֶׁר לֹא תֵאָכֵל` (ויקרא יא:מז)
`וְרָאָה הַכֹּהֵן וְהִנֵּה שְׂאֵת לְבָנָה בָּעוֹר וְהִיא הָפְכָה שֵׂעָר לָבָן וּמִחְיַת בָּשָׂר חַי בַּשְׂאֵת` (ויקרא יג:י)
`וּבְיוֹם הֵרָאוֹת בּוֹ בָּשָׂר חַי יִטְמָא` (ויקרא יג:יד)
`וְרָאָה הַכֹּהֵן אֶת הַבָּשָׂר הַחַי וְטִמְּאוֹ הַבָּשָׂר הַחַי טָמֵא הוּא צָרַעַת הוּא` (ויקרא יג:טו)
`אוֹ כִי יָשׁוּב הַבָּשָׂר הַחַי וְנֶהְפַּךְ לְלָבָן וּבָא אֶל הַכֹּהֵן` (ויקרא יג:טז)
`וְצִוָּה הַכֹּהֵן וְלָקַח לַמִּטַּהֵר שְׁתֵּי צִפֳּרִים חַיּוֹת טְהֹרוֹת וְעֵץ אֶרֶז וּשְׁנִי תוֹלַעַת וְאֵזֹב` (ויקרא יד:ד)
`וְצִוָּה הַכֹּהֵן וְשָׁחַט אֶת הַצִּפּוֹר הָאֶחָת אֶל כְּלִי חֶרֶשׂ עַל מַיִם חַיִּים` (ויקרא יד:ה)
`אֶת הַצִּפֹּר הַחַיָּה יִקַּח אֹתָהּ וְאֶת עֵץ הָאֶרֶז וְאֶת שְׁנִי הַתּוֹלַעַת וְאֶת הָאֵזֹב וְטָבַל אוֹתָם וְאֵת הַצִּפֹּר הַחַיָּה בְּדַם הַצִּפֹּר הַשְּׁחֻטָה עַל הַמַּיִם הַחַיִּים` (ויקרא יד:ו)
`וְהִזָּה עַל הַמִּטַּהֵר מִן הַצָּרַעַת שֶׁבַע פְּעָמִים וְטִהֲרוֹ וְשִׁלַּח אֶת הַצִּפֹּר הַחַיָּה עַל פְּנֵי הַשָּׂדֶה` (ויקרא יד:ז)
`וְשָׁחַט אֶת הַצִּפֹּר הָאֶחָת אֶל כְּלִי חֶרֶשׂ עַל מַיִם חַיִּים` (ויקרא יד:נ)
`וְלָקַח אֶת עֵץ הָאֶרֶז וְאֶת הָאֵזֹב וְאֵת שְׁנִי הַתּוֹלַעַת וְאֵת הַצִּפֹּר הַחַיָּה וְטָבַל אֹתָם בְּדַם הַצִּפֹּר הַשְּׁחוּטָה וּבַמַּיִם הַחַיִּים וְהִזָּה אֶל הַבַּיִת שֶׁבַע פְּעָמִים` (ויקרא יד:נא)
`וְחִטֵּא אֶת הַבַּיִת בְּדַם הַצִּפּוֹר וּבַמַּיִם הַחַיִּים וּבַצִּפֹּר הַחַיָּה וּבְעֵץ הָאֶרֶז וּבָאֵזֹב וּבִשְׁנִי הַתּוֹלָעַת` (ויקרא יד:נב)
`וְשִׁלַּח אֶת הַצִּפֹּר הַחַיָּה אֶל מִחוּץ לָעִיר אֶל פְּנֵי הַשָּׂדֶה וְכִפֶּר עַל הַבַּיִת וְטָהֵר` (ויקרא יד:נג)
`וְכִי יִטְהַר הַזָּב מִזּוֹבוֹ וְסָפַר לוֹ שִׁבְעַת יָמִים לְטָהֳרָתוֹ וְכִבֶּס בְּגָדָיו וְרָחַץ בְּשָׂרוֹ בְּמַיִם חַיִּים וְטָהֵר` (ויקרא טו:יג)
`וְהַשָּׂעִיר אֲשֶׁר עָלָה עָלָיו הַגּוֹרָל לַעֲזָאזֵל יָעֳמַד חַי לִפְנֵי יְהוָה לְכַפֵּר עָלָיו לְשַׁלַּח אֹתוֹ לַעֲזָאזֵל הַמִּדְבָּרָה` (ויקרא טז:י)
`וְכִלָּה מִכַּפֵּר אֶת הַקֹּדֶשׁ וְאֶת אֹהֶל מוֹעֵד וְאֶת הַמִּזְבֵּחַ וְהִקְרִיב אֶת הַשָּׂעִיר הֶחָי` (ויקרא טז:כ)
`וְסָמַךְ אַהֲרֹן אֶת שְׁתֵּי יָדוֹ עַל רֹאשׁ הַשָּׂעִיר הַחַי וְהִתְוַדָּה עָלָיו אֶת כָּל עֲוֺנֹת בְּנֵי יִשְׂרָאֵל וְאֶת כָּל פִּשְׁעֵיהֶם לְכָל חַטֹּאתָם וְנָתַן אֹתָם עַל רֹאשׁ הַשָּׂעִיר וְשִׁלַּח בְּיַד אִישׁ עִתִּי הַמִּדְבָּרָה` (ויקרא טז:כא)
`וְאִישׁ אִישׁ מִבְּנֵי יִשְׂרָאֵל וּמִן הַגֵּר הַגָּר בְּתוֹכָם אֲשֶׁר יָצוּד צֵיד חַיָּה אוֹ עוֹף אֲשֶׁר יֵאָכֵל וְשָׁפַךְ אֶת דָּמוֹ וְכִסָּהוּ בֶּעָפָר` (ויקרא יז:יג)
`וְאִשָּׁה אֶל אֲחֹתָהּ לֹא תִקָּח לִצְרֹר לְגַלּוֹת עֶרְוָתָהּ עָלֶיהָ בְּחַיֶּיהָ` (ויקרא יח:יח)
`וְלִבְהֶמְתְּךָ וְלַחַיָּה אֲשֶׁר בְּאַרְצֶךָ תִּהְיֶה כָל תְּבוּאָתָהּ לֶאֱכֹל` (ויקרא כה:ז)
`וְנָתַתִּי שָׁלוֹם בָּאָרֶץ וּשְׁכַבְתֶּם וְאֵין מַחֲרִיד וְהִשְׁבַּתִּי חַיָּה רָעָה מִן הָאָרֶץ וְחֶרֶב לֹא תַעֲבֹר בְּאַרְצְכֶם` (ויקרא כו:ו)
`וְהִשְׁלַחְתִּי בָכֶם אֶת חַיַּת הַשָּׂדֶה וְשִׁכְּלָה אֶתְכֶם וְהִכְרִיתָה אֶת בְּהֶמְתְּכֶם וְהִמְעִיטָה אֶתְכֶם וְנָשַׁמּוּ דַּרְכֵיכֶם` (ויקרא כו:כב)
`וְאוּלָם חַי אָנִי וְיִמָּלֵא כְבוֹד יְהוָה אֶת כָּל הָאָרֶץ` (במדבר יד:כא)
`אֱמֹר אֲלֵהֶם חַי אָנִי נְאֻם יְהוָה אִם לֹא כַּאֲשֶׁר דִּבַּרְתֶּם בְּאָזְנָי כֵּן אֶעֱשֶׂה לָכֶם` (במדבר יד:כח)
`וְאִם בְּרִיאָה יִבְרָא יְהוָה וּפָצְתָה הָאֲדָמָה אֶת פִּיהָ וּבָלְעָה אֹתָם וְאֶת כָּל אֲשֶׁר לָהֶם וְיָרְדוּ חַיִּים שְׁאֹלָה וִידַעְתֶּם כִּי נִאֲצוּ הָאֲנָשִׁים הָאֵלֶּה אֶת יְהוָה` (במדבר טז:ל)
`וַיֵּרְדוּ הֵם וְכָל אֲשֶׁר לָהֶם חַיִּים שְׁאֹלָה וַתְּכַס עֲלֵיהֶם הָאָרֶץ וַיֹּאבְדוּ מִתּוֹךְ הַקָּהָל` (במדבר טז:לג)
`וַיַּעֲמֹד בֵּין הַמֵּתִים וּבֵין הַחַיִּים וַתֵּעָצַר הַמַּגֵּפָה` (במדבר טז:מח)
`וְלָקְחוּ לַטָּמֵא מֵעֲפַר שְׂרֵפַת הַחַטָּאת וְנָתַן עָלָיו מַיִם חַיִּים אֶל כֶּלִי` (במדבר יט:יז)
`וְהָיוּ הֶעָרִים לָהֶם לָשָׁבֶת וּמִגְרְשֵׁיהֶם יִהְיוּ לִבְהֶמְתָּם וְלִרְכֻשָׁם וּלְכֹל חַיָּתָם` (במדבר לה:ג)
`וְאַתֶּם הַדְּבֵקִים בַּיהוָה אֱלֹהֵיכֶם חַיִּים כֻּלְּכֶם הַיּוֹם` (דברים ד:ד)
`רַק הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד פֶּן תִּשְׁכַּח אֶת הַדְּבָרִים אֲשֶׁר רָאוּ עֵינֶיךָ וּפֶן יָסוּרוּ מִלְּבָבְךָ כֹּל יְמֵי חַיֶּיךָ וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ` (דברים ד:ט)
`יוֹם אֲשֶׁר עָמַדְתָּ לִפְנֵי יְהוָה אֱלֹהֶיךָ בְּחֹרֵב בֶּאֱמֹר יְהוָה אֵלַי הַקְהֶל לִי אֶת הָעָם וְאַשְׁמִעֵם אֶת דְּבָרָי אֲשֶׁר יִלְמְדוּן לְיִרְאָה אֹתִי כָּל הַיָּמִים אֲשֶׁר הֵם חַיִּים עַל הָאֲדָמָה וְאֶת בְּנֵיהֶם יְלַמֵּדוּן` (דברים ד:י)
`לֹא אֶת אֲבֹתֵינוּ כָּרַת יְהוָה אֶת הַבְּרִית הַזֹּאת כִּי אִתָּנוּ אֲנַחְנוּ אֵלֶּה פֹה הַיּוֹם כֻּלָּנוּ חַיִּים` (דברים ה:ג)
`כִּי מִי כָל בָּשָׂר אֲשֶׁר שָׁמַע קוֹל אֱלֹהִים חַיִּים מְדַבֵּר מִתּוֹךְ הָאֵשׁ כָּמֹנוּ וַיֶּחִי` (דברים ה:כו)
`לְמַעַן תִּירָא אֶת יְהוָה אֱלֹהֶיךָ לִשְׁמֹר אֶת כָּל חֻקֹּתָיו וּמִצְוֺתָיו אֲשֶׁר אָנֹכִי מְצַוֶּךָ אַתָּה וּבִנְךָ וּבֶן בִּנְךָ כֹּל יְמֵי חַיֶּיךָ וּלְמַעַן יַאֲרִכֻן יָמֶיךָ` (דברים ו:ב)
`וְנָשַׁל יְהוָה אֱלֹהֶיךָ אֶת הַגּוֹיִם הָאֵל מִפָּנֶיךָ מְעַט מְעָט לֹא תוּכַל כַּלֹּתָם מַהֵר פֶּן תִּרְבֶּה עָלֶיךָ חַיַּת הַשָּׂדֶה` (דברים ז:כב)
`אֵלֶּה הַחֻקִּים וְהַמִּשְׁפָּטִים אֲשֶׁר תִּשְׁמְרוּן לַעֲשׂוֹת בָּאָרֶץ אֲשֶׁר נָתַן יְהוָה אֱלֹהֵי אֲבֹתֶיךָ לְךָ לְרִשְׁתָּהּ כָּל הַיָּמִים אֲשֶׁר אַתֶּם חַיִּים עַל הָאֲדָמָה` (דברים יב:א)
`לֹא תֹאכַל עָלָיו חָמֵץ שִׁבְעַת יָמִים תֹּאכַל עָלָיו מַצּוֹת לֶחֶם עֹנִי כִּי בְחִפָּזוֹן יָצָאתָ מֵאֶרֶץ מִצְרַיִם לְמַעַן תִּזְכֹּר אֶת יוֹם צֵאתְךָ מֵאֶרֶץ מִצְרַיִם כֹּל יְמֵי חַיֶּיךָ` (דברים טז:ג)
`וְהָיְתָה עִמּוֹ וְקָרָא בוֹ כָּל יְמֵי חַיָּיו לְמַעַן יִלְמַד לְיִרְאָה אֶת יְהוָה אֱלֹהָיו לִשְׁמֹר אֶת כָּל דִּבְרֵי הַתּוֹרָה הַזֹּאת וְאֶת הַחֻקִּים הָאֵלֶּה לַעֲשֹׂתָם` (דברים יז:יט)
`וְהָיוּ חַיֶּיךָ תְּלֻאִים לְךָ מִנֶּגֶד וּפָחַדְתָּ לַיְלָה וְיוֹמָם וְלֹא תַאֲמִין בְּחַיֶּיךָ` (דברים כח:סו)
`וּמָל יְהוָה אֱלֹהֶיךָ אֶת לְבָבְךָ וְאֶת לְבַב זַרְעֶךָ לְאַהֲבָה אֶת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ לְמַעַן חַיֶּיךָ` (דברים ל:ו)
`רְאֵה נָתַתִּי לְפָנֶיךָ הַיּוֹם אֶת הַחַיִּים וְאֶת הַטּוֹב וְאֶת הַמָּוֶת וְאֶת הָרָע` (דברים ל:טו)
`הַעִידֹתִי בָכֶם הַיּוֹם אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ הַחַיִּים וְהַמָּוֶת נָתַתִּי לְפָנֶיךָ הַבְּרָכָה וְהַקְּלָלָה וּבָחַרְתָּ בַּחַיִּים לְמַעַן תִּחְיֶה אַתָּה וְזַרְעֶךָ` (דברים ל:יט)
`לְאַהֲבָה אֶת יְהוָה אֱלֹהֶיךָ לִשְׁמֹעַ בְּקֹלוֹ וּלְדָבְקָה בוֹ כִּי הוּא חַיֶּיךָ וְאֹרֶךְ יָמֶיךָ לָשֶׁבֶת עַל הָאֲדָמָה אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֶיךָ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב לָתֵת לָהֶם` (דברים ל:כ)
`וּבְנֵיהֶם אֲשֶׁר לֹא יָדְעוּ יִשְׁמְעוּ וְלָמְדוּ לְיִרְאָה אֶת יְהוָה אֱלֹהֵיכֶם כָּל הַיָּמִים אֲשֶׁר אַתֶּם חַיִּים עַל הָאֲדָמָה אֲשֶׁר אַתֶּם עֹבְרִים אֶת הַיַּרְדֵּן שָׁמָּה לְרִשְׁתָּהּ` (דברים לא:יג)
`כִּי אָנֹכִי יָדַעְתִּי אֶת מֶרְיְךָ וְאֶת עָרְפְּךָ הַקָּשֶׁה הֵן בְּעוֹדֶנִּי חַי עִמָּכֶם הַיּוֹם מַמְרִים הֱיִתֶם עִם יְהֹוָה וְאַף כִּי אַחֲרֵי מוֹתִי` (דברים לא:כז)
`כִּי אֶשָּׂא אֶל שָׁמַיִם יָדִי וְאָמַרְתִּי חַי אָנֹכִי לְעֹלָם` (דברים לב:מ)
`כִּי לֹא דָבָר רֵק הוּא מִכֶּם כִּי הוּא חַיֵּיכֶם וּבַדָּבָר הַזֶּה תַּאֲרִיכוּ יָמִים עַל הָאֲדָמָה אֲשֶׁר אַתֶּם עֹבְרִים אֶת הַיַּרְדֵּן שָׁמָּה לְרִשְׁתָּהּ` (דברים לב:מז)
`לֹא יִתְיַצֵּב אִישׁ לְפָנֶיךָ כֹּל יְמֵי חַיֶּיךָ כַּאֲשֶׁר הָיִיתִי עִם מֹשֶׁה אֶהְיֶה עִמָּךְ לֹא אַרְפְּךָ וְלֹא אֶעֶזְבֶךָּ` (יהושע א:ה)
`וַיֹּאמֶר יְהוֹשֻׁעַ בְּזֹאת תֵּדְעוּן כִּי אֵל חַי בְּקִרְבְּכֶם וְהוֹרֵשׁ יוֹרִישׁ מִפְּנֵיכֶם אֶת הַכְּנַעֲנִי וְאֶת הַחִתִּי וְאֶת הַחִוִּי וְאֶת הַפְּרִזִּי וְאֶת הַגִּרְגָּשִׁי וְהָאֱמֹרִי וְהַיְבוּסִי` (יהושע ג:י)
`בַּיּוֹם הַהוּא גִּדַּל יְהוָה אֶת יְהוֹשֻׁעַ בְּעֵינֵי כָּל יִשְׂרָאֵל וַיִּרְאוּ אֹתוֹ כַּאֲשֶׁר יָרְאוּ אֶת מֹשֶׁה כָּל יְמֵי חַיָּיו` (יהושע ד:יד)
`וְאֶת מֶלֶךְ הָעַי תָּפְשׂוּ חָי וַיַּקְרִבוּ אֹתוֹ אֶל יְהוֹשֻׁעַ` (יהושע ח:כג)
`וַיֹּאמַר אַחַי בְּנֵי אִמִּי הֵם חַי יְהוָה לוּ הַחֲיִתֶם אוֹתָם לֹא הָרַגְתִּי אֶתְכֶם` (שופטים ח:יט)
`וַיֹּאמֶר שִׁמְשׁוֹן תָּמוֹת נַפְשִׁי עִם פְּלִשְׁתִּים וַיֵּט בְּכֹחַ וַיִּפֹּל הַבַּיִת עַל הַסְּרָנִים וְעַל כָּל הָעָם אֲשֶׁר בּוֹ וַיִּהְיוּ הַמֵּתִים אֲשֶׁר הֵמִית בְּמוֹתוֹ רַבִּים מֵאֲשֶׁר הֵמִית בְּחַיָּיו` (שופטים טז:ל)
`וַתִּדֹּר נֶדֶר וַתֹּאמַר יְהוָה צְבָאוֹת אִם רָאֹה תִרְאֶה בָּעֳנִי אֲמָתֶךָ וּזְכַרְתַּנִי וְלֹא תִשְׁכַּח אֶת אֲמָתֶךָ וְנָתַתָּה לַאֲמָתְךָ זֶרַע אֲנָשִׁים וּנְתַתִּיו לַיהוָה כָּל יְמֵי חַיָּיו וּמוֹרָה לֹא יַעֲלֶה עַל רֹאשׁוֹ` (שמואל-א א:יא)
`וַתֹּאמֶר בִּי אֲדֹנִי חֵי נַפְשְׁךָ אֲדֹנִי אֲנִי הָאִשָּׁה הַנִּצֶּבֶת עִמְּכָה בָּזֶה לְהִתְפַּלֵּל אֶל יְהוָה` (שמואל-א א:כו)
`גַּם בְּטֶרֶם יַקְטִרוּן אֶת הַחֵלֶב וּבָא נַעַר הַכֹּהֵן וְאָמַר לָאִישׁ הַזֹּבֵחַ תְּנָה בָשָׂר לִצְלוֹת לַכֹּהֵן וְלֹא יִקַּח מִמְּךָ בָּשָׂר מְבֻשָּׁל כִּי אִם חָי` (שמואל-א ב:טו)
`וַיִּשְׁפֹּט שְׁמוּאֵל אֶת יִשְׂרָאֵל כֹּל יְמֵי חַיָּיו` (שמואל-א ז:טו)
`כִּי חַי יְהוָה הַמּוֹשִׁיעַ אֶת יִשְׂרָאֵל כִּי אִם יֶשְׁנוֹ בְּיוֹנָתָן בְּנִי כִּי מוֹת יָמוּת וְאֵין עֹנֵהוּ מִכָּל הָעָם` (שמואל-א יד:לט)
`וַיֹּאמֶר הָעָם אֶל שָׁאוּל הֲיוֹנָתָן יָמוּת אֲשֶׁר עָשָׂה הַיְשׁוּעָה הַגְּדוֹלָה הַזֹּאת בְּיִשְׂרָאֵל חָלִילָה חַי יְהוָה אִם יִפֹּל מִשַּׂעֲרַת רֹאשׁוֹ אַרְצָה כִּי עִם אֱלֹהִים עָשָׂה הַיּוֹם הַזֶּה וַיִּפְדּוּ הָעָם אֶת יוֹנָתָן וְלֹא מֵת` (שמואל-א יד:מה)
`וַיִּתְפֹּשׂ אֶת אֲגַג מֶלֶךְ עֲמָלֵק חָי וְאֶת כָּל הָעָם הֶחֱרִים לְפִי חָרֶב` (שמואל-א טו:ח)
`וַיֹּאמֶר דָּוִד אֶל הָאֲנָשִׁים הָעֹמְדִים עִמּוֹ לֵאמֹר מַה יֵּעָשֶׂה לָאִישׁ אֲשֶׁר יַכֶּה אֶת הַפְּלִשְׁתִּי הַלָּז וְהֵסִיר חֶרְפָּה מֵעַל יִשְׂרָאֵל כִּי מִי הַפְּלִשְׁתִּי הֶעָרֵל הַזֶּה כִּי חֵרֵף מַעַרְכוֹת אֱלֹהִים חַיִּים` (שמואל-א יז:כו)
`גַּם אֶת הָאֲרִי גַּם הַדּוֹב הִכָּה עַבְדֶּךָ וְהָיָה הַפְּלִשְׁתִּי הֶעָרֵל הַזֶּה כְּאַחַד מֵהֶם כִּי חֵרֵף מַעַרְכֹת אֱלֹהִים חַיִּים` (שמואל-א יז:לו)
`הַיּוֹם הַזֶּה יְסַגֶּרְךָ יְהוָה בְּיָדִי וְהִכִּיתִךָ וַהֲסִרֹתִי אֶת רֹאשְׁךָ מֵעָלֶיךָ וְנָתַתִּי פֶּגֶר מַחֲנֵה פְלִשְׁתִּים הַיּוֹם הַזֶּה לְעוֹף הַשָּׁמַיִם וּלְחַיַּת הָאָרֶץ וְיֵדְעוּ כָּל הָאָרֶץ כִּי יֵשׁ אֱלֹהִים לְיִשְׂרָאֵל` (שמואל-א יז:מו)
`וְכִרְאוֹת שָׁאוּל אֶת דָּוִד יֹצֵא לִקְרַאת הַפְּלִשְׁתִּי אָמַר אֶל אַבְנֵר שַׂר הַצָּבָא בֶּן מִי זֶה הַנַּעַר אַבְנֵר וַיֹּאמֶר אַבְנֵר חֵי נַפְשְׁךָ הַמֶּלֶךְ אִם יָדָעְתִּי` (שמואל-א יז:נה)
`וַיֹּאמֶר דָּוִד אֶל שָׁאוּל מִי אָנֹכִי וּמִי חַיַּי מִשְׁפַּחַת אָבִי בְּיִשְׂרָאֵל כִּי אֶהְיֶה חָתָן לַמֶּלֶךְ` (שמואל-א יח:יח)
`וַיִּשְׁמַע שָׁאוּל בְּקוֹל יְהוֹנָתָן וַיִּשָּׁבַע שָׁאוּל חַי יְהוָה אִם יוּמָת` (שמואל-א יט:ו)
`וַיִּשָּׁבַע עוֹד דָּוִד וַיֹּאמֶר יָדֹעַ יָדַע אָבִיךָ כִּי מָצָאתִי חֵן בְּעֵינֶיךָ וַיֹּאמֶר אַל יֵדַע זֹאת יְהוֹנָתָן פֶּן יֵעָצֵב וְאוּלָם חַי יְהוָה וְחֵי נַפְשֶׁךָ כִּי כְפֶשַׂע בֵּינִי וּבֵין הַמָּוֶת` (שמואל-א כ:ג)
`וְלֹא אִם עוֹדֶנִּי חָי וְלֹא תַעֲשֶׂה עִמָּדִי חֶסֶד יְהוָה וְלֹא אָמוּת` (שמואל-א כ:יד)
`וְהִנֵּה אֶשְׁלַח אֶת הַנַּעַר לֵךְ מְצָא אֶת הַחִצִּים אִם אָמֹר אֹמַר לַנַּעַר הִנֵּה הַחִצִּים מִמְּךָ וָהֵנָּה קָחֶנּוּ וָבֹאָה כִּי שָׁלוֹם לְךָ וְאֵין דָּבָר חַי יְהוָה` (שמואל-א כ:כא)
`כִּי כָל הַיָּמִים אֲשֶׁר בֶּן יִשַׁי חַי עַל הָאֲדָמָה לֹא תִכּוֹן אַתָּה וּמַלְכוּתֶךָ וְעַתָּה שְׁלַח וְקַח אֹתוֹ אֵלַי כִּי בֶן מָוֶת הוּא` (שמואל-א כ:לא)
`וַאֲמַרְתֶּם כֹּה לֶחָי וְאַתָּה שָׁלוֹם וּבֵיתְךָ שָׁלוֹם וְכֹל אֲשֶׁר לְךָ שָׁלוֹם` (שמואל-א כה:ו)
`וְעַתָּה אֲדֹנִי חַי יְהוָה וְחֵי נַפְשְׁךָ אֲשֶׁר מְנָעֲךָ יְהוָה מִבּוֹא בְדָמִים וְהוֹשֵׁעַ יָדְךָ לָךְ וְעַתָּה יִהְיוּ כְנָבָל אֹיְבֶיךָ וְהַמְבַקְשִׁים אֶל אֲדֹנִי רָעָה` (שמואל-א כה:כו)
`וַיָּקָם אָדָם לִרְדָפְךָ וּלְבַקֵּשׁ אֶת נַפְשֶׁךָ וְהָיְתָה נֶפֶשׁ אֲדֹנִי צְרוּרָה בִּצְרוֹר הַחַיִּים אֵת יְהוָה אֱלֹהֶיךָ וְאֵת נֶפֶשׁ אֹיְבֶיךָ יְקַלְּעֶנָּה בְּתוֹךְ כַּף הַקָּלַע` (שמואל-א כה:כט)
`וְאוּלָם חַי יְהוָה אֱלֹהֵי יִשְׂרָאֵל אֲשֶׁר מְנָעַנִי מֵהָרַע אֹתָךְ כִּי לוּלֵי מִהַרְתְּ וַתָּבֹאתִי לִקְרָאתִי כִּי אִם נוֹתַר לְנָבָל עַד אוֹר הַבֹּקֶר מַשְׁתִּין בְּקִיר` (שמואל-א כה:לד)
`וַיֹּאמֶר דָּוִד חַי יְהוָה כִּי אִם יְהוָה יִגָּפֶנּוּ אוֹ יוֹמוֹ יָבוֹא וָמֵת אוֹ בַמִּלְחָמָה יֵרֵד וְנִסְפָּה` (שמואל-א כו:י)
`לֹא טוֹב הַדָּבָר הַזֶּה אֲשֶׁר עָשִׂיתָ חַי יְהוָה כִּי בְנֵי מָוֶת אַתֶּם אֲשֶׁר לֹא שְׁמַרְתֶּם עַל אֲדֹנֵיכֶם עַל מְשִׁיחַ יְהוָה וְעַתָּה רְאֵה אֵי חֲנִית הַמֶּלֶךְ וְאֶת צַפַּחַת הַמַּיִם אֲשֶׁר מְרַאֲשֹׁתוֹ` (שמואל-א כו:טז)
`וַיִּשָּׁבַע לָהּ שָׁאוּל בַּיהוָה לֵאמֹר חַי יְהוָה אִם יִקְּרֵךְ עָוֺן בַּדָּבָר הַזֶּה` (שמואל-א כח:י)
`וַיִּקְרָא אָכִישׁ אֶל דָּוִד וַיֹּאמֶר אֵלָיו חַי יְהוָה כִּי יָשָׁר אַתָּה וְטוֹב בְּעֵינַי צֵאתְךָ וּבֹאֲךָ אִתִּי בַּמַּחֲנֶה כִּי לֹא מָצָאתִי בְךָ רָעָה מִיּוֹם בֹּאֲךָ אֵלַי עַד הַיּוֹם הַזֶּה וּבְעֵינֵי הַסְּרָנִים לֹא טוֹב אָתָּה` (שמואל-א כט:ו)
`שָׁאוּל וִיהוֹנָתָן הַנֶּאֱהָבִים וְהַנְּעִימִם בְּחַיֵּיהֶם וּבְמוֹתָם לֹא נִפְרָדוּ מִנְּשָׁרִים קַלּוּ מֵאֲרָיוֹת גָּבֵרוּ` (שמואל-ב א:כג)
`וַיֹּאמֶר יוֹאָב חַי הָאֱלֹהִים כִּי לוּלֵא דִּבַּרְתָּ כִּי אָז מֵהַבֹּקֶר נַעֲלָה הָעָם אִישׁ מֵאַחֲרֵי אָחִיו` (שמואל-ב ב:כז)
`וַיַּעַן דָּוִד אֶת רֵכָב וְאֶת בַּעֲנָה אָחִיו בְּנֵי רִמּוֹן הַבְּאֵרֹתִי וַיֹּאמֶר לָהֶם חַי יְהוָה אֲשֶׁר פָּדָה אֶת נַפְשִׁי מִכָּל צָרָה` (שמואל-ב ד:ט)
`וַיֹּאמֶר אוּרִיָּה אֶל דָּוִד הָאָרוֹן וְיִשְׂרָאֵל וִיהוּדָה יֹשְׁבִים בַּסֻּכּוֹת וַאדֹנִי יוֹאָב וְעַבְדֵי אֲדֹנִי עַל פְּנֵי הַשָּׂדֶה חֹנִים וַאֲנִי אָבוֹא אֶל בֵּיתִי לֶאֱכֹל וְלִשְׁתּוֹת וְלִשְׁכַּב עִם אִשְׁתִּי חַיֶּךָ וְחֵי נַפְשֶׁךָ אִם אֶעֱשֶׂה אֶת הַדָּבָר הַזֶּה` (שמואל-ב יא:יא)
`וַיִּחַר אַף דָּוִד בָּאִישׁ מְאֹד וַיֹּאמֶר אֶל נָתָן חַי יְהוָה כִּי בֶן מָוֶת הָאִישׁ הָעֹשֶׂה זֹאת` (שמואל-ב יב:ה)
`וַיְהִי בַּיּוֹם הַשְּׁבִיעִי וַיָּמָת הַיָּלֶד וַיִּרְאוּ עַבְדֵי דָוִד לְהַגִּיד לוֹ כִּי מֵת הַיֶּלֶד כִּי אָמְרוּ הִנֵּה בִהְיוֹת הַיֶּלֶד חַי דִּבַּרְנוּ אֵלָיו וְלֹא שָׁמַע בְּקוֹלֵנוּ וְאֵיךְ נֹאמַר אֵלָיו מֵת הַיֶּלֶד וְעָשָׂה רָעָה` (שמואל-ב יב:יח)
`וַיֹּאמְרוּ עֲבָדָיו אֵלָיו מָה הַדָּבָר הַזֶּה אֲשֶׁר עָשִׂיתָה בַּעֲבוּר הַיֶּלֶד חַי צַמְתָּ וַתֵּבְךְּ וְכַאֲשֶׁר מֵת הַיֶּלֶד קַמְתָּ וַתֹּאכַל לָחֶם` (שמואל-ב יב:כא)
`וַיֹּאמֶר בְּעוֹד הַיֶּלֶד חַי צַמְתִּי וָאֶבְכֶּה כִּי אָמַרְתִּי מִי יוֹדֵעַ יְחָנַּנִי יְהוָה וְחַי הַיָּלֶד` (שמואל-ב יב:כב)
`וַתֹּאמֶר יִזְכָּר נָא הַמֶּלֶךְ אֶת יְהוָה אֱלֹהֶיךָ מֵהַרְבִּית גֹּאֵל הַדָּם לְשַׁחֵת וְלֹא יַשְׁמִידוּ אֶת בְּנִי וַיֹּאמֶר חַי יְהוָה אִם יִפֹּל מִשַּׂעֲרַת בְּנֵךְ אָרְצָה` (שמואל-ב יד:יא)
`וַיֹּאמֶר הַמֶּלֶךְ הֲיַד יוֹאָב אִתָּךְ בְּכָל זֹאת וַתַּעַן הָאִשָּׁה וַתֹּאמֶר חֵי נַפְשְׁךָ אֲדֹנִי הַמֶּלֶךְ אִם אִשׁ לְהֵמִין וּלְהַשְׂמִיל מִכֹּל אֲשֶׁר דִּבֶּר אֲדֹנִי הַמֶּלֶךְ כִּי עַבְדְּךָ יוֹאָב הוּא צִוָּנִי וְהוּא שָׂם בְּפִי שִׁפְחָתְךָ אֵת כָּל הַדְּבָרִים הָאֵלֶּה` (שמואל-ב יד:יט)
`וַיַּעַן אִתַּי אֶת הַמֶּלֶךְ וַיֹּאמַר חַי יְהוָה וְחֵי אֲדֹנִי הַמֶּלֶךְ כִּי אִם בִּמְקוֹם אֲשֶׁר יִהְיֶה שָּׁם אֲדֹנִי הַמֶּלֶךְ אִם לְמָוֶת אִם לְחַיִּים כִּי שָׁם יִהְיֶה עַבְדֶּךָ` (שמואל-ב טו:כא)
`וַיֹּאמֶר יוֹאָב לֹא כֵן אֹחִילָה לְפָנֶיךָ וַיִּקַּח שְׁלֹשָׁה שְׁבָטִים בְּכַפּוֹ וַיִּתְקָעֵם בְּלֵב אַבְשָׁלוֹם עוֹדֶנּוּ חַי בְּלֵב הָאֵלָה` (שמואל-ב יח:יד)
`וְאַבְשָׁלֹם לָקַח וַיַּצֶּב לוֹ בְחַיָּו אֶת מַצֶּבֶת אֲשֶׁר בְּעֵמֶק הַמֶּלֶךְ כִּי אָמַר אֵין לִי בֵן בַּעֲבוּר הַזְכִּיר שְׁמִי וַיִּקְרָא לַמַּצֶּבֶת עַל שְׁמוֹ וַיִּקָּרֵא לָהּ יַד אַבְשָׁלֹם עַד הַיּוֹם הַזֶּה` (שמואל-ב יח:יח)
`לְאַהֲבָה אֶת שֹׂנְאֶיךָ וְלִשְׂנֹא אֶת אֹהֲבֶיךָ כִּי הִגַּדְתָּ הַיּוֹם כִּי אֵין לְךָ שָׂרִים וַעֲבָדִים כִּי יָדַעְתִּי הַיּוֹם כִּי לֹא אַבְשָׁלוֹם חַי וְכֻלָּנוּ הַיּוֹם מֵתִים כִּי אָז יָשָׁר בְּעֵינֶיךָ` (שמואל-ב יט:ו)
`וַיֹּאמֶר בַּרְזִלַּי אֶל הַמֶּלֶךְ כַּמָּה יְמֵי שְׁנֵי חַיַּי כִּי אֶעֱלֶה אֶת הַמֶּלֶךְ יְרוּשָׁלִָם` (שמואל-ב יט:לד)
`וַתִּקַּח רִצְפָּה בַת אַיָּה אֶת הַשַּׂק וַתַּטֵּהוּ לָהּ אֶל הַצּוּר מִתְּחִלַּת קָצִיר עַד נִתַּךְ מַיִם עֲלֵיהֶם מִן הַשָּׁמָיִם וְלֹא נָתְנָה עוֹף הַשָּׁמַיִם לָנוּחַ עֲלֵיהֶם יוֹמָם וְאֶת חַיַּת הַשָּׂדֶה לָיְלָה` (שמואל-ב כא:י)
`חַי יְהוָה וּבָרוּךְ צוּרִי וְיָרֻם אֱלֹהֵי צוּר יִשְׁעִי` (שמואל-ב כב:מז)
`וַיֵּרְדוּ שְׁלֹשִׁים מֵהַשְּׁלֹשִׁים רֹאשׁ וַיָּבֹאוּ אֶל קָצִיר אֶל דָּוִד אֶל מְעָרַת עֲדֻלָּם וְחַיַּת פְּלִשְׁתִּים חֹנָה בְּעֵמֶק רְפָאִים` (שמואל-ב כג:יג)
`וּבְנָיָהוּ בֶן יְהוֹיָדָע בֶּן אִישׁ חַי רַב פְּעָלִים מִקַּבְצְאֵל הוּא הִכָּה אֵת שְׁנֵי אֲרִאֵל מוֹאָב וְהוּא יָרַד וְהִכָּה אֶת הָאֲרִיַּה בְּתוֹךְ הַבֹּאר בְּיוֹם הַשָּׁלֶג` (שמואל-ב כג:כ)
`וַיִּשָּׁבַע הַמֶּלֶךְ וַיֹּאמַר חַי יְהוָה אֲשֶׁר פָּדָה אֶת נַפְשִׁי מִכָּל צָרָה` (מלכים-א א:כט)
`וְעַתָּה חַי יְהוָה אֲשֶׁר הֱכִינַנִי וַיּוֹשִׁיבִינִי עַל כִּסֵּא דָּוִד אָבִי וַאֲשֶׁר עָשָׂה לִי בַּיִת כַּאֲשֶׁר דִּבֵּר כִּי הַיּוֹם יוּמַת אֲדֹנִיָּהוּ` (מלכים-א ב:כד)
`וַתֹּאמֶר הָאִשָּׁה הָאַחֶרֶת לֹא כִי בְּנִי הַחַי וּבְנֵךְ הַמֵּת וְזֹאת אֹמֶרֶת לֹא כִי בְּנֵךְ הַמֵּת וּבְנִי הֶחָי וַתְּדַבֵּרְנָה לִפְנֵי הַמֶּלֶךְ` (מלכים-א ג:כב)
`וַיֹּאמֶר הַמֶּלֶךְ זֹאת אֹמֶרֶת זֶה בְּנִי הַחַי וּבְנֵךְ הַמֵּת וְזֹאת אֹמֶרֶת לֹא כִי בְּנֵךְ הַמֵּת וּבְנִי הֶחָי` (מלכים-א ג:כג)
`וַיֹּאמֶר הַמֶּלֶךְ גִּזְרוּ אֶת הַיֶּלֶד הַחַי לִשְׁנָיִם וּתְנוּ אֶת הַחֲצִי לְאַחַת וְאֶת הַחֲצִי לְאֶחָת` (מלכים-א ג:כה)
`וַתֹּאמֶר הָאִשָּׁה אֲשֶׁר בְּנָהּ הַחַי אֶל הַמֶּלֶךְ כִּי נִכְמְרוּ רַחֲמֶיהָ עַל בְּנָהּ וַתֹּאמֶר בִּי אֲדֹנִי תְּנוּ לָהּ אֶת הַיָּלוּד הַחַי וְהָמֵת אַל תְּמִיתֻהוּ וְזֹאת אֹמֶרֶת גַּם לִי גַם לָךְ לֹא יִהְיֶה גְּזֹרוּ` (מלכים-א ג:כו)
`וַיַּעַן הַמֶּלֶךְ וַיֹּאמֶר תְּנוּ לָהּ אֶת הַיָּלוּד הַחַי וְהָמֵת לֹא תְמִיתֻהוּ הִיא אִמּוֹ` (מלכים-א ג:כז)
`וּשְׁלֹמֹה הָיָה מוֹשֵׁל בְּכָל הַמַּמְלָכוֹת מִן הַנָּהָר אֶרֶץ פְּלִשְׁתִּים וְעַד גְּבוּל מִצְרָיִם מַגִּשִׁים מִנְחָה וְעֹבְדִים אֶת שְׁלֹמֹה כָּל יְמֵי חַיָּיו` (מלכים-א ד:כא)
`לְמַעַן יִרָאוּךָ כָּל הַיָּמִים אֲשֶׁר הֵם חַיִּים עַל פְּנֵי הָאֲדָמָה אֲשֶׁר נָתַתָּה לַאֲבֹתֵינוּ` (מלכים-א ח:מ)
`וְלֹא אֶקַּח אֶת כָּל הַמַּמְלָכָה מִיָּדוֹ כִּי נָשִׂיא אֲשִׁתֶנּוּ כֹּל יְמֵי חַיָּיו לְמַעַן דָּוִד עַבְדִּי אֲשֶׁר בָּחַרְתִּי אֹתוֹ אֲשֶׁר שָׁמַר מִצְוֺתַי וְחֻקֹּתָי` (מלכים-א יא:לד)
`וַיִּוָּעַץ הַמֶּלֶךְ רְחַבְעָם אֶת הַזְּקֵנִים אֲשֶׁר הָיוּ עֹמְדִים אֶת פְּנֵי שְׁלֹמֹה אָבִיו בִּהְיֹתוֹ חַי לֵאמֹר אֵיךְ אַתֶּם נוֹעָצִים לְהָשִׁיב אֶת הָעָם הַזֶּה דָּבָר` (מלכים-א יב:ו)
`אֲשֶׁר עָשָׂה דָוִד אֶת הַיָּשָׁר בְּעֵינֵי יְהוָה וְלֹא סָר מִכֹּל אֲשֶׁר צִוָּהוּ כֹּל יְמֵי חַיָּיו רַק בִּדְבַר אוּרִיָּה הַחִתִּי` (מלכים-א טו:ה)
`וּמִלְחָמָה הָיְתָה בֵין רְחַבְעָם וּבֵין יָרָבְעָם כָּל יְמֵי חַיָּיו` (מלכים-א טו:ו)
`וַיֹּאמֶר אֵלִיָּהוּ הַתִּשְׁבִּי מִתֹּשָׁבֵי גִלְעָד אֶל אַחְאָב חַי יְהוָה אֱלֹהֵי יִשְׂרָאֵל אֲשֶׁר עָמַדְתִּי לְפָנָיו אִם יִהְיֶה הַשָּׁנִים הָאֵלֶּה טַל וּמָטָר כִּי אִם לְפִי דְבָרִי` (מלכים-א יז:א)
`וַתֹּאמֶר חַי יְהוָה אֱלֹהֶיךָ אִם יֶשׁ לִי מָעוֹג כִּי אִם מְלֹא כַף קֶמַח בַּכַּד וּמְעַט שֶׁמֶן בַּצַּפָּחַת וְהִנְנִי מְקֹשֶׁשֶׁת שְׁנַיִם עֵצִים וּבָאתִי וַעֲשִׂיתִיהוּ לִי וְלִבְנִי וַאֲכַלְנֻהוּ וָמָתְנוּ` (מלכים-א יז:יב)
`וַיִּקַּח אֵלִיָּהוּ אֶת הַיֶּלֶד וַיֹּרִדֵהוּ מִן הָעֲלִיָּה הַבַּיְתָה וַיִּתְּנֵהוּ לְאִמּוֹ וַיֹּאמֶר אֵלִיָּהוּ רְאִי חַי בְּנֵךְ` (מלכים-א יז:כג)
`חַי יְהוָה אֱלֹהֶיךָ אִם יֶשׁ גּוֹי וּמַמְלָכָה אֲשֶׁר לֹא שָׁלַח אֲדֹנִי שָׁם לְבַקֶּשְׁךָ וְאָמְרוּ אָיִן וְהִשְׁבִּיעַ אֶת הַמַּמְלָכָה וְאֶת הַגּוֹי כִּי לֹא יִמְצָאֶכָּה` (מלכים-א יח:י)
`וַיֹּאמֶר אֵלִיָּהוּ חַי יְהוָה צְבָאוֹת אֲשֶׁר עָמַדְתִּי לְפָנָיו כִּי הַיּוֹם אֵרָאֶה אֵלָיו` (מלכים-א יח:טו)
`וַיֹּאמֶר אִם לְשָׁלוֹם יָצָאוּ תִּפְשׂוּם חַיִּים וְאִם לְמִלְחָמָה יָצָאוּ חַיִּים תִּפְשׂוּם` (מלכים-א כ:יח)
`וַיַּחְגְּרוּ שַׂקִּים בְּמָתְנֵיהֶם וַחֲבָלִים בְּרָאשֵׁיהֶם וַיָּבֹאוּ אֶל מֶלֶךְ יִשְׂרָאֵל וַיֹּאמְרוּ עַבְדְּךָ בֶן הֲדַד אָמַר תְּחִי נָא נַפְשִׁי וַיֹּאמֶר הַעוֹדֶנּוּ חַי אָחִי הוּא` (מלכים-א כ:לב)
`וַיְהִי כִּשְׁמֹעַ אִיזֶבֶל כִּי סֻקַּל נָבוֹת וַיָּמֹת וַתֹּאמֶר אִיזֶבֶל אֶל אַחְאָב קוּם רֵשׁ אֶת כֶּרֶם נָבוֹת הַיִּזְרְעֵאלִי אֲשֶׁר מֵאֵן לָתֶת לְךָ בְכֶסֶף כִּי אֵין נָבוֹת חַי כִּי מֵת` (מלכים-א כא:טו)
`וַיֹּאמֶר מִיכָיְהוּ חַי יְהוָה כִּי אֶת אֲשֶׁר יֹאמַר יְהוָה אֵלַי אֹתוֹ אֲדַבֵּר` (מלכים-א כב:יד)
`וַיֹּאמֶר אֵלִיָּהוּ אֶל אֱלִישָׁע שֵׁב נָא פֹה כִּי יְהוָה שְׁלָחַנִי עַד בֵּית אֵל וַיֹּאמֶר אֱלִישָׁע חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיֵּרְדוּ בֵּית אֵל` (מלכים-ב ב:ב)
`וַיֹּאמֶר לוֹ אֵלִיָּהוּ אֱלִישָׁע שֵׁב נָא פֹה כִּי יְהוָה שְׁלָחַנִי יְרִיחוֹ וַיֹּאמֶר חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיָּבֹאוּ יְרִיחוֹ` (מלכים-ב ב:ד)
`וַיֹּאמֶר לוֹ אֵלִיָּהוּ שֵׁב נָא פֹה כִּי יְהוָה שְׁלָחַנִי הַיַּרְדֵּנָה וַיֹּאמֶר חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיֵּלְכוּ שְׁנֵיהֶם` (מלכים-ב ב:ו)
`וַיֹּאמֶר אֱלִישָׁע חַי יְהוָה צְבָאוֹת אֲשֶׁר עָמַדְתִּי לְפָנָיו כִּי לוּלֵי פְּנֵי יְהוֹשָׁפָט מֶלֶךְ יְהוּדָה אֲנִי נֹשֵׂא אִם אַבִּיט אֵלֶיךָ וְאִם אֶרְאֶךָּ` (מלכים-ב ג:יד)
`וַיֹּאמֶר לַמּוֹעֵד הַזֶּה כָּעֵת חַיָּה אַתִּי חֹבֶקֶת בֵּן וַתֹּאמֶר אַל אֲדֹנִי אִישׁ הָאֱלֹהִים אַל תְּכַזֵּב בְּשִׁפְחָתֶךָ` (מלכים-ב ד:טז)
`וַתַּהַר הָאִשָּׁה וַתֵּלֶד בֵּן לַמּוֹעֵד הַזֶּה כָּעֵת חַיָּה אֲשֶׁר דִּבֶּר אֵלֶיהָ אֱלִישָׁע` (מלכים-ב ד:יז)
`וַתֹּאמֶר אֵם הַנַּעַר חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיָּקָם וַיֵּלֶךְ אַחֲרֶיהָ` (מלכים-ב ד:ל)
`וַיֹּאמֶר חַי יְהוָה אֲשֶׁר עָמַדְתִּי לְפָנָיו אִם אֶקָּח וַיִּפְצַר בּוֹ לָקַחַת וַיְמָאֵן` (מלכים-ב ה:טז)
`וַיֹּאמֶר גֵּיחֲזִי נַעַר אֱלִישָׁע אִישׁ הָאֱלֹהִים הִנֵּה חָשַׂךְ אֲדֹנִי אֶת נַעֲמָן הָאֲרַמִּי הַזֶּה מִקַּחַת מִיָּדוֹ אֵת אֲשֶׁר הֵבִיא חַי יְהוָה כִּי אִם רַצְתִּי אַחֲרָיו וְלָקַחְתִּי מֵאִתּוֹ מְאוּמָה` (מלכים-ב ה:כ)
`וַיָּקָם הַמֶּלֶךְ לַיְלָה וַיֹּאמֶר אֶל עֲבָדָיו אַגִּידָה נָּא לָכֶם אֵת אֲשֶׁר עָשׂוּ לָנוּ אֲרָם יָדְעוּ כִּי רְעֵבִים אֲנַחְנוּ וַיֵּצְאוּ מִן הַמַּחֲנֶה לְהֵחָבֵה בְהַשָּׂדֶה לֵאמֹר כִּי יֵצְאוּ מִן הָעִיר וְנִתְפְּשֵׂם חַיִּים וְאֶל הָעִיר נָבֹא` (מלכים-ב ז:יב)
`וַיֹּאמֶר תִּפְשׂוּם חַיִּים וַיִּתְפְּשׂוּם חַיִּים וַיִּשְׁחָטוּם אֶל בּוֹר בֵּית עֵקֶד אַרְבָּעִים וּשְׁנַיִם אִישׁ וְלֹא הִשְׁאִיר אִישׁ מֵהֶם` (מלכים-ב י:יד)
`וַיִּשְׁלַח יְהוֹאָשׁ מֶלֶךְ יִשְׂרָאֵל אֶל אֲמַצְיָהוּ מֶלֶךְ יְהוּדָה לֵאמֹר הַחוֹחַ אֲשֶׁר בַּלְּבָנוֹן שָׁלַח אֶל הָאֶרֶז אֲשֶׁר בַּלְּבָנוֹן לֵאמֹר תְּנָה אֶת בִּתְּךָ לִבְנִי לְאִשָּׁה וַתַּעֲבֹר חַיַּת הַשָּׂדֶה אֲשֶׁר בַּלְּבָנוֹן וַתִּרְמֹס אֶת הַחוֹחַ` (מלכים-ב יד:ט)
`אוּלַי יִשְׁמַע יְהוָה אֱלֹהֶיךָ אֵת כָּל דִּבְרֵי רַב שָׁקֵה אֲשֶׁר שְׁלָחוֹ מֶלֶךְ אַשּׁוּר אֲדֹנָיו לְחָרֵף אֱלֹהִים חַי וְהוֹכִיחַ בַּדְּבָרִים אֲשֶׁר שָׁמַע יְהוָה אֱלֹהֶיךָ וְנָשָׂאתָ תְפִלָּה בְּעַד הַשְּׁאֵרִית הַנִּמְצָאָה` (מלכים-ב יט:ד)
`הַטֵּה יְהוָה אָזְנְךָ וּשֲׁמָע פְּקַח יְהוָה עֵינֶיךָ וּרְאֵה וּשְׁמַע אֵת דִּבְרֵי סַנְחֵרִיב אֲשֶׁר שְׁלָחוֹ לְחָרֵף אֱלֹהִים חָי` (מלכים-ב יט:טז)
`וְשִׁנָּא אֵת בִּגְדֵי כִלְאוֹ וְאָכַל לֶחֶם תָּמִיד לְפָנָיו כָּל יְמֵי חַיָּיו` (מלכים-ב כה:כט)
`וַאֲרֻחָתוֹ אֲרֻחַת תָּמִיד נִתְּנָה לּוֹ מֵאֵת הַמֶּלֶךְ דְּבַר יוֹם בְּיוֹמוֹ כֹּל יְמֵי חַיָּו` (מלכים-ב כה:ל)
`וְהָיָה הַנִּשְׁאָר בְּצִיּוֹן וְהַנּוֹתָר בִּירוּשָׁלִַם קָדוֹשׁ יֵאָמֶר לוֹ כָּל הַכָּתוּב לַחַיִּים בִּירוּשָׁלִָם` (ישעיהו ד:ג)
`וְכִי יֹאמְרוּ אֲלֵיכֶם דִּרְשׁוּ אֶל הָאֹבוֹת וְאֶל הַיִּדְּעֹנִים הַמְצַפְצְפִים וְהַמַּהְגִּים הֲלוֹא עַם אֶל אֱלֹהָיו יִדְרֹשׁ בְּעַד הַחַיִּים אֶל הַמֵּתִים` (ישעיהו ח:יט)
`לֹא יִהְיֶה שָׁם אַרְיֵה וּפְרִיץ חַיּוֹת בַּל יַעֲלֶנָּה לֹא תִמָּצֵא שָׁם וְהָלְכוּ גְּאוּלִים` (ישעיהו לה:ט)
`אוּלַי יִשְׁמַע יְהוָה אֱלֹהֶיךָ אֵת דִּבְרֵי רַב שָׁקֵה אֲשֶׁר שְׁלָחוֹ מֶלֶךְ אַשּׁוּר אֲדֹנָיו לְחָרֵף אֱלֹהִים חַי וְהוֹכִיחַ בַּדְּבָרִים אֲשֶׁר שָׁמַע יְהוָה אֱלֹהֶיךָ וְנָשָׂאתָ תְפִלָּה בְּעַד הַשְּׁאֵרִית הַנִּמְצָאָה` (ישעיהו לז:ד)
`הַטֵּה יְהוָה אָזְנְךָ וּשְׁמָע פְּקַח יְהוָה עֵינֶךָ וּרְאֵה וּשְׁמַע אֵת כָּל דִּבְרֵי סַנְחֵרִיב אֲשֶׁר שָׁלַח לְחָרֵף אֱלֹהִים חָי` (ישעיהו לז:יז)
`אָמַרְתִּי לֹא אֶרְאֶה יָהּ יָהּ בְּאֶרֶץ הַחַיִּים לֹא אַבִּיט אָדָם עוֹד עִם יוֹשְׁבֵי חָדֶל` (ישעיהו לח:יא)
`דּוֹרִי נִסַּע וְנִגְלָה מִנִּי כְּאֹהֶל רֹעִי קִפַּדְתִּי כָאֹרֵג חַיַּי מִדַּלָּה יְבַצְּעֵנִי מִיּוֹם עַד לַיְלָה תַּשְׁלִימֵנִי` (ישעיהו לח:יב)
`אֲדֹנָי עֲלֵיהֶם יִחְיוּ וּלְכָל בָּהֶן חַיֵּי רוּחִי וְתַחֲלִימֵנִי וְהַחֲיֵנִי` (ישעיהו לח:טז)
`חַי חַי הוּא יוֹדֶךָ כָּמוֹנִי הַיּוֹם אָב לְבָנִים יוֹדִיעַ אֶל אֲמִתֶּךָ` (ישעיהו לח:יט)
`יְהוָה לְהוֹשִׁיעֵנִי וּנְגִנוֹתַי נְנַגֵּן כָּל יְמֵי חַיֵּינוּ עַל בֵּית יְהוָה` (ישעיהו לח:כ)
`וּלְבָנוֹן אֵין דֵּי בָּעֵר וְחַיָּתוֹ אֵין דֵּי עוֹלָה` (ישעיהו מ:טז)
`תְּכַבְּדֵנִי חַיַּת הַשָּׂדֶה תַּנִּים וּבְנוֹת יַעֲנָה כִּי נָתַתִּי בַמִּדְבָּר מַיִם נְהָרוֹת בִּישִׁימֹן לְהַשְׁקוֹת עַמִּי בְחִירִי` (ישעיהו מג:כ)
`כָּרַע בֵּל קֹרֵס נְבוֹ הָיוּ עֲצַבֵּיהֶם לַחַיָּה וְלַבְּהֵמָה נְשֻׂאֹתֵיכֶם עֲמוּסוֹת מַשָּׂא לַעֲיֵפָה` (ישעיהו מו:א)
`שְׂאִי סָבִיב עֵינַיִךְ וּרְאִי כֻּלָּם נִקְבְּצוּ בָאוּ לָךְ חַי אָנִי נְאֻם יְהוָה כִּי כֻלָּם כָּעֲדִי תִלְבָּשִׁי וּתְקַשְּׁרִים כַּכַּלָּה` (ישעיהו מט:יח)
`מֵעֹצֶר וּמִמִּשְׁפָּט לֻקָּח וְאֶת דּוֹרוֹ מִי יְשׂוֹחֵחַ כִּי נִגְזַר מֵאֶרֶץ חַיִּים מִפֶּשַׁע עַמִּי נֶגַע לָמוֹ` (ישעיהו נג:ח)
`כֹּל חַיְתוֹ שָׂדָי אֵתָיוּ לֶאֱכֹל כָּל חַיְתוֹ בַּיָּעַר` (ישעיהו נו:ט)
`בְּרֹב דַּרְכֵּךְ יָגַעַתְּ לֹא אָמַרְתְּ נוֹאָשׁ חַיַּת יָדֵךְ מָצָאת עַל כֵּן לֹא חָלִית` (ישעיהו נז:י)
`כִּי שְׁתַּיִם רָעוֹת עָשָׂה עַמִּי אֹתִי עָזְבוּ מְקוֹר מַיִם חַיִּים לַחְצֹב לָהֶם בֹּארוֹת בֹּארֹת נִשְׁבָּרִים אֲשֶׁר לֹא יָכִלוּ הַמָּיִם` (ירמיהו ב:יג)
`וְנִשְׁבַּעְתָּ חַי יְהוָה בֶּאֱמֶת בְּמִשְׁפָּט וּבִצְדָקָה וְהִתְבָּרְכוּ בוֹ גּוֹיִם וּבוֹ יִתְהַלָּלוּ` (ירמיהו ד:ב)
`וְאִם חַי יְהֹוָה יֹאמֵרוּ לָכֵן לַשֶּׁקֶר יִשָּׁבֵעוּ` (ירמיהו ה:ב)
`וְנִבְחַר מָוֶת מֵחַיִּים לְכֹל הַשְּׁאֵרִית הַנִּשְׁאָרִים מִן הַמִּשְׁפָּחָה הָרָעָה הַזֹּאת בְּכָל הַמְּקֹמוֹת הַנִּשְׁאָרִים אֲשֶׁר הִדַּחְתִּים שָׁם נְאֻם יְהוָה צְבָאוֹת` (ירמיהו ח:ג)
`וַיהוָה אֱלֹהִים אֱמֶת הוּא אֱלֹהִים חַיִּים וּמֶלֶךְ עוֹלָם מִקִּצְפּוֹ תִּרְעַשׁ הָאָרֶץ וְלֹא יָכִלוּ גוֹיִם זַעְמוֹ` (ירמיהו י:י)
`וַאֲנִי כְּכֶבֶשׂ אַלּוּף יוּבַל לִטְבוֹחַ וְלֹא יָדַעְתִּי כִּי עָלַי חָשְׁבוּ מַחֲשָׁבוֹת נַשְׁחִיתָה עֵץ בְּלַחְמוֹ וְנִכְרְתֶנּוּ מֵאֶרֶץ חַיִּים וּשְׁמוֹ לֹא יִזָּכֵר עוֹד` (ירמיהו יא:יט)
`הַעַיִט צָבוּעַ נַחֲלָתִי לִי הַעַיִט סָבִיב עָלֶיהָ לְכוּ אִסְפוּ כָּל חַיַּת הַשָּׂדֶה הֵתָיוּ לְאָכְלָה` (ירמיהו יב:ט)
`וְהָיָה אִם לָמֹד יִלְמְדוּ אֶת דַּרְכֵי עַמִּי לְהִשָּׁבֵעַ בִּשְׁמִי חַי יְהוָה כַּאֲשֶׁר לִמְּדוּ אֶת עַמִּי לְהִשָּׁבֵעַ בַּבָּעַל וְנִבְנוּ בְּתוֹךְ עַמִּי` (ירמיהו יב:טז)
`לָכֵן הִנֵּה יָמִים בָּאִים נְאֻם יְהוָה וְלֹא יֵאָמֵר עוֹד חַי יְהוָה אֲשֶׁר הֶעֱלָה אֶת בְּנֵי יִשְׂרָאֵל מֵאֶרֶץ מִצְרָיִם` (ירמיהו טז:יד)
`כִּי אִם חַי יְהוָה אֲשֶׁר הֶעֱלָה אֶת בְּנֵי יִשְׂרָאֵל מֵאֶרֶץ צָפוֹן וּמִכֹּל הָאֲרָצוֹת אֲשֶׁר הִדִּיחָם שָׁמָּה וַהֲשִׁבֹתִים עַל אַדְמָתָם אֲשֶׁר נָתַתִּי לַאֲבוֹתָם` (ירמיהו טז:טו)
`מִקְוֵה יִשְׂרָאֵל יְהוָה כָּל עֹזְבֶיךָ יֵבֹשׁוּ יִסּוֹרַי בָּאָרֶץ יִכָּתֵבוּ כִּי עָזְבוּ מְקוֹר מַיִם חַיִּים אֶת יְהוָה` (ירמיהו יז:יג)
`וְאֶל הָעָם הַזֶּה תֹּאמַר כֹּה אָמַר יְהוָה הִנְנִי נֹתֵן לִפְנֵיכֶם אֶת דֶּרֶךְ הַחַיִּים וְאֶת דֶּרֶךְ הַמָּוֶת` (ירמיהו כא:ח)
`חַי אָנִי נְאֻם יְהוָה כִּי אִם יִהְיֶה כָּנְיָהוּ בֶן יְהוֹיָקִים מֶלֶךְ יְהוּדָה חוֹתָם עַל יַד יְמִינִי כִּי מִשָּׁם אֶתְּקֶנְךָּ` (ירמיהו כב:כד)
`לָכֵן הִנֵּה יָמִים בָּאִים נְאֻם יְהוָה וְלֹא יֹאמְרוּ עוֹד חַי יְהוָה אֲשֶׁר הֶעֱלָה אֶת בְּנֵי יִשְׂרָאֵל מֵאֶרֶץ מִצְרָיִם` (ירמיהו כג:ז)
`כִּי אִם חַי יְהוָה אֲשֶׁר הֶעֱלָה וַאֲשֶׁר הֵבִיא אֶת זֶרַע בֵּית יִשְׂרָאֵל מֵאֶרֶץ צָפוֹנָה וּמִכֹּל הָאֲרָצוֹת אֲשֶׁר הִדַּחְתִּים שָׁם וְיָשְׁבוּ עַל אַדְמָתָם` (ירמיהו כג:ח)
`וּמַשָּׂא יְהוָה לֹא תִזְכְּרוּ עוֹד כִּי הַמַּשָּׂא יִהְיֶה לְאִישׁ דְּבָרוֹ וַהֲפַכְתֶּם אֶת דִּבְרֵי אֱלֹהִים חַיִּים יְהוָה צְבָאוֹת אֱלֹהֵינוּ` (ירמיהו כג:לו)
`וְעַתָּה אָנֹכִי נָתַתִּי אֶת כָּל הָאֲרָצוֹת הָאֵלֶּה בְּיַד נְבוּכַדְנֶאצַּר מֶלֶךְ בָּבֶל עַבְדִּי וְגַם אֶת חַיַּת הַשָּׂדֶה נָתַתִּי לוֹ לְעָבְדוֹ` (ירמיהו כז:ו)
`כִּי כֹה אָמַר יְהוָה צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל עֹל בַּרְזֶל נָתַתִּי עַל צַוַּאר כָּל הַגּוֹיִם הָאֵלֶּה לַעֲבֹד אֶת נְבֻכַדְנֶאצַּר מֶלֶךְ בָּבֶל וַעֲבָדֻהוּ וְגַם אֶת חַיַּת הַשָּׂדֶה נָתַתִּי לוֹ` (ירמיהו כח:יד)
`וַיִּשָּׁבַע הַמֶּלֶךְ צִדְקִיָּהוּ אֶל יִרְמְיָהוּ בַּסֵּתֶר לֵאמֹר חַי יְהוָה את אֲשֶׁר עָשָׂה לָנוּ אֶת הַנֶּפֶשׁ הַזֹּאת אִם אֲמִיתֶךָ וְאִם אֶתֶּנְךָ בְּיַד הָאֲנָשִׁים הָאֵלֶּה אֲשֶׁר מְבַקְשִׁים אֶת נַפְשֶׁךָ` (ירמיהו לח:טז)
`לָכֵן שִׁמְעוּ דְבַר יְהוָה כָּל יְהוּדָה הַיֹּשְׁבִים בְּאֶרֶץ מִצְרָיִם הִנְנִי נִשְׁבַּעְתִּי בִּשְׁמִי הַגָּדוֹל אָמַר יְהוָה אִם יִהְיֶה עוֹד שְׁמִי נִקְרָא בְּפִי כָּל אִישׁ יְהוּדָה אֹמֵר חַי אֲדֹנָי יְהוִה בְּכָל אֶרֶץ מִצְרָיִם` (ירמיהו מד:כו)
`חַי אָנִי נְאֻם הַמֶּלֶךְ יְהוָה צְבָאוֹת שְׁמוֹ כִּי כְּתָבוֹר בֶּהָרִים וּכְכַרְמֶל בַּיָּם יָבוֹא` (ירמיהו מו:יח)
`וְשִׁנָּה אֵת בִּגְדֵי כִלְאוֹ וְאָכַל לֶחֶם לְפָנָיו תָּמִיד כָּל יְמֵי חַיָּו` (ירמיהו נב:לג)
`וַאֲרֻחָתוֹ אֲרֻחַת תָּמִיד נִתְּנָה לּוֹ מֵאֵת מֶלֶךְ בָּבֶל דְּבַר יוֹם בְּיוֹמוֹ עַד יוֹם מוֹתוֹ כֹּל יְמֵי חַיָּיו` (ירמיהו נב:לד)
`וּמִתּוֹכָהּ דְּמוּת אַרְבַּע חַיּוֹת וְזֶה מַרְאֵיהֶן דְּמוּת אָדָם לָהֵנָּה` (יחזקאל א:ה)
`וּדְמוּת הַחַיּוֹת מַרְאֵיהֶם כְּגַחֲלֵי אֵשׁ בֹּעֲרוֹת כְּמַרְאֵה הַלַּפִּדִים הִיא מִתְהַלֶּכֶת בֵּין הַחַיּוֹת וְנֹגַהּ לָאֵשׁ וּמִן הָאֵשׁ יוֹצֵא בָרָק` (יחזקאל א:יג)
`וְהַחַיּוֹת רָצוֹא וָשׁוֹב כְּמַרְאֵה הַבָּזָק` (יחזקאל א:יד)
`וָאֵרֶא הַחַיּוֹת וְהִנֵּה אוֹפַן אֶחָד בָּאָרֶץ אֵצֶל הַחַיּוֹת לְאַרְבַּעַת פָּנָיו` (יחזקאל א:טו)
`וּבְלֶכֶת הַחַיּוֹת יֵלְכוּ הָאוֹפַנִּים אֶצְלָם וּבְהִנָּשֵׂא הַחַיּוֹת מֵעַל הָאָרֶץ יִנָּשְׂאוּ הָאוֹפַנִּים` (יחזקאל א:יט)
`עַל אֲשֶׁר יִהְיֶה שָּׁם הָרוּחַ לָלֶכֶת יֵלֵכוּ שָׁמָּה הָרוּחַ לָלֶכֶת וְהָאוֹפַנִּים יִנָּשְׂאוּ לְעֻמָּתָם כִּי רוּחַ הַחַיָּה בָּאוֹפַנִּים` (יחזקאל א:כ)
`בְּלֶכְתָּם יֵלֵכוּ וּבְעָמְדָם יַעֲמֹדוּ וּבְהִנָּשְׂאָם מֵעַל הָאָרֶץ יִנָּשְׂאוּ הָאוֹפַנִּים לְעֻמָּתָם כִּי רוּחַ הַחַיָּה בָּאוֹפַנִּים` (יחזקאל א:כא)
`וּדְמוּת עַל רָאשֵׁי הַחַיָּה רָקִיעַ כְּעֵין הַקֶּרַח הַנּוֹרָא נָטוּי עַל רָאשֵׁיהֶם מִלְמָעְלָה` (יחזקאל א:כב)
`וְקוֹל כַּנְפֵי הַחַיּוֹת מַשִּׁיקוֹת אִשָּׁה אֶל אֲחוֹתָהּ וְקוֹל הָאוֹפַנִּים לְעֻמָּתָם וְקוֹל רַעַשׁ גָּדוֹל` (יחזקאל ג:יג)
`לָכֵן חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם לֹא יַעַן אֶת מִקְדָּשִׁי טִמֵּאת בְּכָל שִׁקּוּצַיִךְ וּבְכָל תּוֹעֲבֹתָיִךְ וְגַם אֲנִי אֶגְרַע וְלֹא תָחוֹס עֵינִי וְגַם אֲנִי לֹא אֶחְמוֹל` (יחזקאל ה:יא)
`וְשִׁלַּחְתִּי עֲלֵיכֶם רָעָב וְחַיָּה רָעָה וְשִׁכְּלֻךְ וְדֶבֶר וָדָם יַעֲבָר בָּךְ וְחֶרֶב אָבִיא עָלַיִךְ אֲנִי יְהוָה דִּבַּרְתִּי` (יחזקאל ה:יז)
`כִּי הַמּוֹכֵר אֶל הַמִּמְכָּר לֹא יָשׁוּב וְעוֹד בַּחַיִּים חַיָּתָם כִּי חָזוֹן אֶל כָּל הֲמוֹנָהּ לֹא יָשׁוּב וְאִישׁ בַּעֲוֺנוֹ חַיָּתוֹ לֹא יִתְחַזָּקוּ` (יחזקאל ז:יג)
`וַיֵּרֹמּוּ הַכְּרוּבִים הִיא הַחַיָּה אֲשֶׁר רָאִיתִי בִּנְהַר כְּבָר` (יחזקאל י:טו)
`בְּעָמְדָם יַעֲמֹדוּ וּבְרוֹמָם יֵרוֹמּוּ אוֹתָם כִּי רוּחַ הַחַיָּה בָּהֶם` (יחזקאל י:יז)
`הִיא הַחַיָּה אֲשֶׁר רָאִיתִי תַּחַת אֱלֹהֵי יִשְׂרָאֵל בִּנְהַר כְּבָר וָאֵדַע כִּי כְרוּבִים הֵמָּה` (יחזקאל י:כ)
`לוּ חַיָּה רָעָה אַעֲבִיר בָּאָרֶץ וְשִׁכְּלָתָּה וְהָיְתָה שְׁמָמָה מִבְּלִי עוֹבֵר מִפְּנֵי הַחַיָּה` (יחזקאל יד:טו)
`שְׁלֹשֶׁת הָאֲנָשִׁים הָאֵלֶּה בְּתוֹכָהּ חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם בָּנִים וְאִם בָּנוֹת יַצִּילוּ הֵמָּה לְבַדָּם יִנָּצֵלוּ וְהָאָרֶץ תִּהְיֶה שְׁמָמָה` (יחזקאל יד:טז)
`וּשְׁלֹשֶׁת הָאֲנָשִׁים הָאֵלֶּה בְּתוֹכָהּ חַי אָנִי נְאֻם אֲדֹנָי יְהוִה לֹא יַצִּילוּ בָּנִים וּבָנוֹת כִּי הֵם לְבַדָּם יִנָּצֵלוּ` (יחזקאל יד:יח)
`וְנֹחַ דָּנִאֵל וְאִיּוֹב בְּתוֹכָהּ חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם בֵּן אִם בַּת יַצִּילוּ הֵמָּה בְצִדְקָתָם יַצִּילוּ נַפְשָׁם` (יחזקאל יד:כ)
`כִּי כֹה אָמַר אֲדֹנָי יְהֹוִה אַף כִּי אַרְבַּעַת שְׁפָטַי הָרָעִים חֶרֶב וְרָעָב וְחַיָּה רָעָה וָדֶבֶר שִׁלַּחְתִּי אֶל יְרוּשָׁלִָם לְהַכְרִית מִמֶּנָּה אָדָם וּבְהֵמָה` (יחזקאל יד:כא)
`חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם עָשְׂתָה סְדֹם אֲחוֹתֵךְ הִיא וּבְנוֹתֶיהָ כַּאֲשֶׁר עָשִׂית אַתְּ וּבְנוֹתָיִךְ` (יחזקאל טז:מח)
`חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם לֹא בִּמְקוֹם הַמֶּלֶךְ הַמַּמְלִיךְ אֹתוֹ אֲשֶׁר בָּזָה אֶת אָלָתוֹ וַאֲשֶׁר הֵפֵר אֶת בְּרִיתוֹ אִתּוֹ בְתוֹךְ בָּבֶל יָמוּת` (יחזקאל יז:טז)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה חַי אָנִי אִם לֹא אָלָתִי אֲשֶׁר בָּזָה וּבְרִיתִי אֲשֶׁר הֵפִיר וּנְתַתִּיו בְּרֹאשׁוֹ` (יחזקאל יז:יט)
`חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם יִהְיֶה לָכֶם עוֹד מְשֹׁל הַמָּשָׁל הַזֶּה בְּיִשְׂרָאֵל` (יחזקאל יח:ג)
`בֶּן אָדָם דַּבֵּר אֶת זִקְנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם כֹּה אָמַר אֲדֹנָי יְהוִה הֲלִדְרֹשׁ אֹתִי אַתֶּם בָּאִים חַי אָנִי אִם אִדָּרֵשׁ לָכֶם נְאֻם אֲדֹנָי יְהוִה` (יחזקאל כ:ג)
`וּבִשְׂאֵת מַתְּנֹתֵיכֶם בְּהַעֲבִיר בְּנֵיכֶם בָּאֵשׁ אַתֶּם נִטְמְאִים לְכָל גִּלּוּלֵיכֶם עַד הַיּוֹם וַאֲנִי אִדָּרֵשׁ לָכֶם בֵּית יִשְׂרָאֵל חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם אִדָּרֵשׁ לָכֶם` (יחזקאל כ:לא)
`חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם לֹא בְּיָד חֲזָקָה וּבִזְרוֹעַ נְטוּיָה וּבְחֵמָה שְׁפוּכָה אֶמְלוֹךְ עֲלֵיכֶם` (יחזקאל כ:לג)
`וְהוֹרַדְתִּיךְ אֶת יוֹרְדֵי בוֹר אֶל עַם עוֹלָם וְהוֹשַׁבְתִּיךְ בְּאֶרֶץ תַּחְתִּיּוֹת כָּחֳרָבוֹת מֵעוֹלָם אֶת יוֹרְדֵי בוֹר לְמַעַן לֹא תֵשֵׁבִי וְנָתַתִּי צְבִי בְּאֶרֶץ חַיִּים` (יחזקאל כו:כ)
`וּנְטַשְׁתִּיךָ הַמִּדְבָּרָה אוֹתְךָ וְאֵת כָּל דְּגַת יְאֹרֶיךָ עַל פְּנֵי הַשָּׂדֶה תִּפּוֹל לֹא תֵאָסֵף וְלֹא תִקָּבֵץ לְחַיַּת הָאָרֶץ וּלְעוֹף הַשָּׁמַיִם נְתַתִּיךָ לְאָכְלָה` (יחזקאל כט:ה)
`בִּסְעַפֹּתָיו קִנְנוּ כָּל עוֹף הַשָּׁמַיִם וְתַחַת פֹּארֹתָיו יָלְדוּ כֹּל חַיַּת הַשָּׂדֶה וּבְצִלּוֹ יֵשְׁבוּ כֹּל גּוֹיִם רַבִּים` (יחזקאל לא:ו)
`עַל מַפַּלְתּוֹ יִשְׁכְּנוּ כָּל עוֹף הַשָּׁמָיִם וְאֶל פֹּארֹתָיו הָיוּ כֹּל חַיַּת הַשָּׂדֶה` (יחזקאל לא:יג)
`וּנְטַשְׁתִּיךָ בָאָרֶץ עַל פְּנֵי הַשָּׂדֶה אֲטִילֶךָ וְהִשְׁכַּנְתִּי עָלֶיךָ כָּל עוֹף הַשָּׁמַיִם וְהִשְׂבַּעְתִּי מִמְּךָ חַיַּת כָּל הָאָרֶץ` (יחזקאל לב:ד)
`אֲשֶׁר נִתְּנוּ קִבְרֹתֶיהָ בְּיַרְכְּתֵי בוֹר וַיְהִי קְהָלָהּ סְבִיבוֹת קְבֻרָתָהּ כֻּלָּם חֲלָלִים נֹפְלִים בַּחֶרֶב אֲשֶׁר נָתְנוּ חִתִּית בְּאֶרֶץ חַיִּים` (יחזקאל לב:כג)
`שָׁם עֵילָם וְכָל הֲמוֹנָהּ סְבִיבוֹת קְבֻרָתָהּ כֻּלָּם חֲלָלִים הַנֹּפְלִים בַּחֶרֶב אֲ‍שֶׁר יָרְדוּ עֲרֵלִים אֶל אֶרֶץ תַּחְתִּיּוֹת אֲשֶׁר נָתְנוּ חִתִּיתָם בְּאֶרֶץ חַיִּים וַיִּשְׂאוּ כְלִמָּתָם אֶת יוֹרְדֵי בוֹר` (יחזקאל לב:כד)
`בְּתוֹךְ חֲלָלִים נָתְנוּ מִשְׁכָּב לָהּ בְּכָל הֲמוֹנָהּ סְבִיבוֹתָיו קִבְרֹתֶהָ כֻּלָּם עֲרֵלִים חַלְלֵי חֶרֶב כִּי נִתַּן חִתִּיתָם בְּאֶרֶץ חַיִּים וַיִּשְׂאוּ כְלִמָּתָם אֶת יוֹרְדֵי בוֹר בְּתוֹךְ חֲלָלִים נִתָּן` (יחזקאל לב:כה)
`שָׁם מֶשֶׁךְ תֻּבַל וְכָל הֲמוֹנָהּ סְבִיבוֹתָיו קִבְרוֹתֶיהָ כֻּלָּם עֲרֵלִים מְחֻלְלֵי חֶרֶב כִּי נָתְנוּ חִתִּיתָם בְּאֶרֶץ חַיִּים` (יחזקאל לב:כו)
`וְלֹא יִשְׁכְּבוּ אֶת גִּבּוֹרִים נֹפְלִים מֵעֲרֵלִים אֲשֶׁר יָרְדוּ שְׁאוֹל בִּכְלֵי מִלְחַמְתָּם וַיִּתְּנוּ אֶת חַרְבוֹתָם תַּחַת רָאשֵׁיהֶם וַתְּהִי עֲוֺנֹתָם עַל עַצְמוֹתָם כִּי חִתִּית גִּבּוֹרִים בְּאֶרֶץ חַיִּים` (יחזקאל לב:כז)
`כִּי נָתַתִּי אֶת חִתִּיתוֹ בְּאֶרֶץ חַיִּים וְהֻשְׁכַּב בְּתוֹךְ עֲרֵלִים אֶת חַלְלֵי חֶרֶב פַּרְעֹה וְכָל הֲמוֹנֹה נְאֻם אֲדֹנָי יְהוִה` (יחזקאל לב:לב)
`אֱמֹר אֲלֵיהֶם חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם אֶחְפֹּץ בְּמוֹת הָרָשָׁע כִּי אִם בְּשׁוּב רָשָׁע מִדַּרְכּוֹ וְחָיָה שׁוּבוּ שׁוּבוּ מִדַּרְכֵיכֶם הָרָעִים וְלָמָּה תָמוּתוּ בֵּית יִשְׂרָאֵל` (יחזקאל לג:יא)
`חֲבֹל יָשִׁיב רָשָׁע גְּזֵלָה יְשַׁלֵּם בְּחֻקּוֹת הַחַיִּים הָלַךְ לְבִלְתִּי עֲשׂוֹת עָוֶל חָיוֹ יִחְיֶה לֹא יָמוּת` (יחזקאל לג:טו)
`כֹּה תֹאמַר אֲלֵהֶם כֹּה אָמַר אֲדֹנָי יְהוִה חַי אָנִי אִם לֹא אֲשֶׁר בֶּחֳרָבוֹת בַּחֶרֶב יִפֹּלוּ וַאֲשֶׁר עַל פְּנֵי הַשָּׂדֶה לַחַיָּה נְתַתִּיו לְאָכְלוֹ וַאֲשֶׁר בַּמְּצָדוֹת וּבַמְּעָרוֹת בַּדֶּבֶר יָמוּתוּ` (יחזקאל לג:כז)
`וַתְּפוּצֶינָה מִבְּלִי רֹעֶה וַתִּהְיֶינָה לְאָכְלָה לְכָל חַיַּת הַשָּׂדֶה וַתְּפוּצֶינָה` (יחזקאל לד:ה)
`חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם לֹא יַעַן הֱיוֹת צֹאנִי לָבַז וַתִּהְיֶינָה צֹאנִי לְאָכְלָה לְכָל חַיַּת הַשָּׂדֶה מֵאֵין רֹעֶה וְלֹא דָרְשׁוּ רֹעַי אֶת צֹאנִי וַיִּרְעוּ הָרֹעִים אוֹתָם וְאֶת צֹאנִי לֹא רָעוּ` (יחזקאל לד:ח)
`וְכָרַתִּי לָהֶם בְּרִית שָׁלוֹם וְהִשְׁבַּתִּי חַיָּה רָעָה מִן הָאָרֶץ וְיָשְׁבוּ בַמִּדְבָּר לָבֶטַח וְיָשְׁנוּ בַּיְּעָרִים` (יחזקאל לד:כה)
`וְלֹא יִהְיוּ עוֹד בַּז לַגּוֹיִם וְחַיַּת הָאָרֶץ לֹא תֹאכְלֵם וְיָשְׁבוּ לָבֶטַח וְאֵין מַחֲרִיד` (יחזקאל לד:כח)
`לָכֵן חַי אָנִי נְאֻם אֲדֹנָי יְהוִה כִּי לְדָם אֶעֶשְׂךָ וְדָם יִרְדֲּפֶךָ אִם לֹא דָם שָׂנֵאתָ וְדָם יִרְדֲּפֶךָ` (יחזקאל לה:ו)
`לָכֵן חַי אָנִי נְאֻם אֲדֹנָי יְהוִה וְעָשִׂיתִי כְּאַפְּךָ וּכְקִנְאָתְךָ אֲשֶׁר עָשִׂיתָה מִשִּׂנְאָתֶיךָ בָּם וְנוֹדַעְתִּי בָם כַּאֲשֶׁר אֶשְׁפְּטֶךָ` (יחזקאל לה:יא)
`וְרָעֲשׁוּ מִפָּנַי דְּגֵי הַיָּם וְעוֹף הַשָּׁמַיִם וְחַיַּת הַשָּׂדֶה וְכָל הָרֶמֶשׂ הָרֹמֵשׂ עַל הָאֲדָמָה וְכֹל הָאָדָם אֲשֶׁר עַל פְּנֵי הָאֲדָמָה וְנֶהֶרְסוּ הֶהָרִים וְנָפְלוּ הַמַּדְרֵגוֹת וְכָל חוֹמָה לָאָרֶץ תִּפּוֹל` (יחזקאל לח:כ)
`עַל הָרֵי יִשְׂרָאֵל תִּפּוֹל אַתָּה וְכָל אֲגַפֶּיךָ וְעַמִּים אֲשֶׁר אִתָּךְ לְעֵיט צִפּוֹר כָּל כָּנָף וְחַיַּת הַשָּׂדֶה נְתַתִּיךָ לְאָכְלָה` (יחזקאל לט:ד)
`וְאַתָּה בֶן אָדָם כֹּה אָמַר אֲדֹנָי יְהֹוִה אֱמֹר לְצִפּוֹר כָּל כָּנָף וּלְכֹל חַיַּת הַשָּׂדֶה הִקָּבְצוּ וָבֹאוּ הֵאָסְפוּ מִסָּבִיב עַל זִבְחִי אֲשֶׁר אֲנִי זֹבֵחַ לָכֶם זֶבַח גָּדוֹל עַל הָרֵי יִשְׂרָאֵל וַאֲכַלְתֶּם בָּשָׂר וּשְׁתִיתֶם דָּם` (יחזקאל לט:יז)
`וְהָיָה כָל נֶפֶשׁ חַיָּה אֲ‍שֶׁר יִשְׁרֹץ אֶל כָּל אֲשֶׁר יָבוֹא שָׁם נַחֲלַיִם יִחְיֶה וְהָיָה הַדָּגָה רַבָּה מְאֹד כִּי בָאוּ שָׁמָּה הַמַּיִם הָאֵלֶּה וְיֵרָפְאוּ וָחָי כֹּל אֲשֶׁר יָבוֹא שָׁמָּה הַנָּחַל` (יחזקאל מז:ט)
`וְהָיָה מִסְפַּר בְּנֵי יִשְׂרָאֵל כְּחוֹל הַיָּם אֲשֶׁר לֹא יִמַּד וְלֹא יִסָּפֵר וְהָיָה בִּמְקוֹם אֲשֶׁר יֵאָמֵר לָהֶם לֹא עַמִּי אַתֶּם יֵאָמֵר לָהֶם בְּנֵי אֵל חָי` (הושע א:י)
`וַהֲשִׁמֹּתִי גַּפְנָהּ וּתְאֵנָתָהּ אֲשֶׁר אָמְרָה אֶתְנָה הֵמָּה לִי אֲשֶׁר נָתְנוּ לִי מְאַהֲבָי וְשַׂמְתִּים לְיַעַר וַאֲכָלָתַם חַיַּת הַשָּׂדֶה` (הושע ב:יב)
`וְכָרַתִּי לָהֶם בְּרִית בַּיּוֹם הַהוּא עִם חַיַּת הַשָּׂדֶה וְעִם עוֹף הַשָּׁמַיִם וְרֶמֶשׂ הָאֲדָמָה וְקֶשֶׁת וְחֶרֶב וּמִלְחָמָה אֶשְׁבּוֹר מִן הָאָרֶץ וְהִשְׁכַּבְתִּים לָבֶטַח` (הושע ב:יח)
`עַל כֵּן תֶּאֱבַל הָאָרֶץ וְאֻמְלַל כָּל יוֹשֵׁב בָּהּ בְּחַיַּת הַשָּׂדֶה וּבְעוֹף הַשָּׁמָיִם וְגַם דְּגֵי הַיָּם יֵאָסֵפוּ` (הושע ד:ג)
`אִם זֹנֶה אַתָּה יִשְׂרָאֵל אַל יֶאְשַׁם יְהוּדָה וְאַל תָּבֹאוּ הַגִּלְגָּל וְאַל תַּעֲלוּ בֵּית אָוֶן וְאַל תִּשָּׁבְעוּ חַי יְהוָה` (הושע ד:טו)
`אֶפְגְּשֵׁם כְּדֹב שַׁכּוּל וְאֶקְרַע סְגוֹר לִבָּם וְאֹכְלֵם שָׁם כְּלָבִיא חַיַּת הַשָּׂדֶה תְּבַקְּעֵם` (הושע יג:ח)
`הַנִּשְׁבָּעִים בְּאַשְׁמַת שֹׁמְרוֹן וְאָמְרוּ חֵי אֱלֹהֶיךָ דָּן וְחֵי דֶּרֶךְ בְּאֵר שָׁבַע וְנָפְלוּ וְלֹא יָקוּמוּ עוֹד` (עמוס ח:יד)
`לְקִצְבֵי הָרִים יָרַדְתִּי הָאָרֶץ בְּרִחֶיהָ בַעֲדִי לְעוֹלָם וַתַּעַל מִשַּׁחַת חַיַּי יְהוָה אֱלֹהָי` (יונה ב:ו)
`וְעַתָּה יְהוָה קַח נָא אֶת נַפְשִׁי מִמֶּנִּי כִּי טוֹב מוֹתִי מֵחַיָּי` (יונה ד:ג)
`וַיְהִי כִּזְרֹחַ הַשֶּׁמֶשׁ וַיְמַן אֱלֹהִים רוּחַ קָדִים חֲרִישִׁית וַתַּךְ הַשֶּׁמֶשׁ עַל רֹאשׁ יוֹנָה וַיִּתְעַלָּף וַיִּשְׁאַל אֶת נַפְשׁוֹ לָמוּת וַיֹּאמֶר טוֹב מוֹתִי מֵחַיָּי` (יונה ד:ח)
`לָכֵן חַי אָנִי נְאֻם יְהוָה צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל כִּי מוֹאָב כִּסְדֹם תִּהְיֶה וּבְנֵי עַמּוֹן כַּעֲמֹרָה מִמְשַׁק חָרוּל וּמִכְרֵה מֶלַח וּשְׁמָמָה עַד עוֹלָם שְׁאֵרִית עַמִּי יְבָזּוּם וְיֶתֶר גּוֹי יִנְחָלוּם` (צפניה ב:ט)
`וְרָבְצוּ בְתוֹכָהּ עֲדָרִים כָּל חַיְתוֹ גוֹי גַּם קָאַת גַּם קִפֹּד בְּכַפְתֹּרֶיהָ יָלִינוּ קוֹל יְשׁוֹרֵר בַּחַלּוֹן חֹרֶב בַּסַּף כִּי אַרְזָה עֵרָה` (צפניה ב:יד)
`זֹאת הָעִיר הָעַלִּיזָה הַיּוֹשֶׁבֶת לָבֶטַח הָאֹמְרָה בִּלְבָבָהּ אֲנִי וְאַפְסִי עוֹד אֵיךְ הָיְתָה לְשַׁמָּה מַרְבֵּץ לַחַיָּה כֹּל עוֹבֵר עָלֶיהָ יִשְׁרֹק יָנִיעַ יָדוֹ` (צפניה ב:טו)
`וְהָיָה בַּיּוֹם הַהוּא יֵצְאוּ מַיִם חַיִּים מִירוּשָׁלִַם חֶצְיָם אֶל הַיָּם הַקַּדְמוֹנִי וְחֶצְיָם אֶל הַיָּם הָאַחֲרוֹן בַּקַּיִץ וּבָחֹרֶף יִהְיֶה` (זכריה יד:ח)
`בְּרִיתִי הָיְתָה אִתּוֹ הַחַיִּים וְהַשָּׁלוֹם וָאֶתְּנֵם לוֹ מוֹרָא וַיִּירָאֵנִי וּמִפְּנֵי שְׁמִי נִחַת הוּא` (מלאכי ב:ה)
`לְמַעַן יִירָאוּךָ לָלֶכֶת בִּדְרָכֶיךָ כָּל הַיָּמִים אֲשֶׁר הֵם חַיִּים עַל פְּנֵי הָאֲדָמָה אֲשֶׁר נָתַתָּה לַאֲבֹתֵינוּ` (דברי-הימים-ב ו:לא)
`וַיִּוָּעַץ הַמֶּלֶךְ רְחַבְעָם אֶת הַזְּקֵנִים אֲשֶׁר הָיוּ עֹמְדִים לִפְנֵי שְׁלֹמֹה אָבִיו בִּהְיֹתוֹ חַי לֵאמֹר אֵיךְ אַתֶּם נוֹעָצִים לְהָשִׁיב לָעָם הַזֶּה דָּבָר` (דברי-הימים-ב י:ו)
`וַיֹּאמֶר מִיכָיְהוּ חַי יְהוָה כִּי אֶת אֲשֶׁר יֹאמַר אֱלֹהַי אֹתוֹ אֲדַבֵּר` (דברי-הימים-ב יח:יג)
`וַעֲשֶׂרֶת אֲלָפִים חַיִּים שָׁבוּ בְּנֵי יְהוּדָה וַיְבִיאוּם לְרֹאשׁ הַסָּלַע וַיַּשְׁלִיכוּם מֵרֹאשׁ הַסֶּלַע וְכֻלָּם נִבְקָעוּ` (דברי-הימים-ב כה:יב)
`וַיִּשְׁלַח יוֹאָשׁ מֶלֶךְ יִשְׂרָאֵל אֶל אֲמַצְיָהוּ מֶלֶךְ יְהוּדָה לֵאמֹר הַחוֹחַ אֲשֶׁר בַּלְּבָנוֹן שָׁלַח אֶל הָאֶרֶז אֲשֶׁר בַּלְּבָנוֹן לֵאמֹר תְּנָה אֶת בִּתְּךָ לִבְנִי לְאִשָּׁה וַתַּעֲבֹר חַיַּת הַשָּׂדֶה אֲשֶׁר בַּלְּבָנוֹן וַתִּרְמֹס אֶת הַחוֹחַ` (דברי-הימים-ב כה:יח)
`יִרַדֹּף אוֹיֵב נַפְשִׁי וְיַשֵּׂג וְיִרְמֹס לָאָרֶץ חַיָּי וּכְבוֹדִי לֶעָפָר יַשְׁכֵּן סֶלָה` (תהילים ז:ה)
`תּוֹדִיעֵנִי אֹרַח חַיִּים שֹׂבַע שְׂמָחוֹת אֶת פָּנֶיךָ נְעִמוֹת בִּימִינְךָ נֶצַח` (תהילים טז:יא)
`מִמְתִים יָדְךָ יְהוָה מִמְתִים מֵחֶלֶד חֶלְקָם בַּחַיִּים וּצְפִינְךָ תְּמַלֵּא בִטְנָם יִשְׂבְּעוּ בָנִים וְהִנִּיחוּ יִתְרָם לְעוֹלְלֵיהֶם` (תהילים יז:יד)
`חַי יְהוָה וּבָרוּךְ צוּרִי וְיָרוּם אֱלוֹהֵי יִשְׁעִי` (תהילים יח:מו)
`חַיִּים שָׁאַל מִמְּךָ נָתַתָּה לּוֹ אֹרֶךְ יָמִים עוֹלָם וָעֶד` (תהילים כא:ד)
`אַךְ טוֹב וָחֶסֶד יִרְדְּפוּנִי כָּל יְמֵי חַיָּי וְשַׁבְתִּי בְּבֵית יְהוָה לְאֹרֶךְ יָמִים` (תהילים כג:ו)
`אַל תֶּאֱסֹף עִם חַטָּאִים נַפְשִׁי וְעִם אַנְשֵׁי דָמִים חַיָּי` (תהילים כו:ט)
`לְדָוִד יְהוָה אוֹרִי וְיִשְׁעִי מִמִּי אִירָא יְהוָה מָעוֹז חַיַּי מִמִּי אֶפְחָד` (תהילים כז:א)
`אַחַת שָׁאַלְתִּי מֵאֵת יְהוָה אוֹתָהּ אֲבַקֵּשׁ שִׁבְתִּי בְּבֵית יְהוָה כָּל יְמֵי חַיַּי לַחֲזוֹת בְּנֹעַם יְהוָה וּלְבַקֵּר בְּהֵיכָלוֹ` (תהילים כז:ד)
`לׅׄוּלֵׅׄאׅׄ הֶאֱמַנְתִּי לִרְאוֹת בְּטוּב יְהוָה בְּאֶרֶץ חַיִּים` (תהילים כז:יג)
`כִּי רֶגַע בְּאַפּוֹ חַיִּים בִּרְצוֹנוֹ בָּעֶרֶב יָלִין בֶּכִי וְלַבֹּקֶר רִנָּה` (תהילים ל:ה)
`כִּי כָלוּ בְיָגוֹן חַיַּי וּשְׁנוֹתַי בַּאֲנָחָה כָּשַׁל בַּעֲוֺנִי כֹחִי וַעֲצָמַי עָשֵׁשׁוּ` (תהילים לא:י)
`מִי הָאִישׁ הֶחָפֵץ חַיִּים אֹהֵב יָמִים לִרְאוֹת טוֹב` (תהילים לד:יב)
`כִּי עִמְּךָ מְקוֹר חַיִּים בְּאוֹרְךָ נִרְאֶה אוֹר` (תהילים לו:ט)
`וְאֹיְבַי חַיִּים עָצֵמוּ וְרַבּוּ שֹׂנְאַי שָׁקֶר` (תהילים לח:יט)
`צָמְאָה נַפְשִׁי לֵאלֹהִים לְאֵל חָי מָתַי אָבוֹא וְאֵרָאֶה פְּנֵי אֱלֹהִים` (תהילים מב:ב)
`יוֹמָם יְצַוֶּה יְהוָה חַסְדּוֹ וּבַלַּיְלָה שִׁירָהּ עִמִּי תְּפִלָּה לְאֵל חַיָּי` (תהילים מב:ח)
`כִּי נַפְשׁוֹ בְּחַיָּיו יְבָרֵךְ וְיוֹדֻךָ כִּי תֵיטִיב לָךְ` (תהילים מט:יח)
`כִּי לִי כָל חַיְתוֹ יָעַר בְּהֵמוֹת בְּהַרְרֵי אָלֶף` (תהילים נ:י)
`גַּם אֵל יִתָּצְךָ לָנֶצַח יַחְתְּךָ וְיִסָּחֲךָ מֵאֹהֶל וְשֵׁרֶשְׁךָ מֵאֶרֶץ חַיִּים סֶלָה` (תהילים נב:ה)
`יַשִּׁימָוֶת עָלֵימוֹ יֵרְדוּ שְׁאוֹל חַיִּים כִּי רָעוֹת בִּמְגוּרָם בְּקִרְבָּם` (תהילים נה:טו)
`כִּי הִצַּלְתָּ נַפְשִׁי מִמָּוֶת הֲלֹא רַגְלַי מִדֶּחִי לְהִתְהַלֵּךְ לִפְנֵי אֱלֹהִים בְּאוֹר הַחַיִּים` (תהילים נו:יג)
`בְּטֶרֶם יָבִינוּ סִּירֹתֵיכֶם אָטָד כְּמוֹ חַי כְּמוֹ חָרוֹן יִשְׂעָרֶנּוּ` (תהילים נח:ט)
`כִּי טוֹב חַסְדְּךָ מֵחַיִּים שְׂפָתַי יְשַׁבְּחוּנְךָ` (תהילים סג:ג)
`כֵּן אֲבָרֶכְךָ בְחַיָּי בְּשִׁמְךָ אֶשָּׂא כַפָּי` (תהילים סג:ד)
`לַמְנַצֵּחַ מִזְמוֹר לְדָוִד שְׁמַע אֱלֹהִים קוֹלִי בְשִׂיחִי מִפַּחַד אוֹיֵב תִּצֹּר חַיָּי` (תהילים סד:א)
`הַשָּׂם נַפְשֵׁנוּ בַּחַיִּים וְלֹא נָתַן לַמּוֹט רַגְלֵנוּ` (תהילים סו:ט)
`חַיָּתְךָ יָשְׁבוּ בָהּ תָּכִין בְּטוֹבָתְךָ לֶעָנִי אֱלֹהִים` (תהילים סח:י)
`גְּעַר חַיַּת קָנֶה עֲדַת אַבִּירִים בְּעֶגְלֵי עַמִּים מִתְרַפֵּס בְּרַצֵּי כָסֶף בִּזַּר עַמִּים קְרָבוֹת יֶחְפָּצוּ` (תהילים סח:ל)
`יִמָּחוּ מִסֵּפֶר חַיִּים וְעִם צַדִּיקִים אַל יִכָּתֵבוּ` (תהילים סט:כח)
`אַל תִּתֵּן לְחַיַּת נֶפֶשׁ תּוֹרֶךָ חַיַּת עֲנִיֶּיךָ אַל תִּשְׁכַּח לָנֶצַח` (תהילים עד:יט)
`יְפַלֵּס נָתִיב לְאַפּוֹ לֹא חָשַׂךְ מִמָּוֶת נַפְשָׁם וְחַיָּתָם לַדֶּבֶר הִסְגִּיר` (תהילים עח:נ)
`נָתְנוּ אֶת נִבְלַת עֲבָדֶיךָ מַאֲכָל לְעוֹף הַשָּׁמָיִם בְּשַׂר חֲסִידֶיךָ לְחַיְתוֹ אָרֶץ` (תהילים עט:ב)
`נִכְסְפָה וְגַם כָּלְתָה נַפְשִׁי לְחַצְרוֹת יְהוָה לִבִּי וּבְשָׂרִי יְרַנְּנוּ אֶל אֵל חָי` (תהילים פד:ב)
`כִּי שָׂבְעָה בְרָעוֹת נַפְשִׁי וְחַיַּי לִשְׁאוֹל הִגִּיעוּ` (תהילים פח:ג)
`הַגּוֹאֵל מִשַּׁחַת חַיָּיְכִי הַמְעַטְּרֵכִי חֶסֶד וְרַחֲמִים` (תהילים קג:ד)
`יַשְׁקוּ כָּל חַיְתוֹ שָׂדָי יִשְׁבְּרוּ פְרָאִים צְמָאָם` (תהילים קד:יא)
`תָּשֶׁת חֹשֶׁךְ וִיהִי לָיְלָה בּוֹ תִרְמֹשׂ כָּל חַיְתוֹ יָעַר` (תהילים קד:כ)
`זֶה הַיָּם גָּדוֹל וּרְחַב יָדָיִם שָׁם רֶמֶשׂ וְאֵין מִסְפָּר חַיּוֹת קְטַנּוֹת עִם גְּדֹלוֹת` (תהילים קד:כה)
`אָשִׁירָה לַיהוָה בְּחַיָּי אֲזַמְּרָה לֵאלֹהַי בְּעוֹדִי` (תהילים קד:לג)
`אֶתְהַלֵּךְ לִפְנֵי יְהוָה בְּאַרְצוֹת הַחַיִּים` (תהילים קיו:ט)
`אֲזַי חַיִּים בְּלָעוּנוּ בַּחֲרוֹת אַפָּם בָּנוּ` (תהילים קכד:ג)
`יְבָרֶכְךָ יְהוָה מִצִּיּוֹן וּרְאֵה בְּטוּב יְרוּשָׁלִָם כֹּל יְמֵי חַיֶּיךָ` (תהילים קכח:ה)
`כְּטַל חֶרְמוֹן שֶׁיֹּרֵד עַל הַרְרֵי צִיּוֹן כִּי שָׁם צִוָּה יְהוָה אֶת הַבְּרָכָה חַיִּים עַד הָעוֹלָם` (תהילים קלג:ג)
`זָעַקְתִּי אֵלֶיךָ יְהוָה אָמַרְתִּי אַתָּה מַחְסִי חֶלְקִי בְּאֶרֶץ הַחַיִּים` (תהילים קמב:ה)
`וְאַל תָּבוֹא בְמִשְׁפָּט אֶת עַבְדֶּךָ כִּי לֹא יִצְדַּק לְפָנֶיךָ כָל חָי` (תהילים קמג:ב)
`כִּי רָדַף אוֹיֵב נַפְשִׁי דִּכָּא לָאָרֶץ חַיָּתִי הוֹשִׁיבַנִי בְמַחֲשַׁכִּים כְּמֵתֵי עוֹלָם` (תהילים קמג:ג)
`פּוֹתֵחַ אֶת יָדֶךָ וּמַשְׂבִּיעַ לְכָל חַי רָצוֹן` (תהילים קמה:טז)
`אֲהַלְלָה יְהוָה בְּחַיָּי אֲזַמְּרָה לֵאלֹהַי בְּעוֹדִי` (תהילים קמו:ב)
`הַחַיָּה וְכָל בְּהֵמָה רֶמֶשׂ וְצִפּוֹר כָּנָף` (תהילים קמח:י)
`לָמָּה יִתֵּן לְעָמֵל אוֹר וְחַיִּים לְמָרֵי נָפֶשׁ` (איוב ג:כ)
`לְשֹׁד וּלְכָפָן תִּשְׂחָק וּמֵחַיַּת הָאָרֶץ אַל תִּירָא` (איוב ה:כב)
`כִּי עִם אַבְנֵי הַשָּׂדֶה בְרִיתֶךָ וְחַיַּת הַשָּׂדֶה הָשְׁלְמָה לָךְ` (איוב ה:כג)
`זְכֹר כִּי רוּחַ חַיָּי לֹא תָשׁוּב עֵינִי לִרְאוֹת טוֹב` (איוב ז:ז)
`תָּם אָנִי לֹא אֵדַע נַפְשִׁי אֶמְאַס חַיָּי` (איוב ט:כא)
`נָקְטָה נַפְשִׁי בְּחַיָּי אֶעֶזְבָה עָלַי שִׂיחִי אֲדַבְּרָה בְּמַר נַפְשִׁי` (איוב י:א)
`חַיִּים וָחֶסֶד עָשִׂיתָ עִמָּדִי וּפְקֻדָּתְךָ שָׁמְרָה רוּחִי` (איוב י:יב)
`אֲשֶׁר בְּיָדוֹ נֶפֶשׁ כָּל חָי וְרוּחַ כָּל בְּשַׂר אִישׁ` (איוב יב:י)
`וַאֲנִי יָדַעְתִּי גֹּאֲלִי חָי וְאַחֲרוֹן עַל עָפָר יָקוּם` (איוב יט:כה)
`וּמָשַׁךְ אַבִּירִים בְּכֹחוֹ יָקוּם וְלֹא יַאֲמִין בַּחַיִּין` (איוב כד:כב)
`חַי אֵל הֵסִיר מִשְׁפָּטִי וְשַׁדַּי הֵמַר נַפְשִׁי` (איוב כז:ב)
`לֹא יָדַע אֱנוֹשׁ עֶרְכָּהּ וְלֹא תִמָּצֵא בְּאֶרֶץ הַחַיִּים` (איוב כח:יג)
`וְנֶעֶלְמָה מֵעֵינֵי כָל חָי וּמֵעוֹף הַשָּׁמַיִם נִסְתָּרָה` (איוב כח:כא)
`כִּי יָדַעְתִּי מָוֶת תְּשִׁיבֵנִי וּבֵית מוֹעֵד לְכָל חָי` (איוב ל:כג)
`יַחְשֹׂךְ נַפְשׁוֹ מִנִּי שָׁחַת וְחַיָּתוֹ מֵעֲבֹר בַּשָּׁלַח` (איוב לג:יח)
`וְזִהֲמַתּוּ חַיָּתוֹ לָחֶם וְנַפְשׁוֹ מַאֲכַל תַּאֲוָה` (איוב לג:כ)
`וַתִּקְרַב לַשַּׁחַת נַפְשׁוֹ וְחַיָּתוֹ לַמְמִתִים` (איוב לג:כב)
`פָּדָה נַפְשִׁי מֵעֲבֹר בַּשָּׁחַת וְחַיָּתִי בָּאוֹר תִּרְאֶה` (איוב לג:כח)
`לְהָשִׁיב נַפְשׁוֹ מִנִּי שָׁחַת לֵאוֹר בְּאוֹר הַחַיִּים` (איוב לג:ל)
`תָּמֹת בַּנֹּעַר נַפְשָׁם וְחַיָּתָם בַּקְּדֵשִׁים` (איוב לו:יד)
`וַתָּבֹא חַיָּה בְמוֹ אָרֶב וּבִמְעוֹנֹתֶיהָ תִשְׁכֹּן` (איוב לז:ח)
`הֲתָצוּד לְלָבִיא טָרֶף וְחַיַּת כְּפִירִים תְּמַלֵּא` (איוב לח:לט)
`וַתִּשְׁכַּח כִּי רֶגֶל תְּזוּרֶהָ וְחַיַּת הַשָּׂדֶה תְּדוּשֶׁהָ` (איוב לט:טו)
`כִּי בוּל הָרִים יִשְׂאוּ לוֹ וְכָל חַיַּת הַשָּׂדֶה יְשַׂחֲקוּ שָׁם` (איוב מ:כ)
`נִבְלָעֵם כִּשְׁאוֹל חַיִּים וּתְמִימִים כְּיוֹרְדֵי בוֹר` (משלי א:יב)
`כָּל בָּאֶיהָ לֹא יְשׁוּבוּן וְלֹא יַשִּׂיגוּ אָרְחוֹת חַיִּים` (משלי ב:יט)
`כִּי אֹרֶךְ יָמִים וּשְׁנוֹת חַיִּים וְשָׁלוֹם יוֹסִיפוּ לָךְ` (משלי ג:ב)
`עֵץ חַיִּים הִיא לַמַּחֲזִיקִים בָּהּ וְתֹמְכֶיהָ מְאֻשָּׁר` (משלי ג:יח)
`וְיִהְיוּ חַיִּים לְנַפְשֶׁךָ וְחֵן לְגַרְגְּרֹתֶיךָ` (משלי ג:כב)
`שְׁמַע בְּנִי וְקַח אֲמָרָי וְיִרְבּוּ לְךָ שְׁנוֹת חַיִּים` (משלי ד:י)
`הַחֲזֵק בַּמּוּסָר אַל תֶּרֶף נִצְּרֶהָ כִּי הִיא חַיֶּיךָ` (משלי ד:יג)
`כִּי חַיִּים הֵם לְמֹצְאֵיהֶם וּלְכָל בְּשָׂרוֹ מַרְפֵּא` (משלי ד:כב)
`מִכָּל מִשְׁמָר נְצֹר לִבֶּךָ כִּי מִמֶּנּוּ תּוֹצְאוֹת חַיִּים` (משלי ד:כג)
`אֹרַח חַיִּים פֶּן תְּפַלֵּס נָעוּ מַעְגְּלֹתֶיהָ לֹא תֵדָע` (משלי ה:ו)
`כִּי נֵר מִצְוָה וְתוֹרָה אוֹר וְדֶרֶךְ חַיִּים תּוֹכְחוֹת מוּסָר` (משלי ו:כג)
`כִּי מֹצְאִי מֹצְאֵי חַיִּים וַיָּפֶק רָצוֹן מֵיְהוָה` (משלי ח:לה)
`כִּי בִי יִרְבּוּ יָמֶיךָ וְיוֹסִיפוּ לְּךָ שְׁנוֹת חַיִּים` (משלי ט:יא)
`מְקוֹר חַיִּים פִּי צַדִּיק וּפִי רְשָׁעִים יְכַסֶּה חָמָס` (משלי י:יא)
`פְּעֻלַּת צַדִּיק לְחַיִּים תְּבוּאַת רָשָׁע לְחַטָּאת` (משלי י:טז)
`אֹרַח לְחַיִּים שׁוֹמֵר מוּסָר וְעוֹזֵב תּוֹכַחַת מַתְעֶה` (משלי י:יז)
`כֵּן צְדָקָה לְחַיִּים וּמְרַדֵּף רָעָה לְמוֹתוֹ` (משלי יא:יט)
`פְּרִי צַדִּיק עֵץ חַיִּים וְלֹקֵחַ נְפָשׂוֹת חָכָם` (משלי יא:ל)
`בְּאֹרַח צְדָקָה חַיִּים וְדֶרֶךְ נְתִיבָה אַל מָוֶת` (משלי יב:כח)
`תּוֹחֶלֶת מְמֻשָּׁכָה מַחֲלָה לֵב וְעֵץ חַיִּים תַּאֲוָה בָאָה` (משלי יג:יב)
`תּוֹרַת חָכָם מְקוֹר חַיִּים לָסוּר מִמֹּקְשֵׁי מָוֶת` (משלי יג:יד)
`יִרְאַת יְהוָה מְקוֹר חַיִּים לָסוּר מִמֹּקְשֵׁי מָוֶת` (משלי יד:כז)
`חַיֵּי בְשָׂרִים לֵב מַרְפֵּא וּרְקַב עֲצָמוֹת קִנְאָה` (משלי יד:ל)
`מַרְפֵּא לָשׁוֹן עֵץ חַיִּים וְסֶלֶף בָּהּ שֶׁבֶר בְּרוּחַ` (משלי טו:ד)
`אֹרַח חַיִּים לְמַעְלָה לְמַשְׂכִּיל לְמַעַן סוּר מִשְּׁאוֹל מָטָּה` (משלי טו:כד)
`אֹזֶן שֹׁמַעַת תּוֹכַחַת חַיִּים בְּקֶרֶב חֲכָמִים תָּלִין` (משלי טו:לא)
`בְּאוֹר פְּנֵי מֶלֶךְ חַיִּים וּרְצוֹנוֹ כְּעָב מַלְקוֹשׁ` (משלי טז:טו)
`מְקוֹר חַיִּים שֵׂכֶל בְּעָלָיו וּמוּסַר אֱוִלִים אִוֶּלֶת` (משלי טז:כב)
`מָוֶת וְחַיִּים בְּיַד לָשׁוֹן וְאֹהֲבֶיהָ יֹאכַל פִּרְיָהּ` (משלי יח:כא)
`יִרְאַת יְהוָה לְחַיִּים וְשָׂבֵעַ יָלִין בַּל יִפָּקֶד רָע` (משלי יט:כג)
`רֹדֵף צְדָקָה וָחָסֶד יִמְצָא חַיִּים צְדָקָה וְכָבוֹד` (משלי כא:כא)
`עֵקֶב עֲנָוָה יִרְאַת יְהוָה עֹשֶׁר וְכָבוֹד וְחַיִּים` (משלי כב:ד)
`וְדֵי חֲלֵב עִזִּים לְלַחְמְךָ לְלֶחֶם בֵּיתֶךָ וְחַיִּים לְנַעֲרוֹתֶיךָ` (משלי כז:כז)
`גְּמָלַתְהוּ טוֹב וְלֹא רָע כֹּל יְמֵי חַיֶּיה` (משלי לא:יב)
`וַתֹּאמֶר נָעֳמִי לְכַלָּתָהּ בָּרוּךְ הוּא לַיהוָה אֲשֶׁר לֹא עָזַב חַסְדּוֹ אֶת הַחַיִּים וְאֶת הַמֵּתִים וַתֹּאמֶר לָהּ נָעֳמִי קָרוֹב לָנוּ הָאִישׁ מִגֹּאֲלֵנוּ הוּא` (רות ב:כ)
`לִינִי הַלַּיְלָה וְהָיָה בַבֹּקֶר אִם יִגְאָלֵךְ טוֹב יִגְאָל וְאִם לֹא יַחְפֹּץ לְגָאֳלֵךְ וּגְאַלְתִּיךְ אָנֹכִי חַי יְהוָה שִׁכְבִי עַד הַבֹּקֶר` (רות ג:יג)
`מַעְיַן גַּנִּים בְּאֵר מַיִם חַיִּים וְנֹזְלִים מִן לְבָנוֹן` (שיר-השירים ד:טו)
`תַּרְתִּי בְלִבִּי לִמְשׁוֹךְ בַּיַּיִן אֶת בְּשָׂרִי וְלִבִּי נֹהֵג בַּחָכְמָה וְלֶאֱחֹז בְּסִכְלוּת עַד אֲשֶׁר אֶרְאֶה אֵי זֶה טוֹב לִבְנֵי הָאָדָם אֲשֶׁר יַעֲשׂוּ תַּחַת הַשָּׁמַיִם מִסְפַּר יְמֵי חַיֵּיהֶם` (קהלת ב:ג)
`וְשָׂנֵאתִי אֶת הַחַיִּים כִּי רַע עָלַי הַמַּעֲשֶׂה שֶׁנַּעֲשָׂה תַּחַת הַשָּׁמֶשׁ כִּי הַכֹּל הֶבֶל וּרְעוּת רוּחַ` (קהלת ב:יז)
`יָדַעְתִּי כִּי אֵין טוֹב בָּם כִּי אִם לִשְׂמוֹחַ וְלַעֲשׂוֹת טוֹב בְּחַיָּיו` (קהלת ג:יב)
`וְשַׁבֵּחַ אֲנִי אֶת הַמֵּתִים שֶׁכְּבָר מֵתוּ מִן הַחַיִּים אֲשֶׁר הֵמָּה חַיִּים עֲדֶנָה` (קהלת ד:ב)
`רָאִיתִי אֶת כָּל הַחַיִּים הַמְהַלְּכִים תַּחַת הַשָּׁמֶשׁ עִם הַיֶּלֶד הַשֵּׁנִי אֲשֶׁר יַעֲמֹד תַּחְתָּיו` (קהלת ד:טו)
`הִנֵּה אֲשֶׁר רָאִיתִי אָנִי טוֹב אֲשֶׁר יָפֶה לֶאֶכוֹל וְלִשְׁתּוֹת וְלִרְאוֹת טוֹבָה בְּכָל עֲמָלוֹ שֶׁיַּעֲמֹל תַּחַת הַשֶּׁמֶשׁ מִסְפַּר יְמֵי חַיָּו אֲשֶׁר נָתַן לוֹ הָאֱלֹהִים כִּי הוּא חֶלְקוֹ` (קהלת ה:יח)
`כִּי לֹא הַרְבֵּה יִזְכֹּר אֶת יְמֵי חַיָּיו כִּי הָאֱלֹהִים מַעֲנֶה בְּשִׂמְחַת לִבּוֹ` (קהלת ה:כ)
`כִּי מַה יּוֹתֵר לֶחָכָם מִן הַכְּסִיל מַה לֶּעָנִי יוֹדֵעַ לַהֲלֹךְ נֶגֶד הַחַיִּים` (קהלת ו:ח)
`כִּי מִי יוֹדֵעַ מַה טּוֹב לָאָדָם בַּחַיִּים מִסְפַּר יְמֵי חַיֵּי הֶבְלוֹ וְיַעֲשֵׂם כַּצֵּל אֲשֶׁר מִי יַגִּיד לָאָדָם מַה יִּהְיֶה אַחֲרָיו תַּחַת הַשָּׁמֶשׁ` (קהלת ו:יב)
`טוֹב לָלֶכֶת אֶל בֵּית אֵבֶל מִלֶּכֶת אֶל בֵּית מִשְׁתֶּה בַּאֲשֶׁר הוּא סוֹף כָּל הָאָדָם וְהַחַי יִתֵּן אֶל לִבּוֹ` (קהלת ז:ב)
`וְשִׁבַּחְתִּי אֲנִי אֶת הַשִּׂמְחָה אֲשֶׁר אֵין טוֹב לָאָדָם תַּחַת הַשֶּׁמֶשׁ כִּי אִם לֶאֱכוֹל וְלִשְׁתּוֹת וְלִשְׂמוֹחַ וְהוּא יִלְוֶנּוּ בַעֲמָלוֹ יְמֵי חַיָּיו אֲשֶׁר נָתַן לוֹ הָאֱלֹהִים תַּחַת הַשָּׁמֶשׁ` (קהלת ח:טו)
`זֶה רָע בְּכֹל אֲשֶׁר נַעֲשָׂה תַּחַת הַשֶּׁמֶשׁ כִּי מִקְרֶה אֶחָד לַכֹּל וְגַם לֵב בְּנֵי הָאָדָם מָלֵא רָע וְהוֹלֵלוֹת בִּלְבָבָם בְּחַיֵּיהֶם וְאַחֲרָיו אֶל הַמֵּתִים` (קהלת ט:ג)
`כִּי מִי אֲשֶׁר יִבָּחֵר אֶל כָּל הַחַיִּים יֵשׁ בִּטָּחוֹן כִּי לְכֶלֶב חַי הוּא טוֹב מִן הָאַרְיֵה הַמֵּת` (קהלת ט:ד)
`כִּי הַחַיִּים יוֹדְעִים שֶׁיָּמֻתוּ וְהַמֵּתִים אֵינָם יוֹדְעִים מְאוּמָה וְאֵין עוֹד לָהֶם שָׂכָר כִּי נִשְׁכַּח זִכְרָם` (קהלת ט:ה)
`רְאֵה חַיִּים עִם אִשָּׁה אֲשֶׁר אָהַבְתָּ כָּל יְמֵי חַיֵּי הֶבְלֶךָ אֲשֶׁר נָתַן לְךָ תַּחַת הַשֶּׁמֶשׁ כֹּל יְמֵי הֶבְלֶךָ כִּי הוּא חֶלְקְךָ בַּחַיִּים וּבַעֲמָלְךָ אֲשֶׁר אַתָּה עָמֵל תַּחַת הַשָּׁמֶשׁ` (קהלת ט:ט)
`לִשְׂחוֹק עֹשִׂים לֶחֶם וְיַיִן יְשַׂמַּח חַיִּים וְהַכֶּסֶף יַעֲנֶה אֶת הַכֹּל` (קהלת י:יט)
`מַה יִּתְאוֹנֵן אָדָם חָי גֶּבֶר עַל חֶטְאוֹ` (איכה ג:לט)
`צָמְתוּ בַבּוֹר חַיָּי וַיַּדּוּ אֶבֶן בִּי` (איכה ג:נג)
`רַבְתָּ אֲדֹנָי רִיבֵי נַפְשִׁי גָּאַלְתָּ חַיָּי` (איכה ג:נח)
`רָאִיתִי אֶת הָאַיִל מְנַגֵּחַ יָמָּה וְצָפוֹנָה וָנֶגְבָּה וְכָל חַיּוֹת לֹא יַעַמְדוּ לְפָנָיו וְאֵין מַצִּיל מִיָּדוֹ וְעָשָׂה כִרְצֹנוֹ וְהִגְדִּיל` (דניאל ח:ד)
`וְרַבִּים מִיְּשֵׁנֵי אַדְמַת עָפָר יָקִיצוּ אֵלֶּה לְחַיֵּי עוֹלָם וְאֵלֶּה לַחֲרָפוֹת לְדִרְאוֹן עוֹלָם` (דניאל יב:ב)
`וָאֶשְׁמַע אֶת הָאִישׁ לְבוּשׁ הַבַּדִּים אֲשֶׁר מִמַּעַל לְמֵימֵי הַיְאֹר וַיָּרֶם יְמִינוֹ וּשְׂמֹאלוֹ אֶל הַשָּׁמַיִם וַיִּשָּׁבַע בְּחֵי הָעוֹלָם כִּי לְמוֹעֵד מוֹעֲדִים וָחֵצִי וּכְכַלּוֹת נַפֵּץ יַד עַם קֹדֶשׁ תִּכְלֶינָה כָל אֵלֶּה` (דניאל יב:ז)
