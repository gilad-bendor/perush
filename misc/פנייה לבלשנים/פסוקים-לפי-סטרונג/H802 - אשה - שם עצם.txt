`וַיִּבֶן יְהוָה אֱלֹהִים אֶת הַצֵּלָע אֲשֶׁר לָקַח מִן הָאָדָם לְאִשָּׁה וַיְבִאֶהָ אֶל הָאָדָם` (בראשית ב:כב)
`וַיֹּאמֶר הָאָדָם זֹאת הַפַּעַם עֶצֶם מֵעֲצָמַי וּבָשָׂר מִבְּשָׂרִי לְזֹאת יִקָּרֵא אִשָּׁה כִּי מֵאִישׁ לֻקֳחָה זֹּאת` (בראשית ב:כג)
`עַל כֵּן יַעֲזָב אִישׁ אֶת אָבִיו וְאֶת אִמּוֹ וְדָבַק בְּאִשְׁתּוֹ וְהָיוּ לְבָשָׂר אֶחָד` (בראשית ב:כד)
`וַיִּהְיוּ שְׁנֵיהֶם עֲרוּמִּים הָאָדָם וְאִשְׁתּוֹ וְלֹא יִתְבֹּשָׁשׁוּ` (בראשית ב:כה)
`וְהַנָּחָשׁ הָיָה עָרוּם מִכֹּל חַיַּת הַשָּׂדֶה אֲשֶׁר עָשָׂה יְהוָה אֱלֹהִים וַיֹּאמֶר אֶל הָאִשָּׁה אַף כִּי אָמַר אֱלֹהִים לֹא תֹאכְלוּ מִכֹּל עֵץ הַגָּן` (בראשית ג:א)
`וַתֹּאמֶר הָאִשָּׁה אֶל הַנָּחָשׁ מִפְּרִי עֵץ הַגָּן נֹאכֵל` (בראשית ג:ב)
`וַיֹּאמֶר הַנָּחָשׁ אֶל הָאִשָּׁה לֹא מוֹת תְּמֻתוּן` (בראשית ג:ד)
`וַתֵּרֶא הָאִשָּׁה כִּי טוֹב הָעֵץ לְמַאֲכָל וְכִי תַאֲוָה הוּא לָעֵינַיִם וְנֶחְמָד הָעֵץ לְהַשְׂכִּיל וַתִּקַּח מִפִּרְיוֹ וַתֹּאכַל וַתִּתֵּן גַּם לְאִישָׁהּ עִמָּהּ וַיֹּאכַל` (בראשית ג:ו)
`וַיִּשְׁמְעוּ אֶת קוֹל יְהוָה אֱלֹהִים מִתְהַלֵּךְ בַּגָּן לְרוּחַ הַיּוֹם וַיִּתְחַבֵּא הָאָדָם וְאִשְׁתּוֹ מִפְּנֵי יְהוָה אֱלֹהִים בְּתוֹךְ עֵץ הַגָּן` (בראשית ג:ח)
`וַיֹּאמֶר הָאָדָם הָאִשָּׁה אֲשֶׁר נָתַתָּה עִמָּדִי הִוא נָתְנָה לִּי מִן הָעֵץ וָאֹכֵל` (בראשית ג:יב)
`וַיֹּאמֶר יְהוָה אֱלֹהִים לָאִשָּׁה מַה זֹּאת עָשִׂית וַתֹּאמֶר הָאִשָּׁה הַנָּחָשׁ הִשִּׁיאַנִי וָאֹכֵל` (בראשית ג:יג)
`וְאֵיבָה אָשִׁית בֵּינְךָ וּבֵין הָאִשָּׁה וּבֵין זַרְעֲךָ וּבֵין זַרְעָהּ הוּא יְשׁוּפְךָ רֹאשׁ וְאַתָּה תְּשׁוּפֶנּוּ עָקֵב` (בראשית ג:טו)
`אֶל הָאִשָּׁה אָמַר הַרְבָּה אַרְבֶּה עִצְּבוֹנֵךְ וְהֵרֹנֵךְ בְּעֶצֶב תֵּלְדִי בָנִים וְאֶל אִישֵׁךְ תְּשׁוּקָתֵךְ וְהוּא יִמְשָׁל בָּךְ` (בראשית ג:טז)
`וּלְאָדָם אָמַר כִּי שָׁמַעְתָּ לְקוֹל אִשְׁתֶּךָ וַתֹּאכַל מִן הָעֵץ אֲשֶׁר צִוִּיתִיךָ לֵאמֹר לֹא תֹאכַל מִמֶּנּוּ אֲרוּרָה הָאֲדָמָה בַּעֲבוּרֶךָ בְּעִצָּבוֹן תֹּאכֲלֶנָּה כֹּל יְמֵי חַיֶּיךָ` (בראשית ג:יז)
`וַיִּקְרָא הָאָדָם שֵׁם אִשְׁתּוֹ חַוָּה כִּי הִוא הָיְתָה אֵם כָּל חָי` (בראשית ג:כ)
`וַיַּעַשׂ יְהוָה אֱלֹהִים לְאָדָם וּלְאִשְׁתּוֹ כָּתְנוֹת עוֹר וַיַּלְבִּשֵׁם` (בראשית ג:כא)
`וְהָאָדָם יָדַע אֶת חַוָּה אִשְׁתּוֹ וַתַּהַר וַתֵּלֶד אֶת קַיִן וַתֹּאמֶר קָנִיתִי אִישׁ אֶת יְהוָה` (בראשית ד:א)
`וַיֵּדַע קַיִן אֶת אִשְׁתּוֹ וַתַּהַר וַתֵּלֶד אֶת חֲנוֹךְ וַיְהִי בֹּנֶה עִיר וַיִּקְרָא שֵׁם הָעִיר כְּשֵׁם בְּנוֹ חֲנוֹךְ` (בראשית ד:יז)
`וַיִּקַּח לוֹ לֶמֶךְ שְׁתֵּי נָשִׁים שֵׁם הָאַחַת עָדָה וְשֵׁם הַשֵּׁנִית צִלָּה` (בראשית ד:יט)
`וַיֹּאמֶר לֶמֶךְ לְנָשָׁיו עָדָה וְצִלָּה שְׁמַעַן קוֹלִי נְשֵׁי לֶמֶךְ הַאְזֵנָּה אִמְרָתִי כִּי אִישׁ הָרַגְתִּי לְפִצְעִי וְיֶלֶד לְחַבֻּרָתִי` (בראשית ד:כג)
`וַיֵּדַע אָדָם עוֹד אֶת אִשְׁתּוֹ וַתֵּלֶד בֵּן וַתִּקְרָא אֶת שְׁמוֹ שֵׁת כִּי שָׁת לִי אֱלֹהִים זֶרַע אַחֵר תַּחַת הֶבֶל כִּי הֲרָגוֹ קָיִן` (בראשית ד:כה)
`וַיִּרְאוּ בְנֵי הָאֱלֹהִים אֶת בְּנוֹת הָאָדָם כִּי טֹבֹת הֵנָּה וַיִּקְחוּ לָהֶם נָשִׁים מִכֹּל אֲשֶׁר בָּחָרוּ` (בראשית ו:ב)
`וַהֲקִמֹתִי אֶת בְּרִיתִי אִתָּךְ וּבָאתָ אֶל הַתֵּבָה אַתָּה וּבָנֶיךָ וְאִשְׁתְּךָ וּנְשֵׁי בָנֶיךָ אִתָּךְ` (בראשית ו:יח)
`מִכֹּל הַבְּהֵמָה הַטְּהוֹרָה תִּקַּח לְךָ שִׁבְעָה שִׁבְעָה אִישׁ וְאִשְׁתּוֹ וּמִן הַבְּהֵמָה אֲשֶׁר לֹא טְהֹרָה הִוא שְׁנַיִם אִישׁ וְאִשְׁתּוֹ` (בראשית ז:ב)
`וַיָּבֹא נֹחַ וּבָנָיו וְאִשְׁתּוֹ וּנְשֵׁי בָנָיו אִתּוֹ אֶל הַתֵּבָה מִפְּנֵי מֵי הַמַּבּוּל` (בראשית ז:ז)
`בְּעֶצֶם הַיּוֹם הַזֶּה בָּא נֹחַ וְשֵׁם וְחָם וָיֶפֶת בְּנֵי נֹחַ וְאֵשֶׁת נֹחַ וּשְׁלֹשֶׁת נְשֵׁי בָנָיו אִתָּם אֶל הַתֵּבָה` (בראשית ז:יג)
`צֵא מִן הַתֵּבָה אַתָּה וְאִשְׁתְּךָ וּבָנֶיךָ וּנְשֵׁי בָנֶיךָ אִתָּךְ` (בראשית ח:טז)
`וַיֵּצֵא נֹחַ וּבָנָיו וְאִשְׁתּוֹ וּנְשֵׁי בָנָיו אִתּוֹ` (בראשית ח:יח)
`וַיִּקַּח אַבְרָם וְנָחוֹר לָהֶם נָשִׁים שֵׁם אֵשֶׁת אַבְרָם שָׂרָי וְשֵׁם אֵשֶׁת נָחוֹר מִלְכָּה בַּת הָרָן אֲבִי מִלְכָּה וַאֲבִי יִסְכָּה` (בראשית יא:כט)
`וַיִּקַּח תֶּרַח אֶת אַבְרָם בְּנוֹ וְאֶת לוֹט בֶּן הָרָן בֶּן בְּנוֹ וְאֵת שָׂרַי כַּלָּתוֹ אֵשֶׁת אַבְרָם בְּנוֹ וַיֵּצְאוּ אִתָּם מֵאוּר כַּשְׂדִּים לָלֶכֶת אַרְצָה כְּנַעַן וַיָּבֹאוּ עַד חָרָן וַיֵּשְׁבוּ שָׁם` (בראשית יא:לא)
`וַיִּקַּח אַבְרָם אֶת שָׂרַי אִשְׁתּוֹ וְאֶת לוֹט בֶּן אָחִיו וְאֶת כָּל רְכוּשָׁם אֲשֶׁר רָכָשׁוּ וְאֶת הַנֶּפֶשׁ אֲשֶׁר עָשׂוּ בְחָרָן וַיֵּצְאוּ לָלֶכֶת אַרְצָה כְּנַעַן וַיָּבֹאוּ אַרְצָה כְּנָעַן` (בראשית יב:ה)
`וַיְהִי כַּאֲשֶׁר הִקְרִיב לָבוֹא מִצְרָיְמָה וַיֹּאמֶר אֶל שָׂרַי אִשְׁתּוֹ הִנֵּה נָא יָדַעְתִּי כִּי אִשָּׁה יְפַת מַרְאֶה אָתְּ` (בראשית יב:יא)
`וְהָיָה כִּי יִרְאוּ אֹתָךְ הַמִּצְרִים וְאָמְרוּ אִשְׁתּוֹ זֹאת וְהָרְגוּ אֹתִי וְאֹתָךְ יְחַיּוּ` (בראשית יב:יב)
`וַיְהִי כְּבוֹא אַבְרָם מִצְרָיְמָה וַיִּרְאוּ הַמִּצְרִים אֶת הָאִשָּׁה כִּי יָפָה הִוא מְאֹד` (בראשית יב:יד)
`וַיִּרְאוּ אֹתָהּ שָׂרֵי פַרְעֹה וַיְהַלְלוּ אֹתָהּ אֶל פַּרְעֹה וַתֻּקַּח הָאִשָּׁה בֵּית פַּרְעֹה` (בראשית יב:טו)
`וַיְנַגַּע יְהוָה אֶת פַּרְעֹה נְגָעִים גְּדֹלִים וְאֶת בֵּיתוֹ עַל דְּבַר שָׂרַי אֵשֶׁת אַבְרָם` (בראשית יב:יז)
`וַיִּקְרָא פַרְעֹה לְאַבְרָם וַיֹּאמֶר מַה זֹּאת עָשִׂיתָ לִּי לָמָּה לֹא הִגַּדְתָּ לִּי כִּי אִשְׁתְּךָ הִוא` (בראשית יב:יח)
`לָמָה אָמַרְתָּ אֲחֹתִי הִוא וָאֶקַּח אֹתָהּ לִי לְאִשָּׁה וְעַתָּה הִנֵּה אִשְׁתְּךָ קַח וָלֵךְ` (בראשית יב:יט)
`וַיְצַו עָלָיו פַּרְעֹה אֲנָשִׁים וַיְשַׁלְּחוּ אֹתוֹ וְאֶת אִשְׁתּוֹ וְאֶת כָּל אֲשֶׁר לוֹ` (בראשית יב:כ)
`וַיַּעַל אַבְרָם מִמִּצְרַיִם הוּא וְאִשְׁתּוֹ וְכָל אֲשֶׁר לוֹ וְלוֹט עִמּוֹ הַנֶּגְבָּה` (בראשית יג:א)
`וַיָּשֶׁב אֵת כָּל הָרְכֻשׁ וְגַם אֶת לוֹט אָחִיו וּרְכֻשׁוֹ הֵשִׁיב וְגַם אֶת הַנָּשִׁים וְאֶת הָעָם` (בראשית יד:טז)
`וְשָׂרַי אֵשֶׁת אַבְרָם לֹא יָלְדָה לוֹ וְלָהּ שִׁפְחָה מִצְרִית וּשְׁמָהּ הָגָר` (בראשית טז:א)
`וַתִּקַּח שָׂרַי אֵשֶׁת אַבְרָם אֶת הָגָר הַמִּצְרִית שִׁפְחָתָהּ מִקֵּץ עֶשֶׂר שָׁנִים לְשֶׁבֶת אַבְרָם בְּאֶרֶץ כְּנָעַן וַתִּתֵּן אֹתָהּ לְאַבְרָם אִישָׁהּ לוֹ לְאִשָּׁה` (בראשית טז:ג)
`וַיֹּאמֶר אֱלֹהִים אֶל אַבְרָהָם שָׂרַי אִשְׁתְּךָ לֹא תִקְרָא אֶת שְׁמָהּ שָׂרָי כִּי שָׂרָה שְׁמָהּ` (בראשית יז:טו)
`וַיֹּאמֶר אֱלֹהִים אֲבָל שָׂרָה אִשְׁתְּךָ יֹלֶדֶת לְךָ בֵּן וְקָרָאתָ אֶת שְׁמוֹ יִצְחָק וַהֲקִמֹתִי אֶת בְּרִיתִי אִתּוֹ לִבְרִית עוֹלָם לְזַרְעוֹ אַחֲרָיו` (בראשית יז:יט)
`וַיֹּאמְרוּ אֵׄלָׄיׄוׄ אַיֵּה שָׂרָה אִשְׁתֶּךָ וַיֹּאמֶר הִנֵּה בָאֹהֶל` (בראשית יח:ט)
`וַיֹּאמֶר שׁוֹב אָשׁוּב אֵלֶיךָ כָּעֵת חַיָּה וְהִנֵּה בֵן לְשָׂרָה אִשְׁתֶּךָ וְשָׂרָה שֹׁמַעַת פֶּתַח הָאֹהֶל וְהוּא אַחֲרָיו` (בראשית יח:י)
`וְאַבְרָהָם וְשָׂרָה זְקֵנִים בָּאִים בַּיָּמִים חָדַל לִהְיוֹת לְשָׂרָה אֹרַח כַּנָּשִׁים` (בראשית יח:יא)
`וּכְמוֹ הַשַּׁחַר עָלָה וַיָּאִיצוּ הַמַּלְאָכִים בְּלוֹט לֵאמֹר קוּם קַח אֶת אִשְׁתְּךָ וְאֶת שְׁתֵּי בְנֹתֶיךָ הַנִּמְצָאֹת פֶּן תִּסָּפֶה בַּעֲוֺן הָעִיר` (בראשית יט:טו)
`וַיִּתְמַהְמָהּ וַיַּחֲזִקוּ הָאֲנָשִׁים בְּיָדוֹ וּבְיַד אִשְׁתּוֹ וּבְיַד שְׁתֵּי בְנֹתָיו בְּחֶמְלַת יְהוָה עָלָיו וַיֹּצִאֻהוּ וַיַּנִּחֻהוּ מִחוּץ לָעִיר` (בראשית יט:טז)
`וַתַּבֵּט אִשְׁתּוֹ מֵאַחֲרָיו וַתְּהִי נְצִיב מֶלַח` (בראשית יט:כו)
`וַיֹּאמֶר אַבְרָהָם אֶל שָׂרָה אִשְׁתּוֹ אֲחֹתִי הִוא וַיִּשְׁלַח אֲבִימֶלֶךְ מֶלֶךְ גְּרָר וַיִּקַּח אֶת שָׂרָה` (בראשית כ:ב)
`וַיָּבֹא אֱלֹהִים אֶל אֲבִימֶלֶךְ בַּחֲלוֹם הַלָּיְלָה וַיֹּאמֶר לוֹ הִנְּךָ מֵת עַל הָאִשָּׁה אֲשֶׁר לָקַחְתָּ וְהִוא בְּעֻלַת בָּעַל` (בראשית כ:ג)
`וְעַתָּה הָשֵׁב אֵשֶׁת הָאִישׁ כִּי נָבִיא הוּא וְיִתְפַּלֵּל בַּעַדְךָ וֶחְיֵה וְאִם אֵינְךָ מֵשִׁיב דַּע כִּי מוֹת תָּמוּת אַתָּה וְכָל אֲשֶׁר לָךְ` (בראשית כ:ז)
`וַיֹּאמֶר אַבְרָהָם כִּי אָמַרְתִּי רַק אֵין יִרְאַת אֱלֹהִים בַּמָּקוֹם הַזֶּה וַהֲרָגוּנִי עַל דְּבַר אִשְׁתִּי` (בראשית כ:יא)
`וְגַם אָמְנָה אֲחֹתִי בַת אָבִי הִוא אַךְ לֹא בַת אִמִּי וַתְּהִי לִי לְאִשָּׁה` (בראשית כ:יב)
`וַיִּקַּח אֲבִימֶלֶךְ צֹאן וּבָקָר וַעֲבָדִים וּשְׁפָחֹת וַיִּתֵּן לְאַבְרָהָם וַיָּשֶׁב לוֹ אֵת שָׂרָה אִשְׁתּוֹ` (בראשית כ:יד)
`וַיִּתְפַּלֵּל אַבְרָהָם אֶל הָאֱלֹהִים וַיִּרְפָּא אֱלֹהִים אֶת אֲבִימֶלֶךְ וְאֶת אִשְׁתּוֹ וְאַמְהֹתָיו וַיֵּלֵדוּ` (בראשית כ:יז)
`כִּי עָצֹר עָצַר יְהוָה בְּעַד כָּל רֶחֶם לְבֵית אֲבִימֶלֶךְ עַל דְּבַר שָׂרָה אֵשֶׁת אַבְרָהָם` (בראשית כ:יח)
`וַיֵּשֶׁב בְּמִדְבַּר פָּארָן וַתִּקַּח לוֹ אִמּוֹ אִשָּׁה מֵאֶרֶץ מִצְרָיִם` (בראשית כא:כא)
`וְאַחֲרֵי כֵן קָבַר אַבְרָהָם אֶת שָׂרָה אִשְׁתּוֹ אֶל מְעָרַת שְׂדֵה הַמַּכְפֵּלָה עַל פְּנֵי מַמְרֵא הִוא חֶבְרוֹן בְּאֶרֶץ כְּנָעַן` (בראשית כג:יט)
`וְאַשְׁבִּיעֲךָ בַּיהוָה אֱלֹהֵי הַשָּׁמַיִם וֵאלֹהֵי הָאָרֶץ אֲשֶׁר לֹא תִקַּח אִשָּׁה לִבְנִי מִבְּנוֹת הַכְּנַעֲנִי אֲשֶׁר אָנֹכִי יוֹשֵׁב בְּקִרְבּוֹ` (בראשית כד:ג)
`כִּי אֶל אַרְצִי וְאֶל מוֹלַדְתִּי תֵּלֵךְ וְלָקַחְתָּ אִשָּׁה לִבְנִי לְיִצְחָק` (בראשית כד:ד)
`וַיֹּאמֶר אֵלָיו הָעֶבֶד אוּלַי לֹא תֹאבֶה הָאִשָּׁה לָלֶכֶת אַחֲרַי אֶל הָאָרֶץ הַזֹּאת הֶהָשֵׁב אָשִׁיב אֶת בִּנְךָ אֶל הָאָרֶץ אֲשֶׁר יָצָאתָ מִשָּׁם` (בראשית כד:ה)
`יְהוָה אֱלֹהֵי הַשָּׁמַיִם אֲשֶׁר לְקָחַנִי מִבֵּית אָבִי וּמֵאֶרֶץ מוֹלַדְתִּי וַאֲשֶׁר דִּבֶּר לִי וַאֲשֶׁר נִשְׁבַּע לִי לֵאמֹר לְזַרְעֲךָ אֶתֵּן אֶת הָאָרֶץ הַזֹּאת הוּא יִשְׁלַח מַלְאָכוֹ לְפָנֶיךָ וְלָקַחְתָּ אִשָּׁה לִבְנִי מִשָּׁם` (בראשית כד:ז)
`וְאִם לֹא תֹאבֶה הָאִשָּׁה לָלֶכֶת אַחֲרֶיךָ וְנִקִּיתָ מִשְּׁבֻעָתִי זֹאת רַק אֶת בְּנִי לֹא תָשֵׁב שָׁמָּה` (בראשית כד:ח)
`וַיְהִי הוּא טֶרֶם כִּלָּה לְדַבֵּר וְהִנֵּה רִבְקָה יֹצֵאת אֲשֶׁר יֻלְּדָה לִבְתוּאֵל בֶּן מִלְכָּה אֵשֶׁת נָחוֹר אֲחִי אַבְרָהָם וְכַדָּהּ עַל שִׁכְמָהּ` (בראשית כד:טו)
`וַתֵּלֶד שָׂרָה אֵשֶׁת אֲדֹנִי בֵן לַאדֹנִי אַחֲרֵי זִקְנָתָהּ וַיִּתֶּן לּוֹ אֶת כָּל אֲשֶׁר לוֹ` (בראשית כד:לו)
`וַיַּשְׁבִּעֵנִי אֲדֹנִי לֵאמֹר לֹא תִקַּח אִשָּׁה לִבְנִי מִבְּנוֹת הַכְּנַעֲנִי אֲשֶׁר אָנֹכִי יֹשֵׁב בְּאַרְצוֹ` (בראשית כד:לז)
`אִם לֹא אֶל בֵּית אָבִי תֵּלֵךְ וְאֶל מִשְׁפַּחְתִּי וְלָקַחְתָּ אִשָּׁה לִבְנִי` (בראשית כד:לח)
`וָאֹמַר אֶל אֲדֹנִי אֻלַי לֹא תֵלֵךְ הָאִשָּׁה אַחֲרָי` (בראשית כד:לט)
`וַיֹּאמֶר אֵלָי יְהוָה אֲשֶׁר הִתְהַלַּכְתִּי לְפָנָיו יִשְׁלַח מַלְאָכוֹ אִתָּךְ וְהִצְלִיחַ דַּרְכֶּךָ וְלָקַחְתָּ אִשָּׁה לִבְנִי מִמִּשְׁפַּחְתִּי וּמִבֵּית אָבִי` (בראשית כד:מ)
`וְאָמְרָה אֵלַי גַּם אַתָּה שְׁתֵה וְגַם לִגְמַלֶּיךָ אֶשְׁאָב הִוא הָאִשָּׁה אֲשֶׁר הֹכִיחַ יְהוָה לְבֶן אֲדֹנִי` (בראשית כד:מד)
`הִנֵּה רִבְקָה לְפָנֶיךָ קַח וָלֵךְ וּתְהִי אִשָּׁה לְבֶן אֲדֹנֶיךָ כַּאֲשֶׁר דִּבֶּר יְהוָה` (בראשית כד:נא)
`וַיְבִאֶהָ יִצְחָק הָאֹהֱלָה שָׂרָה אִמּוֹ וַיִּקַּח אֶת רִבְקָה וַתְּהִי לוֹ לְאִשָּׁה וַיֶּאֱהָבֶהָ וַיִּנָּחֵם יִצְחָק אַחֲרֵי אִמּוֹ` (בראשית כד:סז)
`וַיֹּסֶף אַבְרָהָם וַיִּקַּח אִשָּׁה וּשְׁמָהּ קְטוּרָה` (בראשית כה:א)
`הַשָּׂדֶה אֲשֶׁר קָנָה אַבְרָהָם מֵאֵת בְּנֵי חֵת שָׁמָּה קֻבַּר אַבְרָהָם וְשָׂרָה אִשְׁתּוֹ` (בראשית כה:י)
`וַיְהִי יִצְחָק בֶּן אַרְבָּעִים שָׁנָה בְּקַחְתּוֹ אֶת רִבְקָה בַּת בְּתוּאֵל הָאֲרַמִּי מִפַּדַּן אֲרָם אֲחוֹת לָבָן הָאֲרַמִּי לוֹ לְאִשָּׁה` (בראשית כה:כ)
`וַיֶּעְתַּר יִצְחָק לַיהוָה לְנֹכַח אִשְׁתּוֹ כִּי עֲקָרָה הִוא וַיֵּעָתֶר לוֹ יְהוָה וַתַּהַר רִבְקָה אִשְׁתּוֹ` (בראשית כה:כא)
`וַיִּשְׁאֲלוּ אַנְשֵׁי הַמָּקוֹם לְאִשְׁתּוֹ וַיֹּאמֶר אֲחֹתִי הִוא כִּי יָרֵא לֵאמֹר אִשְׁתִּי פֶּן יַהַרְגֻנִי אַנְשֵׁי הַמָּקוֹם עַל רִבְקָה כִּי טוֹבַת מַרְאֶה הִיא` (בראשית כו:ז)
`וַיְהִי כִּי אָרְכוּ לוֹ שָׁם הַיָּמִים וַיַּשְׁקֵף אֲבִימֶלֶךְ מֶלֶךְ פְּלִשְׁתִּים בְּעַד הַחַלּוֹן וַיַּרְא וְהִנֵּה יִצְחָק מְצַחֵק אֵת רִבְקָה אִשְׁתּוֹ` (בראשית כו:ח)
`וַיִּקְרָא אֲבִימֶלֶךְ לְיִצְחָק וַיֹּאמֶר אַךְ הִנֵּה אִשְׁתְּךָ הִוא וְאֵיךְ אָמַרְתָּ אֲחֹתִי הִוא וַיֹּאמֶר אֵלָיו יִצְחָק כִּי אָמַרְתִּי פֶּן אָמוּת עָלֶיהָ` (בראשית כו:ט)
`וַיֹּאמֶר אֲבִימֶלֶךְ מַה זֹּאת עָשִׂיתָ לָּנוּ כִּמְעַט שָׁכַב אַחַד הָעָם אֶת אִשְׁתֶּךָ וְהֵבֵאתָ עָלֵינוּ אָשָׁם` (בראשית כו:י)
`וַיְצַו אֲבִימֶלֶךְ אֶת כָּל הָעָם לֵאמֹר הַנֹּגֵעַ בָּאִישׁ הַזֶּה וּבְאִשְׁתּוֹ מוֹת יוּמָת` (בראשית כו:יא)
`וַיְהִי עֵשָׂו בֶּן אַרְבָּעִים שָׁנָה וַיִּקַּח אִשָּׁה אֶת יְהוּדִית בַּת בְּאֵרִי הַחִתִּי וְאֶת בָּשְׂמַת בַּת אֵילֹן הַחִתִּי` (בראשית כו:לד)
`וַתֹּאמֶר רִבְקָה אֶל יִצְחָק קַצְתִּי בְחַיַּי מִפְּנֵי בְּנוֹת חֵת אִם לֹקֵחַ יַעֲקֹב אִשָּׁה מִבְּנוֹת חֵת כָּאֵלֶּה מִבְּנוֹת הָאָרֶץ לָמָּה לִּי חַיִּים` (בראשית כז:מו)
`וַיִּקְרָא יִצְחָק אֶל יַעֲקֹב וַיְבָרֶךְ אֹתוֹ וַיְצַוֵּהוּ וַיֹּאמֶר לוֹ לֹא תִקַּח אִשָּׁה מִבְּנוֹת כְּנָעַן` (בראשית כח:א)
`קוּם לֵךְ פַּדֶּנָה אֲרָם בֵּיתָה בְתוּאֵל אֲבִי אִמֶּךָ וְקַח לְךָ מִשָּׁם אִשָּׁה מִבְּנוֹת לָבָן אֲחִי אִמֶּךָ` (בראשית כח:ב)
`וַיַּרְא עֵשָׂו כִּי בֵרַךְ יִצְחָק אֶת יַעֲקֹב וְשִׁלַּח אֹתוֹ פַּדֶּנָה אֲרָם לָקַחַת לוֹ מִשָּׁם אִשָּׁה בְּבָרֲכוֹ אֹתוֹ וַיְצַו עָלָיו לֵאמֹר לֹא תִקַּח אִשָּׁה מִבְּנוֹת כְּנָעַן` (בראשית כח:ו)
`וַיֵּלֶךְ עֵשָׂו אֶל יִשְׁמָעֵאל וַיִּקַּח אֶת מָחֲלַת בַּת יִשְׁמָעֵאל בֶּן אַבְרָהָם אֲחוֹת נְבָיוֹת עַל נָשָׁיו לוֹ לְאִשָּׁה` (בראשית כח:ט)
`וַיֹּאמֶר יַעֲקֹב אֶל לָבָן הָבָה אֶת אִשְׁתִּי כִּי מָלְאוּ יָמָי וְאָבוֹאָה אֵלֶיהָ` (בראשית כט:כא)
`וַיַּעַשׂ יַעֲקֹב כֵּן וַיְמַלֵּא שְׁבֻעַ זֹאת וַיִּתֶּן לוֹ אֶת רָחֵל בִּתּוֹ לוֹ לְאִשָּׁה` (בראשית כט:כח)
`וַתִּתֶּן לוֹ אֶת בִּלְהָה שִׁפְחָתָהּ לְאִשָּׁה וַיָּבֹא אֵלֶיהָ יַעֲקֹב` (בראשית ל:ד)
`וַתֵּרֶא לֵאָה כִּי עָמְדָה מִלֶּדֶת וַתִּקַּח אֶת זִלְפָּה שִׁפְחָתָהּ וַתִּתֵּן אֹתָהּ לְיַעֲקֹב לְאִשָּׁה` (בראשית ל:ט)
`תְּנָה אֶת נָשַׁי וְאֶת יְלָדַי אֲשֶׁר עָבַדְתִּי אֹתְךָ בָּהֵן וְאֵלֵכָה כִּי אַתָּה יָדַעְתָּ אֶת עֲבֹדָתִי אֲשֶׁר עֲבַדְתִּיךָ` (בראשית ל:כו)
`וַיָּקָם יַעֲקֹב וַיִּשָּׂא אֶת בָּנָיו וְאֶת נָשָׁיו עַל הַגְּמַלִּים` (בראשית לא:יז)
`וַתֹּאמֶר אֶל אָבִיהָ אַל יִחַר בְּעֵינֵי אֲדֹנִי כִּי לוֹא אוּכַל לָקוּם מִפָּנֶיךָ כִּי דֶרֶךְ נָשִׁים לִי וַיְחַפֵּשׂ וְלֹא מָצָא אֶת הַתְּרָפִים` (בראשית לא:לה)
`אִם תְּעַנֶּה אֶת בְּנֹתַי וְאִם תִּקַּח נָשִׁים עַל בְּנֹתַי אֵין אִישׁ עִמָּנוּ רְאֵה אֱלֹהִים עֵד בֵּינִי וּבֵינֶךָ` (בראשית לא:נ)
`וַיָּקָם בַּלַּיְלָה הוּא וַיִּקַּח אֶת שְׁתֵּי נָשָׁיו וְאֶת שְׁתֵּי שִׁפְחֹתָיו וְאֶת אַחַד עָשָׂר יְלָדָיו וַיַּעֲבֹר אֵת מַעֲבַר יַבֹּק` (בראשית לב:כב)
`וַיִּשָּׂא אֶת עֵינָיו וַיַּרְא אֶת הַנָּשִׁים וְאֶת הַיְלָדִים וַיֹּאמֶר מִי אֵלֶּה לָּךְ וַיֹּאמַר הַיְלָדִים אֲשֶׁר חָנַן אֱלֹהִים אֶת עַבְדֶּךָ` (בראשית לג:ה)
`וַיֹּאמֶר שְׁכֶם אֶל חֲמוֹר אָבִיו לֵאמֹר קַח לִי אֶת הַיַּלְדָּה הַזֹּאת לְאִשָּׁה` (בראשית לד:ד)
`וַיְדַבֵּר חֲמוֹר אִתָּם לֵאמֹר שְׁכֶם בְּנִי חָשְׁקָה נַפְשׁוֹ בְּבִתְּכֶם תְּנוּ נָא אֹתָהּ לוֹ לְאִשָּׁה` (בראשית לד:ח)
`הַרְבּוּ עָלַי מְאֹד מֹהַר וּמַתָּן וְאֶתְּנָה כַּאֲשֶׁר תֹּאמְרוּ אֵלָי וּתְנוּ לִי אֶת הַנַּעֲרָ לְאִשָּׁה` (בראשית לד:יב)
`הָאֲנָשִׁים הָאֵלֶּה שְׁלֵמִים הֵם אִתָּנוּ וְיֵשְׁבוּ בָאָרֶץ וְיִסְחֲרוּ אֹתָהּ וְהָאָרֶץ הִנֵּה רַחֲבַת יָדַיִם לִפְנֵיהֶם אֶת בְּנֹתָם נִקַּח לָנוּ לְנָשִׁים וְאֶת בְּנֹתֵינוּ נִתֵּן לָהֶם` (בראשית לד:כא)
`וְאֶת כָּל חֵילָם וְאֶת כָּל טַפָּם וְאֶת נְשֵׁיהֶם שָׁבוּ וַיָּבֹזּוּ וְאֵת כָּל אֲשֶׁר בַּבָּיִת` (בראשית לד:כט)
`עֵשָׂו לָקַח אֶת נָשָׁיו מִבְּנוֹת כְּנָעַן אֶת עָדָה בַּת אֵילוֹן הַחִתִּי וְאֶת אָהֳלִיבָמָה בַּת עֲנָה בַּת צִבְעוֹן הַחִוִּי` (בראשית לו:ב)
`וַיִּקַּח עֵשָׂו אֶת נָשָׁיו וְאֶת בָּנָיו וְאֶת בְּנֹתָיו וְאֶת כָּל נַפְשׁוֹת בֵּיתוֹ וְאֶת מִקְנֵהוּ וְאֶת כָּל בְּהֶמְתּוֹ וְאֵת כָּל קִנְיָנוֹ אֲשֶׁר רָכַשׁ בְּאֶרֶץ כְּנָעַן וַיֵּלֶךְ אֶל אֶרֶץ מִפְּנֵי יַעֲקֹב אָחִיו` (בראשית לו:ו)
`אֵלֶּה שְׁמוֹת בְּנֵי עֵשָׂו אֱלִיפַז בֶּן עָדָה אֵשֶׁת עֵשָׂו רְעוּאֵל בֶּן בָּשְׂמַת אֵשֶׁת עֵשָׂו` (בראשית לו:י)
`וְתִמְנַע הָיְתָה פִילֶגֶשׁ לֶאֱלִיפַז בֶּן עֵשָׂו וַתֵּלֶד לֶאֱלִיפַז אֶת עֲמָלֵק אֵלֶּה בְּנֵי עָדָה אֵשֶׁת עֵשָׂו` (בראשית לו:יב)
`וְאֵלֶּה בְּנֵי רְעוּאֵל נַחַת וָזֶרַח שַׁמָּה וּמִזָּה אֵלֶּה הָיוּ בְּנֵי בָשְׂמַת אֵשֶׁת עֵשָׂו` (בראשית לו:יג)
`וְאֵלֶּה הָיוּ בְּנֵי אָהֳלִיבָמָה בַת עֲנָה בַּת צִבְעוֹן אֵשֶׁת עֵשָׂו וַתֵּלֶד לְעֵשָׂו אֶת יְעִישׁ וְאֶת יַעְלָם וְאֶת קֹרַח` (בראשית לו:יד)
`וְאֵלֶּה בְּנֵי רְעוּאֵל בֶּן עֵשָׂו אַלּוּף נַחַת אַלּוּף זֶרַח אַלּוּף שַׁמָּה אַלּוּף מִזָּה אֵלֶּה אַלּוּפֵי רְעוּאֵל בְּאֶרֶץ אֱדוֹם אֵלֶּה בְּנֵי בָשְׂמַת אֵשֶׁת עֵשָׂו` (בראשית לו:יז)
`וְאֵלֶּה בְּנֵי אָהֳלִיבָמָה אֵשֶׁת עֵשָׂו אַלּוּף יְעוּשׁ אַלּוּף יַעְלָם אַלּוּף קֹרַח אֵלֶּה אַלּוּפֵי אָהֳלִיבָמָה בַּת עֲנָה אֵשֶׁת עֵשָׂו` (בראשית לו:יח)
`וַיָּמָת בַּעַל חָנָן בֶּן עַכְבּוֹר וַיִּמְלֹךְ תַּחְתָּיו הֲדַר וְשֵׁם עִירוֹ פָּעוּ וְשֵׁם אִשְׁתּוֹ מְהֵיטַבְאֵל בַּת מַטְרֵד בַּת מֵי זָהָב` (בראשית לו:לט)
`אֵלֶּה תֹּלְדוֹת יַעֲקֹב יוֹסֵף בֶּן שְׁבַע עֶשְׂרֵה שָׁנָה הָיָה רֹעֶה אֶת אֶחָיו בַּצֹּאן וְהוּא נַעַר אֶת בְּנֵי בִלְהָה וְאֶת בְּנֵי זִלְפָּה נְשֵׁי אָבִיו וַיָּבֵא יוֹסֵף אֶת דִּבָּתָם רָעָה אֶל אֲבִיהֶם` (בראשית לז:ב)
`וַיִּקַּח יְהוּדָה אִשָּׁה לְעֵר בְּכוֹרוֹ וּשְׁמָהּ תָּמָר` (בראשית לח:ו)
`וַיֹּאמֶר יְהוּדָה לְאוֹנָן בֹּא אֶל אֵשֶׁת אָחִיךָ וְיַבֵּם אֹתָהּ וְהָקֵם זֶרַע לְאָחִיךָ` (בראשית לח:ח)
`וַיֵּדַע אוֹנָן כִּי לֹּא לוֹ יִהְיֶה הַזָּרַע וְהָיָה אִם בָּא אֶל אֵשֶׁת אָחִיו וְשִׁחֵת אַרְצָה לְבִלְתִּי נְתָן זֶרַע לְאָחִיו` (בראשית לח:ט)
`וַיִּרְבּוּ הַיָּמִים וַתָּמָת בַּת שׁוּעַ אֵשֶׁת יְהוּדָה וַיִּנָּחֶם יְהוּדָה וַיַּעַל עַל גֹּזֲזֵי צֹאנוֹ הוּא וְחִירָה רֵעֵהוּ הָעֲדֻלָּמִי תִּמְנָתָה` (בראשית לח:יב)
`וַתָּסַר בִּגְדֵי אַלְמְנוּתָהּ מֵעָלֶיהָ וַתְּכַס בַּצָּעִיף וַתִּתְעַלָּף וַתֵּשֶׁב בְּפֶתַח עֵינַיִם אֲשֶׁר עַל דֶּרֶךְ תִּמְנָתָה כִּי רָאֲתָה כִּי גָדַל שֵׁלָה וְהִוא לֹא נִתְּנָה לוֹ לְאִשָּׁה` (בראשית לח:יד)
`וַיִּשְׁלַח יְהוּדָה אֶת גְּדִי הָעִזִּים בְּיַד רֵעֵהוּ הָעֲדֻלָּמִי לָקַחַת הָעֵרָבוֹן מִיַּד הָאִשָּׁה וְלֹא מְצָאָהּ` (בראשית לח:כ)
`וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה וַתִּשָּׂא אֵשֶׁת אֲדֹנָיו אֶת עֵינֶיהָ אֶל יוֹסֵף וַתֹּאמֶר שִׁכְבָה עִמִּי` (בראשית לט:ז)
`וַיְמָאֵן וַיֹּאמֶר אֶל אֵשֶׁת אֲדֹנָיו הֵן אֲדֹנִי לֹא יָדַע אִתִּי מַה בַּבָּיִת וְכֹל אֲשֶׁר יֶשׁ לוֹ נָתַן בְּיָדִי` (בראשית לט:ח)
`אֵינֶנּוּ גָדוֹל בַּבַּיִת הַזֶּה מִמֶּנִּי וְלֹא חָשַׂךְ מִמֶּנִּי מְאוּמָה כִּי אִם אוֹתָךְ בַּאֲשֶׁר אַתְּ אִשְׁתּוֹ וְאֵיךְ אֶעֱשֶׂה הָרָעָה הַגְּדֹלָה הַזֹּאת וְחָטָאתִי לֵאלֹהִים` (בראשית לט:ט)
`וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת דִּבְרֵי אִשְׁתּוֹ אֲשֶׁר דִּבְּרָה אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂהּ לִי עַבְדֶּךָ וַיִּחַר אַפּוֹ` (בראשית לט:יט)
`וַיִּקְרָא פַרְעֹה שֵׁם יוֹסֵף צָפְנַת פַּעְנֵחַ וַיִּתֶּן לוֹ אֶת אָסְנַת בַּת פּוֹטִי פֶרַע כֹּהֵן אֹן לְאִשָּׁה וַיֵּצֵא יוֹסֵף עַל אֶרֶץ מִצְרָיִם` (בראשית מא:מה)
`וַיֹּאמֶר עַבְדְּךָ אָבִי אֵלֵינוּ אַתֶּם יְדַעְתֶּם כִּי שְׁנַיִם יָלְדָה לִּי אִשְׁתִּי` (בראשית מד:כז)
`וְאַתָּה צֻוֵּיתָה זֹאת עֲשׂוּ קְחוּ לָכֶם מֵאֶרֶץ מִצְרַיִם עֲגָלוֹת לְטַפְּכֶם וְלִנְשֵׁיכֶם וּנְשָׂאתֶם אֶת אֲבִיכֶם וּבָאתֶם` (בראשית מה:יט)
`וַיָּקָם יַעֲקֹב מִבְּאֵר שָׁבַע וַיִּשְׂאוּ בְנֵי יִשְׂרָאֵל אֶת יַעֲקֹב אֲבִיהֶם וְאֶת טַפָּם וְאֶת נְשֵׁיהֶם בָּעֲגָלוֹת אֲשֶׁר שָׁלַח פַּרְעֹה לָשֵׂאת אֹתוֹ` (בראשית מו:ה)
`בְּנֵי רָחֵל אֵשֶׁת יַעֲקֹב יוֹסֵף וּבִנְיָמִן` (בראשית מו:יט)
`כָּל הַנֶּפֶשׁ הַבָּאָה לְיַעֲקֹב מִצְרַיְמָה יֹצְאֵי יְרֵכוֹ מִלְּבַד נְשֵׁי בְנֵי יַעֲקֹב כָּל נֶפֶשׁ שִׁשִּׁים וָשֵׁשׁ` (בראשית מו:כו)
`שָׁמָּה קָבְרוּ אֶת אַבְרָהָם וְאֵת שָׂרָה אִשְׁתּוֹ שָׁמָּה קָבְרוּ אֶת יִצְחָק וְאֵת רִבְקָה אִשְׁתּוֹ וְשָׁמָּה קָבַרְתִּי אֶת לֵאָה` (בראשית מט:לא)
`וַתֹּאמַרְןָ הַמְיַלְּדֹת אֶל פַּרְעֹה כִּי לֹא כַנָּשִׁים הַמִּצְרִיֹּת הָעִבְרִיֹּת כִּי חָיוֹת הֵנָּה בְּטֶרֶם תָּבוֹא אֲלֵהֶן הַמְיַלֶּדֶת וְיָלָדוּ` (שמות א:יט)
`וַתַּהַר הָאִשָּׁה וַתֵּלֶד בֵּן וַתֵּרֶא אֹתוֹ כִּי טוֹב הוּא וַתִּצְפְּנֵהוּ שְׁלֹשָׁה יְרָחִים` (שמות ב:ב)
`וַתֹּאמֶר אֲחֹתוֹ אֶל בַּת פַּרְעֹה הַאֵלֵךְ וְקָרָאתִי לָךְ אִשָּׁה מֵינֶקֶת מִן הָעִבְרִיֹּת וְתֵינִק לָךְ אֶת הַיָּלֶד` (שמות ב:ז)
`וַתֹּאמֶר לָהּ בַּת פַּרְעֹה הֵילִיכִי אֶת הַיֶּלֶד הַזֶּה וְהֵינִקִהוּ לִי וַאֲנִי אֶתֵּן אֶת שְׂכָרֵךְ וַתִּקַּח הָאִשָּׁה הַיֶּלֶד וַתְּנִיקֵהוּ` (שמות ב:ט)
`וְשָׁאֲלָה אִשָּׁה מִשְּׁכֶנְתָּהּ וּמִגָּרַת בֵּיתָהּ כְּלֵי כֶסֶף וּכְלֵי זָהָב וּשְׂמָלֹת וְשַׂמְתֶּם עַל בְּנֵיכֶם וְעַל בְּנֹתֵיכֶם וְנִצַּלְתֶּם אֶת מִצְרָיִם` (שמות ג:כב)
`וַיִּקַּח מֹשֶׁה אֶת אִשְׁתּוֹ וְאֶת בָּנָיו וַיַּרְכִּבֵם עַל הַחֲמֹר וַיָּשָׁב אַרְצָה מִצְרָיִם וַיִּקַּח מֹשֶׁה אֶת מַטֵּה הָאֱלֹהִים בְּיָדוֹ` (שמות ד:כ)
`וַיִּקַּח עַמְרָם אֶת יוֹכֶבֶד דֹּדָתוֹ לוֹ לְאִשָּׁה וַתֵּלֶד לוֹ אֶת אַהֲרֹן וְאֶת מֹשֶׁה וּשְׁנֵי חַיֵּי עַמְרָם שֶׁבַע וּשְׁלֹשִׁים וּמְאַת שָׁנָה` (שמות ו:כ)
`וַיִּקַּח אַהֲרֹן אֶת אֱלִישֶׁבַע בַּת עַמִּינָדָב אֲחוֹת נַחְשׁוֹן לוֹ לְאִשָּׁה וַתֵּלֶד לוֹ אֶת נָדָב וְאֶת אֲבִיהוּא אֶת אֶלְעָזָר וְאֶת אִיתָמָר` (שמות ו:כג)
`וְאֶלְעָזָר בֶּן אַהֲרֹן לָקַח לוֹ מִבְּנוֹת פּוּטִיאֵל לוֹ לְאִשָּׁה וַתֵּלֶד לוֹ אֶת פִּינְחָס אֵלֶּה רָאשֵׁי אֲבוֹת הַלְוִיִּם לְמִשְׁפְּחֹתָם` (שמות ו:כה)
`דַּבֶּר נָא בְּאָזְנֵי הָעָם וְיִשְׁאֲלוּ אִישׁ מֵאֵת רֵעֵהוּ וְאִשָּׁה מֵאֵת רְעוּתָהּ כְּלֵי כֶסֶף וּכְלֵי זָהָב` (שמות יא:ב)
`וַתִּקַּח מִרְיָם הַנְּבִיאָה אֲחוֹת אַהֲרֹן אֶת הַתֹּף בְּיָדָהּ וַתֵּצֶאןָ כָל הַנָּשִׁים אַחֲרֶיהָ בְּתֻפִּים וּבִמְחֹלֹת` (שמות טו:כ)
`וַיִּקַּח יִתְרוֹ חֹתֵן מֹשֶׁה אֶת צִפֹּרָה אֵשֶׁת מֹשֶׁה אַחַר שִׁלּוּחֶיהָ` (שמות יח:ב)
`וַיָּבֹא יִתְרוֹ חֹתֵן מֹשֶׁה וּבָנָיו וְאִשְׁתּוֹ אֶל מֹשֶׁה אֶל הַמִּדְבָּר אֲשֶׁר הוּא חֹנֶה שָׁם הַר הָאֱלֹהִים` (שמות יח:ה)
`וַיֹּאמֶר אֶל מֹשֶׁה אֲנִי חֹתֶנְךָ יִתְרוֹ בָּא אֵלֶיךָ וְאִשְׁתְּךָ וּשְׁנֵי בָנֶיהָ עִמָּהּ` (שמות יח:ו)
`וַיֹּאמֶר אֶל הָעָם הֱיוּ נְכֹנִים לִשְׁלֹשֶׁת יָמִים אַל תִּגְּשׁוּ אֶל אִשָּׁה` (שמות יט:טו)
`לֹא תַחְמֹד בֵּית רֵעֶךָ לֹא תַחְמֹד אֵשֶׁת רֵעֶךָ וְעַבְדּוֹ וַאֲמָתוֹ וְשׁוֹרוֹ וַחֲמֹרוֹ וְכֹל אֲשֶׁר לְרֵעֶךָ` (שמות כ:יז)
`אִם בְּגַפּוֹ יָבֹא בְּגַפּוֹ יֵצֵא אִם בַּעַל אִשָּׁה הוּא וְיָצְאָה אִשְׁתּוֹ עִמּוֹ` (שמות כא:ג)
`אִם אֲדֹנָיו יִתֶּן לוֹ אִשָּׁה וְיָלְדָה לוֹ בָנִים אוֹ בָנוֹת הָאִשָּׁה וִילָדֶיהָ תִּהְיֶה לַאדֹנֶיהָ וְהוּא יֵצֵא בְגַפּוֹ` (שמות כא:ד)
`וְאִם אָמֹר יֹאמַר הָעֶבֶד אָהַבְתִּי אֶת אֲדֹנִי אֶת אִשְׁתִּי וְאֶת בָּנָי לֹא אֵצֵא חָפְשִׁי` (שמות כא:ה)
`וְכִי יִנָּצוּ אֲנָשִׁים וְנָגְפוּ אִשָּׁה הָרָה וְיָצְאוּ יְלָדֶיהָ וְלֹא יִהְיֶה אָסוֹן עָנוֹשׁ יֵעָנֵשׁ כַּאֲשֶׁר יָשִׁית עָלָיו בַּעַל הָאִשָּׁה וְנָתַן בִּפְלִלִים` (שמות כא:כב)
`וְכִי יִגַּח שׁוֹר אֶת אִישׁ אוֹ אֶת אִשָּׁה וָמֵת סָקוֹל יִסָּקֵל הַשּׁוֹר וְלֹא יֵאָכֵל אֶת בְּשָׂרוֹ וּבַעַל הַשּׁוֹר נָקִי` (שמות כא:כח)
`וְאִם שׁוֹר נַגָּח הוּא מִתְּמֹל שִׁלְשֹׁם וְהוּעַד בִּבְעָלָיו וְלֹא יִשְׁמְרֶנּוּ וְהֵמִית אִישׁ אוֹ אִשָּׁה הַשּׁוֹר יִסָּקֵל וְגַם בְּעָלָיו יוּמָת` (שמות כא:כט)
`וְכִי יְפַתֶּה אִישׁ בְּתוּלָה אֲשֶׁר לֹא אֹרָשָׂה וְשָׁכַב עִמָּהּ מָהֹר יִמְהָרֶנָּה לּוֹ לְאִשָּׁה` (שמות כב:טז)
`וְחָרָה אַפִּי וְהָרַגְתִּי אֶתְכֶם בֶּחָרֶב וְהָיוּ נְשֵׁיכֶם אַלְמָנוֹת וּבְנֵיכֶם יְתֹמִים` (שמות כב:כד)
`חֲמֵשׁ הַיְרִיעֹת תִּהְיֶיןָ חֹבְרֹת אִשָּׁה אֶל אֲחֹתָהּ וְחָמֵשׁ יְרִיעֹת חֹבְרֹת אִשָּׁה אֶל אֲחֹתָהּ` (שמות כו:ג)
`חֲמִשִּׁים לֻלָאֹת תַּעֲשֶׂה בַּיְרִיעָה הָאֶחָת וַחֲמִשִּׁים לֻלָאֹת תַּעֲשֶׂה בִּקְצֵה הַיְרִיעָה אֲשֶׁר בַּמַּחְבֶּרֶת הַשֵּׁנִית מַקְבִּילֹת הַלֻּלָאֹת אִשָּׁה אֶל אֲחֹתָהּ` (שמות כו:ה)
`וְעָשִׂיתָ חֲמִשִּׁים קַרְסֵי זָהָב וְחִבַּרְתָּ אֶת הַיְרִיעֹת אִשָּׁה אֶל אֲחֹתָהּ בַּקְּרָסִים וְהָיָה הַמִּשְׁכָּן אֶחָד` (שמות כו:ו)
`שְׁתֵּי יָדוֹת לַקֶּרֶשׁ הָאֶחָד מְשֻׁלָּבֹת אִשָּׁה אֶל אֲחֹתָהּ כֵּן תַּעֲשֶׂה לְכֹל קַרְשֵׁי הַמִּשְׁכָּן` (שמות כו:יז)
`וַיֹּאמֶר אֲלֵהֶם אַהֲרֹן פָּרְקוּ נִזְמֵי הַזָּהָב אֲשֶׁר בְּאָזְנֵי נְשֵׁיכֶם בְּנֵיכֶם וּבְנֹתֵיכֶם וְהָבִיאוּ אֵלָי` (שמות לב:ב)
`וַיָּבֹאוּ הָאֲנָשִׁים עַל הַנָּשִׁים כֹּל נְדִיב לֵב הֵבִיאוּ חָח וָנֶזֶם וְטַבַּעַת וְכוּמָז כָּל כְּלִי זָהָב וְכָל אִישׁ אֲשֶׁר הֵנִיף תְּנוּפַת זָהָב לַיהוָה` (שמות לה:כב)
`וְכָל אִשָּׁה חַכְמַת לֵב בְּיָדֶיהָ טָווּ וַיָּבִיאוּ מַטְוֶה אֶת הַתְּכֵלֶת וְאֶת הָאַרְגָּמָן אֶת תּוֹלַעַת הַשָּׁנִי וְאֶת הַשֵּׁשׁ` (שמות לה:כה)
`וְכָל הַנָּשִׁים אֲשֶׁר נָשָׂא לִבָּן אֹתָנָה בְּחָכְמָה טָווּ אֶת הָעִזִּים` (שמות לה:כו)
`כָּל אִישׁ וְאִשָּׁה אֲשֶׁר נָדַב לִבָּם אֹתָם לְהָבִיא לְכָל הַמְּלָאכָה אֲשֶׁר צִוָּה יְהוָה לַעֲשׂוֹת בְּיַד מֹשֶׁה הֵבִיאוּ בְנֵי יִשְׂרָאֵל נְדָבָה לַיהוָה` (שמות לה:כט)
`וַיְצַו מֹשֶׁה וַיַּעֲבִירוּ קוֹל בַּמַּחֲנֶה לֵאמֹר אִישׁ וְאִשָּׁה אַל יַעֲשׂוּ עוֹד מְלָאכָה לִתְרוּמַת הַקֹּדֶשׁ וַיִּכָּלֵא הָעָם מֵהָבִיא` (שמות לו:ו)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל לֵאמֹר אִשָּׁה כִּי תַזְרִיעַ וְיָלְדָה זָכָר וְטָמְאָה שִׁבְעַת יָמִים כִּימֵי נִדַּת דְּוֺתָהּ תִּטְמָא` (ויקרא יב:ב)
`וְאִישׁ אוֹ אִשָּׁה כִּי יִהְיֶה בוֹ נָגַע בְּרֹאשׁ אוֹ בְזָקָן` (ויקרא יג:כט)
`וְאִישׁ אוֹ אִשָּׁה כִּי יִהְיֶה בְעוֹר בְּשָׂרָם בֶּהָרֹת בֶּהָרֹת לְבָנֹת` (ויקרא יג:לח)
`וְאִשָּׁה אֲשֶׁר יִשְׁכַּב אִישׁ אֹתָהּ שִׁכְבַת זָרַע וְרָחֲצוּ בַמַּיִם וְטָמְאוּ עַד הָעָרֶב` (ויקרא טו:יח)
`וְאִשָּׁה כִּי תִהְיֶה זָבָה דָּם יִהְיֶה זֹבָהּ בִּבְשָׂרָהּ שִׁבְעַת יָמִים תִּהְיֶה בְנִדָּתָהּ וְכָל הַנֹּגֵעַ בָּהּ יִטְמָא עַד הָעָרֶב` (ויקרא טו:יט)
`וְאִשָּׁה כִּי יָזוּב זוֹב דָּמָהּ יָמִים רַבִּים בְּלֹא עֶת נִדָּתָהּ אוֹ כִי תָזוּב עַל נִדָּתָהּ כָּל יְמֵי זוֹב טֻמְאָתָהּ כִּימֵי נִדָּתָהּ תִּהְיֶה טְמֵאָה הִוא` (ויקרא טו:כה)
`עֶרְוַת אֵשֶׁת אָבִיךָ לֹא תְגַלֵּה עֶרְוַת אָבִיךָ הִוא` (ויקרא יח:ח)
`עֶרְוַת בַּת אֵשֶׁת אָבִיךָ מוֹלֶדֶת אָבִיךָ אֲחוֹתְךָ הִוא לֹא תְגַלֶּה עֶרְוָתָהּ` (ויקרא יח:יא)
`עֶרְוַת אֲחִי אָבִיךָ לֹא תְגַלֵּה אֶל אִשְׁתּוֹ לֹא תִקְרָב דֹּדָתְךָ הִוא` (ויקרא יח:יד)
`עֶרְוַת כַּלָּתְךָ לֹא תְגַלֵּה אֵשֶׁת בִּנְךָ הִוא לֹא תְגַלֶּה עֶרְוָתָהּ` (ויקרא יח:טו)
`עֶרְוַת אֵשֶׁת אָחִיךָ לֹא תְגַלֵּה עֶרְוַת אָחִיךָ הִוא` (ויקרא יח:טז)
`עֶרְוַת אִשָּׁה וּבִתָּהּ לֹא תְגַלֵּה אֶת בַּת בְּנָהּ וְאֶת בַּת בִּתָּהּ לֹא תִקַּח לְגַלּוֹת עֶרְוָתָהּ שַׁאֲרָה הֵנָּה זִמָּה הִוא` (ויקרא יח:יז)
`וְאִשָּׁה אֶל אֲחֹתָהּ לֹא תִקָּח לִצְרֹר לְגַלּוֹת עֶרְוָתָהּ עָלֶיהָ בְּחַיֶּיהָ` (ויקרא יח:יח)
`וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב לְגַלּוֹת עֶרְוָתָהּ` (ויקרא יח:יט)
`וְאֶל אֵשֶׁת עֲמִיתְךָ לֹא תִתֵּן שְׁכָבְתְּךָ לְזָרַע לְטָמְאָה בָהּ` (ויקרא יח:כ)
`וְאֶת זָכָר לֹא תִשְׁכַּב מִשְׁכְּבֵי אִשָּׁה תּוֹעֵבָה הִוא` (ויקרא יח:כב)
`וּבְכָל בְּהֵמָה לֹא תִתֵּן שְׁכָבְתְּךָ לְטָמְאָה בָהּ וְאִשָּׁה לֹא תַעֲמֹד לִפְנֵי בְהֵמָה לְרִבְעָהּ תֶּבֶל הוּא` (ויקרא יח:כג)
`וְאִישׁ כִּי יִשְׁכַּב אֶת אִשָּׁה שִׁכְבַת זֶרַע וְהִוא שִׁפְחָה נֶחֱרֶפֶת לְאִישׁ וְהָפְדֵּה לֹא נִפְדָּתָה אוֹ חֻפְשָׁה לֹא נִתַּן לָהּ בִּקֹּרֶת תִּהְיֶה לֹא יוּמְתוּ כִּי לֹא חֻפָּשָׁה` (ויקרא יט:כ)
`וְאִישׁ אֲשֶׁר יִנְאַף אֶת אֵשֶׁת אִישׁ אֲשֶׁר יִנְאַף אֶת אֵשֶׁת רֵעֵהוּ מוֹת יוּמַת הַנֹּאֵף וְהַנֹּאָפֶת` (ויקרא כ:י)
`וְאִישׁ אֲשֶׁר יִשְׁכַּב אֶת אֵשֶׁת אָבִיו עֶרְוַת אָבִיו גִּלָּה מוֹת יוּמְתוּ שְׁנֵיהֶם דְּמֵיהֶם בָּם` (ויקרא כ:יא)
`וְאִישׁ אֲשֶׁר יִשְׁכַּב אֶת זָכָר מִשְׁכְּבֵי אִשָּׁה תּוֹעֵבָה עָשׂוּ שְׁנֵיהֶם מוֹת יוּמָתוּ דְּמֵיהֶם בָּם` (ויקרא כ:יג)
`וְאִישׁ אֲשֶׁר יִקַּח אֶת אִשָּׁה וְאֶת אִמָּהּ זִמָּה הִוא בָּאֵשׁ יִשְׂרְפוּ אֹתוֹ וְאֶתְהֶן וְלֹא תִהְיֶה זִמָּה בְּתוֹכְכֶם` (ויקרא כ:יד)
`וְאִשָּׁה אֲשֶׁר תִּקְרַב אֶל כָּל בְּהֵמָה לְרִבְעָה אֹתָהּ וְהָרַגְתָּ אֶת הָאִשָּׁה וְאֶת הַבְּהֵמָה מוֹת יוּמָתוּ דְּמֵיהֶם בָּם` (ויקרא כ:טז)
`וְאִישׁ אֲשֶׁר יִשְׁכַּב אֶת אִשָּׁה דָּוָה וְגִלָּה אֶת עֶרְוָתָהּ אֶת מְקֹרָהּ הֶעֱרָה וְהִיא גִּלְּתָה אֶת מְקוֹר דָּמֶיהָ וְנִכְרְתוּ שְׁנֵיהֶם מִקֶּרֶב עַמָּם` (ויקרא כ:יח)
`וְאִישׁ אֲשֶׁר יִקַּח אֶת אֵשֶׁת אָחִיו נִדָּה הִוא עֶרְוַת אָחִיו גִּלָּה עֲרִירִים יִהְיוּ` (ויקרא כ:כא)
`וְאִישׁ אוֹ אִשָּׁה כִּי יִהְיֶה בָהֶם אוֹב אוֹ יִדְּעֹנִי מוֹת יוּמָתוּ בָּאֶבֶן יִרְגְּמוּ אֹתָם דְּמֵיהֶם בָּם` (ויקרא כ:כז)
`אִשָּׁה זֹנָה וַחֲלָלָה לֹא יִקָּחוּ וְאִשָּׁה גְּרוּשָׁה מֵאִישָׁהּ לֹא יִקָּחוּ כִּי קָדֹשׁ הוּא לֵאלֹהָיו` (ויקרא כא:ז)
`וְהוּא אִשָּׁה בִבְתוּלֶיהָ יִקָּח` (ויקרא כא:יג)
`אַלְמָנָה וּגְרוּשָׁה וַחֲלָלָה זֹנָה אֶת אֵלֶּה לֹא יִקָּח כִּי אִם בְּתוּלָה מֵעַמָּיו יִקַּח אִשָּׁה` (ויקרא כא:יד)
`וַיֵּצֵא בֶּן אִשָּׁה יִשְׂרְאֵלִית וְהוּא בֶּן אִישׁ מִצְרִי בְּתוֹךְ בְּנֵי יִשְׂרָאֵל וַיִּנָּצוּ בַּמַּחֲנֶה בֶּן הַיִּשְׂרְאֵלִית וְאִישׁ הַיִּשְׂרְאֵלִי` (ויקרא כד:י)
`וַיִּקֹּב בֶּן הָאִשָּׁה הַיִּשְׂרְאֵלִית אֶת הַשֵּׁם וַיְקַלֵּל וַיָּבִיאוּ אֹתוֹ אֶל מֹשֶׁה וְשֵׁם אִמּוֹ שְׁלֹמִית בַּת דִּבְרִי לְמַטֵּה דָן` (ויקרא כד:יא)
`בְּשִׁבְרִי לָכֶם מַטֵּה לֶחֶם וְאָפוּ עֶשֶׂר נָשִׁים לַחְמְכֶם בְּתַנּוּר אֶחָד וְהֵשִׁיבוּ לַחְמְכֶם בַּמִּשְׁקָל וַאֲכַלְתֶּם וְלֹא תִשְׂבָּעוּ` (ויקרא כו:כו)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל אִישׁ אוֹ אִשָּׁה כִּי יַעֲשׂוּ מִכָּל חַטֹּאת הָאָדָם לִמְעֹל מַעַל בַּיהוָה וְאָשְׁמָה הַנֶּפֶשׁ הַהִוא` (במדבר ה:ו)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אִישׁ אִישׁ כִּי תִשְׂטֶה אִשְׁתּוֹ וּמָעֲלָה בוֹ מָעַל` (במדבר ה:יב)
`וְעָבַר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת אִשְׁתּוֹ וְהִוא נִטְמָאָה אוֹ עָבַר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת אִשְׁתּוֹ וְהִיא לֹא נִטְמָאָה` (במדבר ה:יד)
`וְהֵבִיא הָאִישׁ אֶת אִשְׁתּוֹ אֶל הַכֹּהֵן וְהֵבִיא אֶת קָרְבָּנָהּ עָלֶיהָ עֲשִׂירִת הָאֵיפָה קֶמַח שְׂעֹרִים לֹא יִצֹק עָלָיו שֶׁמֶן וְלֹא יִתֵּן עָלָיו לְבֹנָה כִּי מִנְחַת קְנָאֹת הוּא מִנְחַת זִכָּרוֹן מַזְכֶּרֶת עָוֺן` (במדבר ה:טו)
`וְהֶעֱמִיד הַכֹּהֵן אֶת הָאִשָּׁה לִפְנֵי יְהוָה וּפָרַע אֶת רֹאשׁ הָאִשָּׁה וְנָתַן עַל כַּפֶּיהָ אֵת מִנְחַת הַזִּכָּרוֹן מִנְחַת קְנָאֹת הִוא וּבְיַד הַכֹּהֵן יִהְיוּ מֵי הַמָּרִים הַמְאָרֲרִים` (במדבר ה:יח)
`וְהִשְׁבִּיעַ אֹתָהּ הַכֹּהֵן וְאָמַר אֶל הָאִשָּׁה אִם לֹא שָׁכַב אִישׁ אֹתָךְ וְאִם לֹא שָׂטִית טֻמְאָה תַּחַת אִישֵׁךְ הִנָּקִי מִמֵּי הַמָּרִים הַמְאָרֲרִים הָאֵלֶּה` (במדבר ה:יט)
`וְהִשְׁבִּיעַ הַכֹּהֵן אֶת הָאִשָּׁה בִּשְׁבֻעַת הָאָלָה וְאָמַר הַכֹּהֵן לָאִשָּׁה יִתֵּן יְהוָה אוֹתָךְ לְאָלָה וְלִשְׁבֻעָה בְּתוֹךְ עַמֵּךְ בְּתֵת יְהוָה אֶת יְרֵכֵךְ נֹפֶלֶת וְאֶת בִּטְנֵךְ צָבָה` (במדבר ה:כא)
`וּבָאוּ הַמַּיִם הַמְאָרְרִים הָאֵלֶּה בְּמֵעַיִךְ לַצְבּוֹת בֶּטֶן וְלַנְפִּל יָרֵךְ וְאָמְרָה הָאִשָּׁה אָמֵן אָמֵן` (במדבר ה:כב)
`וְהִשְׁקָה אֶת הָאִשָּׁה אֶת מֵי הַמָּרִים הַמְאָרֲרִים וּבָאוּ בָהּ הַמַּיִם הַמְאָרֲרִים לְמָרִים` (במדבר ה:כד)
`וְלָקַח הַכֹּהֵן מִיַּד הָאִשָּׁה אֵת מִנְחַת הַקְּנָאֹת וְהֵנִיף אֶת הַמִּנְחָה לִפְנֵי יְהוָה וְהִקְרִיב אֹתָהּ אֶל הַמִּזְבֵּחַ` (במדבר ה:כה)
`וְקָמַץ הַכֹּהֵן מִן הַמִּנְחָה אֶת אַזְכָּרָתָהּ וְהִקְטִיר הַמִּזְבֵּחָה וְאַחַר יַשְׁקֶה אֶת הָאִשָּׁה אֶת הַמָּיִם` (במדבר ה:כו)
`וְהִשְׁקָהּ אֶת הַמַּיִם וְהָיְתָה אִם נִטְמְאָה וַתִּמְעֹל מַעַל בְּאִישָׁהּ וּבָאוּ בָהּ הַמַּיִם הַמְאָרֲרִים לְמָרִים וְצָבְתָה בִטְנָהּ וְנָפְלָה יְרֵכָהּ וְהָיְתָה הָאִשָּׁה לְאָלָה בְּקֶרֶב עַמָּהּ` (במדבר ה:כז)
`וְאִם לֹא נִטְמְאָה הָאִשָּׁה וּטְהֹרָה הִוא וְנִקְּתָה וְנִזְרְעָה זָרַע` (במדבר ה:כח)
`זֹאת תּוֹרַת הַקְּנָאֹת אֲשֶׁר תִּשְׂטֶה אִשָּׁה תַּחַת אִישָׁהּ וְנִטְמָאָה` (במדבר ה:כט)
`אוֹ אִישׁ אֲשֶׁר תַּעֲבֹר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת אִשְׁתּוֹ וְהֶעֱמִיד אֶת הָאִשָּׁה לִפְנֵי יְהוָה וְעָשָׂה לָהּ הַכֹּהֵן אֵת כָּל הַתּוֹרָה הַזֹּאת` (במדבר ה:ל)
`וְנִקָּה הָאִישׁ מֵעָוֺן וְהָאִשָּׁה הַהִוא תִּשָּׂא אֶת עֲוֺנָהּ` (במדבר ה:לא)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אִישׁ אוֹ אִשָּׁה כִּי יַפְלִא לִנְדֹּר נֶדֶר נָזִיר לְהַזִּיר לַיהוָה` (במדבר ו:ב)
`וַתְּדַבֵּר מִרְיָם וְאַהֲרֹן בְּמֹשֶׁה עַל אֹדוֹת הָאִשָּׁה הַכֻּשִׁית אֲשֶׁר לָקָח כִּי אִשָּׁה כֻשִׁית לָקָח` (במדבר יב:א)
`וְלָמָה יְהוָה מֵבִיא אֹתָנוּ אֶל הָאָרֶץ הַזֹּאת לִנְפֹּל בַּחֶרֶב נָשֵׁינוּ וְטַפֵּנוּ יִהְיוּ לָבַז הֲלוֹא טוֹב לָנוּ שׁוּב מִצְרָיְמָה` (במדבר יד:ג)
`וַיֵּעָלוּ מֵעַל מִשְׁכַּן קֹרֶח דָּתָן וַאֲבִירָם מִסָּבִיב וְדָתָן וַאֲבִירָם יָצְאוּ נִצָּבִים פֶּתַח אָהֳלֵיהֶם וּנְשֵׁיהֶם וּבְנֵיהֶם וְטַפָּם` (במדבר טז:כז)
`וַיָּבֹא אַחַר אִישׁ יִשְׂרָאֵל אֶל הַקֻּבָּה וַיִּדְקֹר אֶת שְׁנֵיהֶם אֵת אִישׁ יִשְׂרָאֵל וְאֶת הָאִשָּׁה אֶל קֳבָתָהּ וַתֵּעָצַר הַמַּגֵּפָה מֵעַל בְּנֵי יִשְׂרָאֵל` (במדבר כה:ח)
`וְשֵׁם הָאִשָּׁה הַמֻּכָּה הַמִּדְיָנִית כָּזְבִּי בַת צוּר רֹאשׁ אֻמּוֹת בֵּית אָב בְּמִדְיָן הוּא` (במדבר כה:טו)
`וְשֵׁם אֵשֶׁת עַמְרָם יוֹכֶבֶד בַּת לֵוִי אֲשֶׁר יָלְדָה אֹתָהּ לְלֵוִי בְּמִצְרָיִם וַתֵּלֶד לְעַמְרָם אֶת אַהֲרֹן וְאֶת מֹשֶׁה וְאֵת מִרְיָם אֲחֹתָם` (במדבר כו:נט)
`וְאִשָּׁה כִּי תִדֹּר נֶדֶר לַיהוָה וְאָסְרָה אִסָּר בְּבֵית אָבִיהָ בִּנְעֻרֶיהָ` (במדבר ל:ג)
`אֵלֶּה הַחֻקִּים אֲשֶׁר צִוָּה יְהוָה אֶת מֹשֶׁה בֵּין אִישׁ לְאִשְׁתּוֹ בֵּין אָב לְבִתּוֹ בִּנְעֻרֶיהָ בֵּית אָבִיהָ` (במדבר ל:טז)
`וַיִּשְׁבּוּ בְנֵי יִשְׂרָאֵל אֶת נְשֵׁי מִדְיָן וְאֶת טַפָּם וְאֵת כָּל בְּהֶמְתָּם וְאֶת כָּל מִקְנֵהֶם וְאֶת כָּל חֵילָם בָּזָזוּ` (במדבר לא:ט)
`וְעַתָּה הִרְגוּ כָל זָכָר בַּטָּף וְכָל אִשָּׁה יֹדַעַת אִישׁ לְמִשְׁכַּב זָכָר הֲרֹגוּ` (במדבר לא:יז)
`וְכֹל הַטַּף בַּנָּשִׁים אֲשֶׁר לֹא יָדְעוּ מִשְׁכַּב זָכָר הַחֲיוּ לָכֶם` (במדבר לא:יח)
`וְנֶפֶשׁ אָדָם מִן הַנָּשִׁים אֲשֶׁר לֹא יָדְעוּ מִשְׁכַּב זָכָר כָּל נֶפֶשׁ שְׁנַיִם וּשְׁלֹשִׁים אָלֶף` (במדבר לא:לה)
`טַפֵּנוּ נָשֵׁינוּ מִקְנֵנוּ וְכָל בְּהֶמְתֵּנוּ יִהְיוּ שָׁם בְּעָרֵי הַגִּלְעָד` (במדבר לב:כו)
`וְהָיוּ לְאֶחָד מִבְּנֵי שִׁבְטֵי בְנֵי יִשְׂרָאֵל לְנָשִׁים וְנִגְרְעָה נַחֲלָתָן מִנַּחֲלַת אֲבֹתֵינוּ וְנוֹסַף עַל נַחֲלַת הַמַּטֶּה אֲשֶׁר תִּהְיֶינָה לָהֶם וּמִגֹּרַל נַחֲלָתֵנוּ יִגָּרֵעַ` (במדבר לו:ג)
`זֶה הַדָּבָר אֲשֶׁר צִוָּה יְהוָה לִבְנוֹת צְלָפְחָד לֵאמֹר לַטּוֹב בְּעֵינֵיהֶם תִּהְיֶינָה לְנָשִׁים אַךְ לְמִשְׁפַּחַת מַטֵּה אֲבִיהֶם תִּהְיֶינָה לְנָשִׁים` (במדבר לו:ו)
`וְכָל בַּת יֹרֶשֶׁת נַחֲלָה מִמַּטּוֹת בְּנֵי יִשְׂרָאֵל לְאֶחָד מִמִּשְׁפַּחַת מַטֵּה אָבִיהָ תִּהְיֶה לְאִשָּׁה לְמַעַן יִירְשׁוּ בְּנֵי יִשְׂרָאֵל אִישׁ נַחֲלַת אֲבֹתָיו` (במדבר לו:ח)
`וַתִּהְיֶינָה מַחְלָה תִרְצָה וְחָגְלָה וּמִלְכָּה וְנֹעָה בְּנוֹת צְלָפְחָד לִבְנֵי דֹדֵיהֶן לְנָשִׁים` (במדבר לו:יא)
`מִמִּשְׁפְּחֹת בְּנֵי מְנַשֶּׁה בֶן יוֹסֵף הָיוּ לְנָשִׁים וַתְּהִי נַחֲלָתָן עַל מַטֵּה מִשְׁפַּחַת אֲבִיהֶן` (במדבר לו:יב)
`וַנִּלְכֹּד אֶת כָּל עָרָיו בָּעֵת הַהִוא וַנַּחֲרֵם אֶת כָּל עִיר מְתִם וְהַנָּשִׁים וְהַטָּף לֹא הִשְׁאַרְנוּ שָׂרִיד` (דברים ב:לד)
`וַנַּחֲרֵם אוֹתָם כַּאֲשֶׁר עָשִׂינוּ לְסִיחֹן מֶלֶךְ חֶשְׁבּוֹן הַחֲרֵם כָּל עִיר מְתִם הַנָּשִׁים וְהַטָּף` (דברים ג:ו)
`רַק נְשֵׁיכֶם וְטַפְּכֶם וּמִקְנֵכֶם יָדַעְתִּי כִּי מִקְנֶה רַב לָכֶם יֵשְׁבוּ בְּעָרֵיכֶם אֲשֶׁר נָתַתִּי לָכֶם` (דברים ג:יט)
`וְלֹא תַחְמֹד אֵשֶׁת רֵעֶךָ וְלֹא תִתְאַוֶּה בֵּית רֵעֶךָ שָׂדֵהוּ וְעַבְדּוֹ וַאֲמָתוֹ שׁוֹרוֹ וַחֲמֹרוֹ וְכֹל אֲשֶׁר לְרֵעֶךָ` (דברים ה:כא)
`כִּי יְסִיתְךָ אָחִיךָ בֶן אִמֶּךָ אוֹ בִנְךָ אוֹ בִתְּךָ אוֹ אֵשֶׁת חֵיקֶךָ אוֹ רֵעֲךָ אֲשֶׁר כְּנַפְשְׁךָ בַּסֵּתֶר לֵאמֹר נֵלְכָה וְנַעַבְדָה אֱלֹהִים אֲחֵרִים אֲשֶׁר לֹא יָדַעְתָּ אַתָּה וַאֲבֹתֶיךָ` (דברים יג:ו)
`כִּי יִמָּצֵא בְקִרְבְּךָ בְּאַחַד שְׁעָרֶיךָ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ אִישׁ אוֹ אִשָּׁה אֲשֶׁר יַעֲשֶׂה אֶת הָרַע בְּעֵינֵי יְהוָה אֱלֹהֶיךָ לַעֲבֹר בְּרִיתוֹ` (דברים יז:ב)
`וְהוֹצֵאתָ אֶת הָאִישׁ הַהוּא אוֹ אֶת הָאִשָּׁה הַהִוא אֲשֶׁר עָשׂוּ אֶת הַדָּבָר הָרָע הַזֶּה אֶל שְׁעָרֶיךָ אֶת הָאִישׁ אוֹ אֶת הָאִשָּׁה וּסְקַלְתָּם בָּאֲבָנִים וָמֵתוּ` (דברים יז:ה)
`וְלֹא יַרְבֶּה לּוֹ נָשִׁים וְלֹא יָסוּר לְבָבוֹ וְכֶסֶף וְזָהָב לֹא יַרְבֶּה לּוֹ מְאֹד` (דברים יז:יז)
`וּמִי הָאִישׁ אֲשֶׁר אֵרַשׂ אִשָּׁה וְלֹא לְקָחָהּ יֵלֵךְ וְיָשֹׁב לְבֵיתוֹ פֶּן יָמוּת בַּמִּלְחָמָה וְאִישׁ אַחֵר יִקָּחֶנָּה` (דברים כ:ז)
`רַק הַנָּשִׁים וְהַטַּף וְהַבְּהֵמָה וְכֹל אֲשֶׁר יִהְיֶה בָעִיר כָּל שְׁלָלָהּ תָּבֹז לָךְ וְאָכַלְתָּ אֶת שְׁלַל אֹיְבֶיךָ אֲשֶׁר נָתַן יְהוָה אֱלֹהֶיךָ לָךְ` (דברים כ:יד)
`וְרָאִיתָ בַּשִּׁבְיָה אֵשֶׁת יְפַת תֹּאַר וְחָשַׁקְתָּ בָהּ וְלָקַחְתָּ לְךָ לְאִשָּׁה` (דברים כא:יא)
`וְהֵסִירָה אֶת שִׂמְלַת שִׁבְיָהּ מֵעָלֶיהָ וְיָשְׁבָה בְּבֵיתֶךָ וּבָכְתָה אֶת אָבִיהָ וְאֶת אִמָּהּ יֶרַח יָמִים וְאַחַר כֵּן תָּבוֹא אֵלֶיהָ וּבְעַלְתָּהּ וְהָיְתָה לְךָ לְאִשָּׁה` (דברים כא:יג)
`כִּי תִהְיֶיןָ לְאִישׁ שְׁתֵּי נָשִׁים הָאַחַת אֲהוּבָה וְהָאַחַת שְׂנוּאָה וְיָלְדוּ לוֹ בָנִים הָאֲהוּבָה וְהַשְּׂנוּאָה וְהָיָה הַבֵּן הַבְּכוֹר לַשְּׂנִיאָה` (דברים כא:טו)
`לֹא יִהְיֶה כְלִי גֶבֶר עַל אִשָּׁה וְלֹא יִלְבַּשׁ גֶּבֶר שִׂמְלַת אִשָּׁה כִּי תוֹעֲבַת יְהוָה אֱלֹהֶיךָ כָּל עֹשֵׂה אֵלֶּה` (דברים כב:ה)
`כִּי יִקַּח אִישׁ אִשָּׁה וּבָא אֵלֶיהָ וּשְׂנֵאָהּ` (דברים כב:יג)
`וְשָׂם לָהּ עֲלִילֹת דְּבָרִים וְהוֹצִיא עָלֶיהָ שֵׁם רָע וְאָמַר אֶת הָאִשָּׁה הַזֹּאת לָקַחְתִּי וָאֶקְרַב אֵלֶיהָ וְלֹא מָצָאתִי לָהּ בְּתוּלִים` (דברים כב:יד)
`וְאָמַר אֲבִי הַנַּעַר אֶל הַזְּקֵנִים אֶת בִּתִּי נָתַתִּי לָאִישׁ הַזֶּה לְאִשָּׁה וַיִּשְׂנָאֶהָ` (דברים כב:טז)
`וְעָנְשׁוּ אֹתוֹ מֵאָה כֶסֶף וְנָתְנוּ לַאֲבִי הַנַּעֲרָה כִּי הוֹצִיא שֵׁם רָע עַל בְּתוּלַת יִשְׂרָאֵל וְלוֹ תִהְיֶה לְאִשָּׁה לֹא יוּכַל לְשַּׁלְּחָהּ כָּל יָמָיו` (דברים כב:יט)
`כִּי יִמָּצֵא אִישׁ שֹׁכֵב עִם אִשָּׁה בְעֻלַת בַּעַל וּמֵתוּ גַּם שְׁנֵיהֶם הָאִישׁ הַשֹּׁכֵב עִם הָאִשָּׁה וְהָאִשָּׁה וּבִעַרְתָּ הָרָע מִיִּשְׂרָאֵל` (דברים כב:כב)
`וְהוֹצֵאתֶם אֶת שְׁנֵיהֶם אֶל שַׁעַר הָעִיר הַהִוא וּסְקַלְתֶּם אֹתָם בָּאֲבָנִים וָמֵתוּ אֶת הַנַּעַר עַל דְּבַר אֲשֶׁר לֹא צָעֲקָה בָעִיר וְאֶת הָאִישׁ עַל דְּבַר אֲשֶׁר עִנָּה אֶת אֵשֶׁת רֵעֵהוּ וּבִעַרְתָּ הָרָע מִקִּרְבֶּךָ` (דברים כב:כד)
`וְנָתַן הָאִישׁ הַשֹּׁכֵב עִמָּהּ לַאֲבִי הַנַּעַר חֲמִשִּׁים כָּסֶף וְלוֹ תִהְיֶה לְאִשָּׁה תַּחַת אֲשֶׁר עִנָּהּ לֹא יוּכַל שַׁלְּחָה כָּל יָמָיו` (דברים כב:כט)
`לֹא יִקַּח אִישׁ אֶת אֵשֶׁת אָבִיו וְלֹא יְגַלֶּה כְּנַף אָבִיו` (דברים כב:ל)
`כִּי יִקַּח אִישׁ אִשָּׁה וּבְעָלָהּ וְהָיָה אִם לֹא תִמְצָא חֵן בְּעֵינָיו כִּי מָצָא בָהּ עֶרְוַת דָּבָר וְכָתַב לָהּ סֵפֶר כְּרִיתֻת וְנָתַן בְּיָדָהּ וְשִׁלְּחָהּ מִבֵּיתוֹ` (דברים כד:א)
`וּשְׂנֵאָהּ הָאִישׁ הָאַחֲרוֹן וְכָתַב לָהּ סֵפֶר כְּרִיתֻת וְנָתַן בְּיָדָהּ וְשִׁלְּחָהּ מִבֵּיתוֹ אוֹ כִי יָמוּת הָאִישׁ הָאַחֲרוֹן אֲשֶׁר לְקָחָהּ לוֹ לְאִשָּׁה` (דברים כד:ג)
`לֹא יוּכַל בַּעְלָהּ הָרִאשׁוֹן אֲשֶׁר שִׁלְּחָהּ לָשׁוּב לְקַחְתָּהּ לִהְיוֹת לוֹ לְאִשָּׁה אַחֲרֵי אֲשֶׁר הֻטַּמָּאָה כִּי תוֹעֵבָה הִוא לִפְנֵי יְהוָה וְלֹא תַחֲטִיא אֶת הָאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ נַחֲלָה` (דברים כד:ד)
`כִּי יִקַּח אִישׁ אִשָּׁה חֲדָשָׁה לֹא יֵצֵא בַּצָּבָא וְלֹא יַעֲבֹר עָלָיו לְכָל דָּבָר נָקִי יִהְיֶה לְבֵיתוֹ שָׁנָה אֶחָת וְשִׂמַּח אֶת אִשְׁתּוֹ אֲשֶׁר לָקָח` (דברים כד:ה)
`כִּי יֵשְׁבוּ אַחִים יַחְדָּו וּמֵת אַחַד מֵהֶם וּבֵן אֵין לוֹ לֹא תִהְיֶה אֵשֶׁת הַמֵּת הַחוּצָה לְאִישׁ זָר יְבָמָהּ יָבֹא עָלֶיהָ וּלְקָחָהּ לוֹ לְאִשָּׁה וְיִבְּמָהּ` (דברים כה:ה)
`כִּי יִנָּצוּ אֲנָשִׁים יַחְדָּו אִישׁ וְאָחִיו וְקָרְבָה אֵשֶׁת הָאֶחָד לְהַצִּיל אֶת אִישָׁהּ מִיַּד מַכֵּהוּ וְשָׁלְחָה יָדָהּ וְהֶחֱזִיקָה בִּמְבֻשָׁיו` (דברים כה:יא)
`אָרוּר שֹׁכֵב עִם אֵשֶׁת אָבִיו כִּי גִלָּה כְּנַף אָבִיו וְאָמַר כָּל הָעָם אָמֵן` (דברים כז:כ)
`אִשָּׁה תְאָרֵשׂ וְאִישׁ אַחֵר יִשְׁגָּלֶנָּה בַּיִת תִּבְנֶה וְלֹא תֵשֵׁב בּוֹ כֶּרֶם תִּטַּע וְלֹא תְחַלְּלֶּנּוּ` (דברים כח:ל)
`הָאִישׁ הָרַךְ בְּךָ וְהֶעָנֹג מְאֹד תֵּרַע עֵינוֹ בְאָחִיו וּבְאֵשֶׁת חֵיקוֹ וּבְיֶתֶר בָּנָיו אֲשֶׁר יוֹתִיר` (דברים כח:נד)
`טַפְּכֶם נְשֵׁיכֶם וְגֵרְךָ אֲשֶׁר בְּקֶרֶב מַחֲנֶיךָ מֵחֹטֵב עֵצֶיךָ עַד שֹׁאֵב מֵימֶיךָ` (דברים כט:יא)
`פֶּן יֵשׁ בָּכֶם אִישׁ אוֹ אִשָּׁה אוֹ מִשְׁפָּחָה אוֹ שֵׁבֶט אֲשֶׁר לְבָבוֹ פֹנֶה הַיּוֹם מֵעִם יְהוָה אֱלֹהֵינוּ לָלֶכֶת לַעֲבֹד אֶת אֱלֹהֵי הַגּוֹיִם הָהֵם פֶּן יֵשׁ בָּכֶם שֹׁרֶשׁ פֹּרֶה רֹאשׁ וְלַעֲנָה` (דברים כט:יח)
`הַקְהֵל אֶת הָעָם הָאֲנָשִׁים וְהַנָּשִׁים וְהַטַּף וְגֵרְךָ אֲשֶׁר בִּשְׁעָרֶיךָ לְמַעַן יִשְׁמְעוּ וּלְמַעַן יִלְמְדוּ וְיָרְאוּ אֶת יְהוָה אֱלֹהֵיכֶם וְשָׁמְרוּ לַעֲשׂוֹת אֶת כָּל דִּבְרֵי הַתּוֹרָה הַזֹּאת` (דברים לא:יב)
`נְשֵׁיכֶם טַפְּכֶם וּמִקְנֵיכֶם יֵשְׁבוּ בָּאָרֶץ אֲשֶׁר נָתַן לָכֶם מֹשֶׁה בְּעֵבֶר הַיַּרְדֵּן וְאַתֶּם תַּעַבְרוּ חֲמֻשִׁים לִפְנֵי אֲחֵיכֶם כֹּל גִּבּוֹרֵי הַחַיִל וַעֲזַרְתֶּם אוֹתָם` (יהושע א:יד)
`וַיִּשְׁלַח יְהוֹשֻׁעַ בִּן נוּן מִן הַשִּׁטִּים שְׁנַיִם אֲנָשִׁים מְרַגְּלִים חֶרֶשׁ לֵאמֹר לְכוּ רְאוּ אֶת הָאָרֶץ וְאֶת יְרִיחוֹ וַיֵּלְכוּ וַיָּבֹאוּ בֵּית אִשָּׁה זוֹנָה וּשְׁמָהּ רָחָב וַיִּשְׁכְּבוּ שָׁמָּה` (יהושע ב:א)
`וַתִּקַּח הָאִשָּׁה אֶת שְׁנֵי הָאֲנָשִׁים וַתִּצְפְּנוֹ וַתֹּאמֶר כֵּן בָּאוּ אֵלַי הָאֲנָשִׁים וְלֹא יָדַעְתִּי מֵאַיִן הֵמָּה` (יהושע ב:ד)
`וַיַּחֲרִימוּ אֶת כָּל אֲשֶׁר בָּעִיר מֵאִישׁ וְעַד אִשָּׁה מִנַּעַר וְעַד זָקֵן וְעַד שׁוֹר וָשֶׂה וַחֲמוֹר לְפִי חָרֶב` (יהושע ו:כא)
`וְלִשְׁנַיִם הָאֲנָשִׁים הַמְרַגְּלִים אֶת הָאָרֶץ אָמַר יְהוֹשֻׁעַ בֹּאוּ בֵּית הָאִשָּׁה הַזּוֹנָה וְהוֹצִיאוּ מִשָּׁם אֶת הָאִשָּׁה וְאֶת כָּל אֲשֶׁר לָהּ כַּאֲשֶׁר נִשְׁבַּעְתֶּם לָהּ` (יהושע ו:כב)
`וַיְהִי כָל הַנֹּפְלִים בַּיּוֹם הַהוּא מֵאִישׁ וְעַד אִשָּׁה שְׁנֵים עָשָׂר אָלֶף כֹּל אַנְשֵׁי הָעָי` (יהושע ח:כה)
`לֹא הָיָה דָבָר מִכֹּל אֲשֶׁר צִוָּה מֹשֶׁה אֲשֶׁר לֹא קָרָא יְהוֹשֻׁעַ נֶגֶד כָּל קְהַל יִשְׂרָאֵל וְהַנָּשִׁים וְהַטַּף וְהַגֵּר הַהֹלֵךְ בְּקִרְבָּם` (יהושע ח:לה)
`וַיֹּאמֶר כָּלֵב אֲשֶׁר יַכֶּה אֶת קִרְיַת סֵפֶר וּלְכָדָהּ וְנָתַתִּי לוֹ אֶת עַכְסָה בִתִּי לְאִשָּׁה` (יהושע טו:טז)
`וַיִּלְכְּדָהּ עָתְנִיאֵל בֶּן קְנַז אֲחִי כָלֵב וַיִּתֶּן לוֹ אֶת עַכְסָה בִתּוֹ לְאִשָּׁה` (יהושע טו:יז)
`וַיֹּאמֶר כָּלֵב אֲשֶׁר יַכֶּה אֶת קִרְיַת סֵפֶר וּלְכָדָהּ וְנָתַתִּי לוֹ אֶת עַכְסָה בִתִּי לְאִשָּׁה` (שופטים א:יב)
`וַיִּלְכְּדָהּ עָתְנִיאֵל בֶּן קְנַז אֲחִי כָלֵב הַקָּטֹן מִמֶּנּוּ וַיִּתֶּן לוֹ אֶת עַכְסָה בִתּוֹ לְאִשָּׁה` (שופטים א:יג)
`וַיִּקְחוּ אֶת בְּנוֹתֵיהֶם לָהֶם לְנָשִׁים וְאֶת בְּנוֹתֵיהֶם נָתְנוּ לִבְנֵיהֶם וַיַּעַבְדוּ אֶת אֱלֹהֵיהֶם` (שופטים ג:ו)
`וּדְבוֹרָה אִשָּׁה נְבִיאָה אֵשֶׁת לַפִּידוֹת הִיא שֹׁפְטָה אֶת יִשְׂרָאֵל בָּעֵת הַהִיא` (שופטים ד:ד)
`וַתֹּאמֶר הָלֹךְ אֵלֵךְ עִמָּךְ אֶפֶס כִּי לֹא תִהְיֶה תִּפְאַרְתְּךָ עַל הַדֶּרֶךְ אֲשֶׁר אַתָּה הוֹלֵךְ כִּי בְיַד אִשָּׁה יִמְכֹּר יְהוָה אֶת סִיסְרָא וַתָּקָם דְּבוֹרָה וַתֵּלֶך עִם בָּרָק קֶדְשָׁה` (שופטים ד:ט)
`וְסִיסְרָא נָס בְּרַגְלָיו אֶל אֹהֶל יָעֵל אֵשֶּׁת חֶבֶר הַקֵּינִי כִּי שָׁלוֹם בֵּין יָבִין מֶלֶךְ חָצוֹר וּבֵין בֵּית חֶבֶר הַקֵּינִי` (שופטים ד:יז)
`וַתִּקַּח יָעֵל אֵשֶׁת חֶבֶר אֶת יְתַד הָאֹהֶל וַתָּשֶׂם אֶת הַמַּקֶּבֶת בְּיָדָהּ וַתָּבוֹא אֵלָיו בַּלָּאט וַתִּתְקַע אֶת הַיָּתֵד בְּרַקָּתוֹ וַתִּצְנַח בָּאָרֶץ וְהוּא נִרְדָּם וַיָּעַף וַיָּמֹת` (שופטים ד:כא)
`תְּבֹרַךְ מִנָּשִׁים יָעֵל אֵשֶׁת חֶבֶר הַקֵּינִי מִנָּשִׁים בָּאֹהֶל תְּבֹרָךְ` (שופטים ה:כד)
`וּלְגִדְעוֹן הָיוּ שִׁבְעִים בָּנִים יֹצְאֵי יְרֵכוֹ כִּי נָשִׁים רַבּוֹת הָיוּ לוֹ` (שופטים ח:ל)
`וַיִּכְרְתוּ גַם כָּל הָעָם אִישׁ שׂוֹכֹה וַיֵּלְכוּ אַחֲרֵי אֲבִימֶלֶךְ וַיָּשִׂימוּ עַל הַצְּרִיחַ וַיַּצִּיתוּ עֲלֵיהֶם אֶת הַצְּרִיחַ בָּאֵשׁ וַיָּמֻתוּ גַּם כָּל אַנְשֵׁי מִגְדַּל שְׁכֶם כְּאֶלֶף אִישׁ וְאִשָּׁה` (שופטים ט:מט)
`וּמִגְדַּל עֹז הָיָה בְתוֹךְ הָעִיר וַיָּנֻסוּ שָׁמָּה כָּל הָאֲנָשִׁים וְהַנָּשִׁים וְכֹל בַּעֲלֵי הָעִיר וַיִּסְגְּרוּ בַּעֲדָם וַיַּעֲלוּ עַל גַּג הַמִּגְדָּל` (שופטים ט:נא)
`וַתַּשְׁלֵךְ אִשָּׁה אַחַת פֶּלַח רֶכֶב עַל רֹאשׁ אֲבִימֶלֶךְ וַתָּרִץ אֶת גֻּלְגָּלְתּוֹ` (שופטים ט:נג)
`וַיִּקְרָא מְהֵרָה אֶל הַנַּעַר נֹשֵׂא כֵלָיו וַיֹּאמֶר לוֹ שְׁלֹף חַרְבְּךָ וּמוֹתְתֵנִי פֶּן יֹאמְרוּ לִי אִשָּׁה הֲרָגָתְהוּ וַיִּדְקְרֵהוּ נַעֲרוֹ וַיָּמֹת` (שופטים ט:נד)
`וְיִפְתָּח הַגִּלְעָדִי הָיָה גִּבּוֹר חַיִל וְהוּא בֶּן אִשָּׁה זוֹנָה וַיּוֹלֶד גִּלְעָד אֶת יִפְתָּח` (שופטים יא:א)
`וַתֵּלֶד אֵשֶׁת גִּלְעָד לוֹ בָּנִים וַיִּגְדְּלוּ בְנֵי הָאִשָּׁה וַיְגָרְשׁוּ אֶת יִפְתָּח וַיֹּאמְרוּ לוֹ לֹא תִנְחַל בְּבֵית אָבִינוּ כִּי בֶּן אִשָּׁה אַחֶרֶת אָתָּה` (שופטים יא:ב)
`וַיְהִי אִישׁ אֶחָד מִצָּרְעָה מִמִּשְׁפַּחַת הַדָּנִי וּשְׁמוֹ מָנוֹחַ וְאִשְׁתּוֹ עֲקָרָה וְלֹא יָלָדָה` (שופטים יג:ב)
`וַיֵּרָא מַלְאַךְ יְהוָה אֶל הָאִשָּׁה וַיֹּאמֶר אֵלֶיהָ הִנֵּה נָא אַתְּ עֲקָרָה וְלֹא יָלַדְתְּ וְהָרִית וְיָלַדְתְּ בֵּן` (שופטים יג:ג)
`וַתָּבֹא הָאִשָּׁה וַתֹּאמֶר לְאִישָׁהּ לֵאמֹר אִישׁ הָאֱלֹהִים בָּא אֵלַי וּמַרְאֵהוּ כְּמַרְאֵה מַלְאַךְ הָאֱלֹהִים נוֹרָא מְאֹד וְלֹא שְׁאִלְתִּיהוּ אֵי מִזֶּה הוּא וְאֶת שְׁמוֹ לֹא הִגִּיד לִי` (שופטים יג:ו)
`וַיִּשְׁמַע הָאֱלֹהִים בְּקוֹל מָנוֹחַ וַיָּבֹא מַלְאַךְ הָאֱלֹהִים עוֹד אֶל הָאִשָּׁה וְהִיא יוֹשֶׁבֶת בַּשָּׂדֶה וּמָנוֹחַ אִישָׁהּ אֵין עִמָּהּ` (שופטים יג:ט)
`וַתְּמַהֵר הָאִשָּׁה וַתָּרָץ וַתַּגֵּד לְאִישָׁהּ וַתֹּאמֶר אֵלָיו הִנֵּה נִרְאָה אֵלַי הָאִישׁ אֲשֶׁר בָּא בַיּוֹם אֵלָי` (שופטים יג:י)
`וַיָּקָם וַיֵּלֶךְ מָנוֹחַ אַחֲרֵי אִשְׁתּוֹ וַיָּבֹא אֶל הָאִישׁ וַיֹּאמֶר לוֹ הַאַתָּה הָאִישׁ אֲשֶׁר דִּבַּרְתָּ אֶל הָאִשָּׁה וַיֹּאמֶר אָנִי` (שופטים יג:יא)
`וַיֹּאמֶר מַלְאַךְ יְהוָה אֶל מָנוֹחַ מִכֹּל אֲשֶׁר אָמַרְתִּי אֶל הָאִשָּׁה תִּשָּׁמֵר` (שופטים יג:יג)
`וַיִּקַּח מָנוֹחַ אֶת גְּדִי הָעִזִּים וְאֶת הַמִּנְחָה וַיַּעַל עַל הַצּוּר לַיהוָה וּמַפְלִא לַעֲשׂוֹת וּמָנוֹחַ וְאִשְׁתּוֹ רֹאִים` (שופטים יג:יט)
`וַיְהִי בַעֲלוֹת הַלַּהַב מֵעַל הַמִּזְבֵּחַ הַשָּׁמַיְמָה וַיַּעַל מַלְאַךְ יְהוָה בְּלַהַב הַמִּזְבֵּחַ וּמָנוֹחַ וְאִשְׁתּוֹ רֹאִים וַיִּפְּלוּ עַל פְּנֵיהֶם אָרְצָה` (שופטים יג:כ)
`וְלֹא יָסַף עוֹד מַלְאַךְ יְהוָה לְהֵרָאֹה אֶל מָנוֹחַ וְאֶל אִשְׁתּוֹ אָז יָדַע מָנוֹחַ כִּי מַלְאַךְ יְהוָה הוּא` (שופטים יג:כא)
`וַיֹּאמֶר מָנוֹחַ אֶל אִשְׁתּוֹ מוֹת נָמוּת כִּי אֱלֹהִים רָאִינוּ` (שופטים יג:כב)
`וַתֹּאמֶר לוֹ אִשְׁתּוֹ לוּ חָפֵץ יְהוָה לַהֲמִיתֵנוּ לֹא לָקַח מִיָּדֵנוּ עֹלָה וּמִנְחָה וְלֹא הֶרְאָנוּ אֶת כָּל אֵלֶּה וְכָעֵת לֹא הִשְׁמִיעָנוּ כָּזֹאת` (שופטים יג:כג)
`וַתֵּלֶד הָאִשָּׁה בֵּן וַתִּקְרָא אֶת שְׁמוֹ שִׁמְשׁוֹן וַיִּגְדַּל הַנַּעַר וַיְבָרְכֵהוּ יְהוָה` (שופטים יג:כד)
`וַיֵּרֶד שִׁמְשׁוֹן תִּמְנָתָה וַיַּרְא אִשָּׁה בְּתִמְנָתָה מִבְּנוֹת פְּלִשְׁתִּים` (שופטים יד:א)
`וַיַּעַל וַיַּגֵּד לְאָבִיו וּלְאִמּוֹ וַיֹּאמֶר אִשָּׁה רָאִיתִי בְתִמְנָתָה מִבְּנוֹת פְּלִשְׁתִּים וְעַתָּה קְחוּ אוֹתָהּ לִּי לְאִשָּׁה` (שופטים יד:ב)
`וַיֹּאמֶר לוֹ אָבִיו וְאִמּוֹ הַאֵין בִּבְנוֹת אַחֶיךָ וּבְכָל עַמִּי אִשָּׁה כִּי אַתָּה הוֹלֵךְ לָקַחַת אִשָּׁה מִפְּלִשְׁתִּים הָעֲרֵלִים וַיֹּאמֶר שִׁמְשׁוֹן אֶל אָבִיו אוֹתָהּ קַח לִי כִּי הִיא יָשְׁרָה בְעֵינָי` (שופטים יד:ג)
`וַיֵּרֶד וַיְדַבֵּר לָאִשָּׁה וַתִּישַׁר בְּעֵינֵי שִׁמְשׁוֹן` (שופטים יד:ז)
`וַיֵּרֶד אָבִיהוּ אֶל הָאִשָּׁה וַיַּעַשׂ שָׁם שִׁמְשׁוֹן מִשְׁתֶּה כִּי כֵּן יַעֲשׂוּ הַבַּחוּרִים` (שופטים יד:י)
`וַיְהִי בַּיּוֹם הַשְּׁבִיעִי וַיֹּאמְרוּ לְאֵשֶׁת שִׁמְשׁוֹן פַּתִּי אֶת אִישֵׁךְ וְיַגֶּד לָנוּ אֶת הַחִידָה פֶּן נִשְׂרֹף אוֹתָךְ וְאֶת בֵּית אָבִיךְ בָּאֵשׁ הַלְיָרְשֵׁנוּ קְרָאתֶם לָנוּ הֲלֹא` (שופטים יד:טו)
`וַתֵּבְךְּ אֵשֶׁת שִׁמְשׁוֹן עָלָיו וַתֹּאמֶר רַק שְׂנֵאתַנִי וְלֹא אֲהַבְתָּנִי הַחִידָה חַדְתָּ לִבְנֵי עַמִּי וְלִי לֹא הִגַּדְתָּה וַיֹּאמֶר לָהּ הִנֵּה לְאָבִי וּלְאִמִּי לֹא הִגַּדְתִּי וְלָךְ אַגִּיד` (שופטים יד:טז)
`וַתְּהִי אֵשֶׁת שִׁמְשׁוֹן לְמֵרֵעֵהוּ אֲשֶׁר רֵעָה לוֹ` (שופטים יד:כ)
`וַיְהִי מִיָּמִים בִּימֵי קְצִיר חִטִּים וַיִּפְקֹד שִׁמְשׁוֹן אֶת אִשְׁתּוֹ בִּגְדִי עִזִּים וַיֹּאמֶר אָבֹאָה אֶל אִשְׁתִּי הֶחָדְרָה וְלֹא נְתָנוֹ אָבִיהָ לָבוֹא` (שופטים טו:א)
`וַיֹּאמְרוּ פְלִשְׁתִּים מִי עָשָׂה זֹאת וַיֹּאמְרוּ שִׁמְשׁוֹן חֲתַן הַתִּמְנִי כִּי לָקַח אֶת אִשְׁתּוֹ וַיִּתְּנָהּ לְמֵרֵעֵהוּ וַיַּעֲלוּ פְלִשְׁתִּים וַיִּשְׂרְפוּ אוֹתָהּ וְאֶת אָבִיהָ בָּאֵשׁ` (שופטים טו:ו)
`וַיֵּלֶךְ שִׁמְשׁוֹן עַזָּתָה וַיַּרְא שָׁם אִשָּׁה זוֹנָה וַיָּבֹא אֵלֶיהָ` (שופטים טז:א)
`וַיְהִי אַחֲרֵי כֵן וַיֶּאֱהַב אִשָּׁה בְּנַחַל שֹׂרֵק וּשְׁמָהּ דְּלִילָה` (שופטים טז:ד)
`וְהַבַּיִת מָלֵא הָאֲנָשִׁים וְהַנָּשִׁים וְשָׁמָּה כֹּל סַרְנֵי פְלִשְׁתִּים וְעַל הַגָּג כִּשְׁלֹשֶׁת אֲלָפִים אִישׁ וְאִשָּׁה הָרֹאִים בִּשְׂחוֹק שִׁמְשׁוֹן` (שופטים טז:כז)
`וַיְהִי בַּיָּמִים הָהֵם וּמֶלֶךְ אֵין בְּיִשְׂרָאֵל וַיְהִי אִישׁ לֵוִי גָּר בְּיַרְכְּתֵי הַר אֶפְרַיִם וַיִּקַּח לוֹ אִשָּׁה פִילֶגֶשׁ מִבֵּית לֶחֶם יְהוּדָה` (שופטים יט:א)
`וַתָּבֹא הָאִשָּׁה לִפְנוֹת הַבֹּקֶר וַתִּפֹּל פֶּתַח בֵּית הָאִישׁ אֲשֶׁר אֲדוֹנֶיהָ שָּׁם עַד הָאוֹר` (שופטים יט:כו)
`וַיָּקָם אֲדֹנֶיהָ בַּבֹּקֶר וַיִּפְתַּח דַּלְתוֹת הַבַּיִת וַיֵּצֵא לָלֶכֶת לְדַרְכּוֹ וְהִנֵּה הָאִשָּׁה פִילַגְשׁוֹ נֹפֶלֶת פֶּתַח הַבַּיִת וְיָדֶיהָ עַל הַסַּף` (שופטים יט:כז)
`וַיַּעַן הָאִישׁ הַלֵּוִי אִישׁ הָאִשָּׁה הַנִּרְצָחָה וַיֹּאמַר הַגִּבְעָתָה אֲשֶׁר לְבִנְיָמִן בָּאתִי אֲנִי וּפִילַגְשִׁי לָלוּן` (שופטים כ:ד)
`וְאִישׁ יִשְׂרָאֵל נִשְׁבַּע בַּמִּצְפָּה לֵאמֹר אִישׁ מִמֶּנּוּ לֹא יִתֵּן בִּתּוֹ לְבִנְיָמִן לְאִשָּׁה` (שופטים כא:א)
`מַה נַּעֲשֶׂה לָהֶם לַנּוֹתָרִים לְנָשִׁים וַאֲנַחְנוּ נִשְׁבַּעְנוּ בַיהוָה לְבִלְתִּי תֵּת לָהֶם מִבְּנוֹתֵינוּ לְנָשִׁים` (שופטים כא:ז)
`וַיִּשְׁלְחוּ שָׁם הָעֵדָה שְׁנֵים עָשָׂר אֶלֶף אִישׁ מִבְּנֵי הֶחָיִל וַיְצַוּוּ אוֹתָם לֵאמֹר לְכוּ וְהִכִּיתֶם אֶת יוֹשְׁבֵי יָבֵשׁ גִּלְעָד לְפִי חֶרֶב וְהַנָּשִׁים וְהַטָּף` (שופטים כא:י)
`וְזֶה הַדָּבָר אֲשֶׁר תַּעֲשׂוּ כָּל זָכָר וְכָל אִשָּׁה יֹדַעַת מִשְׁכַּב זָכָר תַּחֲרִימוּ` (שופטים כא:יא)
`וַיָּשָׁב בִּנְיָמִן בָּעֵת הַהִיא וַיִּתְּנוּ לָהֶם הַנָּשִׁים אֲשֶׁר חִיּוּ מִנְּשֵׁי יָבֵשׁ גִּלְעָד וְלֹא מָצְאוּ לָהֶם כֵּן` (שופטים כא:יד)
`וַיֹּאמְרוּ זִקְנֵי הָעֵדָה מַה נַּעֲשֶׂה לַנּוֹתָרִים לְנָשִׁים כִּי נִשְׁמְדָה מִבִּנְיָמִן אִשָּׁה` (שופטים כא:טז)
`וַאֲנַחְנוּ לֹא נוּכַל לָתֵת לָהֶם נָשִׁים מִבְּנוֹתֵינוּ כִּי נִשְׁבְּעוּ בְנֵי יִשְׂרָאֵל לֵאמֹר אָרוּר נֹתֵן אִשָּׁה לְבִנְיָמִן` (שופטים כא:יח)
`וּרְאִיתֶם וְהִנֵּה אִם יֵצְאוּ בְנוֹת שִׁילוֹ לָחוּל בַּמְּחֹלוֹת וִיצָאתֶם מִן הַכְּרָמִים וַחֲטַפְתֶּם לָכֶם אִישׁ אִשְׁתּוֹ מִבְּנוֹת שִׁילוֹ וַהֲלַכְתֶּם אֶרֶץ בִּנְיָמִן` (שופטים כא:כא)
`וְהָיָה כִּי יָבֹאוּ אֲבוֹתָם אוֹ אֲחֵיהֶם לָרוֹב אֵלֵינוּ וְאָמַרְנוּ אֲלֵיהֶם חָנּוּנוּ אוֹתָם כִּי לֹא לָקַחְנוּ אִישׁ אִשְׁתּוֹ בַּמִּלְחָמָה כִּי לֹא אַתֶּם נְתַתֶּם לָהֶם כָּעֵת תֶּאְשָׁמוּ` (שופטים כא:כב)
`וַיַּעֲשׂוּ כֵן בְּנֵי בִנְיָמִן וַיִּשְׂאוּ נָשִׁים לְמִסְפָּרָם מִן הַמְּחֹלְלוֹת אֲשֶׁר גָּזָלוּ וַיֵּלְכוּ וַיָּשׁוּבוּ אֶל נַחֲלָתָם וַיִּבְנוּ אֶת הֶעָרִים וַיֵּשְׁבוּ בָּהֶם` (שופטים כא:כג)
`וְלוֹ שְׁתֵּי נָשִׁים שֵׁם אַחַת חַנָּה וְשֵׁם הַשֵּׁנִית פְּנִנָּה וַיְהִי לִפְנִנָּה יְלָדִים וּלְחַנָּה אֵין יְלָדִים` (שמואל-א א:ב)
`וַיְהִי הַיּוֹם וַיִּזְבַּח אֶלְקָנָה וְנָתַן לִפְנִנָּה אִשְׁתּוֹ וּלְכָל בָּנֶיהָ וּבְנוֹתֶיהָ מָנוֹת` (שמואל-א א:ד)
`וַתַּעַן חַנָּה וַתֹּאמֶר לֹא אֲדֹנִי אִשָּׁה קְשַׁת רוּחַ אָנֹכִי וְיַיִן וְשֵׁכָר לֹא שָׁתִיתִי וָאֶשְׁפֹּךְ אֶת נַפְשִׁי לִפְנֵי יְהוָה` (שמואל-א א:טו)
`וַתֹּאמֶר תִּמְצָא שִׁפְחָתְךָ חֵן בְּעֵינֶיךָ וַתֵּלֶךְ הָאִשָּׁה לְדַרְכָּהּ וַתֹּאכַל וּפָנֶיהָ לֹא הָיוּ לָהּ עוֹד` (שמואל-א א:יח)
`וַיַּשְׁכִּמוּ בַבֹּקֶר וַיִּשְׁתַּחֲווּ לִפְנֵי יְהוָה וַיָּשֻׁבוּ וַיָּבֹאוּ אֶל בֵּיתָם הָרָמָתָה וַיֵּדַע אֶלְקָנָה אֶת חַנָּה אִשְׁתּוֹ וַיִּזְכְּרֶהָ יְהוָה` (שמואל-א א:יט)
`וַיֹּאמֶר לָהּ אֶלְקָנָה אִישָׁהּ עֲשִׂי הַטּוֹב בְּעֵינַיִךְ שְׁבִי עַד גָּמְלֵךְ אֹתוֹ אַךְ יָקֵם יְהוָה אֶת דְּבָרוֹ וַתֵּשֶׁב הָאִשָּׁה וַתֵּינֶק אֶת בְּנָהּ עַד גָּמְלָהּ אֹתוֹ` (שמואל-א א:כג)
`וַתֹּאמֶר בִּי אֲדֹנִי חֵי נַפְשְׁךָ אֲדֹנִי אֲנִי הָאִשָּׁה הַנִּצֶּבֶת עִמְּכָה בָּזֶה לְהִתְפַּלֵּל אֶל יְהוָה` (שמואל-א א:כו)
`וּבֵרַךְ עֵלִי אֶת אֶלְקָנָה וְאֶת אִשְׁתּוֹ וְאָמַר יָשֵׂם יְהוָה לְךָ זֶרַע מִן הָאִשָּׁה הַזֹּאת תַּחַת הַשְּׁאֵלָה אֲשֶׁר שָׁאַל לַיהוָה וְהָלְכוּ לִמְקֹמוֹ` (שמואל-א ב:כ)
`וְעֵלִי זָקֵן מְאֹד וְשָׁמַע אֵת כָּל אֲשֶׁר יַעֲשׂוּן בָּנָיו לְכָל יִשְׂרָאֵל וְאֵת אֲשֶׁר יִשְׁכְּבוּן אֶת הַנָּשִׁים הַצֹּבְאוֹת פֶּתַח אֹהֶל מוֹעֵד` (שמואל-א ב:כב)
`וְכַלָּתוֹ אֵשֶׁת פִּינְחָס הָרָה לָלַת וַתִּשְׁמַע אֶת הַשְּׁמֻעָה אֶל הִלָּקַח אֲרוֹן הָאֱלֹהִים וּמֵת חָמִיהָ וְאִישָׁהּ וַתִּכְרַע וַתֵּלֶד כִּי נֶהֶפְכוּ עָלֶיהָ צִרֶיהָ` (שמואל-א ד:יט)
`וְשֵׁם אֵשֶׁת שָׁאוּל אֲחִינֹעַם בַּת אֲחִימָעַץ וְשֵׁם שַׂר צְבָאוֹ אֲבִינֵר בֶּן נֵר דּוֹד שָׁאוּל` (שמואל-א יד:נ)
`עַתָּה לֵךְ וְהִכִּיתָה אֶת עֲמָלֵק וְהַחֲרַמְתֶּם אֶת כָּל אֲשֶׁר לוֹ וְלֹא תַחְמֹל עָלָיו וְהֵמַתָּה מֵאִישׁ עַד אִשָּׁה מֵעֹלֵל וְעַד יוֹנֵק מִשּׁוֹר וְעַד שֶׂה מִגָּמָל וְעַד חֲמוֹר` (שמואל-א טו:ג)
`וַיֹּאמֶר שְׁמוּאֵל כַּאֲשֶׁר שִׁכְּלָה נָשִׁים חַרְבֶּךָ כֵּן תִּשְׁכַּל מִנָּשִׁים אִמֶּךָ וַיְשַׁסֵּף שְׁמוּאֵל אֶת אֲגָג לִפְנֵי יְהוָה בַּגִּלְגָּל` (שמואל-א טו:לג)
`וַיְהִי בְּבוֹאָם בְּשׁוּב דָּוִד מֵהַכּוֹת אֶת הַפְּלִשְׁתִּי וַתֵּצֶאנָה הַנָּשִׁים מִכָּל עָרֵי יִשְׂרָאֵל לָשׁוּר וְהַמְּחֹלוֹת לִקְרַאת שָׁאוּל הַמֶּלֶךְ בְּתֻפִּים בְּשִׂמְחָה וּבְשָׁלִשִׁים` (שמואל-א יח:ו)
`וַתַּעֲנֶינָה הַנָּשִׁים הַמְשַׂחֲקוֹת וַתֹּאמַרְןָ הִכָּה שָׁאוּל בְּאַלְפּוֹ וְדָוִד בְּרִבְבֹתָיו` (שמואל-א יח:ז)
`וַיֹּאמֶר שָׁאוּל אֶל דָּוִד הִנֵּה בִתִּי הַגְּדוֹלָה מֵרַב אֹתָהּ אֶתֶּן לְךָ לְאִשָּׁה אַךְ הֱיֵה לִּי לְבֶן חַיִל וְהִלָּחֵם מִלְחֲמוֹת יְהוָה וְשָׁאוּל אָמַר אַל תְּהִי יָדִי בּוֹ וּתְהִי בוֹ יַד פְּלִשְׁתִּים` (שמואל-א יח:יז)
`וַיְהִי בְּעֵת תֵּת אֶת מֵרַב בַּת שָׁאוּל לְדָוִד וְהִיא נִתְּנָה לְעַדְרִיאֵל הַמְּחֹלָתִי לְאִשָּׁה` (שמואל-א יח:יט)
`וַיָּקָם דָּוִד וַיֵּלֶךְ הוּא וַאֲנָשָׁיו וַיַּךְ בַּפְּלִשְׁתִּים מָאתַיִם אִישׁ וַיָּבֵא דָוִד אֶת עָרְלֹתֵיהֶם וַיְמַלְאוּם לַמֶּלֶךְ לְהִתְחַתֵּן בַּמֶּלֶךְ וַיִּתֶּן לוֹ שָׁאוּל אֶת מִיכַל בִּתּוֹ לְאִשָּׁה` (שמואל-א יח:כז)
`וַיִּשְׁלַח שָׁאוּל מַלְאָכִים אֶל בֵּית דָּוִד לְשָׁמְרוֹ וְלַהֲמִיתוֹ בַּבֹּקֶר וַתַּגֵּד לְדָוִד מִיכַל אִשְׁתּוֹ לֵאמֹר אִם אֵינְךָ מְמַלֵּט אֶת נַפְשְׁךָ הַלַּיְלָה מָחָר אַתָּה מוּמָת` (שמואל-א יט:יא)
`וַיַּעַן הַכֹּהֵן אֶת דָּוִד וַיֹּאמֶר אֵין לֶחֶם חֹל אֶל תַּחַת יָדִי כִּי אִם לֶחֶם קֹדֶשׁ יֵשׁ אִם נִשְׁמְרוּ הַנְּעָרִים אַךְ מֵאִשָּׁה` (שמואל-א כא:ד)
`וַיַּעַן דָּוִד אֶת הַכֹּהֵן וַיֹּאמֶר לוֹ כִּי אִם אִשָּׁה עֲצֻרָה לָנוּ כִּתְמוֹל שִׁלְשֹׁם בְּצֵאתִי וַיִּהְיוּ כְלֵי הַנְּעָרִים קֹדֶשׁ וְהוּא דֶּרֶךְ חֹל וְאַף כִּי הַיּוֹם יִקְדַּשׁ בַּכֶּלִי` (שמואל-א כא:ה)
`וְאֵת נֹב עִיר הַכֹּהֲנִים הִכָּה לְפִי חֶרֶב מֵאִישׁ וְעַד אִשָּׁה מֵעוֹלֵל וְעַד יוֹנֵק וְשׁוֹר וַחֲמוֹר וָשֶׂה לְפִי חָרֶב` (שמואל-א כב:יט)
`וְשֵׁם הָאִישׁ נָבָל וְשֵׁם אִשְׁתּוֹ אֲבִגָיִל וְהָאִשָּׁה טוֹבַת שֶׂכֶל וִיפַת תֹּאַר וְהָאִישׁ קָשֶׁה וְרַע מַעֲלָלִים וְהוּא כְלִבּוֹ` (שמואל-א כה:ג)
`וְלַאֲבִיגַיִל אֵשֶׁת נָבָל הִגִּיד נַעַר אֶחָד מֵהַנְּעָרִים לֵאמֹר הִנֵּה שָׁלַח דָּוִד מַלְאָכִים מֵהַמִּדְבָּר לְבָרֵךְ אֶת אֲדֹנֵינוּ וַיָּעַט בָּהֶם` (שמואל-א כה:יד)
`וַיְהִי בַבֹּקֶר בְּצֵאת הַיַּיִן מִנָּבָל וַתַּגֶּד לוֹ אִשְׁתּוֹ אֶת הַדְּבָרִים הָאֵלֶּה וַיָּמָת לִבּוֹ בְּקִרְבּוֹ וְהוּא הָיָה לְאָבֶן` (שמואל-א כה:לז)
`וַיִּשְׁמַע דָּוִד כִּי מֵת נָבָל וַיֹּאמֶר בָּרוּךְ יְהוָה אֲשֶׁר רָב אֶת רִיב חֶרְפָּתִי מִיַּד נָבָל וְאֶת עַבְדּוֹ חָשַׂךְ מֵרָעָה וְאֵת רָעַת נָבָל הֵשִׁיב יְהוָה בְּרֹאשׁוֹ וַיִּשְׁלַח דָּוִד וַיְדַבֵּר בַּאֲבִיגַיִל לְקַחְתָּהּ לוֹ לְאִשָּׁה` (שמואל-א כה:לט)
`וַיָּבֹאוּ עַבְדֵי דָוִד אֶל אֲבִיגַיִל הַכַּרְמֶלָה וַיְדַבְּרוּ אֵלֶיהָ לֵאמֹר דָּוִד שְׁלָחָנוּ אֵלַיִךְ לְקַחְתֵּךְ לוֹ לְאִשָּׁה` (שמואל-א כה:מ)
`וַתְּמַהֵר וַתָּקָם אֲבִיגַיִל וַתִּרְכַּב עַל הַחֲמוֹר וְחָמֵשׁ נַעֲרֹתֶיהָ הַהֹלְכוֹת לְרַגְלָהּ וַתֵּלֶךְ אַחֲרֵי מַלְאֲכֵי דָוִד וַתְּהִי לוֹ לְאִשָּׁה` (שמואל-א כה:מב)
`וְאֶת אֲחִינֹעַם לָקַח דָּוִד מִיִּזְרְעֶאל וַתִּהְיֶיןָ גַּם שְׁתֵּיהֶן לוֹ לְנָשִׁים` (שמואל-א כה:מג)
`וְשָׁאוּל נָתַן אֶת מִיכַל בִּתּוֹ אֵשֶׁת דָּוִד לְפַלְטִי בֶן לַיִשׁ אֲשֶׁר מִגַּלִּים` (שמואל-א כה:מד)
`וַיֵּשֶׁב דָּוִד עִם אָכִישׁ בְּגַת הוּא וַאֲנָשָׁיו אִישׁ וּבֵיתוֹ דָּוִד וּשְׁתֵּי נָשָׁיו אֲחִינֹעַם הַיִּזְרְעֵאלִית וַאֲבִיגַיִל אֵשֶׁת נָבָל הַכַּרְמְלִית` (שמואל-א כז:ג)
`וְהִכָּה דָוִד אֶת הָאָרֶץ וְלֹא יְחַיֶּה אִישׁ וְאִשָּׁה וְלָקַח צֹאן וּבָקָר וַחֲמֹרִים וּגְמַלִּים וּבְגָדִים וַיָּשָׁב וַיָּבֹא אֶל אָכִישׁ` (שמואל-א כז:ט)

`וַיֹּאמֶר שָׁאוּל לַעֲבָדָיו בַּקְּשׁוּ לִי אֵשֶׁת בַּעֲלַת אוֹב וְאֵלְכָה אֵלֶיהָ וְאֶדְרְשָׁה בָּהּ וַיֹּאמְרוּ עֲבָדָיו אֵלָיו הִנֵּה אֵשֶׁת בַּעֲלַת אוֹב בְּעֵין דּוֹר` (שמואל-א כח:ז)
`וַיִּתְחַפֵּשׂ שָׁאוּל וַיִּלְבַּשׁ בְּגָדִים אֲחֵרִים וַיֵּלֶךְ הוּא וּשְׁנֵי אֲנָשִׁים עִמּוֹ וַיָּבֹאוּ אֶל הָאִשָּׁה לָיְלָה וַיֹּאמֶר קָסוֹמִי נָא לִי בָּאוֹב וְהַעֲלִי לִי אֵת אֲשֶׁר אֹמַר אֵלָיִךְ` (שמואל-א כח:ח)
`וַתֹּאמֶר הָאִשָּׁה אֵלָיו הִנֵּה אַתָּה יָדַעְתָּ אֵת אֲשֶׁר עָשָׂה שָׁאוּל אֲשֶׁר הִכְרִית אֶת הָאֹבוֹת וְאֶת הַיִּדְּעֹנִי מִן הָאָרֶץ וְלָמָה אַתָּה מִתְנַקֵּשׁ בְּנַפְשִׁי לַהֲמִיתֵנִי` (שמואל-א כח:ט)
`וַתֹּאמֶר הָאִשָּׁה אֶת מִי אַעֲלֶה לָּךְ וַיֹּאמֶר אֶת שְׁמוּאֵל הַעֲלִי לִי` (שמואל-א כח:יא)
`וַתֵּרֶא הָאִשָּׁה אֶת שְׁמוּאֵל וַתִּזְעַק בְּקוֹל גָּדוֹל וַתֹּאמֶר הָאִשָּׁה אֶל שָׁאוּל לֵאמֹר לָמָּה רִמִּיתָנִי וְאַתָּה שָׁאוּל` (שמואל-א כח:יב)
`וַיֹּאמֶר לָהּ הַמֶּלֶךְ אַל תִּירְאִי כִּי מָה רָאִית וַתֹּאמֶר הָאִשָּׁה אֶל שָׁאוּל אֱלֹהִים רָאִיתִי עֹלִים מִן הָאָרֶץ` (שמואל-א כח:יג)
`וַתָּבוֹא הָאִשָּׁה אֶל שָׁאוּל וַתֵּרֶא כִּי נִבְהַל מְאֹד וַתֹּאמֶר אֵלָיו הִנֵּה שָׁמְעָה שִׁפְחָתְךָ בְּקוֹלֶךָ וָאָשִׂים נַפְשִׁי בְּכַפִּי וָאֶשְׁמַע אֶת דְּבָרֶיךָ אֲשֶׁר דִּבַּרְתָּ אֵלָי` (שמואל-א כח:כא)
`וַיְמָאֵן וַיֹּאמֶר לֹא אֹכַל וַיִּפְרְצוּ בוֹ עֲבָדָיו וְגַם הָאִשָּׁה וַיִּשְׁמַע לְקֹלָם וַיָּקָם מֵהָאָרֶץ וַיֵּשֶׁב אֶל הַמִּטָּה` (שמואל-א כח:כג)
`וְלָאִשָּׁה עֵגֶל מַרְבֵּק בַּבַּיִת וַתְּמַהֵר וַתִּזְבָּחֵהוּ וַתִּקַּח קֶמַח וַתָּלָשׁ וַתֹּפֵהוּ מַצּוֹת` (שמואל-א כח:כד)
`וַיִּשְׁבּוּ אֶת הַנָּשִׁים אֲשֶׁר בָּהּ מִקָּטֹן וְעַד גָּדוֹל לֹא הֵמִיתוּ אִישׁ וַיִּנְהֲגוּ וַיֵּלְכוּ לְדַרְכָּם` (שמואל-א ל:ב)
`וַיָּבֹא דָוִד וַאֲנָשָׁיו אֶל הָעִיר וְהִנֵּה שְׂרוּפָה בָּאֵשׁ וּנְשֵׁיהֶם וּבְנֵיהֶם וּבְנֹתֵיהֶם נִשְׁבּוּ` (שמואל-א ל:ג)
`וּשְׁתֵּי נְשֵׁי דָוִד נִשְׁבּוּ אֲחִינֹעַם הַיִּזְרְעֵלִית וַאֲבִיגַיִל אֵשֶׁת נָבָל הַכַּרְמְלִי` (שמואל-א ל:ה)
`וַיַּצֵּל דָּוִד אֵת כָּל אֲשֶׁר לָקְחוּ עֲמָלֵק וְאֶת שְׁתֵּי נָשָׁיו הִצִּיל דָּוִד` (שמואל-א ל:יח)
`וַיַּעַן כָּל אִישׁ רָע וּבְלִיַּעַל מֵהָאֲנָשִׁים אֲשֶׁר הָלְכוּ עִם דָּוִד וַיֹּאמְרוּ יַעַן אֲשֶׁר לֹא הָלְכוּ עִמִּי לֹא נִתֵּן לָהֶם מֵהַשָּׁלָל אֲשֶׁר הִצַּלְנוּ כִּי אִם אִישׁ אֶת אִשְׁתּוֹ וְאֶת בָּנָיו וְיִנְהֲגוּ וְיֵלֵכוּ` (שמואל-א ל:כב)
`צַר לִי עָלֶיךָ אָחִי יְהוֹנָתָן נָעַמְתָּ לִּי מְאֹד נִפְלְאַתָה אַהֲבָתְךָ לִי מֵאַהֲבַת נָשִׁים` (שמואל-ב א:כו)
`וַיַּעַל שָׁם דָּוִד וְגַם שְׁתֵּי נָשָׁיו אֲחִינֹעַם הַיִּזְרְעֵלִית וַאֲבִיגַיִל אֵשֶׁת נָבָל הַכַּרְמְלִי` (שמואל-ב ב:ב)
`וּמִשְׁנֵהוּ כִלְאָב לַאֲבִיגֵל אֵשֶׁת נָבָל הַכַּרְמְלִי וְהַשְּׁלִשִׁי אַבְשָׁלוֹם בֶּן מַעֲכָה בַּת תַּלְמַי מֶלֶךְ גְּשׁוּר` (שמואל-ב ג:ג)
`וְהַשִּׁשִּׁי יִתְרְעָם לְעֶגְלָה אֵשֶׁת דָּוִד אֵלֶּה יֻלְּדוּ לְדָוִד בְּחֶבְרוֹן` (שמואל-ב ג:ה)
`וַיִּחַר לְאַבְנֵר מְאֹד עַל דִּבְרֵי אִישׁ בֹּשֶׁת וַיֹּאמֶר הֲרֹאשׁ כֶּלֶב אָנֹכִי אֲשֶׁר לִיהוּדָה הַיּוֹם אֶעֱשֶׂה חֶסֶד עִם בֵּית שָׁאוּל אָבִיךָ אֶל אֶחָיו וְאֶל מֵרֵעֵהוּ וְלֹא הִמְצִיתִךָ בְּיַד דָּוִד וַתִּפְקֹד עָלַי עֲוֺן הָאִשָּׁה הַיּוֹם` (שמואל-ב ג:ח)
`וַיִּשְׁלַח דָּוִד מַלְאָכִים אֶל אִישׁ בֹּשֶׁת בֶּן שָׁאוּל לֵאמֹר תְּנָה אֶת אִשְׁתִּי אֶת מִיכַל אֲשֶׁר אֵרַשְׂתִּי לִי בְּמֵאָה עָרְלוֹת פְּלִשְׁתִּים` (שמואל-ב ג:יד)
`וַיִּקַּח דָּוִד עוֹד פִּלַגְשִׁים וְנָשִׁים מִירוּשָׁלִַם אַחֲרֵי בֹּאוֹ מֵחֶבְרוֹן וַיִּוָּלְדּוּ עוֹד לְדָוִד בָּנִים וּבָנוֹת` (שמואל-ב ה:יג)
`וַיְחַלֵּק לְכָל הָעָם לְכָל הֲמוֹן יִשְׂרָאֵל לְמֵאִישׁ וְעַד אִשָּׁה לְאִישׁ חַלַּת לֶחֶם אַחַת וְאֶשְׁפָּר אֶחָד וַאֲשִׁישָׁה אֶחָת וַיֵּלֶךְ כָּל הָעָם אִישׁ לְבֵיתוֹ` (שמואל-ב ו:יט)
`וַיְהִי לְעֵת הָעֶרֶב וַיָּקָם דָּוִד מֵעַל מִשְׁכָּבוֹ וַיִּתְהַלֵּךְ עַל גַּג בֵּית הַמֶּלֶךְ וַיַּרְא אִשָּׁה רֹחֶצֶת מֵעַל הַגָּג וְהָאִשָּׁה טוֹבַת מַרְאֶה מְאֹד` (שמואל-ב יא:ב)
`וַיִּשְׁלַח דָּוִד וַיִּדְרֹשׁ לָאִשָּׁה וַיֹּאמֶר הֲלוֹא זֹאת בַּת שֶׁבַע בַּת אֱלִיעָם אֵשֶׁת אוּרִיָּה הַחִתִּי` (שמואל-ב יא:ג)
`וַתַּהַר הָאִשָּׁה וַתִּשְׁלַח וַתַּגֵּד לְדָוִד וַתֹּאמֶר הָרָה אָנֹכִי` (שמואל-ב יא:ה)
`וַיֹּאמֶר אוּרִיָּה אֶל דָּוִד הָאָרוֹן וְיִשְׂרָאֵל וִיהוּדָה יֹשְׁבִים בַּסֻּכּוֹת וַאדֹנִי יוֹאָב וְעַבְדֵי אֲדֹנִי עַל פְּנֵי הַשָּׂדֶה חֹנִים וַאֲנִי אָבוֹא אֶל בֵּיתִי לֶאֱכֹל וְלִשְׁתּוֹת וְלִשְׁכַּב עִם אִשְׁתִּי חַיֶּךָ וְחֵי נַפְשֶׁךָ אִם אֶעֱשֶׂה אֶת הַדָּבָר הַזֶּה` (שמואל-ב יא:יא)
`מִי הִכָּה אֶת אֲבִימֶלֶךְ בֶּן יְרֻבֶּשֶׁת הֲלוֹא אִשָּׁה הִשְׁלִיכָה עָלָיו פֶּלַח רֶכֶב מֵעַל הַחוֹמָה וַיָּמָת בְּתֵבֵץ לָמָּה נִגַּשְׁתֶּם אֶל הַחוֹמָה וְאָמַרְתָּ גַּם עַבְדְּךָ אוּרִיָּה הַחִתִּי מֵת` (שמואל-ב יא:כא)
`וַתִּשְׁמַע אֵשֶׁת אוּרִיָּה כִּי מֵת אוּרִיָּה אִישָׁהּ וַתִּסְפֹּד עַל בַּעְלָהּ` (שמואל-ב יא:כו)
`וַיַּעֲבֹר הָאֵבֶל וַיִּשְׁלַח דָּוִד וַיַּאַסְפָהּ אֶל בֵּיתוֹ וַתְּהִי לוֹ לְאִשָּׁה וַתֵּלֶד לוֹ בֵּן וַיֵּרַע הַדָּבָר אֲשֶׁר עָשָׂה דָוִד בְּעֵינֵי יְהוָה` (שמואל-ב יא:כז)
`וָאֶתְּנָה לְךָ אֶת בֵּית אֲדֹנֶיךָ וְאֶת נְשֵׁי אֲדֹנֶיךָ בְּחֵיקֶךָ וָאֶתְּנָה לְךָ אֶת בֵּית יִשְׂרָאֵל וִיהוּדָה וְאִם מְעָט וְאֹסִפָה לְּךָ כָּהֵנָּה וְכָהֵנָּה` (שמואל-ב יב:ח)
`מַדּוּעַ בָּזִיתָ אֶת דְּבַר יְהוָה לַעֲשׂוֹת הָרַע בְּעֵינוֹ אֵת אוּרִיָּה הַחִתִּי הִכִּיתָ בַחֶרֶב וְאֶת אִשְׁתּוֹ לָקַחְתָּ לְּךָ לְאִשָּׁה וְאֹתוֹ הָרַגְתָּ בְּחֶרֶב בְּנֵי עַמּוֹן` (שמואל-ב יב:ט)
`וְעַתָּה לֹא תָסוּר חֶרֶב מִבֵּיתְךָ עַד עוֹלָם עֵקֶב כִּי בְזִתָנִי וַתִּקַּח אֶת אֵשֶׁת אוּרִיָּה הַחִתִּי לִהְיוֹת לְךָ לְאִשָּׁה` (שמואל-ב יב:י)
`כֹּה אָמַר יְהוָה הִנְנִי מֵקִים עָלֶיךָ רָעָה מִבֵּיתֶךָ וְלָקַחְתִּי אֶת נָשֶׁיךָ לְעֵינֶיךָ וְנָתַתִּי לְרֵעֶיךָ וְשָׁכַב עִם נָשֶׁיךָ לְעֵינֵי הַשֶּׁמֶשׁ הַזֹּאת` (שמואל-ב יב:יא)
`וַיֵּלֶךְ נָתָן אֶל בֵּיתוֹ וַיִּגֹּף יְהוָה אֶת הַיֶּלֶד אֲשֶׁר יָלְדָה אֵשֶׁת אוּרִיָּה לְדָוִד וַיֵּאָנַשׁ` (שמואל-ב יב:טו)
`וַיְנַחֵם דָּוִד אֵת בַּת שֶׁבַע אִשְׁתּוֹ וַיָּבֹא אֵלֶיהָ וַיִּשְׁכַּב עִמָּהּ וַתֵּלֶד בֵּן וַיִּקְרָא אֶת שְׁמוֹ שְׁלֹמֹה וַיהוָה אֲהֵבוֹ` (שמואל-ב יב:כד)
`וַיִּשְׁלַח יוֹאָב תְּקוֹעָה וַיִּקַּח מִשָּׁם אִשָּׁה חֲכָמָה וַיֹּאמֶר אֵלֶיהָ הִתְאַבְּלִי נָא וְלִבְשִׁי נָא בִגְדֵי אֵבֶל וְאַל תָּסוּכִי שֶׁמֶן וְהָיִית כְּאִשָּׁה זֶה יָמִים רַבִּים מִתְאַבֶּלֶת עַל מֵת` (שמואל-ב יד:ב)
`וַתֹּאמֶר הָאִשָּׁה הַתְּקֹעִית אֶל הַמֶּלֶךְ וַתִּפֹּל עַל אַפֶּיהָ אַרְצָה וַתִּשְׁתָּחוּ וַתֹּאמֶר הוֹשִׁעָה הַמֶּלֶךְ` (שמואל-ב יד:ד)
`וַיֹּאמֶר לָהּ הַמֶּלֶךְ מַה לָּךְ וַתֹּאמֶר אֲבָל אִשָּׁה אַלְמָנָה אָנִי וַיָּמָת אִישִׁי` (שמואל-ב יד:ה)
`וַיֹּאמֶר הַמֶּלֶךְ אֶל הָאִשָּׁה לְכִי לְבֵיתֵךְ וַאֲנִי אֲצַוֶּה עָלָיִךְ` (שמואל-ב יד:ח)
`וַתֹּאמֶר הָאִשָּׁה הַתְּקוֹעִית אֶל הַמֶּלֶךְ עָלַי אֲדֹנִי הַמֶּלֶךְ הֶעָוֺן וְעַל בֵּית אָבִי וְהַמֶּלֶךְ וְכִסְאוֹ נָקִי` (שמואל-ב יד:ט)
`וַתֹּאמֶר הָאִשָּׁה תְּדַבֶּר נָא שִׁפְחָתְךָ אֶל אֲדֹנִי הַמֶּלֶךְ דָּבָר וַיֹּאמֶר דַּבֵּרִי` (שמואל-ב יד:יב)
`וַתֹּאמֶר הָאִשָּׁה וְלָמָּה חָשַׁבְתָּה כָּזֹאת עַל עַם אֱלֹהִים וּמִדַּבֵּר הַמֶּלֶךְ הַדָּבָר הַזֶּה כְּאָשֵׁם לְבִלְתִּי הָשִׁיב הַמֶּלֶךְ אֶת נִדְּחוֹ` (שמואל-ב יד:יג)
`וַיַּעַן הַמֶּלֶךְ וַיֹּאמֶר אֶל הָאִשָּׁה אַל נָא תְכַחֲדִי מִמֶּנִּי דָּבָר אֲשֶׁר אָנֹכִי שֹׁאֵל אֹתָךְ וַתֹּאמֶר הָאִשָּׁה יְדַבֶּר נָא אֲדֹנִי הַמֶּלֶךְ` (שמואל-ב יד:יח)
`וַיֹּאמֶר הַמֶּלֶךְ הֲיַד יוֹאָב אִתָּךְ בְּכָל זֹאת וַתַּעַן הָאִשָּׁה וַתֹּאמֶר חֵי נַפְשְׁךָ אֲדֹנִי הַמֶּלֶךְ אִם אִשׁ לְהֵמִין וּלְהַשְׂמִיל מִכֹּל אֲשֶׁר דִּבֶּר אֲדֹנִי הַמֶּלֶךְ כִּי עַבְדְּךָ יוֹאָב הוּא צִוָּנִי וְהוּא שָׂם בְּפִי שִׁפְחָתְךָ אֵת כָּל הַדְּבָרִים הָאֵלֶּה` (שמואל-ב יד:יט)
`וַיִּוָּלְדוּ לְאַבְשָׁלוֹם שְׁלוֹשָׁה בָנִים וּבַת אַחַת וּשְׁמָהּ תָּמָר הִיא הָיְתָה אִשָּׁה יְפַת מַרְאֶה` (שמואל-ב יד:כז)
`וַיֵּצֵא הַמֶּלֶךְ וְכָל בֵּיתוֹ בְּרַגְלָיו וַיַּעֲזֹב הַמֶּלֶךְ אֵת עֶשֶׂר נָשִׁים פִּלַגְשִׁים לִשְׁמֹר הַבָּיִת` (שמואל-ב טו:טז)
`וַתִּקַּח הָאִשָּׁה וַתִּפְרֹשׂ אֶת הַמָּסָךְ עַל פְּנֵי הַבְּאֵר וַתִּשְׁטַח עָלָיו הָרִפוֹת וְלֹא נוֹדַע דָּבָר` (שמואל-ב יז:יט)
`וַיָּבֹאוּ עַבְדֵי אַבְשָׁלוֹם אֶל הָאִשָּׁה הַבַּיְתָה וַיֹּאמְרוּ אַיֵּה אֲחִימַעַץ וִיהוֹנָתָן וַתֹּאמֶר לָהֶם הָאִשָּׁה עָבְרוּ מִיכַל הַמָּיִם וַיְבַקְשׁוּ וְלֹא מָצָאוּ וַיָּשֻׁבוּ יְרוּשָׁלִָם` (שמואל-ב יז:כ)
`וַיָּבֹא יוֹאָב אֶל הַמֶּלֶךְ הַבָּיִת וַיֹּאמֶר הֹבַשְׁתָּ הַיּוֹם אֶת פְּנֵי כָל עֲבָדֶיךָ הַמְמַלְּטִים אֶת נַפְשְׁךָ הַיּוֹם וְאֵת נֶפֶשׁ בָּנֶיךָ וּבְנֹתֶיךָ וְנֶפֶשׁ נָשֶׁיךָ וְנֶפֶשׁ פִּלַגְשֶׁיךָ` (שמואל-ב יט:ה)
`וַיָּבֹא דָוִד אֶל בֵּיתוֹ יְרוּשָׁלִַם וַיִּקַּח הַמֶּלֶךְ אֵת עֶשֶׂר נָשִׁים פִּלַגְשִׁים אֲשֶׁר הִנִּיחַ לִשְׁמֹר הַבַּיִת וַיִּתְּנֵם בֵּית מִשְׁמֶרֶת וַיְכַלְכְּלֵם וַאֲלֵיהֶם לֹא בָא וַתִּהְיֶינָה צְרֻרוֹת עַד יוֹם מֻתָן אַלְמְנוּת חַיּוּת` (שמואל-ב כ:ג)
`וַתִּקְרָא אִשָּׁה חֲכָמָה מִן הָעִיר שִׁמְעוּ שִׁמְעוּ אִמְרוּ נָא אֶל יוֹאָב קְרַב עַד הֵנָּה וַאֲדַבְּרָה אֵלֶיךָ` (שמואל-ב כ:טז)
`וַיִּקְרַב אֵלֶיהָ וַתֹּאמֶר הָאִשָּׁה הַאַתָּה יוֹאָב וַיֹּאמֶר אָנִי וַתֹּאמֶר לוֹ שְׁמַע דִּבְרֵי אֲמָתֶךָ וַיֹּאמֶר שֹׁמֵעַ אָנֹכִי` (שמואל-ב כ:יז)
`לֹא כֵן הַדָּבָר כִּי אִישׁ מֵהַר אֶפְרַיִם שֶׁבַע בֶּן בִּכְרִי שְׁמוֹ נָשָׂא יָדוֹ בַּמֶּלֶךְ בְּדָוִד תְּנוּ אֹתוֹ לְבַדּוֹ וְאֵלְכָה מֵעַל הָעִיר וַתֹּאמֶר הָאִשָּׁה אֶל יוֹאָב הִנֵּה רֹאשׁוֹ מֻשְׁלָךְ אֵלֶיךָ בְּעַד הַחוֹמָה` (שמואל-ב כ:כא)
`וַתָּבוֹא הָאִשָּׁה אֶל כָּל הָעָם בְּחָכְמָתָהּ וַיִּכְרְתוּ אֶת רֹאשׁ שֶׁבַע בֶּן בִּכְרִי וַיַּשְׁלִכוּ אֶל יוֹאָב וַיִּתְקַע בַּשּׁוֹפָר וַיָּפֻצוּ מֵעַל הָעִיר אִישׁ לְאֹהָלָיו וְיוֹאָב שָׁב יְרוּשָׁלִַם אֶל הַמֶּלֶךְ` (שמואל-ב כ:כב)
`וַיֹּאמֶר אִמְרִי נָא לִשְׁלֹמֹה הַמֶּלֶךְ כִּי לֹא יָשִׁיב אֶת פָּנָיִךְ וְיִתֶּן לִי אֶת אֲבִישַׁג הַשּׁוּנַמִּית לְאִשָּׁה` (מלכים-א ב:יז)
`וַתֹּאמֶר יֻתַּן אֶת אֲבִישַׁג הַשֻּׁנַמִּית לַאֲדֹנִיָּהוּ אָחִיךָ לְאִשָּׁה` (מלכים-א ב:כא)
`אָז תָּבֹאנָה שְׁתַּיִם נָשִׁים זֹנוֹת אֶל הַמֶּלֶךְ וַתַּעֲמֹדְנָה לְפָנָיו` (מלכים-א ג:טז)
`וַתֹּאמֶר הָאִשָּׁה הָאַחַת בִּי אֲדֹנִי אֲנִי וְהָאִשָּׁה הַזֹּאת יֹשְׁבֹת בְּבַיִת אֶחָד וָאֵלֵד עִמָּהּ בַּבָּיִת` (מלכים-א ג:יז)
`וַיְהִי בַּיּוֹם הַשְּׁלִישִׁי לְלִדְתִּי וַתֵּלֶד גַּם הָאִשָּׁה הַזֹּאת וַאֲנַחְנוּ יַחְדָּו אֵין זָר אִתָּנוּ בַּבַּיִת זוּלָתִי שְׁתַּיִם אֲנַחְנוּ בַּבָּיִת` (מלכים-א ג:יח)
`וַיָּמָת בֶּן הָאִשָּׁה הַזֹּאת לָיְלָה אֲשֶׁר שָׁכְבָה עָלָיו` (מלכים-א ג:יט)
`וַתֹּאמֶר הָאִשָּׁה הָאַחֶרֶת לֹא כִי בְּנִי הַחַי וּבְנֵךְ הַמֵּת וְזֹאת אֹמֶרֶת לֹא כִי בְּנֵךְ הַמֵּת וּבְנִי הֶחָי וַתְּדַבֵּרְנָה לִפְנֵי הַמֶּלֶךְ` (מלכים-א ג:כב)
`וַתֹּאמֶר הָאִשָּׁה אֲשֶׁר בְּנָהּ הַחַי אֶל הַמֶּלֶךְ כִּי נִכְמְרוּ רַחֲמֶיהָ עַל בְּנָהּ וַתֹּאמֶר בִּי אֲדֹנִי תְּנוּ לָהּ אֶת הַיָּלוּד הַחַי וְהָמֵת אַל תְּמִיתֻהוּ וְזֹאת אֹמֶרֶת גַּם לִי גַם לָךְ לֹא יִהְיֶה גְּזֹרוּ` (מלכים-א ג:כו)
`בֶּן אֲבִינָדָב כָּל נָפַת דֹּאר טָפַת בַּת שְׁלֹמֹה הָיְתָה לּוֹ לְאִשָּׁה` (מלכים-א ד:יא)
`אֲחִימַעַץ בְּנַפְתָּלִי גַּם הוּא לָקַח אֶת בָּשְׂמַת בַּת שְׁלֹמֹה לְאִשָּׁה` (מלכים-א ד:טו)
`בֶּן אִשָּׁה אַלְמָנָה הוּא מִמַּטֵּה נַפְתָּלִי וְאָבִיו אִישׁ צֹרִי חֹרֵשׁ נְחֹשֶׁת וַיִּמָּלֵא אֶת הַחָכְמָה וְאֶת הַתְּבוּנָה וְאֶת הַדַּעַת לַעֲשׂוֹת כָּל מְלָאכָה בַּנְּחֹשֶׁת וַיָּבוֹא אֶל הַמֶּלֶךְ שְׁלֹמֹה וַיַּעַשׂ אֶת כָּל מְלַאכְתּוֹ` (מלכים-א ז:יד)
`פַּרְעֹה מֶלֶךְ מִצְרַיִם עָלָה וַיִּלְכֹּד אֶת גֶּזֶר וַיִּשְׂרְפָהּ בָּאֵשׁ וְאֶת הַכְּנַעֲנִי הַיֹּשֵׁב בָּעִיר הָרָג וַיִּתְּנָהּ שִׁלֻּחִים לְבִתּוֹ אֵשֶׁת שְׁלֹמֹה` (מלכים-א ט:טז)
`וְהַמֶּלֶךְ שְׁלֹמֹה אָהַב נָשִׁים נָכְרִיּוֹת רַבּוֹת וְאֶת בַּת פַּרְעֹה מוֹאֲבִיּוֹת עַמֳּנִיּוֹת אֲדֹמִיֹּת צֵדְנִיֹּת חִתִּיֹּת` (מלכים-א יא:א)
`וַיְהִי לוֹ נָשִׁים שָׂרוֹת שְׁבַע מֵאוֹת וּפִלַגְשִׁים שְׁלֹשׁ מֵאוֹת וַיַּטּוּ נָשָׁיו אֶת לִבּוֹ` (מלכים-א יא:ג)
`וַיְהִי לְעֵת זִקְנַת שְׁלֹמֹה נָשָׁיו הִטּוּ אֶת לְבָבוֹ אַחֲרֵי אֱלֹהִים אֲחֵרִים וְלֹא הָיָה לְבָבוֹ שָׁלֵם עִם יְהוָה אֱלֹהָיו כִּלְבַב דָּוִיד אָבִיו` (מלכים-א יא:ד)
`וְכֵן עָשָׂה לְכָל נָשָׁיו הַנָּכְרִיּוֹת מַקְטִירוֹת וּמְזַבְּחוֹת לֵאלֹהֵיהֶן` (מלכים-א יא:ח)
`וַיִּמְצָא הֲדַד חֵן בְּעֵינֵי פַרְעֹה מְאֹד וַיִּתֶּן לוֹ אִשָּׁה אֶת אֲחוֹת אִשְׁתּוֹ אֲחוֹת תַּחְפְּנֵיס הַגְּבִירָה` (מלכים-א יא:יט)
`וְיָרָבְעָם בֶּן נְבָט אֶפְרָתִי מִן הַצְּרֵדָה וְשֵׁם אִמּוֹ צְרוּעָה אִשָּׁה אַלְמָנָה עֶבֶד לִשְׁלֹמֹה וַיָּרֶם יָד בַּמֶּלֶךְ` (מלכים-א יא:כו)
`וַיֹּאמֶר יָרָבְעָם לְאִשְׁתּוֹ קוּמִי נָא וְהִשְׁתַּנִּית וְלֹא יֵדְעוּ כִּי אַתִּי אֵשֶׁת יָרָבְעָם וְהָלַכְתְּ שִׁלֹה הִנֵּה שָׁם אֲחִיָּה הַנָּבִיא הוּא דִבֶּר עָלַי לְמֶלֶךְ עַל הָעָם הַזֶּה` (מלכים-א יד:ב)
`וַתַּעַשׂ כֵּן אֵשֶׁת יָרָבְעָם וַתָּקָם וַתֵּלֶךְ שִׁלֹה וַתָּבֹא בֵּית אֲחִיָּה וַאֲחִיָּהוּ לֹא יָכֹל לִרְאוֹת כִּי קָמוּ עֵינָיו מִשֵּׂיבוֹ` (מלכים-א יד:ד)
`וַיהוָה אָמַר אֶל אֲחִיָּהוּ הִנֵּה אֵשֶׁת יָרָבְעָם בָּאָה לִדְרֹשׁ דָּבָר מֵעִמְּךָ אֶל בְּנָהּ כִּי חֹלֶה הוּא כָּזֹה וְכָזֶה תְּדַבֵּר אֵלֶיהָ וִיהִי כְבֹאָהּ וְהִיא מִתְנַכֵּרָה` (מלכים-א יד:ה)
`וַיְהִי כִשְׁמֹעַ אֲחִיָּהוּ אֶת קוֹל רַגְלֶיהָ בָּאָה בַפֶּתַח וַיֹּאמֶר בֹּאִי אֵשֶׁת יָרָבְעָם לָמָּה זֶּה אַתְּ מִתְנַכֵּרָה וְאָנֹכִי שָׁלוּחַ אֵלַיִךְ קָשָׁה` (מלכים-א יד:ו)
`וַתָּקָם אֵשֶׁת יָרָבְעָם וַתֵּלֶךְ וַתָּבֹא תִרְצָתָה הִיא בָּאָה בְסַף הַבַּיִת וְהַנַּעַר מֵת` (מלכים-א יד:יז)
`וַיְהִי הֲנָקֵל לֶכְתּוֹ בְּחַטֹּאות יָרָבְעָם בֶּן נְבָט וַיִּקַּח אִשָּׁה אֶת אִיזֶבֶל בַּת אֶתְבַּעַל מֶלֶךְ צִידֹנִים וַיֵּלֶךְ וַיַּעֲבֹד אֶת הַבַּעַל וַיִּשְׁתַּחוּ לוֹ` (מלכים-א טז:לא)
`קוּם לֵךְ צָרְפַתָה אֲשֶׁר לְצִידוֹן וְיָשַׁבְתָּ שָׁם הִנֵּה צִוִּיתִי שָׁם אִשָּׁה אַלְמָנָה לְכַלְכְּלֶךָ` (מלכים-א יז:ט)
`וַיָּקָם וַיֵּלֶךְ צָרְפַתָה וַיָּבֹא אֶל פֶּתַח הָעִיר וְהִנֵּה שָׁם אִשָּׁה אַלְמָנָה מְקֹשֶׁשֶׁת עֵצִים וַיִּקְרָא אֵלֶיהָ וַיֹּאמַר קְחִי נָא לִי מְעַט מַיִם בַּכְּלִי וְאֶשְׁתֶּה` (מלכים-א יז:י)
`וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה חָלָה בֶּן הָאִשָּׁה בַּעֲלַת הַבָּיִת וַיְהִי חָלְיוֹ חָזָק מְאֹד עַד אֲשֶׁר לֹא נוֹתְרָה בּוֹ נְשָׁמָה` (מלכים-א יז:יז)
`וַתֹּאמֶר הָאִשָּׁה אֶל אֵלִיָּהוּ עַתָּה זֶה יָדַעְתִּי כִּי אִישׁ אֱלֹהִים אָתָּה וּדְבַר יְהוָה בְּפִיךָ אֱמֶת` (מלכים-א יז:כד)
`וַיֹּאמֶר לוֹ כֹּה אָמַר בֶּן הֲדַד כַּסְפְּךָ וּזְהָבְךָ לִי הוּא וְנָשֶׁיךָ וּבָנֶיךָ הַטּוֹבִים לִי הֵם` (מלכים-א כ:ג)
`וַיָּשֻׁבוּ הַמַּלְאָכִים וַיֹּאמְרוּ כֹּה אָמַר בֶּן הֲדַד לֵאמֹר כִּי שָׁלַחְתִּי אֵלֶיךָ לֵאמֹר כַּסְפְּךָ וּזְהָבְךָ וְנָשֶׁיךָ וּבָנֶיךָ לִי תִתֵּן` (מלכים-א כ:ה)
`וַיִּקְרָא מֶלֶךְ יִשְׂרָאֵל לְכָל זִקְנֵי הָאָרֶץ וַיֹּאמֶר דְּעוּ נָא וּרְאוּ כִּי רָעָה זֶה מְבַקֵּשׁ כִּי שָׁלַח אֵלַי לְנָשַׁי וּלְבָנַי וּלְכַסְפִּי וְלִזְהָבִי וְלֹא מָנַעְתִּי מִמֶּנּוּ` (מלכים-א כ:ז)
`וַתָּבֹא אֵלָיו אִיזֶבֶל אִשְׁתּוֹ וַתְּדַבֵּר אֵלָיו מַה זֶּה רוּחֲךָ סָרָה וְאֵינְךָ אֹכֵל לָחֶם` (מלכים-א כא:ה)
`וַתֹּאמֶר אֵלָיו אִיזֶבֶל אִשְׁתּוֹ אַתָּה עַתָּה תַּעֲשֶׂה מְלוּכָה עַל יִשְׂרָאֵל קוּם אֱכָל לֶחֶם וְיִטַב לִבֶּךָ אֲנִי אֶתֵּן לְךָ אֶת כֶּרֶם נָבוֹת הַיִּזְרְעֵאלִי` (מלכים-א כא:ז)
`רַק לֹא הָיָה כְאַחְאָב אֲשֶׁר הִתְמַכֵּר לַעֲשׂוֹת הָרַע בְּעֵינֵי יְהוָה אֲשֶׁר הֵסַתָּה אֹתוֹ אִיזֶבֶל אִשְׁתּוֹ` (מלכים-א כא:כה)
`וְאִשָּׁה אַחַת מִנְּשֵׁי בְנֵי הַנְּבִיאִים צָעֲקָה אֶל אֱלִישָׁע לֵאמֹר עַבְדְּךָ אִישִׁי מֵת וְאַתָּה יָדַעְתָּ כִּי עַבְדְּךָ הָיָה יָרֵא אֶת יְהוָה וְהַנֹּשֶׁה בָּא לָקַחַת אֶת שְׁנֵי יְלָדַי לוֹ לַעֲבָדִים` (מלכים-ב ד:א)
`וַיְהִי הַיּוֹם וַיַּעֲבֹר אֱלִישָׁע אֶל שׁוּנֵם וְשָׁם אִשָּׁה גְדוֹלָה וַתַּחֲזֶק בּוֹ לֶאֱכָל לָחֶם וַיְהִי מִדֵּי עָבְרוֹ יָסֻר שָׁמָּה לֶאֱכָל לָחֶם` (מלכים-ב ד:ח)
`וַתַּהַר הָאִשָּׁה וַתֵּלֶד בֵּן לַמּוֹעֵד הַזֶּה כָּעֵת חַיָּה אֲשֶׁר דִּבֶּר אֵלֶיהָ אֱלִישָׁע` (מלכים-ב ד:יז)
`וַאֲרָם יָצְאוּ גְדוּדִים וַיִּשְׁבּוּ מֵאֶרֶץ יִשְׂרָאֵל נַעֲרָה קְטַנָּה וַתְּהִי לִפְנֵי אֵשֶׁת נַעֲמָן` (מלכים-ב ה:ב)
`וַיְהִי מֶלֶךְ יִשְׂרָאֵל עֹבֵר עַל הַחֹמָה וְאִשָּׁה צָעֲקָה אֵלָיו לֵאמֹר הוֹשִׁיעָה אֲדֹנִי הַמֶּלֶךְ` (מלכים-ב ו:כו)
`וַיֹּאמֶר לָהּ הַמֶּלֶךְ מַה לָּךְ וַתֹּאמֶר הָאִשָּׁה הַזֹּאת אָמְרָה אֵלַי תְּנִי אֶת בְּנֵךְ וְנֹאכְלֶנּוּ הַיּוֹם וְאֶת בְּנִי נֹאכַל מָחָר` (מלכים-ב ו:כח)
`וַיְהִי כִשְׁמֹעַ הַמֶּלֶךְ אֶת דִּבְרֵי הָאִשָּׁה וַיִּקְרַע אֶת בְּגָדָיו וְהוּא עֹבֵר עַל הַחֹמָה וַיַּרְא הָעָם וְהִנֵּה הַשַּׂק עַל בְּשָׂרוֹ מִבָּיִת` (מלכים-ב ו:ל)
`וֶאֱלִישָׁע דִּבֶּר אֶל הָאִשָּׁה אֲשֶׁר הֶחֱיָה אֶת בְּנָהּ לֵאמֹר קוּמִי וּלְכִי אַתִּי וּבֵיתֵךְ וְגוּרִי בַּאֲשֶׁר תָּגוּרִי כִּי קָרָא יְהוָה לָרָעָב וְגַם בָּא אֶל הָאָרֶץ שֶׁבַע שָׁנִים` (מלכים-ב ח:א)
`וַתָּקָם הָאִשָּׁה וַתַּעַשׂ כִּדְבַר אִישׁ הָאֱלֹהִים וַתֵּלֶךְ הִיא וּבֵיתָהּ וַתָּגָר בְּאֶרֶץ פְּלִשְׁתִּים שֶׁבַע שָׁנִים` (מלכים-ב ח:ב)
`וַיְהִי מִקְצֵה שֶׁבַע שָׁנִים וַתָּשָׁב הָאִשָּׁה מֵאֶרֶץ פְּלִשְׁתִּים וַתֵּצֵא לִצְעֹק אֶל הַמֶּלֶךְ אֶל בֵּיתָהּ וְאֶל שָׂדָהּ` (מלכים-ב ח:ג)
`וַיְהִי הוּא מְסַפֵּר לַמֶּלֶךְ אֵת אֲשֶׁר הֶחֱיָה אֶת הַמֵּת וְהִנֵּה הָאִשָּׁה אֲשֶׁר הֶחֱיָה אֶת בְּנָהּ צֹעֶקֶת אֶל הַמֶּלֶךְ עַל בֵּיתָהּ וְעַל שָׂדָהּ וַיֹּאמֶר גֵּחֲזִי אֲדֹנִי הַמֶּלֶךְ זֹאת הָאִשָּׁה וְזֶה בְּנָהּ אֲשֶׁר הֶחֱיָה אֱלִישָׁע` (מלכים-ב ח:ה)
`וַיִּשְׁאַל הַמֶּלֶךְ לָאִשָּׁה וַתְּסַפֶּר לוֹ וַיִּתֶּן לָהּ הַמֶּלֶךְ סָרִיס אֶחָד לֵאמֹר הָשֵׁיב אֶת כָּל אֲשֶׁר לָהּ וְאֵת כָּל תְּבוּאֹת הַשָּׂדֶה מִיּוֹם עָזְבָה אֶת הָאָרֶץ וְעַד עָתָּה` (מלכים-ב ח:ו)
`וַיֵּלֶךְ בְּדֶרֶךְ מַלְכֵי יִשְׂרָאֵל כַּאֲשֶׁר עָשׂוּ בֵּית אַחְאָב כִּי בַּת אַחְאָב הָיְתָה לּוֹ לְאִשָּׁה וַיַּעַשׂ הָרַע בְּעֵינֵי יְהוָה` (מלכים-ב ח:יח)
`וַיִּשְׁלַח יְהוֹאָשׁ מֶלֶךְ יִשְׂרָאֵל אֶל אֲמַצְיָהוּ מֶלֶךְ יְהוּדָה לֵאמֹר הַחוֹחַ אֲשֶׁר בַּלְּבָנוֹן שָׁלַח אֶל הָאֶרֶז אֲשֶׁר בַּלְּבָנוֹן לֵאמֹר תְּנָה אֶת בִּתְּךָ לִבְנִי לְאִשָּׁה וַתַּעֲבֹר חַיַּת הַשָּׂדֶה אֲשֶׁר בַּלְּבָנוֹן וַתִּרְמֹס אֶת הַחוֹחַ` (מלכים-ב יד:ט)
`וַיֵּלֶךְ חִלְקִיָּהוּ הַכֹּהֵן וַאֲחִיקָם וְעַכְבּוֹר וְשָׁפָן וַעֲשָׂיָה אֶל חֻלְדָּה הַנְּבִיאָה אֵשֶׁת שַׁלֻּם בֶּן תִּקְוָה בֶּן חַרְחַס שֹׁמֵר הַבְּגָדִים וְהִיא יֹשֶׁבֶת בִּירוּשָׁלִַם בַּמִּשְׁנֶה וַיְדַבְּרוּ אֵלֶיהָ` (מלכים-ב כב:יד)
`וַיִּתֹּץ אֶת בָּתֵּי הַקְּדֵשִׁים אֲשֶׁר בְּבֵית יְהוָה אֲשֶׁר הַנָּשִׁים אֹרְגוֹת שָׁם בָּתִּים לָאֲשֵׁרָה` (מלכים-ב כג:ז)
`וַיֶּגֶל אֶת יְהוֹיָכִין בָּבֶלָה וְאֶת אֵם הַמֶּלֶךְ וְאֶת נְשֵׁי הַמֶּלֶךְ וְאֶת סָרִיסָיו וְאֵת אֱוִלֵי הָאָרֶץ הוֹלִיךְ גּוֹלָה מִירוּשָׁלִַם בָּבֶלָה` (מלכים-ב כד:טו)
`עַמִּי נֹגְשָׂיו מְעוֹלֵל וְנָשִׁים מָשְׁלוּ בוֹ עַמִּי מְאַשְּׁרֶיךָ מַתְעִים וְדֶרֶךְ אֹרְחֹתֶיךָ בִּלֵּעוּ` (ישעיהו ג:יב)
`וְהֶחֱזִיקוּ שֶׁבַע נָשִׁים בְּאִישׁ אֶחָד בַּיּוֹם הַהוּא לֵאמֹר לַחְמֵנוּ נֹאכֵל וְשִׂמְלָתֵנוּ נִלְבָּשׁ רַק יִקָּרֵא שִׁמְךָ עָלֵינוּ אֱסֹף חֶרְפָּתֵנוּ` (ישעיהו ד:א)
`וְעֹלְלֵיהֶם יְרֻטְּשׁוּ לְעֵינֵיהֶם יִשַּׁסּוּ בָּתֵּיהֶם וּנְשֵׁיהֶם תִּשָּׁגַלְנָה` (ישעיהו יג:טז)
`בַּיּוֹם הַהוּא יִהְיֶה מִצְרַיִם כַּנָּשִׁים וְחָרַד וּפָחַד מִפְּנֵי תְּנוּפַת יַד יְהוָה צְבָאוֹת אֲשֶׁר הוּא מֵנִיף עָלָיו` (ישעיהו יט:טז)
`בִּיבֹשׁ קְצִירָהּ תִּשָּׁבַרְנָה נָשִׁים בָּאוֹת מְאִירוֹת אוֹתָהּ כִּי לֹא עַם בִּינוֹת הוּא עַל כֵּן לֹא יְרַחֲמֶנּוּ עֹשֵׂהוּ וְיֹצְרוֹ לֹא יְחֻנֶּנּוּ` (ישעיהו כז:יא)
`נָשִׁים שַׁאֲנַנּוֹת קֹמְנָה שְׁמַעְנָה קוֹלִי בָּנוֹת בֹּטחוֹת הַאְזֵנָּה אִמְרָתִי` (ישעיהו לב:ט)
`שָׁמָּה קִנְּנָה קִפּוֹז וַתְּמַלֵּט וּבָקְעָה וְדָגְרָה בְצִלָּהּ אַךְ שָׁם נִקְבְּצוּ דַיּוֹת אִשָּׁה רְעוּתָהּ` (ישעיהו לד:טו)
`דִּרְשׁוּ מֵעַל סֵפֶר יְהוָה וּקְרָאוּ אַחַת מֵהֵנָּה לֹא נֶעְדָּרָה אִשָּׁה רְעוּתָהּ לֹא פָקָדוּ כִּי פִי הוּא צִוָּה וְרוּחוֹ הוּא קִבְּצָן` (ישעיהו לד:טז)
`הוֹי אֹמֵר לְאָב מַה תּוֹלִיד וּלְאִשָּׁה מַה תְּחִילִין` (ישעיהו מה:י)
`הֲתִשְׁכַּח אִשָּׁה עוּלָהּ מֵרַחֵם בֶּן בִּטְנָהּ גַּם אֵלֶּה תִשְׁכַּחְנָה וְאָנֹכִי לֹא אֶשְׁכָּחֵךְ` (ישעיהו מט:טו)
`כִּי כְאִשָּׁה עֲזוּבָה וַעֲצוּבַת רוּחַ קְרָאָךְ יְהוָה וְאֵשֶׁת נְעוּרִים כִּי תִמָּאֵס אָמַר אֱלֹהָיִךְ` (ישעיהו נד:ו)
`לֵאמֹר הֵן יְשַׁלַּח אִישׁ אֶת אִשְׁתּוֹ וְהָלְכָה מֵאִתּוֹ וְהָיְתָה לְאִישׁ אַחֵר הֲיָשׁוּב אֵלֶיהָ עוֹד הֲלוֹא חָנוֹף תֶּחֱנַף הָאָרֶץ הַהִיא וְאַתְּ זָנִית רֵעִים רַבִּים וְשׁוֹב אֵלַי נְאֻם יְהֹוָה` (ירמיהו ג:א)
`וַיִּמָּנְעוּ רְבִבִים וּמַלְקוֹשׁ לוֹא הָיָה וּמֵצַח אִשָּׁה זוֹנָה הָיָה לָךְ מֵאַנְתְּ הִכָּלֵם` (ירמיהו ג:ג)
`אָכֵן בָּגְדָה אִשָּׁה מֵרֵעָהּ כֵּן בְּגַדְתֶּם בִּי בֵּית יִשְׂרָאֵל נְאֻם יְהוָה` (ירמיהו ג:כ)
`סוּסִים מְיֻזָּנִים מַשְׁכִּים הָיוּ אִישׁ אֶל אֵשֶׁת רֵעֵהוּ יִצְהָלוּ` (ירמיהו ה:ח)
`וְאֵת חֲמַת יְהוָה מָלֵאתִי נִלְאֵיתִי הָכִיל שְׁפֹךְ עַל עוֹלָל בַּחוּץ וְעַל סוֹד בַּחוּרִים יַחְדָּו כִּי גַם אִישׁ עִם אִשָּׁה יִלָּכֵדוּ זָקֵן עִם מְלֵא יָמִים` (ירמיהו ו:יא)
`וְנָסַבּוּ בָתֵּיהֶם לַאֲחֵרִים שָׂדוֹת וְנָשִׁים יַחְדָּו כִּי אַטֶּה אֶת יָדִי עַל יֹשְׁבֵי הָאָרֶץ נְאֻם יְהוָה` (ירמיהו ו:יב)
`הַבָּנִים מְלַקְּטִים עֵצִים וְהָאָבוֹת מְבַעֲרִים אֶת הָאֵשׁ וְהַנָּשִׁים לָשׁוֹת בָּצֵק לַעֲשׂוֹת כַּוָּנִים לִמְלֶכֶת הַשָּׁמַיִם וְהַסֵּךְ נְסָכִים לֵאלֹהִים אֲחֵרִים לְמַעַן הַכְעִסֵנִי` (ירמיהו ז:יח)
`לָכֵן אֶתֵּן אֶת נְשֵׁיהֶם לַאֲחֵרִים שְׂדוֹתֵיהֶם לְיוֹרְשִׁים כִּי מִקָּטֹן וְעַד גָּדוֹל כֻּלֹּה בֹּצֵעַ בָּצַע מִנָּבִיא וְעַד כֹּהֵן כֻּלֹּה עֹשֶׂה שָּׁקֶר` (ירמיהו ח:י)
`כִּי שְׁמַעְנָה נָשִׁים דְּבַר יְהוָה וְתִקַּח אָזְנְכֶם דְּבַר פִּיו וְלַמֵּדְנָה בְנוֹתֵיכֶם נֶהִי וְאִשָּׁה רְעוּתָהּ קִינָה` (ירמיהו ט:כ)
`מַה תֹּאמְרִי כִּי יִפְקֹד עָלַיִךְ וְאַתְּ לִמַּדְתְּ אֹתָם עָלַיִךְ אַלֻּפִים לְרֹאשׁ הֲלוֹא חֲבָלִים יֹאחֱזוּךְ כְּמוֹ אֵשֶׁת לֵדָה` (ירמיהו יג:כא)
`וְהָעָם אֲשֶׁר הֵמָּה נִבְּאִים לָהֶם יִהְיוּ מֻשְׁלָכִים בְּחֻצוֹת יְרוּשָׁלִַם מִפְּנֵי הָרָעָב וְהַחֶרֶב וְאֵין מְקַבֵּר לָהֵמָּה הֵמָּה נְשֵׁיהֶם וּבְנֵיהֶם וּבְנֹתֵיהֶם וְשָׁפַכְתִּי עֲלֵיהֶם אֶת רָעָתָם` (ירמיהו יד:טז)
`לֹא תִקַּח לְךָ אִשָּׁה וְלֹא יִהְיוּ לְךָ בָּנִים וּבָנוֹת בַּמָּקוֹם הַזֶּה` (ירמיהו טז:ב)
`לָכֵן תֵּן אֶת בְּנֵיהֶם לָרָעָב וְהַגִּרֵם עַל יְדֵי חֶרֶב וְתִהְיֶנָה נְשֵׁיהֶם שַׁכֻּלוֹת וְאַלְמָנוֹת וְאַנְשֵׁיהֶם יִהְיוּ הֲרֻגֵי מָוֶת בַּחוּרֵיהֶם מֻכֵּי חֶרֶב בַּמִּלְחָמָה` (ירמיהו יח:כא)
`קְחוּ נָשִׁים וְהוֹלִידוּ בָּנִים וּבָנוֹת וּקְחוּ לִבְנֵיכֶם נָשִׁים וְאֶת בְּנוֹתֵיכֶם תְּנוּ לַאֲנָשִׁים וְתֵלַדְנָה בָּנִים וּבָנוֹת וּרְבוּ שָׁם וְאַל תִּמְעָטוּ` (ירמיהו כט:ו)
`יַעַן אֲשֶׁר עָשׂוּ נְבָלָה בְּיִשְׂרָאֵל וַיְנַאֲפוּ אֶת נְשֵׁי רֵעֵיהֶם וַיְדַבְּרוּ דָבָר בִּשְׁמִי שֶׁקֶר אֲשֶׁר לוֹא צִוִּיתִם וְאָנֹכִי הוּיֹדֵעַ וָעֵד נְאֻם יְהוָה` (ירמיהו כט:כג)
`וַנִּשְׁמַע בְּקוֹל יְהוֹנָדָב בֶּן רֵכָב אָבִינוּ לְכֹל אֲשֶׁר צִוָּנוּ לְבִלְתִּי שְׁתוֹת יַיִן כָּל יָמֵינוּ אֲנַחְנוּ נָשֵׁינוּ בָּנֵינוּ וּבְנֹתֵינוּ` (ירמיהו לה:ח)
`וְהִנֵּה כָל הַנָּשִׁים אֲשֶׁר נִשְׁאֲרוּ בְּבֵית מֶלֶךְ יְהוּדָה מוּצָאוֹת אֶל שָׂרֵי מֶלֶךְ בָּבֶל וְהֵנָּה אֹמְרוֹת הִסִּיתוּךָ וְיָכְלוּ לְךָ אַנְשֵׁי שְׁלֹמֶךָ הָטְבְּעוּ בַבֹּץ רַגְלֶךָ נָסֹגוּ אָחוֹר` (ירמיהו לח:כב)
`וְאֶת כָּל נָשֶׁיךָ וְאֶת בָּנֶיךָ מוֹצִאִים אֶל הַכַּשְׂדִּים וְאַתָּה לֹא תִמָּלֵט מִיָּדָם כִּי בְיַד מֶלֶךְ בָּבֶל תִּתָּפֵשׂ וְאֶת הָעִיר הַזֹּאת תִּשְׂרֹף בָּאֵשׁ` (ירמיהו לח:כג)
`וַיִּשְׁמְעוּ כָל שָׂרֵי הַחֲיָלִים אֲשֶׁר בַּשָּׂדֶה הֵמָּה וְאַנְשֵׁיהֶם כִּי הִפְקִיד מֶלֶךְ בָּבֶל אֶת גְּדַלְיָהוּ בֶן אֲחִיקָם בָּאָרֶץ וְכִי הִפְקִיד אִתּוֹ אֲנָשִׁים וְנָשִׁים וָטָף וּמִדַּלַּת הָאָרֶץ מֵאֲשֶׁר לֹא הָגְלוּ בָּבֶלָה` (ירמיהו מ:ז)
`וַיִּקַּח יוֹחָנָן בֶּן קָרֵחַ וְכָל שָׂרֵי הַחֲיָלִים אֲשֶׁר אִתּוֹ אֵת כָּל שְׁאֵרִית הָעָם אֲשֶׁר הֵשִׁיב מֵאֵת יִשְׁמָעֵאל בֶּן נְתַנְיָה מִן הַמִּצְפָּה אַחַר הִכָּה אֶת גְּדַלְיָה בֶּן אֲחִיקָם גְּבָרִים אַנְשֵׁי הַמִּלְחָמָה וְנָשִׁים וְטַף וְסָרִסִים אֲשֶׁר הֵשִׁיב מִגִּבְעוֹן` (ירמיהו מא:טז)
`אֶת הַגְּבָרִים וְאֶת הַנָּשִׁים וְאֶת הַטַּף וְאֶת בְּנוֹת הַמֶּלֶךְ וְאֵת כָּל הַנֶּפֶשׁ אֲשֶׁר הִנִּיחַ נְבוּזַרְאֲדָן רַב טַבָּחִים אֶת גְּדַלְיָהוּ בֶּן אֲחִיקָם בֶּן שָׁפָן וְאֵת יִרְמְיָהוּ הַנָּבִיא וְאֶת בָּרוּךְ בֶּן נֵרִיָּהוּ` (ירמיהו מג:ו)
`וְעַתָּה כֹּה אָמַר יְהוָה אֱלֹהֵי צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל לָמָה אַתֶּם עֹשִׂים רָעָה גְדוֹלָה אֶל נַפְשֹׁתֵכֶם לְהַכְרִית לָכֶם אִישׁ וְאִשָּׁה עוֹלֵל וְיוֹנֵק מִתּוֹךְ יְהוּדָה לְבִלְתִּי הוֹתִיר לָכֶם שְׁאֵרִית` (ירמיהו מד:ז)
`הַשְׁכַחְתֶּם אֶת רָעוֹת אֲבוֹתֵיכֶם וְאֶת רָעוֹת מַלְכֵי יְהוּדָה וְאֵת רָעוֹת נָשָׁיו וְאֵת רָעֹתֵכֶם וְאֵת רָעֹת נְשֵׁיכֶם אֲשֶׁר עָשׂוּ בְּאֶרֶץ יְהוּדָה וּבְחֻצוֹת יְרוּשָׁלִָם` (ירמיהו מד:ט)
`וַיַּעֲנוּ אֶת יִרְמְיָהוּ כָּל הָאֲנָשִׁים הַיֹּדְעִים כִּי מְקַטְּרוֹת נְשֵׁיהֶם לֵאלֹהִים אֲחֵרִים וְכָל הַנָּשִׁים הָעֹמְדוֹת קָהָל גָּדוֹל וְכָל הָעָם הַיֹּשְׁבִים בְּאֶרֶץ מִצְרַיִם בְּפַתְרוֹס לֵאמֹר` (ירמיהו מד:טו)
`וַיֹּאמֶר יִרְמְיָהוּ אֶל כָּל הָעָם עַל הַגְּבָרִים וְעַל הַנָּשִׁים וְעַל כָּל הָעָם הָעֹנִים אֹתוֹ דָּבָר לֵאמֹר` (ירמיהו מד:כ)
`וַיֹּאמֶר יִרְמְיָהוּ אֶל כָּל הָעָם וְאֶל כָּל הַנָּשִׁים שִׁמְעוּ דְּבַר יְהוָה כָּל יְהוּדָה אֲשֶׁר בְּאֶרֶץ מִצְרָיִם` (ירמיהו מד:כד)
`כֹּה אָמַר יְהוָה צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל לֵאמֹר אַתֶּם וּנְשֵׁיכֶם וַתְּדַבֵּרְנָה בְּפִיכֶם וּבִידֵיכֶם מִלֵּאתֶם לֵאמֹר עָשֹׂה נַעֲשֶׂה אֶת נְדָרֵינוּ אֲשֶׁר נָדַרְנוּ לְקַטֵּר לִמְלֶכֶת הַשָּׁמַיִם וּלְהַסֵּךְ לָהּ נְסָכִים הָקֵים תָּקִימְנָה אֶת נִדְרֵיכֶם וְעָשֹׂה תַעֲשֶׂינָה אֶת נִדְרֵיכֶם` (ירמיהו מד:כה)
`נִלְכְּדָה הַקְּרִיּוֹת וְהַמְּצָדוֹת נִתְפָּשָׂה וְהָיָה לֵב גִּבּוֹרֵי מוֹאָב בַּיּוֹם הַהוּא כְּלֵב אִשָּׁה מְצֵרָה` (ירמיהו מח:מא)
`הִנֵּה כַנֶּשֶׁר יַעֲלֶה וְיִדְאֶה וְיִפְרֹשׂ כְּנָפָיו עַל בָּצְרָה וְהָיָה לֵב גִּבּוֹרֵי אֱדוֹם בַּיּוֹם הַהוּא כְּלֵב אִשָּׁה מְצֵרָה` (ירמיהו מט:כב)
`חֶרֶב אֶל סוּסָיו וְאֶל רִכְבּוֹ וְאֶל כָּל הָעֶרֶב אֲשֶׁר בְּתוֹכָהּ וְהָיוּ לְנָשִׁים חֶרֶב אֶל אוֹצְרֹתֶיהָ וּבֻזָּזוּ` (ירמיהו נ:לז)
`וְנִפַּצְתִּי בְךָ אִישׁ וְאִשָּׁה וְנִפַּצְתִּי בְךָ זָקֵן וָנָעַר וְנִפַּצְתִּי בְךָ בָּחוּר וּבְתוּלָה` (ירמיהו נא:כב)
`חָדְלוּ גִבּוֹרֵי בָבֶל לְהִלָּחֵם יָשְׁבוּ בַּמְּצָדוֹת נָשְׁתָה גְבוּרָתָם הָיוּ לְנָשִׁים הִצִּיתוּ מִשְׁכְּנֹתֶיהָ נִשְׁבְּרוּ בְרִיחֶיהָ` (ירמיהו נא:ל)
`חֹבְרֹת אִשָּׁה אֶל אֲחוֹתָהּ כַּנְפֵיהֶם לֹא יִסַּבּוּ בְלֶכְתָּן אִישׁ אֶל עֵבֶר פָּנָיו יֵלֵכוּ` (יחזקאל א:ט)
`וְתַחַת הָרָקִיעַ כַּנְפֵיהֶם יְשָׁרוֹת אִשָּׁה אֶל אֲחוֹתָהּ לְאִישׁ שְׁתַּיִם מְכַסּוֹת לָהֵנָּה וּלְאִישׁ שְׁתַּיִם מְכַסּוֹת לָהֵנָּה אֵת גְּוִיֹּתֵיהֶם` (יחזקאל א:כג)
`וְקוֹל כַּנְפֵי הַחַיּוֹת מַשִּׁיקוֹת אִשָּׁה אֶל אֲחוֹתָהּ וְקוֹל הָאוֹפַנִּים לְעֻמָּתָם וְקוֹל רַעַשׁ גָּדוֹל` (יחזקאל ג:יג)
`וַיָּבֵא אֹתִי אֶל פֶּתַח שַׁעַר בֵּית יְהוָה אֲשֶׁר אֶל הַצָּפוֹנָה וְהִנֵּה שָׁם הַנָּשִׁים יֹשְׁבוֹת מְבַכּוֹת אֶת הַתַּמּוּז` (יחזקאל ח:יד)
`זָקֵן בָּחוּר וּבְתוּלָה וְטַף וְנָשִׁים תַּהַרְגוּ לְמַשְׁחִית וְעַל כָּל אִישׁ אֲשֶׁר עָלָיו הַתָּו אַל תִּגַּשׁוּ וּמִמִּקְדָּשִׁי תָּחֵלּוּ וַיָּחֵלּוּ בָּאֲנָשִׁים הַזְּקֵנִים אֲשֶׁר לִפְנֵי הַבָּיִת` (יחזקאל ט:ו)
`מָה אֲמֻלָה לִבָּתֵךְ נְאֻם אֲדֹנָי יְהוִה בַּעֲשׂוֹתֵךְ אֶת כָּל אֵלֶּה מַעֲשֵׂה אִשָּׁה זוֹנָה שַׁלָּטֶת` (יחזקאל טז:ל)
`הָאִשָּׁה הַמְּנָאָפֶת תַּחַת אִישָׁהּ תִּקַּח אֶת זָרִים` (יחזקאל טז:לב)
`וַיְהִי בָךְ הֵפֶךְ מִן הַנָּשִׁים בְּתַזְנוּתַיִךְ וְאַחֲרַיִךְ לֹא זוּנָּה וּבְתִתֵּךְ אֶתְנָן וְאֶתְנַן לֹא נִתַּן לָךְ וַתְּהִי לְהֶפֶךְ` (יחזקאל טז:לד)
`וְשָׂרְפוּ בָתַּיִךְ בָּאֵשׁ וְעָשׂוּ בָךְ שְׁפָטִים לְעֵינֵי נָשִׁים רַבּוֹת וְהִשְׁבַּתִּיךְ מִזּוֹנָה וְגַם אֶתְנַן לֹא תִתְּנִי עוֹד` (יחזקאל טז:מא)
`אֶל הֶהָרִים לֹא אָכָל וְעֵינָיו לֹא נָשָׂא אֶל גִּלּוּלֵי בֵּית יִשְׂרָאֵל וְאֶת אֵשֶׁת רֵעֵהוּ לֹא טִמֵּא וְאֶל אִשָּׁה נִדָּה לֹא יִקְרָב` (יחזקאל יח:ו)
`וְהוּא אֶת כָּל אֵלֶּה לֹא עָשָׂה כִּי גַם אֶל הֶהָרִים אָכַל וְאֶת אֵשֶׁת רֵעֵהוּ טִמֵּא` (יחזקאל יח:יא)
`עַל הֶהָרִים לֹא אָכָל וְעֵינָיו לֹא נָשָׂא אֶל גִּלּוּלֵי בֵּית יִשְׂרָאֵל אֶת אֵשֶׁת רֵעֵהוּ לֹא טִמֵּא` (יחזקאל יח:טו)
`וְאִישׁ אֶת אֵשֶׁת רֵעֵהוּ עָשָׂה תּוֹעֵבָה וְאִישׁ אֶת כַּלָּתוֹ טִמֵּא בְזִמָּה וְאִישׁ אֶת אֲחֹתוֹ בַת אָבִיו עִנָּה בָךְ` (יחזקאל כב:יא)
`בֶּן אָדָם שְׁתַּיִם נָשִׁים בְּנוֹת אֵם אַחַת הָיוּ` (יחזקאל כג:ב)
`הֵמָּה גִּלּוּ עֶרְוָתָהּ בָּנֶיהָ וּבְנוֹתֶיהָ לָקָחוּ וְאוֹתָהּ בַּחֶרֶב הָרָגוּ וַתְּהִי שֵׁם לַנָּשִׁים וּשְׁפוּטִים עָשׂוּ בָהּ` (יחזקאל כג:י)
`וַיָּבוֹא אֵלֶיהָ כְּבוֹא אֶל אִשָּׁה זוֹנָה כֵּן בָּאוּ אֶל אָהֳלָה וְאֶל אָהֳלִיבָה אִשֹּׁת הַזִּמָּה` (יחזקאל כג:מד)
`וְהִשְׁבַּתִּי זִמָּה מִן הָאָרֶץ וְנִוַּסְּרוּ כָּל הַנָּשִׁים וְלֹא תַעֲשֶׂינָה כְּזִמַּתְכֶנָה` (יחזקאל כג:מח)
`וָאֲדַבֵּר אֶל הָעָם בַּבֹּקֶר וַתָּמָת אִשְׁתִּי בָּעָרֶב וָאַעַשׂ בַּבֹּקֶר כַּאֲשֶׁר צֻוֵּיתִי` (יחזקאל כד:יח)
`עֲמַדְתֶּם עַל חַרְבְּכֶם עֲשִׂיתֶן תּוֹעֵבָה וְאִישׁ אֶת אֵשֶׁת רֵעֵהוּ טִמֵּאתֶם וְהָאָרֶץ תִּירָשׁוּ` (יחזקאל לג:כו)
`וְאַלְמָנָה וּגְרוּשָׁה לֹא יִקְחוּ לָהֶם לְנָשִׁים כִּי אִם בְּתוּלֹת מִזֶּרַע בֵּית יִשְׂרָאֵל וְהָאַלְמָנָה אֲשֶׁר תִּהְיֶה אַלְמָנָה מִכֹּהֵן יִקָּחוּ` (יחזקאל מד:כב)
`תְּחִלַּת דִּבֶּר יְהוָה בְּהוֹשֵׁעַ וַיֹּאמֶר יְהוָה אֶל הוֹשֵׁעַ לֵךְ קַח לְךָ אֵשֶׁת זְנוּנִים וְיַלְדֵי זְנוּנִים כִּי זָנֹה תִזְנֶה הָאָרֶץ מֵאַחֲרֵי יְהוָה` (הושע א:ב)
`רִיבוּ בְאִמְּכֶם רִיבוּ כִּי הִיא לֹא אִשְׁתִּי וְאָנֹכִי לֹא אִישָׁהּ וְתָסֵר זְנוּנֶיהָ מִפָּנֶיה וְנַאֲפוּפֶיהָ מִבֵּין שָׁדֶיהָ` (הושע ב:ב)
`וַיֹּאמֶר יְהוָה אֵלַי עוֹד לֵךְ אֱ‍הַב אִשָּׁה אֲהֻבַת רֵעַ וּמְנָאָפֶת כְּאַהֲבַת יְהוָה אֶת בְּנֵי יִשְׂרָאֵל וְהֵם פֹּנִים אֶל אֱלֹהִים אֲחֵרִים וְאֹהֲבֵי אֲשִׁישֵׁי עֲנָבִים` (הושע ג:א)
`וַיִּבְרַח יַעֲקֹב שְׂדֵה אֲרָם וַיַּעֲבֹד יִשְׂרָאֵל בְּאִשָּׁה וּבְאִשָּׁה שָׁמָר` (הושע יב:יב)
`וּפְרָצִים תֵּצֶאנָה אִשָּׁה נֶגְדָּהּ וְהִשְׁלַכְתֶּנָה הַהַרְמוֹנָה נְאֻם יְהוָה` (עמוס ד:ג)
`לָכֵן כֹּה אָמַר יְהוָה אִשְׁתְּךָ בָּעִיר תִּזְנֶה וּבָנֶיךָ וּבְנֹתֶיךָ בַּחֶרֶב יִפֹּלוּ וְאַדְמָתְךָ בַּחֶבֶל תְּחֻלָּק וְאַתָּה עַל אֲדָמָה טְמֵאָה תָּמוּת וְיִשְׂרָאֵל גָּלֹה יִגְלֶה מֵעַל אַדְמָתוֹ` (עמוס ז:יז)
`נְשֵׁי עַמִּי תְּגָרְשׁוּן מִבֵּית תַּעֲנֻגֶיהָ מֵעַל עֹלָלֶיהָ תִּקְחוּ הֲדָרִי לְעוֹלָם` (מיכה ב:ט)
`הִנֵּה עַמֵּךְ נָשִׁים בְּקִרְבֵּךְ לְאֹיְבַיִךְ פָּתוֹחַ נִפְתְּחוּ שַׁעֲרֵי אַרְצֵךְ אָכְלָה אֵשׁ בְּרִיחָיִך` (נחום ג:יג)
`וְהִנֵּה כִּכַּר עֹפֶרֶת נִשֵּׂאת וְזֹאת אִשָּׁה אַחַת יוֹשֶׁבֶת בְּתוֹךְ הָאֵיפָה` (זכריה ה:ז)
`וָאֶשָּׂא עֵינַי וָאֵרֶא וְהִנֵּה שְׁתַּיִם נָשִׁים יוֹצְאוֹת וְרוּחַ בְּכַנְפֵיהֶם וְלָהֵנָּה כְנָפַיִם כְּכַנְפֵי הַחֲסִידָה וַתִּשֶּׂאנָה אֶת הָאֵיפָה בֵּין הָאָרֶץ וּבֵין הַשָּׁמָיִם` (זכריה ה:ט)
`וָאֹמַר לֹא אֶרְעֶה אֶתְכֶם הַמֵּתָה תָמוּת וְהַנִּכְחֶדֶת תִּכָּחֵד וְהַנִּשְׁאָרוֹת תֹּאכַלְנָה אִשָּׁה אֶת בְּשַׂר רְעוּתָהּ` (זכריה יא:ט)
`וְסָפְדָה הָאָרֶץ מִשְׁפָּחוֹת מִשְׁפָּחוֹת לְבָד מִשְׁפַּחַת בֵּית דָּוִיד לְבָד וּנְשֵׁיהֶם לְבָד מִשְׁפַּחַת בֵּית נָתָן לְבָד וּנְשֵׁיהֶם לְבָד` (זכריה יב:יב)
`מִשְׁפַּחַת בֵּית לֵוִי לְבָד וּנְשֵׁיהֶם לְבָד מִשְׁפַּחַת הַשִּׁמְעִי לְבָד וּנְשֵׁיהֶם לְבָד` (זכריה יב:יג)
`כֹּל הַמִּשְׁפָּחוֹת הַנִּשְׁאָרוֹת מִשְׁפָּחֹת מִשְׁפָּחֹת לְבָד וּנְשֵׁיהֶם לְבָד` (זכריה יב:יד)
`וְאָסַפְתִּי אֶת כָּל הַגּוֹיִם אֶל יְרוּשָׁלִַם לַמִּלְחָמָה וְנִלְכְּדָה הָעִיר וְנָשַׁסּוּ הַבָּתִּים וְהַנָּשִׁים תִּשָּׁגַלְנָה וְיָצָא חֲצִי הָעִיר בַּגּוֹלָה וְיֶתֶר הָעָם לֹא יִכָּרֵת מִן הָעִיר` (זכריה יד:ב)
`וַאֲמַרְתֶּם עַל מָה עַל כִּי יְהוָה הֵעִיד בֵּינְךָ וּבֵין אֵשֶׁת נְעוּרֶיךָ אֲשֶׁר אַתָּה בָּגַדְתָּה בָּהּ וְהִיא חֲבֶרְתְּךָ וְאֵשֶׁת בְּרִיתֶךָ` (מלאכי ב:יד)
`וְלֹא אֶחָד עָשָׂה וּשְׁאָר רוּחַ לוֹ וּמָה הָאֶחָד מְבַקֵּשׁ זֶרַע אֱלֹהִים וְנִשְׁמַרְתֶּם בְּרוּחֲכֶם וּבְאֵשֶׁת נְעוּרֶיךָ אַל יִבְגֹּד` (מלאכי ב:טו)
`וַיָּמָת בַּעַל חָנָן וַיִּמְלֹךְ תַּחְתָּיו הֲדַד וְשֵׁם עִירוֹ פָּעִי וְשֵׁם אִשְׁתּוֹ מְהֵיטַבְאֵל בַּת מַטְרֵד בַּת מֵי זָהָב` (דברי-הימים-א א:נ)
`וְכָלֵב בֶּן חֶצְרוֹן הוֹלִיד אֶת עֲזוּבָה אִשָּׁה וְאֶת יְרִיעוֹת וְאֵלֶּה בָנֶיהָ יֵשֶׁר וְשׁוֹבָב וְאַרְדּוֹן` (דברי-הימים-א ב:יח)
`וְאַחַר מוֹת חֶצְרוֹן בְּכָלֵב אֶפְרָתָה וְאֵשֶׁת חֶצְרוֹן אֲבִיָּה וַתֵּלֶד לוֹ אֶת אַשְׁחוּר אֲבִי תְקוֹעַ` (דברי-הימים-א ב:כד)
`וַתְּהִי אִשָּׁה אַחֶרֶת לִירַחְמְאֵל וּשְׁמָהּ עֲטָרָה הִיא אֵם אוֹנָם` (דברי-הימים-א ב:כו)
`וְשֵׁם אֵשֶׁת אֲבִישׁוּר אֲבִיהָיִל וַתֵּלֶד לוֹ אֶת אַחְבָּן וְאֶת מוֹלִיד` (דברי-הימים-א ב:כט)
`וַיִּתֵּן שֵׁשָׁן אֶת בִּתּוֹ לְיַרְחָע עַבְדּוֹ לְאִשָּׁה וַתֵּלֶד לוֹ אֶת עַתָּי` (דברי-הימים-א ב:לה)
`הַחֲמִישִׁי שְׁפַטְיָה לַאֲבִיטָל הַשִּׁשִּׁי יִתְרְעָם לְעֶגְלָה אִשְׁתּוֹ` (דברי-הימים-א ג:ג)
`וּלְאַשְׁחוּר אֲבִי תְקוֹעַ הָיוּ שְׁתֵּי נָשִׁים חֶלְאָה וְנַעֲרָה` (דברי-הימים-א ד:ה)
`וְאִשְׁתּוֹ הַיְהֻדִיָּה יָלְדָה אֶת יֶרֶד אֲבִי גְדוֹר וְאֶת חֶבֶר אֲבִי שׂוֹכוֹ וְאֶת יְקוּתִיאֵל אֲבִי זָנוֹחַ וְאֵלֶּה בְּנֵי בִּתְיָה בַת פַּרְעֹה אֲשֶׁר לָקַח מָרֶד` (דברי-הימים-א ד:יח)
`וּבְנֵי אֵשֶׁת הוֹדִיָּה אֲחוֹת נַחַם אֲבִי קְעִילָה הַגַּרְמִי וְאֶשְׁתְּמֹעַ הַמַּעֲכָתִי` (דברי-הימים-א ד:יט)
`וַעֲלֵיהֶם לְתֹלְדוֹתָם לְבֵית אֲבוֹתָם גְּדוּדֵי צְבָא מִלְחָמָה שְׁלֹשִׁים וְשִׁשָּׁה אָלֶף כִּי הִרְבּוּ נָשִׁים וּבָנִים` (דברי-הימים-א ז:ד)
`וּמָכִיר לָקַח אִשָּׁה לְחֻפִּים וּלְשֻׁפִּים וְשֵׁם אֲחֹתוֹ מַעֲכָה וְשֵׁם הַשֵּׁנִי צְלָפְחָד וַתִּהְיֶנָה לִצְלָפְחָד בָּנוֹת` (דברי-הימים-א ז:טו)
`וַתֵּלֶד מַעֲכָה אֵשֶׁת מָכִיר בֵּן וַתִּקְרָא שְׁמוֹ פֶּרֶשׁ וְשֵׁם אָחִיו שָׁרֶשׁ וּבָנָיו אוּלָם וָרָקֶם` (דברי-הימים-א ז:טז)
`וַיָּבֹא אֶל אִשְׁתּוֹ וַתַּהַר וַתֵּלֶד בֵּן וַיִּקְרָא אֶת שְׁמוֹ בְּרִיעָה כִּי בְרָעָה הָיְתָה בְּבֵיתוֹ` (דברי-הימים-א ז:כג)
`וְשַׁחֲרַיִם הוֹלִיד בִּשְׂדֵה מוֹאָב מִן שִׁלְחוֹ אֹתָם חוּשִׁים וְאֶת בַּעֲרָא נָשָׁיו` (דברי-הימים-א ח:ח)
`וַיּוֹלֶד מִן חֹדֶשׁ אִשְׁתּוֹ אֶת יוֹבָב וְאֶת צִבְיָא וְאֶת מֵישָׁא וְאֶת מַלְכָּם` (דברי-הימים-א ח:ט)
`וּבְגִבְעוֹן יָשְׁבוּ אֲבִי גִבְעוֹן וְשֵׁם אִשְׁתּוֹ מַעֲכָה` (דברי-הימים-א ח:כט)
`וּבְגִבְעוֹן יָשְׁבוּ אֲבִי גִבְעוֹן יְעוּאֵל וְשֵׁם אִשְׁתּוֹ מַעֲכָה` (דברי-הימים-א ט:לה)
`וַיִּקַּח דָּוִיד עוֹד נָשִׁים בִּירוּשָׁלִָם וַיּוֹלֶד דָּוִיד עוֹד בָּנִים וּבָנוֹת` (דברי-הימים-א יד:ג)
`וַיְחַלֵּק לְכָל אִישׁ יִשְׂרָאֵל מֵאִישׁ וְעַד אִשָּׁה לְאִישׁ כִּכַּר לֶחֶם וְאֶשְׁפָּר וַאֲשִׁישָׁה` (דברי-הימים-א טז:ג)
`בֶּן אִשָּׁה מִן בְּנוֹת דָּן וְאָבִיו אִישׁ צֹרִי יוֹדֵעַ לַעֲשׂוֹת בַּזָּהָב וּבַכֶּסֶף בַּנְּחֹשֶׁת בַּבַּרְזֶל בָּאֲבָנִים וּבָעֵצִים בָּאַרְגָּמָן בַּתְּכֵלֶת וּבַבּוּץ וּבַכַּרְמִיל וּלְפַתֵּחַ כָּל פִּתּוּחַ וְלַחְשֹׁב כָּל מַחֲשָׁבֶת אֲשֶׁר יִנָּתֶן לוֹ עִם חֲכָמֶיךָ וְחַכְמֵי אֲדֹנִי דָּוִיד אָבִיךָ` (דברי-הימים-ב ב:יד)
`וְאֶת בַּת פַּרְעֹה הֶעֱלָה שְׁלֹמֹה מֵעִיר דָּוִיד לַבַּיִת אֲשֶׁר בָּנָה לָהּ כִּי אָמַר לֹא תֵשֵׁב אִשָּׁה לִי בְּבֵית דָּוִיד מֶלֶךְ יִשְׂרָאֵל כִּי קֹדֶשׁ הֵמָּה אֲשֶׁר בָּאָה אֲלֵיהֶם אֲרוֹן יְהוָה` (דברי-הימים-ב ח:יא)
`וַיִּקַּח לוֹ רְחַבְעָם אִשָּׁה אֶת מָחֲלַת בֶּן יְרִימוֹת בֶּן דָּוִיד אֲבִיהַיִל בַּת אֱלִיאָב בֶּן יִשָׁי` (דברי-הימים-ב יא:יח)
`וַיֶּאֱהַב רְחַבְעָם אֶת מַעֲכָה בַת אַבְשָׁלוֹם מִכָּל נָשָׁיו וּפִילַגְשָׁיו כִּי נָשִׁים שְׁמוֹנֶה עֶשְׂרֵה נָשָׂא וּפִילַגְשִׁים שִׁשִּׁים וַיּוֹלֶד עֶשְׂרִים וּשְׁמוֹנָה בָּנִים וְשִׁשִּׁים בָּנוֹת` (דברי-הימים-ב יא:כא)
`וַיָּבֶן וַיִּפְרֹץ מִכָּל בָּנָיו לְכָל אַרְצוֹת יְהוּדָה וּבִנְיָמִן לְכֹל עָרֵי הַמְּצֻרוֹת וַיִּתֵּן לָהֶם הַמָּזוֹן לָרֹב וַיִּשְׁאַל הֲמוֹן נָשִׁים` (דברי-הימים-ב יא:כג)
`וַיִּתְחַזֵּק אֲבִיָּהוּ וַיִּשָּׂא לוֹ נָשִׁים אַרְבַּע עֶשְׂרֵה וַיּוֹלֶד עֶשְׂרִים וּשְׁנַיִם בָּנִים וְשֵׁשׁ עֶשְׂרֵה בָּנוֹת` (דברי-הימים-ב יג:כא)
`וְכֹל אֲשֶׁר לֹא יִדְרֹשׁ לַיהוָה אֱלֹהֵי יִשְׂרָאֵל יוּמָת לְמִן קָטֹן וְעַד גָּדוֹל לְמֵאִישׁ וְעַד אִשָּׁה` (דברי-הימים-ב טו:יג)
`וְכָל יְהוּדָה עֹמְדִים לִפְנֵי יְהוָה גַּם טַפָּם נְשֵׁיהֶם וּבְנֵיהֶם` (דברי-הימים-ב כ:יג)
`וַיֵּלֶךְ בְּדֶרֶךְ מַלְכֵי יִשְׂרָאֵל כַּאֲשֶׁר עָשׂוּ בֵּית אַחְאָב כִּי בַּת אַחְאָב הָיְתָה לּוֹ אִשָּׁה וַיַּעַשׂ הָרַע בְּעֵינֵי יְהוָה` (דברי-הימים-ב כא:ו)
`הִנֵּה יְהוָה נֹגֵף מַגֵּפָה גְדוֹלָה בְּעַמֶּךָ וּבְבָנֶיךָ וּבְנָשֶׁיךָ וּבְכָל רְכוּשֶׁךָ` (דברי-הימים-ב כא:יד)
`וַיַּעֲלוּ בִיהוּדָה וַיִּבְקָעוּהָ וַיִּשְׁבּוּ אֵת כָּל הָרְכוּשׁ הַנִּמְצָא לְבֵית הַמֶּלֶךְ וְגַם בָּנָיו וְנָשָׁיו וְלֹא נִשְׁאַר לוֹ בֵּן כִּי אִם יְהוֹאָחָז קְטֹן בָּנָיו` (דברי-הימים-ב כא:יז)
`וַתִּקַּח יְהוֹשַׁבְעַת בַּת הַמֶּלֶךְ אֶת יוֹאָשׁ בֶּן אֲחַזְיָהוּ וַתִּגְנֹב אֹתוֹ מִתּוֹךְ בְּנֵי הַמֶּלֶךְ הַמּוּמָתִים וַתִּתֵּן אֹתוֹ וְאֶת מֵינִקְתּוֹ בַּחֲדַר הַמִּטּוֹת וַתַּסְתִּירֵהוּ יְהוֹשַׁבְעַת בַּת הַמֶּלֶךְ יְהוֹרָם אֵשֶׁת יְהוֹיָדָע הַכֹּהֵן כִּי הִיא הָיְתָה אֲחוֹת אֲחַזְיָהוּ מִפְּנֵי עֲתַלְיָהוּ וְלֹא הֱמִיתָתְהוּ` (דברי-הימים-ב כב:יא)
`וַיִּשָּׂא לוֹ יְהוֹיָדָע נָשִׁים שְׁתָּיִם וַיּוֹלֶד בָּנִים וּבָנוֹת` (דברי-הימים-ב כד:ג)
`וַיִּשְׁלַח יוֹאָשׁ מֶלֶךְ יִשְׂרָאֵל אֶל אֲמַצְיָהוּ מֶלֶךְ יְהוּדָה לֵאמֹר הַחוֹחַ אֲשֶׁר בַּלְּבָנוֹן שָׁלַח אֶל הָאֶרֶז אֲשֶׁר בַּלְּבָנוֹן לֵאמֹר תְּנָה אֶת בִּתְּךָ לִבְנִי לְאִשָּׁה וַתַּעֲבֹר חַיַּת הַשָּׂדֶה אֲשֶׁר בַּלְּבָנוֹן וַתִּרְמֹס אֶת הַחוֹחַ` (דברי-הימים-ב כה:יח)
`וַיִּשְׁבּוּ בְנֵי יִשְׂרָאֵל מֵאֲחֵיהֶם מָאתַיִם אֶלֶף נָשִׁים בָּנִים וּבָנוֹת וְגַם שָׁלָל רָב בָּזְזוּ מֵהֶם וַיָּבִיאוּ אֶת הַשָּׁלָל לְשֹׁמְרוֹן` (דברי-הימים-ב כח:ח)
`וְהִנֵּה נָפְלוּ אֲבוֹתֵינוּ בֶּחָרֶב וּבָנֵינוּ וּבְנוֹתֵינוּ וְנָשֵׁינוּ בַּשְּׁבִי עַל זֹאת` (דברי-הימים-ב כט:ט)
`וּלְהִתְיַחֵשׂ בְּכָל טַפָּם נְשֵׁיהֶם וּבְנֵיהֶם וּבְנוֹתֵיהֶם לְכָל קָהָל כִּי בֶאֱמוּנָתָם יִתְקַדְּשׁוּ קֹדֶשׁ` (דברי-הימים-ב לא:יח)
`וַיֵּלֶךְ חִלְקִיָּהוּ וַאֲשֶׁר הַמֶּלֶךְ אֶל חֻלְדָּה הַנְּבִיאָה אֵשֶׁת שַׁלֻּם בֶּן תּוֹקְהַת בֶּן חַסְרָה שׁוֹמֵר הַבְּגָדִים וְהִיא יוֹשֶׁבֶת בִּירוּשָׁלִַם בַּמִּשְׁנֶה וַיְדַבְּרוּ אֵלֶיהָ כָּזֹאת` (דברי-הימים-ב לד:כב)
`כְּמוֹ שַׁבְּלוּל תֶּמֶס יַהֲלֹךְ נֵפֶל אֵשֶׁת בַּל חָזוּ שָׁמֶשׁ` (תהילים נח:ח)
`יִהְיוּ בָנָיו יְתוֹמִים וְאִשְׁתּוֹ אַלְמָנָה` (תהילים קט:ט)
`אֶשְׁתְּךָ כְּגֶפֶן פֹּרִיָּה בְּיַרְכְּתֵי בֵיתֶךָ בָּנֶיךָ כִּשְׁתִלֵי זֵיתִים סָבִיב לְשֻׁלְחָנֶךָ` (תהילים קכח:ג)
`וַתֹּאמֶר לוֹ אִשְׁתּוֹ עֹדְךָ מַחֲזִיק בְּתֻמָּתֶךָ בָּרֵךְ אֱלֹהִים וָמֻת` (איוב ב:ט)
`אָדָם יְלוּד אִשָּׁה קְצַר יָמִים וּשְׂבַע רֹגֶז` (איוב יד:א)
`מָה אֱנוֹשׁ כִּי יִזְכֶּה וְכִי יִצְדַּק יְלוּד אִשָּׁה` (איוב טו:יד)
`רוּחִי זָרָה לְאִשְׁתִּי וְחַנֹּתִי לִבְנֵי בִטְנִי` (איוב יט:יז)
`וּמַה יִּצְדַּק אֱנוֹשׁ עִם אֵל וּמַה יִּזְכֶּה יְלוּד אִשָּׁה` (איוב כה:ד)
`אִם נִפְתָּה לִבִּי עַל אִשָּׁה וְעַל פֶּתַח רֵעִי אָרָבְתִּי` (איוב לא:ט)
`תִּטְחַן לְאַחֵר אִשְׁתִּי וְעָלֶיהָ יִכְרְעוּן אֲחֵרִין` (איוב לא:י)
`וְלֹא נִמְצָא נָשִׁים יָפוֹת כִּבְנוֹת אִיּוֹב בְּכָל הָאָרֶץ וַיִּתֵּן לָהֶם אֲבִיהֶם נַחֲלָה בְּתוֹךְ אֲחֵיהֶם` (איוב מב:טו)
`לְהַצִּילְךָ מֵאִשָּׁה זָרָה מִנָּכְרִיָּה אֲמָרֶיהָ הֶחֱלִיקָה` (משלי ב:טז)
`יְהִי מְקוֹרְךָ בָרוּךְ וּשְׂמַח מֵאֵשֶׁת נְעוּרֶךָ` (משלי ה:יח)
`לִשְׁמָרְךָ מֵאֵשֶׁת רָע מֵחֶלְקַת לָשׁוֹן נָכְרִיָּה` (משלי ו:כד)
`כִּי בְעַד אִשָּׁה זוֹנָה עַד כִּכַּר לָחֶם וְאֵשֶׁת אִישׁ נֶפֶשׁ יְקָרָה תָצוּד` (משלי ו:כו)
`כֵּן הַבָּא אֶל אֵשֶׁת רֵעֵהוּ לֹא יִנָּקֶה כָּל הַנֹּגֵעַ בָּהּ` (משלי ו:כט)
`נֹאֵף אִשָּׁה חֲסַר לֵב מַשְׁחִית נַפְשׁוֹ הוּא יַעֲשֶׂנָּה` (משלי ו:לב)
`לִשְׁמָרְךָ מֵאִשָּׁה זָרָה מִנָּכְרִיָּה אֲמָרֶיהָ הֶחֱלִיקָה` (משלי ז:ה)
`וְהִנֵּה אִשָּׁה לִקְרָאתוֹ שִׁית זוֹנָה וּנְצֻרַת לֵב` (משלי ז:י)
`אֵשֶׁת כְּסִילוּת הֹמִיָּה פְּתַיּוּת וּבַל יָדְעָה מָּה` (משלי ט:יג)
`אֵשֶׁת חֵן תִּתְמֹךְ כָּבוֹד וְעָרִיצִים יִתְמְכוּ עֹשֶׁר` (משלי יא:טז)
`נֶזֶם זָהָב בְּאַף חֲזִיר אִשָּׁה יָפָה וְסָרַת טָעַם` (משלי יא:כב)
`אֵשֶׁת חַיִל עֲטֶרֶת בַּעְלָהּ וּכְרָקָב בְּעַצְמוֹתָיו מְבִישָׁה` (משלי יב:ד)
`חַכְמוֹת נָשִׁים בָּנְתָה בֵיתָהּ וְאִוֶּלֶת בְּיָדֶיהָ תֶהֶרְסֶנּוּ` (משלי יד:א)
`מָצָא אִשָּׁה מָצָא טוֹב וַיָּפֶק רָצוֹן מֵיְהוָה` (משלי יח:כב)
`הַוֺּת לְאָבִיו בֵּן כְּסִיל וְדֶלֶף טֹרֵד מִדְיְנֵי אִשָּׁה` (משלי יט:יג)
`בַּיִת וָהוֹן נַחֲלַת אָבוֹת וּמֵיְהוָה אִשָּׁה מַשְׂכָּלֶת` (משלי יט:יד)
`טוֹב לָשֶׁבֶת עַל פִּנַּת גָּג מֵאֵשֶׁת מִדְיָנִים וּבֵית חָבֶר` (משלי כא:ט)
`טוֹב שֶׁבֶת בְּאֶרֶץ מִדְבָּר מֵאֵשֶׁת מְדוֹנִים וָכָעַס` (משלי כא:יט)
`טוֹב שֶׁבֶת עַל פִּנַּת גָּג מֵאֵשֶׁת מְדוֹנִים וּבֵית חָבֶר` (משלי כה:כד)
`דֶּלֶף טוֹרֵד בְּיוֹם סַגְרִיר וְאֵשֶׁת מְדוֹנִים נִשְׁתָּוָה` (משלי כז:טו)
`כֵּן דֶּרֶךְ אִשָּׁה מְנָאָפֶת אָכְלָה וּמָחֲתָה פִיהָ וְאָמְרָה לֹא פָעַלְתִּי אָוֶן` (משלי ל:כ)
`אַל תִּתֵּן לַנָּשִׁים חֵילֶךָ וּדְרָכֶיךָ לַמְחוֹת מְלָכִין` (משלי לא:ג)
`אֵשֶׁת חַיִל מִי יִמְצָא וְרָחֹק מִפְּנִינִים מִכְרָהּ` (משלי לא:י)
`שֶׁקֶר הַחֵן וְהֶבֶל הַיֹּפִי אִשָּׁה יִרְאַת יְהוָה הִיא תִתְהַלָּל` (משלי לא:ל)
`וַיְהִי בִּימֵי שְׁפֹט הַשֹּׁפְטִים וַיְהִי רָעָב בָּאָרֶץ וַיֵּלֶךְ אִישׁ מִבֵּית לֶחֶם יְהוּדָה לָגוּר בִּשְׂדֵי מוֹאָב הוּא וְאִשְׁתּוֹ וּשְׁנֵי בָנָיו` (רות א:א)
`וְשֵׁם הָאִישׁ אֱ‍לִימֶלֶךְ וְשֵׁם אִשְׁתּוֹ נָעֳמִי וְשֵׁם שְׁנֵי בָנָיו מַחְלוֹן וְכִלְיוֹן אֶפְרָתִים מִבֵּית לֶחֶם יְהוּדָה וַיָּבֹאוּ שְׂדֵי מוֹאָב וַיִּהְיוּ שָׁם` (רות א:ב)
`וַיִּשְׂאוּ לָהֶם נָשִׁים מֹאֲבִיּוֹת שֵׁם הָאַחַת עָרְפָּה וְשֵׁם הַשֵּׁנִית רוּת וַיֵּשְׁבוּ שָׁם כְּעֶשֶׂר שָׁנִים` (רות א:ד)
`וַיָּמוּתוּ גַם שְׁנֵיהֶם מַחְלוֹן וְכִלְיוֹן וַתִּשָּׁאֵר הָאִשָּׁה מִשְּׁנֵי יְלָדֶיהָ וּמֵאִישָׁהּ` (רות א:ה)
`וַתֹּאמֶר נָעֳמִי לִשְׁתֵּי כַלֹּתֶיהָ לֵכְנָה שֹּׁבְנָה אִשָּׁה לְבֵית אִמָּהּ יַעֲשֶׂה יְהוָה עִמָּכֶם חֶסֶד כַּאֲשֶׁר עֲשִׂיתֶם עִם הַמֵּתִים וְעִמָּדִי` (רות א:ח)
`יִתֵּן יְהוָה לָכֶם וּמְצֶאןָ מְנוּחָה אִשָּׁה בֵּית אִישָׁהּ וַתִּשַּׁק לָהֶן וַתִּשֶּׂאנָה קוֹלָן וַתִּבְכֶּינָה` (רות א:ט)
`וַיְהִי בַּחֲצִי הַלַּיְלָה וַיֶּחֱרַד הָאִישׁ וַיִּלָּפֵת וְהִנֵּה אִשָּׁה שֹׁכֶבֶת מַרְגְּלֹתָיו` (רות ג:ח)
`וְעַתָּה בִּתִּי אַל תִּירְאִי כֹּל אֲשֶׁר תֹּאמְרִי אֶעֱשֶׂה לָּךְ כִּי יוֹדֵעַ כָּל שַׁעַר עַמִּי כִּי אֵשֶׁת חַיִל אָתְּ` (רות ג:יא)
`וַתִּשְׁכַּב מַרְגְּלֹתָו עַד הַבֹּקֶר וַתָּקָם בִּטְרוֹם יַכִּיר אִישׁ אֶת רֵעֵהוּ וַיֹּאמֶר אַל יִוָּדַע כִּי בָאָה הָאִשָּׁה הַגֹּרֶן` (רות ג:יד)
`וַיֹּאמֶר בֹּעַז בְּיוֹם קְנוֹתְךָ הַשָּׂדֶה מִיַּד נָעֳמִי וּמֵאֵת רוּת הַמּוֹאֲבִיָּה אֵשֶׁת הַמֵּת קָנִיתִי לְהָקִים שֵׁם הַמֵּת עַל נַחֲלָתוֹ` (רות ד:ה)
`וְגַם אֶת רוּת הַמֹּאֲבִיָּה אֵשֶׁת מַחְלוֹן קָנִיתִי לִי לְאִשָּׁה לְהָקִים שֵׁם הַמֵּת עַל נַחֲלָתוֹ וְלֹא יִכָּרֵת שֵׁם הַמֵּת מֵעִם אֶחָיו וּמִשַּׁעַר מְקוֹמוֹ עֵדִים אַתֶּם הַיּוֹם` (רות ד:י)
`וַיֹּאמְרוּ כָּל הָעָם אֲשֶׁר בַּשַּׁעַר וְהַזְּקֵנִים עֵדִים יִתֵּן יְהוָה אֶת הָאִשָּׁה הַבָּאָה אֶל בֵּיתֶךָ כְּרָחֵל וּכְלֵאָה אֲשֶׁר בָּנוּ שְׁתֵּיהֶם אֶת בֵּית יִשְׂרָאֵל וַעֲשֵׂה חַיִל בְּאֶפְרָתָה וּקְרָא שֵׁם בְּבֵית לָחֶם` (רות ד:יא)
`וַיִּקַּח בֹּעַז אֶת רוּת וַתְּהִי לוֹ לְאִשָּׁה וַיָּבֹא אֵלֶיהָ וַיִּתֵּן יְהוָה לָהּ הֵרָיוֹן וַתֵּלֶד בֵּן` (רות ד:יג)
`וַתֹּאמַרְנָה הַנָּשִׁים אֶל נָעֳמִי בָּרוּךְ יְהוָה אֲשֶׁר לֹא הִשְׁבִּית לָךְ גֹּאֵל הַיּוֹם וְיִקָּרֵא שְׁמוֹ בְּיִשְׂרָאֵל` (רות ד:יד)
`אִם לֹא תֵדְעִי לָךְ הַיָּפָה בַּנָּשִׁים צְאִי לָךְ בְּעִקְבֵי הַצֹּאן וּרְעִי אֶת גְּדִיֹּתַיִךְ עַל מִשְׁכְּנוֹת הָרֹעִים` (שיר-השירים א:ח)
`מַה דּוֹדֵךְ מִדּוֹד הַיָּפָה בַּנָּשִׁים מַה דּוֹדֵךְ מִדּוֹד שֶׁכָּכָה הִשְׁבַּעְתָּנוּ` (שיר-השירים ה:ט)
`אָנָה הָלַךְ דּוֹדֵךְ הַיָּפָה בַּנָּשִׁים אָנָה פָּנָה דוֹדֵךְ וּנְבַקְשֶׁנּוּ עִמָּךְ` (שיר-השירים ו:א)
`וּמוֹצֶא אֲנִי מַר מִמָּוֶת אֶת הָאִשָּׁה אֲשֶׁר הִיא מְצוֹדִים וַחֲרָמִים לִבָּהּ אֲסוּרִים יָדֶיהָ טוֹב לִפְנֵי הָאֱלֹהִים יִמָּלֵט מִמֶּנָּה וְחוֹטֵא יִלָּכֶד בָּהּ` (קהלת ז:כו)
`אֲשֶׁר עוֹד בִּקְשָׁה נַפְשִׁי וְלֹא מָצָאתִי אָדָם אֶחָד מֵאֶלֶף מָצָאתִי וְאִשָּׁה בְכָל אֵלֶּה לֹא מָצָאתִי` (קהלת ז:כח)
`רְאֵה חַיִּים עִם אִשָּׁה אֲשֶׁר אָהַבְתָּ כָּל יְמֵי חַיֵּי הֶבְלֶךָ אֲשֶׁר נָתַן לְךָ תַּחַת הַשֶּׁמֶשׁ כֹּל יְמֵי הֶבְלֶךָ כִּי הוּא חֶלְקְךָ בַּחַיִּים וּבַעֲמָלְךָ אֲשֶׁר אַתָּה עָמֵל תַּחַת הַשָּׁמֶשׁ` (קהלת ט:ט)
`רְאֵה יְהוָה וְהַבִּיטָה לְמִי עוֹלַלְתָּ כֹּה אִם תֹּאכַלְנָה נָשִׁים פִּרְיָם עֹלֲלֵי טִפֻּחִים אִם יֵהָרֵג בְּמִקְדַּשׁ אֲדֹנָי כֹּהֵן וְנָבִיא` (איכה ב:כ)
`יְדֵי נָשִׁים רַחֲמָנִיּוֹת בִּשְּׁלוּ יַלְדֵיהֶן הָיוּ לְבָרוֹת לָמוֹ בְּשֶׁבֶר בַּת עַמִּי` (איכה ד:י)
`נָשִׁים בְּצִיּוֹן עִנּוּ בְּתֻלֹת בְּעָרֵי יְהוּדָה` (איכה ה:יא)
`גַּם וַשְׁתִּי הַמַּלְכָּה עָשְׂתָה מִשְׁתֵּה נָשִׁים בֵּית הַמַּלְכוּת אֲשֶׁר לַמֶּלֶךְ אֲחַשְׁוֵרוֹשׁ` (אסתר א:ט)
`כִּי יֵצֵא דְבַר הַמַּלְכָּה עַל כָּל הַנָּשִׁים לְהַבְזוֹת בַּעְלֵיהֶן בְּעֵינֵיהֶן בְּאָמְרָם הַמֶּלֶךְ אֲחַשְׁוֵרוֹשׁ אָמַר לְהָבִיא אֶת וַשְׁתִּי הַמַּלְכָּה לְפָנָיו וְלֹא בָאָה` (אסתר א:יז)
`וְנִשְׁמַע פִּתְגָם הַמֶּלֶךְ אֲשֶׁר יַעֲשֶׂה בְּכָל מַלְכוּתוֹ כִּי רַבָּה הִיא וְכָל הַנָּשִׁים יִתְּנוּ יְקָר לְבַעְלֵיהֶן לְמִגָּדוֹל וְעַד קָטָן` (אסתר א:כ)
`וְיַפְקֵד הַמֶּלֶךְ פְּקִידִים בְּכָל מְדִינוֹת מַלְכוּתוֹ וְיִקְבְּצוּ אֶת כָּל נַעֲרָה בְתוּלָה טוֹבַת מַרְאֶה אֶל שׁוּשַׁן הַבִּירָה אֶל בֵּית הַנָּשִׁים אֶל יַד הֵגֶא סְרִיס הַמֶּלֶךְ שֹׁמֵר הַנָּשִׁים וְנָתוֹן תַּמְרוּקֵיהֶן` (אסתר ב:ג)
`וַיְהִי בְּהִשָּׁמַע דְּבַר הַמֶּלֶךְ וְדָתוֹ וּבְהִקָּבֵץ נְעָרוֹת רַבּוֹת אֶל שׁוּשַׁן הַבִּירָה אֶל יַד הֵגָי וַתִּלָּקַח אֶסְתֵּר אֶל בֵּית הַמֶּלֶךְ אֶל יַד הֵגַי שֹׁמֵר הַנָּשִׁים` (אסתר ב:ח)
`וַתִּיטַב הַנַּעֲרָה בְעֵינָיו וַתִּשָּׂא חֶסֶד לְפָנָיו וַיְבַהֵל אֶת תַּמְרוּקֶיהָ וְאֶת מָנוֹתֶהָ לָתֵת לָהּ וְאֵת שֶׁבַע הַנְּעָרוֹת הָרְאֻיוֹת לָתֶת לָהּ מִבֵּית הַמֶּלֶךְ וַיְשַׁנֶּהָ וְאֶת נַעֲרוֹתֶיהָ לְטוֹב בֵּית הַנָּשִׁים` (אסתר ב:ט)
`וּבְכָל יוֹם וָיוֹם מָרְדֳּכַי מִתְהַלֵּךְ לִפְנֵי חֲצַר בֵּית הַנָּשִׁים לָדַעַת אֶת שְׁלוֹם אֶסְתֵּר וּמַה יֵּעָשֶׂה בָּהּ` (אסתר ב:יא)
`וּבְהַגִּיעַ תֹּר נַעֲרָה וְנַעֲרָה לָבוֹא אֶל הַמֶּלֶךְ אֲחַשְׁוֵרוֹשׁ מִקֵּץ הֱיוֹת לָהּ כְּדָת הַנָּשִׁים שְׁנֵים עָשָׂר חֹדֶשׁ כִּי כֵּן יִמְלְאוּ יְמֵי מְרוּקֵיהֶן שִׁשָּׁה חֳדָשִׁים בְּשֶׁמֶן הַמֹּר וְשִׁשָּׁה חֳדָשִׁים בַּבְּשָׂמִים וּבְתַמְרוּקֵי הַנָּשִׁים` (אסתר ב:יב)
`וּבָזֶה הַנַּעֲרָה בָּאָה אֶל הַמֶּלֶךְ אֵת כָּל אֲשֶׁר תֹּאמַר יִנָּתֵן לָהּ לָבוֹא עִמָּהּ מִבֵּית הַנָּשִׁים עַד בֵּית הַמֶּלֶךְ` (אסתר ב:יג)
`בָּעֶרֶב הִיא בָאָה וּבַבֹּקֶר הִיא שָׁבָה אֶל בֵּית הַנָּשִׁים שֵׁנִי אֶל יַד שַׁעֲשְׁגַז סְרִיס הַמֶּלֶךְ שֹׁמֵר הַפִּילַגְשִׁים לֹא תָבוֹא עוֹד אֶל הַמֶּלֶךְ כִּי אִם חָפֵץ בָּהּ הַמֶּלֶךְ וְנִקְרְאָה בְשֵׁם` (אסתר ב:יד)
`וּבְהַגִּיעַ תֹּר אֶסְתֵּר בַּת אֲבִיחַיִל דֹּד מָרְדֳּכַי אֲשֶׁר לָקַח לוֹ לְבַת לָבוֹא אֶל הַמֶּלֶךְ לֹא בִקְשָׁה דָּבָר כִּי אִם אֶת אֲשֶׁר יֹאמַר הֵגַי סְרִיס הַמֶּלֶךְ שֹׁמֵר הַנָּשִׁים וַתְּהִי אֶסְתֵּר נֹשֵׂאת חֵן בְּעֵינֵי כָּל רֹאֶיהָ` (אסתר ב:טו)
`וַיֶּאֱהַב הַמֶּלֶךְ אֶת אֶסְתֵּר מִכָּל הַנָּשִׁים וַתִּשָּׂא חֵן וָחֶסֶד לְפָנָיו מִכָּל הַבְּתוּלֹת וַיָּשֶׂם כֶּתֶר מַלְכוּת בְּרֹאשָׁהּ וַיַּמְלִיכֶהָ תַּחַת וַשְׁתִּי` (אסתר ב:יז)
`וְנִשְׁלוֹחַ סְפָרִים בְּיַד הָרָצִים אֶל כָּל מְדִינוֹת הַמֶּלֶךְ לְהַשְׁמִיד לַהֲרֹג וּלְאַבֵּד אֶת כָּל הַיְּהוּדִים מִנַּעַר וְעַד זָקֵן טַף וְנָשִׁים בְּיוֹם אֶחָד בִּשְׁלוֹשָׁה עָשָׂר לְחֹדֶשׁ שְׁנֵים עָשָׂר הוּא חֹדֶשׁ אֲדָר וּשְׁלָלָם לָבוֹז` (אסתר ג:יג)
`כָּל עַבְדֵי הַמֶּלֶךְ וְעַם מְדִינוֹת הַמֶּלֶךְ יוֹדְעִים אֲשֶׁר כָּל אִישׁ וְאִשָּׁה אֲשֶׁר יָבוֹא אֶל הַמֶּלֶךְ אֶל הֶחָצֵר הַפְּנִימִית אֲשֶׁר לֹא יִקָּרֵא אַחַת דָּתוֹ לְהָמִית לְבַד מֵאֲשֶׁר יוֹשִׁיט לוֹ הַמֶּלֶךְ אֶת שַׁרְבִיט הַזָּהָב וְחָיָה וַאֲנִי לֹא נִקְרֵאתי לָבוֹא אֶל הַמֶּלֶךְ זֶה שְׁלוֹשִׁים יוֹם` (אסתר ד:יא)
`וַיִּתְאַפַּק הָמָן וַיָּבוֹא אֶל בֵּיתוֹ וַיִּשְׁלַח וַיָּבֵא אֶת אֹהֲבָיו וְאֶת זֶרֶשׁ אִשְׁתּוֹ` (אסתר ה:י)
`וַתֹּאמֶר לוֹ זֶרֶשׁ אִשְׁתּוֹ וְכָל אֹהֲבָיו יַעֲשׂוּ עֵץ גָּבֹהַּ חֲמִשִּׁים אַמָּה וּבַבֹּקֶר אֱמֹר לַמֶּלֶךְ וְיִתְלוּ אֶת מָרְדֳּכַי עָלָיו וּבֹא עִם הַמֶּלֶךְ אֶל הַמִּשְׁתֶּה שָׂמֵחַ וַיִּיטַב הַדָּבָר לִפְנֵי הָמָן וַיַּעַשׂ הָעֵץ` (אסתר ה:יד)
`וַיְסַפֵּר הָמָן לְזֶרֶשׁ אִשְׁתּוֹ וּלְכָל אֹהֲבָיו אֵת כָּל אֲשֶׁר קָרָהוּ וַיֹּאמְרוּ לוֹ חֲכָמָיו וְזֶרֶשׁ אִשְׁתּוֹ אִם מִזֶּרַע הַיְּהוּדִים מָרְדֳּכַי אֲשֶׁר הַחִלּוֹתָ לִנְפֹּל לְפָנָיו לֹא תוּכַל לוֹ כִּי נָפוֹל תִּפּוֹל לְפָנָיו` (אסתר ו:יג)
`אֲשֶׁר נָתַן הַמֶּלֶךְ לַיְּהוּדִים אֲשֶׁר בְּכָל עִיר וָעִיר לְהִקָּהֵל וְלַעֲמֹד עַל נַפְשָׁם לְהַשְׁמִיד וְלַהֲרֹג וּלְאַבֵּד אֶת כָּל חֵיל עַם וּמְדִינָה הַצָּרִים אֹתָם טַף וְנָשִׁים וּשְׁלָלָם לָבוֹז` (אסתר ח:יא)
`וְיָשֵׂם פָּנָיו לָבוֹא בְּתֹקֶף כָּל מַלְכוּתוֹ וִישָׁרִים עִמּוֹ וְעָשָׂה וּבַת הַנָּשִׁים יִתֶּן לוֹ לְהַשְׁחִיתָהּ וְלֹא תַעֲמֹד וְלֹא לוֹ תִהְיֶה` (דניאל יא:יז)
`וְעַל אֱלֹהֵי אֲבֹתָיו לֹא יָבִין וְעַל חֶמְדַּת נָשִׁים וְעַל כָּל אֱלוֹהַּ לֹא יָבִין כִּי עַל כֹּל יִתְגַּדָּל` (דניאל יא:לז)
`וּמִבְּנֵי הַכֹּהֲנִים בְּנֵי חֳבַיָּה בְּנֵי הַקּוֹץ בְּנֵי בַרְזִלַּי אֲשֶׁר לָקַח מִבְּנוֹת בַּרְזִלַּי הַגִּלְעָדִי אִשָּׁה וַיִּקָּרֵא עַל שְׁמָם` (עזרא ב:סא)
`וּכְהִתְפַּלֵּל עֶזְרָא וּכְהִתְוַדֹּתוֹ בֹּכֶה וּמִתְנַפֵּל לִפְנֵי בֵּית הָאֱלֹהִים נִקְבְּצוּ אֵלָיו מִיִּשְׂרָאֵל קָהָל רַב מְאֹד אֲנָשִׁים וְנָשִׁים וִילָדִים כִּי בָכוּ הָעָם הַרְבֵּה בֶכֶה` (עזרא י:א)
`וַיַּעַן שְׁכַנְיָה בֶן יְחִיאֵל מִבְּנֵי עוֹלָם וַיֹּאמֶר לְעֶזְרָא אֲנַחְנוּ מָעַלְנוּ בֵאלֹהֵינוּ וַנֹּשֶׁב נָשִׁים נָכְרִיּוֹת מֵעַמֵּי הָאָרֶץ וְעַתָּה יֵשׁ מִקְוֶה לְיִשְׂרָאֵל עַל זֹאת` (עזרא י:ב)
`וְעַתָּה נִכְרָת בְּרִית לֵאלֹהֵינוּ לְהוֹצִיא כָל נָשִׁים וְהַנּוֹלָד מֵהֶם בַּעֲצַת אֲדֹנָי וְהַחֲרֵדִים בְּמִצְוַת אֱלֹהֵינוּ וְכַתּוֹרָה יֵעָשֶׂה` (עזרא י:ג)
`וַיָּקָם עֶזְרָא הַכֹּהֵן וַיֹּאמֶר אֲלֵהֶם אַתֶּם מְעַלְתֶּם וַתֹּשִׁיבוּ נָשִׁים נָכְרִיּוֹת לְהוֹסִיף עַל אַשְׁמַת יִשְׂרָאֵל` (עזרא י:י)
`וְעַתָּה תְּנוּ תוֹדָה לַיהוָה אֱלֹהֵי אֲבֹתֵיכֶם וַעֲשׂוּ רְצוֹנוֹ וְהִבָּדְלוּ מֵעַמֵּי הָאָרֶץ וּמִן הַנָּשִׁים הַנָּכְרִיּוֹת` (עזרא י:יא)
`יַעֲמְדוּ נָא שָׂרֵינוּ לְכָל הַקָּהָל וְכֹל אֲשֶׁר בֶּעָרֵינוּ הַהֹשִׁיב נָשִׁים נָכְרִיּוֹת יָבֹא לְעִתִּים מְזֻמָּנִים וְעִמָּהֶם זִקְנֵי עִיר וָעִיר וְשֹׁפְטֶיהָ עַד לְהָשִׁיב חֲרוֹן אַף אֱלֹהֵינוּ מִמֶּנּוּ עַד לַדָּבָר הַזֶּה` (עזרא י:יד)
`וַיְכַלּוּ בַכֹּל אֲנָשִׁים הַהֹשִׁיבוּ נָשִׁים נָכְרִיּוֹת עַד יוֹם אֶחָד לַחֹדֶשׁ הָרִאשׁוֹן` (עזרא י:יז)
`וַיִּמָּצֵא מִבְּנֵי הַכֹּהֲנִים אֲשֶׁר הֹשִׁיבוּ נָשִׁים נָכְרִיּוֹת מִבְּנֵי יֵשׁוּעַ בֶּן יוֹצָדָק וְאֶחָיו מַעֲשֵׂיָה וֶאֱלִיעֶזֶר וְיָרִיב וּגְדַלְיָה` (עזרא י:יח)
`וַיִּתְּנוּ יָדָם לְהוֹצִיא נְשֵׁיהֶם וַאֲשֵׁמִים אֵיל צֹאן עַל אַשְׁמָתָם` (עזרא י:יט)
`כָּל אֵלֶּה נָשְׂאי נָשִׁים נָכְרִיּוֹת וְיֵשׁ מֵהֶם נָשִׁים וַיָּשִׂימוּ בָּנִים` (עזרא י:מד)
`וָאֵרֶא וָאָקוּם וָאֹמַר אֶל הַחֹרִים וְאֶל הַסְּגָנִים וְאֶל יֶתֶר הָעָם אַל תִּירְאוּ מִפְּנֵיהֶם אֶת אֲדֹנָי הַגָּדוֹל וְהַנּוֹרָא זְכֹרוּ וְהִלָּחֲמוּ עַל אֲחֵיכֶם בְּנֵיכֶם וּבְנֹתֵיכֶם נְשֵׁיכֶם וּבָתֵּיכֶם` (נחמיה ד:יד)
`וַתְּהִי צַעֲקַת הָעָם וּנְשֵׁיהֶם גְּדוֹלָה אֶל אֲחֵיהֶם הַיְּהוּדִים` (נחמיה ה:א)
`וּמִן הַכֹּהֲנִים בְּנֵי חֳבַיָּה בְּנֵי הַקּוֹץ בְּנֵי בַרְזִלַּי אֲשֶׁר לָקַח מִבְּנוֹת בַּרְזִלַּי הַגִּלְעָדִי אִשָּׁה וַיִּקָּרֵא עַל שְׁמָם` (נחמיה ז:סג)
`וַיָּבִיא עֶזְרָא הַכֹּהֵן אֶת הַתּוֹרָה לִפְנֵי הַקָּהָל מֵאִישׁ וְעַד אִשָּׁה וְכֹל מֵבִין לִשְׁמֹעַ בְּיוֹם אֶחָד לַחֹדֶשׁ הַשְּׁבִיעִי` (נחמיה ח:ב)
`וַיִּקְרָא בוֹ לִפְנֵי הָרְחוֹב אֲשֶׁר לִפְנֵי שַׁעַר הַמַּיִם מִן הָאוֹר עַד מַחֲצִית הַיּוֹם נֶגֶד הָאֲנָשִׁים וְהַנָּשִׁים וְהַמְּבִינִים וְאָזְנֵי כָל הָעָם אֶל סֵפֶר הַתּוֹרָה` (נחמיה ח:ג)
`וּשְׁאָר הָעָם הַכֹּהֲנִים הַלְוִיִּם הַשּׁוֹעֲרִים הַמְשֹׁרְרִים הַנְּתִינִים וְכָל הַנִּבְדָּל מֵעַמֵּי הָאֲרָצוֹת אֶל תּוֹרַת הָאֱלֹהִים נְשֵׁיהֶם בְּנֵיהֶם וּבְנֹתֵיהֶם כֹּל יוֹדֵעַ מֵבִין` (נחמיה י:כח)
`וַיִּזְבְּחוּ בַיּוֹם הַהוּא זְבָחִים גְּדוֹלִים וַיִּשְׂמָחוּ כִּי הָאֱלֹהִים שִׂמְּחָם שִׂמְחָה גְדוֹלָה וְגַם הַנָּשִׁים וְהַיְלָדִים שָׂמֵחוּ וַתִּשָּׁמַע שִׂמְחַת יְרוּשָׁלִַם מֵרָחוֹק` (נחמיה יב:מג)
`גַּם בַּיָּמִים הָהֵם רָאִיתִי אֶת הַיְּהוּדִים הֹשִׁיבוּ נָשִׁים אַשְׁדּוֹדִיּוֹת עַמּוֹנִיּוֹת מוֹאֲבִיּוֹת` (נחמיה יג:כג)
`הֲלוֹא עַל אֵלֶּה חָטָא שְׁלֹמֹה מֶלֶךְ יִשְׂרָאֵל וּבַגּוֹיִם הָרַבִּים לֹא הָיָה מֶלֶךְ כָּמֹהוּ וְאָהוּב לֵאלֹהָיו הָיָה וַיִּתְּנֵהוּ אֱלֹהִים מֶלֶךְ עַל כָּל יִשְׂרָאֵל גַּם אוֹתוֹ הֶחֱטִיאוּ הַנָּשִׁים הַנָּכְרִיּוֹת` (נחמיה יג:כו)
`וְלָכֶם הֲנִשְׁמַע לַעֲשֹׂת אֵת כָּל הָרָעָה הַגְּדוֹלָה הַזֹּאת לִמְעֹל בֵּאלֹהֵינוּ לְהֹשִׁיב נָשִׁים נָכְרִיּוֹת` (נחמיה יג:כז)
