`וַיֹּאמֶר יְהוָה אֱלֹהִים הֵן הָאָדָם הָיָה כְּאַחַד מִמֶּנּוּ לָדַעַת טוֹב וָרָע וְעַתָּה פֶּן יִשְׁלַח יָדוֹ וְלָקַח גַּם מֵעֵץ הַחַיִּים וְאָכַל וָחַי לְעֹלָם` (בראשית ג:כב)
`וַיְחִי אָדָם שְׁלֹשִׁים וּמְאַת שָׁנָה וַיּוֹלֶד בִּדְמוּתוֹ כְּצַלְמוֹ וַיִּקְרָא אֶת שְׁמוֹ שֵׁת` (בראשית ה:ג)
`וַיִּהְיוּ כָּל יְמֵי אָדָם אֲשֶׁר חַי תְּשַׁע מֵאוֹת שָׁנָה וּשְׁלֹשִׁים שָׁנָה וַיָּמֹת` (בראשית ה:ה)
`וַיְחִי שֵׁת חָמֵשׁ שָׁנִים וּמְאַת שָׁנָה וַיּוֹלֶד אֶת אֱנוֹשׁ` (בראשית ה:ו)
`וַיְחִי שֵׁת אַחֲרֵי הוֹלִידוֹ אֶת אֱנוֹשׁ שֶׁבַע שָׁנִים וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:ז)
`וַיְחִי אֱנוֹשׁ תִּשְׁעִים שָׁנָה וַיּוֹלֶד אֶת קֵינָן` (בראשית ה:ט)
`וַיְחִי אֱנוֹשׁ אַחֲרֵי הוֹלִידוֹ אֶת קֵינָן חֲמֵשׁ עֶשְׂרֵה שָׁנָה וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:י)
`וַיְחִי קֵינָן שִׁבְעִים שָׁנָה וַיּוֹלֶד אֶת מַהֲלַלְאֵל` (בראשית ה:יב)
`וַיְחִי קֵינָן אַחֲרֵי הוֹלִידוֹ אֶת מַהֲלַלְאֵל אַרְבָּעִים שָׁנָה וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:יג)
`וַיְחִי מַהֲלַלְאֵל חָמֵשׁ שָׁנִים וְשִׁשִּׁים שָׁנָה וַיּוֹלֶד אֶת יָרֶד` (בראשית ה:טו)
`וַיְחִי מַהֲלַלְאֵל אַחֲרֵי הוֹלִידוֹ אֶת יֶרֶד שְׁלֹשִׁים שָׁנָה וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:טז)
`וַיְחִי יֶרֶד שְׁתַּיִם וְשִׁשִּׁים שָׁנָה וּמְאַת שָׁנָה וַיּוֹלֶד אֶת חֲנוֹךְ` (בראשית ה:יח)
`וַיְחִי יֶרֶד אַחֲרֵי הוֹלִידוֹ אֶת חֲנוֹךְ שְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:יט)
`וַיְחִי חֲנוֹךְ חָמֵשׁ וְשִׁשִּׁים שָׁנָה וַיּוֹלֶד אֶת מְתוּשָׁלַח` (בראשית ה:כא)
`וַיְחִי מְתוּשֶׁלַח שֶׁבַע וּשְׁמֹנִים שָׁנָה וּמְאַת שָׁנָה וַיּוֹלֶד אֶת לָמֶךְ` (בראשית ה:כה)
`וַיְחִי מְתוּשֶׁלַח אַחֲרֵי הוֹלִידוֹ אֶת לֶמֶךְ שְׁתַּיִם וּשְׁמוֹנִים שָׁנָה וּשְׁבַע מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:כו)
`וַיְחִי לֶמֶךְ שְׁתַּיִם וּשְׁמֹנִים שָׁנָה וּמְאַת שָׁנָה וַיּוֹלֶד בֵּן` (בראשית ה:כח)
`וַיְחִי לֶמֶךְ אַחֲרֵי הוֹלִידוֹ אֶת נֹחַ חָמֵשׁ וְתִשְׁעִים שָׁנָה וַחֲמֵשׁ מֵאֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית ה:ל)
`וּמִכָּל הָחַי מִכָּל בָּשָׂר שְׁנַיִם מִכֹּל תָּבִיא אֶל הַתֵּבָה לְהַחֲיֹת אִתָּךְ זָכָר וּנְקֵבָה יִהְיוּ` (בראשית ו:יט)
`מֵהָעוֹף לְמִינֵהוּ וּמִן הַבְּהֵמָה לְמִינָהּ מִכֹּל רֶמֶשׂ הָאֲדָמָה לְמִינֵהוּ שְׁנַיִם מִכֹּל יָבֹאוּ אֵלֶיךָ לְהַחֲיוֹת` (בראשית ו:כ)
`גַּם מֵעוֹף הַשָּׁמַיִם שִׁבְעָה שִׁבְעָה זָכָר וּנְקֵבָה לְחַיּוֹת זֶרַע עַל פְּנֵי כָל הָאָרֶץ` (בראשית ז:ג)
`וַיְחִי נֹחַ אַחַר הַמַּבּוּל שְׁלֹשׁ מֵאוֹת שָׁנָה וַחֲמִשִּׁים שָׁנָה` (בראשית ט:כח)
`וַיְחִי שֵׁם אַחֲרֵי הוֹלִידוֹ אֶת אַרְפַּכְשָׁד חֲמֵשׁ מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:יא)
`וְאַרְפַּכְשַׁד חַי חָמֵשׁ וּשְׁלֹשִׁים שָׁנָה וַיּוֹלֶד אֶת שָׁלַח` (בראשית יא:יב)
`וַיְחִי אַרְפַּכְשַׁד אַחֲרֵי הוֹלִידוֹ אֶת שֶׁלַח שָׁלֹשׁ שָׁנִים וְאַרְבַּע מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:יג)
`וְשֶׁלַח חַי שְׁלֹשִׁים שָׁנָה וַיּוֹלֶד אֶת עֵבֶר` (בראשית יא:יד)
`וַיְחִי שֶׁלַח אַחֲרֵי הוֹלִידוֹ אֶת עֵבֶר שָׁלֹשׁ שָׁנִים וְאַרְבַּע מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:טו)
`וַיְחִי עֵבֶר אַרְבַּע וּשְׁלֹשִׁים שָׁנָה וַיּוֹלֶד אֶת פָּלֶג` (בראשית יא:טז)
`וַיְחִי עֵבֶר אַחֲרֵי הוֹלִידוֹ אֶת פֶּלֶג שְׁלֹשִׁים שָׁנָה וְאַרְבַּע מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:יז)
`וַיְחִי פֶלֶג שְׁלֹשִׁים שָׁנָה וַיּוֹלֶד אֶת רְעוּ` (בראשית יא:יח)
`וַיְחִי פֶלֶג אַחֲרֵי הוֹלִידוֹ אֶת רְעוּ תֵּשַׁע שָׁנִים וּמָאתַיִם שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:יט)
`וַיְחִי רְעוּ שְׁתַּיִם וּשְׁלֹשִׁים שָׁנָה וַיּוֹלֶד אֶת שְׂרוּג` (בראשית יא:כ)
`וַיְחִי רְעוּ אַחֲרֵי הוֹלִידוֹ אֶת שְׂרוּג שֶׁבַע שָׁנִים וּמָאתַיִם שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:כא)
`וַיְחִי שְׂרוּג שְׁלֹשִׁים שָׁנָה וַיּוֹלֶד אֶת נָחוֹר` (בראשית יא:כב)
`וַיְחִי שְׂרוּג אַחֲרֵי הוֹלִידוֹ אֶת נָחוֹר מָאתַיִם שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:כג)
`וַיְחִי נָחוֹר תֵּשַׁע וְעֶשְׂרִים שָׁנָה וַיּוֹלֶד אֶת תָּרַח` (בראשית יא:כד)
`וַיְחִי נָחוֹר אַחֲרֵי הוֹלִידוֹ אֶת תֶּרַח תְּשַׁע עֶשְׂרֵה שָׁנָה וּמְאַת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת` (בראשית יא:כה)
`וַיְחִי תֶרַח שִׁבְעִים שָׁנָה וַיּוֹלֶד אֶת אַבְרָם אֶת נָחוֹר וְאֶת הָרָן` (בראשית יא:כו)
`וְהָיָה כִּי יִרְאוּ אֹתָךְ הַמִּצְרִים וְאָמְרוּ אִשְׁתּוֹ זֹאת וְהָרְגוּ אֹתִי וְאֹתָךְ יְחַיּוּ` (בראשית יב:יב)
`אִמְרִי נָא אֲחֹתִי אָתְּ לְמַעַן יִיטַב לִי בַעֲבוּרֵךְ וְחָיְתָה נַפְשִׁי בִּגְלָלֵךְ` (בראשית יב:יג)
`וַיֹּאמֶר אַבְרָהָם אֶל הָאֱלֹהִים לוּ יִשְׁמָעֵאל יִחְיֶה לְפָנֶיךָ` (בראשית יז:יח)
`הִנֵּה נָא מָצָא עַבְדְּךָ חֵן בְּעֵינֶיךָ וַתַּגְדֵּל חַסְדְּךָ אֲשֶׁר עָשִׂיתָ עִמָּדִי לְהַחֲיוֹת אֶת נַפְשִׁי וְאָנֹכִי לֹא אוּכַל לְהִמָּלֵט הָהָרָה פֶּן תִּדְבָּקַנִי הָרָעָה וָמַתִּי` (בראשית יט:יט)
`הִנֵּה נָא הָעִיר הַזֹּאת קְרֹבָה לָנוּס שָׁמָּה וְהִיא מִצְעָר אִמָּלְטָה נָּא שָׁמָּה הֲלֹא מִצְעָר הִוא וּתְחִי נַפְשִׁי` (בראשית יט:כ)
`לְכָה נַשְׁקֶה אֶת אָבִינוּ יַיִן וְנִשְׁכְּבָה עִמּוֹ וּנְחַיֶּה מֵאָבִינוּ זָרַע` (בראשית יט:לב)
`וַיְהִי מִמָּחֳרָת וַתֹּאמֶר הַבְּכִירָה אֶל הַצְּעִירָה הֵן שָׁכַבְתִּי אֶמֶשׁ אֶת אָבִי נַשְׁקֶנּוּ יַיִן גַּם הַלַּיְלָה וּבֹאִי שִׁכְבִי עִמּוֹ וּנְחַיֶּה מֵאָבִינוּ זָרַע` (בראשית יט:לד)
`וְעַתָּה הָשֵׁב אֵשֶׁת הָאִישׁ כִּי נָבִיא הוּא וְיִתְפַּלֵּל בַּעַדְךָ וֶחְיֵה וְאִם אֵינְךָ מֵשִׁיב דַּע כִּי מוֹת תָּמוּת אַתָּה וְכָל אֲשֶׁר לָךְ` (בראשית כ:ז)
`וְאֵלֶּה יְמֵי שְׁנֵי חַיֵּי אַבְרָהָם אֲשֶׁר חָי מְאַת שָׁנָה וְשִׁבְעִים שָׁנָה וְחָמֵשׁ שָׁנִים` (בראשית כה:ז)
`וְעַל חַרְבְּךָ תִחְיֶה וְאֶת אָחִיךָ תַּעֲבֹד וְהָיָה כַּאֲשֶׁר תָּרִיד וּפָרַקְתָּ עֻלּוֹ מֵעַל צַוָּארֶךָ` (בראשית כז:מ)
`עִם אֲשֶׁר תִּמְצָא אֶת אֱלֹהֶיךָ לֹא יִחְיֶה נֶגֶד אַחֵינוּ הַכֶּר לְךָ מָה עִמָּדִי וְקַח לָךְ וְלֹא יָדַע יַעֲקֹב כִּי רָחֵל גְּנָבָתַם` (בראשית לא:לב)
`וַיֹּאמֶר הִנֵּה שָׁמַעְתִּי כִּי יֶשׁ שֶׁבֶר בְּמִצְרָיִם רְדוּ שָׁמָּה וְשִׁבְרוּ לָנוּ מִשָּׁם וְנִחְיֶה וְלֹא נָמוּת` (בראשית מב:ב)
`וַיֹּאמֶר אֲלֵהֶם יוֹסֵף בַּיּוֹם הַשְּׁלִישִׁי זֹאת עֲשׂוּ וִחְיוּ אֶת הָאֱלֹהִים אֲנִי יָרֵא` (בראשית מב:יח)
`וַיֹּאמֶר יְהוּדָה אֶל יִשְׂרָאֵל אָבִיו שִׁלְחָה הַנַּעַר אִתִּי וְנָקוּמָה וְנֵלֵכָה וְנִחְיֶה וְלֹא נָמוּת גַּם אֲנַחְנוּ גַם אַתָּה גַּם טַפֵּנוּ` (בראשית מג:ח)
`וַיִּשְׁלָחֵנִי אֱלֹהִים לִפְנֵיכֶם לָשׂוּם לָכֶם שְׁאֵרִית בָּאָרֶץ וּלְהַחֲיוֹת לָכֶם לִפְלֵיטָה גְּדֹלָה` (בראשית מה:ז)
`וַיְדַבְּרוּ אֵלָיו אֵת כָּל דִּבְרֵי יוֹסֵף אֲשֶׁר דִּבֶּר אֲלֵהֶם וַיַּרְא אֶת הָעֲגָלוֹת אֲשֶׁר שָׁלַח יוֹסֵף לָשֵׂאת אֹתוֹ וַתְּחִי רוּחַ יַעֲקֹב אֲבִיהֶם` (בראשית מה:כז)
`לָמָּה נָמוּת לְעֵינֶיךָ גַּם אֲנַחְנוּ גַּם אַדְמָתֵנוּ קְנֵה אֹתָנוּ וְאֶת אַדְמָתֵנוּ בַּלָּחֶם וְנִהְיֶה אֲנַחְנוּ וְאַדְמָתֵנוּ עֲבָדִים לְפַרְעֹה וְתֶן זֶרַע וְנִחְיֶה וְלֹא נָמוּת וְהָאֲדָמָה לֹא תֵשָׁם` (בראשית מז:יט)
`וַיֹּאמְרוּ הֶחֱיִתָנוּ נִמְצָא חֵן בְּעֵינֵי אֲדֹנִי וְהָיִינוּ עֲבָדִים לְפַרְעֹה` (בראשית מז:כה)
`וַיְחִי יַעֲקֹב בְּאֶרֶץ מִצְרַיִם שְׁבַע עֶשְׂרֵה שָׁנָה וַיְהִי יְמֵי יַעֲקֹב שְׁנֵי חַיָּיו שֶׁבַע שָׁנִים וְאַרְבָּעִים וּמְאַת שָׁנָה` (בראשית מז:כח)
`וְאַתֶּם חֲשַׁבְתֶּם עָלַי רָעָה אֱלֹהִים חֲשָׁבָהּ לְטֹבָה לְמַעַן עֲשֹׂה כַּיּוֹם הַזֶּה לְהַחֲיֹת עַם רָב` (בראשית נ:כ)
`וַיֵּשֶׁב יוֹסֵף בְּמִצְרַיִם הוּא וּבֵית אָבִיו וַיְחִי יוֹסֵף מֵאָה וָעֶשֶׂר שָׁנִים` (בראשית נ:כב)
`וַיֹּאמֶר בְּיַלֶּדְכֶן אֶת הָעִבְרִיּוֹת וּרְאִיתֶן עַל הָאָבְנָיִם אִם בֵּן הוּא וַהֲמִתֶּן אֹתוֹ וְאִם בַּת הִיא וָחָיָה` (שמות א:טז)
`וַתִּירֶאןָ הַמְיַלְּדֹת אֶת הָאֱלֹהִים וְלֹא עָשׂוּ כַּאֲשֶׁר דִּבֶּר אֲלֵיהֶן מֶלֶךְ מִצְרָיִם וַתְּחַיֶּיןָ אֶת הַיְלָדִים` (שמות א:יז)
`וַיִּקְרָא מֶלֶךְ מִצְרַיִם לַמְיַלְּדֹת וַיֹּאמֶר לָהֶן מַדּוּעַ עֲשִׂיתֶן הַדָּבָר הַזֶּה וַתְּחַיֶּיןָ אֶת הַיְלָדִים` (שמות א:יח)
`וַיְצַו פַּרְעֹה לְכָל עַמּוֹ לֵאמֹר כָּל הַבֵּן הַיִּלּוֹד הַיְאֹרָה תַּשְׁלִיכֻהוּ וְכָל הַבַּת תְּחַיּוּן` (שמות א:כב)
`לֹא תִגַּע בּוֹ יָד כִּי סָקוֹל יִסָּקֵל אוֹ יָרֹה יִיָּרֶה אִם בְּהֵמָה אִם אִישׁ לֹא יִחְיֶה בִּמְשֹׁךְ הַיֹּבֵל הֵמָּה יַעֲלוּ בָהָר` (שמות יט:יג)
`מְכַשֵּׁפָה לֹא תְחַיֶּה` (שמות כב:יח)
`וַיֹּאמֶר לֹא תוּכַל לִרְאֹת אֶת פָּנָי כִּי לֹא יִרְאַנִי הָאָדָם וָחָי` (שמות לג:כ)
`וּשְׁמַרְתֶּם אֶת חֻקֹּתַי וְאֶת מִשְׁפָּטַי אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם וָחַי בָּהֶם אֲנִי יְהוָה` (ויקרא יח:ה)
`וְכִי יָמוּךְ אָחִיךָ וּמָטָה יָדוֹ עִמָּךְ וְהֶחֱזַקְתָּ בּוֹ גֵּר וְתוֹשָׁב וָחַי עִמָּךְ` (ויקרא כה:לה)
`אַל תִּקַּח מֵאִתּוֹ נֶשֶׁךְ וְתַרְבִּית וְיָרֵאתָ מֵאֱלֹהֶיךָ וְחֵי אָחִיךָ עִמָּךְ` (ויקרא כה:לו)
`וְזֹאת עֲשׂוּ לָהֶם וְחָיוּ וְלֹא יָמֻתוּ בְּגִשְׁתָּם אֶת קֹדֶשׁ הַקֳּדָשִׁים אַהֲרֹן וּבָנָיו יָבֹאוּ וְשָׂמוּ אוֹתָם אִישׁ אִישׁ עַל עֲבֹדָתוֹ וְאֶל מַשָּׂאוֹ` (במדבר ד:יט)
`וִיהוֹשֻׁעַ בִּן נוּן וְכָלֵב בֶּן יְפֻנֶּה חָיוּ מִן הָאֲנָשִׁים הָהֵם הַהֹלְכִים לָתוּר אֶת הָאָרֶץ` (במדבר יד:לח)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה עֲשֵׂה לְךָ שָׂרָף וְשִׂים אֹתוֹ עַל נֵס וְהָיָה כָּל הַנָּשׁוּךְ וְרָאָה אֹתוֹ וָחָי` (במדבר כא:ח)
`וַיַּעַשׂ מֹשֶׁה נְחַשׁ נְחֹשֶׁת וַיְשִׂמֵהוּ עַל הַנֵּס וְהָיָה אִם נָשַׁךְ הַנָּחָשׁ אֶת אִישׁ וְהִבִּיט אֶל נְחַשׁ הַנְּחֹשֶׁת וָחָי` (במדבר כא:ט)
`וַתִּרְאַנִי הָאָתוֹן וַתֵּט לְפָנַי זֶה שָׁלֹשׁ רְגָלִים אוּלַי נָטְתָה מִפָּנַי כִּי עַתָּה גַּם אֹתְכָה הָרַגְתִּי וְאוֹתָהּ הֶחֱיֵיתִי` (במדבר כב:לג)
`וַיִּשָּׂא מְשָׁלוֹ וַיֹּאמַר אוֹי מִי יִחְיֶה מִשֻּׂמוֹ אֵל` (במדבר כד:כג)
`וַיֹּאמֶר אֲלֵיהֶם מֹשֶׁה הַחִיִּיתֶם כָּל נְקֵבָה` (במדבר לא:טו)
`וְכֹל הַטַּף בַּנָּשִׁים אֲשֶׁר לֹא יָדְעוּ מִשְׁכַּב זָכָר הַחֲיוּ לָכֶם` (במדבר לא:יח)
`וְעַתָּה יִשְׂרָאֵל שְׁמַע אֶל הַחֻקִּים וְאֶל הַמִּשְׁפָּטִים אֲשֶׁר אָנֹכִי מְלַמֵּד אֶתְכֶם לַעֲשׂוֹת לְמַעַן תִּחְיוּ וּבָאתֶם וִירִשְׁתֶּם אֶת הָאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֵי אֲבֹתֵיכֶם נֹתֵן לָכֶם` (דברים ד:א)
`הֲשָׁמַע עָם קוֹל אֱלֹהִים מְדַבֵּר מִתּוֹךְ הָאֵשׁ כַּאֲשֶׁר שָׁמַעְתָּ אַתָּה וַיֶּחִי` (דברים ד:לג)
`לָנֻס שָׁמָּה רוֹצֵחַ אֲשֶׁר יִרְצַח אֶת רֵעֵהוּ בִּבְלִי דַעַת וְהוּא לֹא שֹׂנֵא לוֹ מִתְּמוֹל שִׁלְשׁוֹם וְנָס אֶל אַחַת מִן הֶעָרִים הָאֵל וָחָי` (דברים ד:מב)
`וַתֹּאמְרוּ הֵן הֶרְאָנוּ יְהוָה אֱלֹהֵינוּ אֶת כְּבֹדוֹ וְאֶת גָּדְלוֹ וְאֶת קֹלוֹ שָׁמַעְנוּ מִתּוֹךְ הָאֵשׁ הַיּוֹם הַזֶּה רָאִינוּ כִּי יְדַבֵּר אֱלֹהִים אֶת הָאָדָם וָחָי` (דברים ה:כד)
`כִּי מִי כָל בָּשָׂר אֲשֶׁר שָׁמַע קוֹל אֱלֹהִים חַיִּים מְדַבֵּר מִתּוֹךְ הָאֵשׁ כָּמֹנוּ וַיֶּחִי` (דברים ה:כו)
`בְּכָל הַדֶּרֶךְ אֲשֶׁר צִוָּה יְהוָה אֱלֹהֵיכֶם אֶתְכֶם תֵּלֵכוּ לְמַעַן תִּחְיוּן וְטוֹב לָכֶם וְהַאֲרַכְתֶּם יָמִים בָּאָרֶץ אֲשֶׁר תִּירָשׁוּן` (דברים ה:לג)
`וַיְצַוֵּנוּ יְהוָה לַעֲשׂוֹת אֶת כָּל הַחֻקִּים הָאֵלֶּה לְיִרְאָה אֶת יְהוָה אֱלֹהֵינוּ לְטוֹב לָנוּ כָּל הַיָּמִים לְחַיֹּתֵנוּ כְּהַיּוֹם הַזֶּה` (דברים ו:כד)
`כָּל הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם תִּשְׁמְרוּן לַעֲשׂוֹת לְמַעַן תִּחְיוּן וּרְבִיתֶם וּבָאתֶם וִירִשְׁתֶּם אֶת הָאָרֶץ אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֵיכֶם` (דברים ח:א)
`וַיְעַנְּךָ וַיַּרְעִבֶךָ וַיַּאֲכִלְךָ אֶת הַמָּן אֲשֶׁר לֹא יָדַעְתָּ וְלֹא יָדְעוּן אֲבֹתֶיךָ לְמַעַן הוֹדִעֲךָ כִּי לֹא עַל הַלֶּחֶם לְבַדּוֹ יִחְיֶה הָאָדָם כִּי עַל כָּל מוֹצָא פִי יְהוָה יִחְיֶה הָאָדָם` (דברים ח:ג)
`צֶדֶק צֶדֶק תִּרְדֹּף לְמַעַן תִּחְיֶה וְיָרַשְׁתָּ אֶת הָאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ` (דברים טז:כ)
`וְזֶה דְּבַר הָרֹצֵחַ אֲשֶׁר יָנוּס שָׁמָּה וָחָי אֲשֶׁר יַכֶּה אֶת רֵעֵהוּ בִּבְלִי דַעַת וְהוּא לֹא שֹׂנֵא לוֹ מִתְּמֹל שִׁלְשֹׁם` (דברים יט:ד)
`וַאֲשֶׁר יָבֹא אֶת רֵעֵהוּ בַיַּעַר לַחְטֹב עֵצִים וְנִדְּחָה יָדוֹ בַגַּרְזֶן לִכְרֹת הָעֵץ וְנָשַׁל הַבַּרְזֶל מִן הָעֵץ וּמָצָא אֶת רֵעֵהוּ וָמֵת הוּא יָנוּס אֶל אַחַת הֶעָרִים הָאֵלֶּה וָחָי` (דברים יט:ה)
`רַק מֵעָרֵי הָעַמִּים הָאֵלֶּה אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ נַחֲלָה לֹא תְחַיֶּה כָּל נְשָׁמָה` (דברים כ:טז)
`אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם לְאַהֲבָה אֶת יְהוָה אֱלֹהֶיךָ לָלֶכֶת בִּדְרָכָיו וְלִשְׁמֹר מִצְוֺתָיו וְחֻקֹּתָיו וּמִשְׁפָּטָיו וְחָיִיתָ וְרָבִיתָ וּבֵרַכְךָ יְהוָה אֱלֹהֶיךָ בָּאָרֶץ אֲשֶׁר אַתָּה בָא שָׁמָּה לְרִשְׁתָּהּ` (דברים ל:טז)
`הַעִידֹתִי בָכֶם הַיּוֹם אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ הַחַיִּים וְהַמָּוֶת נָתַתִּי לְפָנֶיךָ הַבְּרָכָה וְהַקְּלָלָה וּבָחַרְתָּ בַּחַיִּים לְמַעַן תִּחְיֶה אַתָּה וְזַרְעֶךָ` (דברים ל:יט)
`רְאוּ עַתָּה כִּי אֲנִי אֲנִי הוּא וְאֵין אֱלֹהִים עִמָּדִי אֲנִי אָמִית וַאֲחַיֶּה מָחַצְתִּי וַאֲנִי אֶרְפָּא וְאֵין מִיָּדִי מַצִּיל` (דברים לב:לט)
`יְחִי רְאוּבֵן וְאַל יָמֹת וִיהִי מְתָיו מִסְפָּר` (דברים לג:ו)
`וְהַחֲיִתֶם אֶת אָבִי וְאֶת אִמִּי וְאֶת אַחַי וְאֶת אַחוֹתַי וְאֵת כָּל אֲשֶׁר לָהֶם וְהִצַּלְתֶּם אֶת נַפְשֹׁתֵינוּ מִמָּוֶת` (יהושע ב:יג)
`וַיְהִי כַּאֲשֶׁר תַּמּוּ כָל הַגּוֹי לְהִמּוֹל וַיֵּשְׁבוּ תַחְתָּם בַּמַּחֲנֶה עַד חֲיוֹתָם` (יהושע ה:ח)
`וְהָיְתָה הָעִיר חֵרֶם הִיא וְכָל אֲשֶׁר בָּהּ לַיהוָה רַק רָחָב הַזּוֹנָה תִּחְיֶה הִיא וְכָל אֲשֶׁר אִתָּהּ בַּבַּיִת כִּי הֶחְבְּאַתָה אֶת הַמַּלְאָכִים אֲשֶׁר שָׁלָחְנוּ` (יהושע ו:יז)
`וְאֶת רָחָב הַזּוֹנָה וְאֶת בֵּית אָבִיהָ וְאֶת כָּל אֲשֶׁר לָהּ הֶחֱיָה יְהוֹשֻׁעַ וַתֵּשֶׁב בְּקֶרֶב יִשְׂרָאֵל עַד הַיּוֹם הַזֶּה כִּי הֶחְבִּיאָה אֶת הַמַּלְאָכִים אֲשֶׁר שָׁלַח יְהוֹשֻׁעַ לְרַגֵּל אֶת יְרִיחוֹ` (יהושע ו:כה)
`וַיַּעַשׂ לָהֶם יְהוֹשֻׁעַ שָׁלוֹם וַיִּכְרֹת לָהֶם בְּרִית לְחַיּוֹתָם וַיִּשָּׁבְעוּ לָהֶם נְשִׂיאֵי הָעֵדָה` (יהושע ט:טו)
`זֹאת נַעֲשֶׂה לָהֶם וְהַחֲיֵה אוֹתָם וְלֹא יִהְיֶה עָלֵינוּ קֶצֶף עַל הַשְּׁבוּעָה אֲשֶׁר נִשְׁבַּעְנוּ לָהֶם` (יהושע ט:כ)
`וַיֹּאמְרוּ אֲלֵיהֶם הַנְּשִׂיאִים יִחְיוּ וַיִּהְיוּ חֹטְבֵי עֵצִים וְשֹׁאֲבֵי מַיִם לְכָל הָעֵדָה כַּאֲשֶׁר דִּבְּרוּ לָהֶם הַנְּשִׂיאִים` (יהושע ט:כא)
`וְעַתָּה הִנֵּה הֶחֱיָה יְהוָה אוֹתִי כַּאֲשֶׁר דִּבֵּר זֶה אַרְבָּעִים וְחָמֵשׁ שָׁנָה מֵאָז דִּבֶּר יְהוָה אֶת הַדָּבָר הַזֶּה אֶל מֹשֶׁה אֲשֶׁר הָלַךְ יִשְׂרָאֵל בַּמִּדְבָּר וְעַתָּה הִנֵּה אָנֹכִי הַיּוֹם בֶּן חָמֵשׁ וּשְׁמוֹנִים שָׁנָה` (יהושע יד:י)
`וַיֹּאמַר אַחַי בְּנֵי אִמִּי הֵם חַי יְהוָה לוּ הַחֲיִתֶם אוֹתָם לֹא הָרַגְתִּי אֶתְכֶם` (שופטים ח:יט)
`וַיִּבְקַע אֱלֹהִים אֶת הַמַּכְתֵּשׁ אֲשֶׁר בַּלֶּחִי וַיֵּצְאוּ מִמֶּנּוּ מַיִם וַיֵּשְׁתְּ וַתָּשָׁב רוּחוֹ וַיֶּחִי עַל כֵּן קָרָא שְׁמָהּ עֵין הַקּוֹרֵא אֲשֶׁר בַּלֶּחִי עַד הַיּוֹם הַזֶּה` (שופטים טו:יט)
`וַיָּשָׁב בִּנְיָמִן בָּעֵת הַהִיא וַיִּתְּנוּ לָהֶם הַנָּשִׁים אֲשֶׁר חִיּוּ מִנְּשֵׁי יָבֵשׁ גִּלְעָד וְלֹא מָצְאוּ לָהֶם כֵּן` (שופטים כא:יד)
`יְהוָה מֵמִית וּמְחַיֶּה מוֹרִיד שְׁאוֹל וַיָּעַל` (שמואל-א ב:ו)
`וַיֹּאמֶר שְׁמוּאֵל אֶל כָּל הָעָם הַרְּאִיתֶם אֲשֶׁר בָּחַר בּוֹ יְהוָה כִּי אֵין כָּמֹהוּ בְּכָל הָעָם וַיָּרִעוּ כָל הָעָם וַיֹּאמְרוּ יְחִי הַמֶּלֶךְ` (שמואל-א י:כד)
`וְהִכָּה דָוִד אֶת הָאָרֶץ וְלֹא יְחַיֶּה אִישׁ וְאִשָּׁה וְלָקַח צֹאן וּבָקָר וַחֲמֹרִים וּגְמַלִּים וּבְגָדִים וַיָּשָׁב וַיָּבֹא אֶל אָכִישׁ` (שמואל-א כז:ט)
`וְאִישׁ וְאִשָּׁה לֹא יְחַיֶּה דָוִד לְהָבִיא גַת לֵאמֹר פֶּן יַגִּדוּ עָלֵינוּ לֵאמֹר כֹּה עָשָׂה דָוִד וְכֹה מִשְׁפָּטוֹ כָּל הַיָּמִים אֲשֶׁר יָשַׁב בִּשְׂדֵה פְלִשְׁתִּים` (שמואל-א כז:יא)
`וָאֶעֱמֹד עָלָיו וַאֲמֹתְתֵהוּ כִּי יָדַעְתִּי כִּי לֹא יִחְיֶה אַחֲרֵי נִפְלוֹ וָאֶקַּח הַנֵּזֶר אֲשֶׁר עַל רֹאשׁוֹ וְאֶצְעָדָה אֲשֶׁר עַל זְרֹעוֹ וָאֲבִיאֵם אֶל אֲדֹנִי הֵנָּה` (שמואל-ב א:י)
`וַיַּךְ אֶת מוֹאָב וַיְמַדְּדֵם בַּחֶבֶל הַשְׁכֵּב אוֹתָם אַרְצָה וַיְמַדֵּד שְׁנֵי חֲבָלִים לְהָמִית וּמְלֹא הַחֶבֶל לְהַחֲיוֹת וַתְּהִי מוֹאָב לְדָוִד לַעֲבָדִים נֹשְׂאֵי מִנְחָה` (שמואל-ב ח:ב)
`וְלָרָשׁ אֵין כֹּל כִּי אִם כִּבְשָׂה אַחַת קְטַנָּה אֲשֶׁר קָנָה וַיְחַיֶּהָ וַתִּגְדַּל עִמּוֹ וְעִם בָּנָיו יַחְדָּו מִפִּתּוֹ תֹאכַל וּמִכֹּסוֹ תִשְׁתֶּה וּבְחֵיקוֹ תִשְׁכָּב וַתְּהִי לוֹ כְּבַת` (שמואל-ב יב:ג)
`וַיֹּאמֶר בְּעוֹד הַיֶּלֶד חַי צַמְתִּי וָאֶבְכֶּה כִּי אָמַרְתִּי מִי יוֹדֵעַ יְחָנַּנִי יְהוָה וְחַי הַיָּלֶד` (שמואל-ב יב:כב)
`וַיְהִי כַּאֲשֶׁר בָּא חוּשַׁי הָאַרְכִּי רֵעֶה דָוִד אֶל אַבְשָׁלוֹם וַיֹּאמֶר חוּשַׁי אֶל אַבְשָׁלֹם יְחִי הַמֶּלֶךְ יְחִי הַמֶּלֶךְ` (שמואל-ב טז:טז)
`כִּי יָרַד הַיּוֹם וַיִּזְבַּח שׁוֹר וּמְרִיא וְצֹאן לָרֹב וַיִּקְרָא לְכָל בְּנֵי הַמֶּלֶךְ וּלְשָׂרֵי הַצָּבָא וּלְאֶבְיָתָר הַכֹּהֵן וְהִנָּם אֹכְלִים וְשֹׁתִים לְפָנָיו וַיֹּאמְרוּ יְחִי הַמֶּלֶךְ אֲדֹנִיָּהוּ` (מלכים-א א:כה)
`וַתִּקֹּד בַּת שֶׁבַע אַפַּיִם אֶרֶץ וַתִּשְׁתַּחוּ לַמֶּלֶךְ וַתֹּאמֶר יְחִי אֲדֹנִי הַמֶּלֶךְ דָּוִד לְעֹלָם` (מלכים-א א:לא)
`וּמָשַׁח אֹתוֹ שָׁם צָדוֹק הַכֹּהֵן וְנָתָן הַנָּבִיא לְמֶלֶךְ עַל יִשְׂרָאֵל וּתְקַעְתֶּם בַּשּׁוֹפָר וַאֲמַרְתֶּם יְחִי הַמֶּלֶךְ שְׁלֹמֹה` (מלכים-א א:לד)
`וַיִּקַּח צָדוֹק הַכֹּהֵן אֶת קֶרֶן הַשֶּׁמֶן מִן הָאֹהֶל וַיִּמְשַׁח אֶת שְׁלֹמֹה וַיִּתְקְעוּ בַּשּׁוֹפָר וַיֹּאמְרוּ כָּל הָעָם יְחִי הַמֶּלֶךְ שְׁלֹמֹה` (מלכים-א א:לט)
`וַיִּשְׁמַע יְהוָה בְּקוֹל אֵלִיָּהוּ וַתָּשָׁב נֶפֶשׁ הַיֶּלֶד עַל קִרְבּוֹ וַיֶּחִי` (מלכים-א יז:כב)
`וַיֹּאמֶר אַחְאָב אֶל עֹבַדְיָהוּ לֵךְ בָּאָרֶץ אֶל כָּל מַעְיְנֵי הַמַּיִם וְאֶל כָּל הַנְּחָלִים אוּלַי נִמְצָא חָצִיר וּנְחַיֶּה סוּס וָפֶרֶד וְלוֹא נַכְרִית מֵהַבְּהֵמָה` (מלכים-א יח:ה)
`וַיֹּאמְרוּ אֵלָיו עֲבָדָיו הִנֵּה נָא שָׁמַעְנוּ כִּי מַלְכֵי בֵּית יִשְׂרָאֵל כִּי מַלְכֵי חֶסֶד הֵם נָשִׂימָה נָּא שַׂקִּים בְּמָתְנֵינוּ וַחֲבָלִים בְּרֹאשֵׁנוּ וְנֵצֵא אֶל מֶלֶךְ יִשְׂרָאֵל אוּלַי יְחַיֶּה אֶת נַפְשֶׁךָ` (מלכים-א כ:לא)
`וַיַּחְגְּרוּ שַׂקִּים בְּמָתְנֵיהֶם וַחֲבָלִים בְּרָאשֵׁיהֶם וַיָּבֹאוּ אֶל מֶלֶךְ יִשְׂרָאֵל וַיֹּאמְרוּ עַבְדְּךָ בֶן הֲדַד אָמַר תְּחִי נָא נַפְשִׁי וַיֹּאמֶר הַעוֹדֶנּוּ חַי אָחִי הוּא` (מלכים-א כ:לב)
`וַיִּפֹּל אֲחַזְיָה בְּעַד הַשְּׂבָכָה בַּעֲלִיָּתוֹ אֲשֶׁר בְּשֹׁמְרוֹן וַיָּחַל וַיִּשְׁלַח מַלְאָכִים וַיֹּאמֶר אֲלֵהֶם לְכוּ דִרְשׁוּ בְּבַעַל זְבוּב אֱלֹהֵי עֶקְרוֹן אִם אֶחְיֶה מֵחֳלִי זֶה` (מלכים-ב א:ב)
`וַתָּבֹא וַתַּגֵּד לְאִישׁ הָאֱלֹהִים וַיֹּאמֶר לְכִי מִכְרִי אֶת הַשֶּׁמֶן וְשַׁלְּמִי אֶת נִשְׁיֵכִי וְאַתְּ בְּנֵיכִי תִחְיִי בַּנּוֹתָר` (מלכים-ב ד:ז)
`וַיְהִי כִּקְרֹא מֶלֶךְ יִשְׂרָאֵל אֶת הַסֵּפֶר וַיִּקְרַע בְּגָדָיו וַיֹּאמֶר הַאֱלֹהִים אָנִי לְהָמִית וּלְהַחֲיוֹת כִּי זֶה שֹׁלֵחַ אֵלַי לֶאֱסֹף אִישׁ מִצָּרַעְתּוֹ כִּי אַךְ דְּעוּ נָא וּרְאוּ כִּי מִתְאַנֶּה הוּא לִי` (מלכים-ב ה:ז)
`אִם אָמַרְנוּ נָבוֹא הָעִיר וְהָרָעָב בָּעִיר וָמַתְנוּ שָׁם וְאִם יָשַׁבְנוּ פֹה וָמָתְנוּ וְעַתָּה לְכוּ וְנִפְּלָה אֶל מַחֲנֵה אֲרָם אִם יְחַיֻּנוּ נִחְיֶה וְאִם יְמִיתֻנוּ וָמָתְנוּ` (מלכים-ב ז:ד)
`וֶאֱלִישָׁע דִּבֶּר אֶל הָאִשָּׁה אֲשֶׁר הֶחֱיָה אֶת בְּנָהּ לֵאמֹר קוּמִי וּלְכִי אַתִּי וּבֵיתֵךְ וְגוּרִי בַּאֲשֶׁר תָּגוּרִי כִּי קָרָא יְהוָה לָרָעָב וְגַם בָּא אֶל הָאָרֶץ שֶׁבַע שָׁנִים` (מלכים-ב ח:א)
`וַיְהִי הוּא מְסַפֵּר לַמֶּלֶךְ אֵת אֲשֶׁר הֶחֱיָה אֶת הַמֵּת וְהִנֵּה הָאִשָּׁה אֲשֶׁר הֶחֱיָה אֶת בְּנָהּ צֹעֶקֶת אֶל הַמֶּלֶךְ עַל בֵּיתָהּ וְעַל שָׂדָהּ וַיֹּאמֶר גֵּחֲזִי אֲדֹנִי הַמֶּלֶךְ זֹאת הָאִשָּׁה וְזֶה בְּנָהּ אֲשֶׁר הֶחֱיָה אֱלִישָׁע` (מלכים-ב ח:ה)
`וַיֹּאמֶר הַמֶּלֶךְ אֶל חֲזָהאֵל קַח בְּיָדְךָ מִנְחָה וְלֵךְ לִקְרַאת אִישׁ הָאֱלֹהִים וְדָרַשְׁתָּ אֶת יְהוָה מֵאוֹתוֹ לֵאמֹר הַאֶחְיֶה מֵחֳלִי זֶה` (מלכים-ב ח:ח)
`וַיֵּלֶךְ חֲזָאֵל לִקְרָאתוֹ וַיִּקַּח מִנְחָה בְיָדוֹ וְכָל טוּב דַּמֶּשֶׂק מַשָּׂא אַרְבָּעִים גָּמָל וַיָּבֹא וַיַּעֲמֹד לְפָנָיו וַיֹּאמֶר בִּנְךָ בֶן הֲדַד מֶלֶךְ אֲרָם שְׁלָחַנִי אֵלֶיךָ לֵאמֹר הַאֶחְיֶה מֵחֳלִי זֶה` (מלכים-ב ח:ט)
`וַיֹּאמֶר אֵלָיו אֱלִישָׁע לֵךְ אֱמָר לֹא חָיֹה תִחְיֶה וְהִרְאַנִי יְהוָה כִּי מוֹת יָמוּת` (מלכים-ב ח:י)
`וַיֵּלֶךְ מֵאֵת אֱלִישָׁע וַיָּבֹא אֶל אֲדֹנָיו וַיֹּאמֶר לוֹ מָה אָמַר לְךָ אֱלִישָׁע וַיֹּאמֶר אָמַר לִי חָיֹה תִחְיֶה` (מלכים-ב ח:יד)
`וְעַתָּה כָל נְבִיאֵי הַבַּעַל כָּל עֹבְדָיו וְכָל כֹּהֲנָיו קִרְאוּ אֵלַי אִישׁ אַל יִפָּקֵד כִּי זֶבַח גָּדוֹל לִי לַבַּעַל כֹּל אֲשֶׁר יִפָּקֵד לֹא יִחְיֶה וְיֵהוּא עָשָׂה בְעָקְבָּה לְמַעַן הַאֲבִיד אֶת עֹבְדֵי הַבָּעַל` (מלכים-ב י:יט)
`וַיּוֹצִא אֶת בֶּן הַמֶּלֶךְ וַיִּתֵּן עָלָיו אֶת הַנֵּזֶר וְאֶת הָעֵדוּת וַיַּמְלִכוּ אֹתוֹ וַיִּמְשָׁחֻהוּ וַיַּכּוּ כָף וַיֹּאמְרוּ יְחִי הַמֶּלֶךְ` (מלכים-ב יא:יב)
`וַיְהִי הֵם קֹבְרִים אִישׁ וְהִנֵּה רָאוּ אֶת הַגְּדוּד וַיַּשְׁלִיכוּ אֶת הָאִישׁ בְּקֶבֶר אֱלִישָׁע וַיֵּלֶךְ וַיִּגַּע הָאִישׁ בְּעַצְמוֹת אֱלִישָׁע וַיְחִי וַיָּקָם עַל רַגְלָיו` (מלכים-ב יג:כא)
`וַיְחִי אֲמַצְיָהוּ בֶן יוֹאָשׁ מֶלֶךְ יְהוּדָה אַחֲרֵי מוֹת יְהוֹאָשׁ בֶּן יְהוֹאָחָז מֶלֶךְ יִשְׂרָאֵל חֲמֵשׁ עֶשְׂרֵה שָׁנָה` (מלכים-ב יד:יז)
`עַד בֹּאִי וְלָקַחְתִּי אֶתְכֶם אֶל אֶרֶץ כְּאַרְצְכֶם אֶרֶץ דָּגָן וְתִירוֹשׁ אֶרֶץ לֶחֶם וּכְרָמִים אֶרֶץ זֵית יִצְהָר וּדְבַשׁ וִחְיוּ וְלֹא תָמֻתוּ וְאַל תִּשְׁמְעוּ אֶל חִזְקִיָּהוּ כִּי יַסִּית אֶתְכֶם לֵאמֹר יְהוָה יַצִּילֵנוּ` (מלכים-ב יח:לב)
`בַּיָּמִים הָהֵם חָלָה חִזְקִיָּהוּ לָמוּת וַיָּבֹא אֵלָיו יְשַׁעְיָהוּ בֶן אָמוֹץ הַנָּבִיא וַיֹּאמֶר אֵלָיו כֹּה אָמַר יְהוָה צַו לְבֵיתֶךָ כִּי מֵת אַתָּה וְלֹא תִחְיֶה` (מלכים-ב כ:א)
`וַיֹּאמֶר יְשַׁעְיָהוּ קְחוּ דְּבֶלֶת תְּאֵנִים וַיִּקְחוּ וַיָּשִׂימוּ עַל הַשְּׁחִין וַיֶּחִי` (מלכים-ב כ:ז)
`וְהָיָה בַּיּוֹם הַהוּא יְחַיֶּה אִישׁ עֶגְלַת בָּקָר וּשְׁתֵּי צֹאן` (ישעיהו ז:כא)
`מֵתִים בַּל יִחְיוּ רְפָאִים בַּל יָקֻמוּ לָכֵן פָּקַדְתָּ וַתַּשְׁמִידֵם וַתְּאַבֵּד כָּל זֵכֶר לָמוֹ` (ישעיהו כו:יד)
`יִחְיוּ מֵתֶיךָ נְבֵלָתִי יְקוּמוּן הָקִיצוּ וְרַנְּנוּ שֹׁכְנֵי עָפָר כִּי טַל אוֹרֹת טַלֶּךָ וָאָרֶץ רְפָאִים תַּפִּיל` (ישעיהו כו:יט)
`בַּיָּמִים הָהֵם חָלָה חִזְקִיָּהוּ לָמוּת וַיָּבוֹא אֵלָיו יְשַׁעְיָהוּ בֶן אָמוֹץ הַנָּבִיא וַיֹּאמֶר אֵלָיו כֹּה אָמַר יְהוָה צַו לְבֵיתֶךָ כִּי מֵת אַתָּה וְלֹא תִחְיֶה` (ישעיהו לח:א)
`מִכְתָּב לְחִזְקִיָּהוּ מֶלֶךְ יְהוּדָה בַּחֲלֹתוֹ וַיְחִי מֵחָלְיוֹ` (ישעיהו לח:ט)
`אֲדֹנָי עֲלֵיהֶם יִחְיוּ וּלְכָל בָּהֶן חַיֵּי רוּחִי וְתַחֲלִימֵנִי וְהַחֲיֵנִי` (ישעיהו לח:טז)
`וַיֹּאמֶר יְשַׁעְיָהוּ יִשְׂאוּ דְּבֶלֶת תְּאֵנִים וְיִמְרְחוּ עַל הַשְּׁחִין וְיֶחִי` (ישעיהו לח:כא)
`הַטּוּ אָזְנְכֶם וּלְכוּ אֵלַי שִׁמְעוּ וּתְחִי נַפְשְׁכֶם וְאֶכְרְתָה לָכֶם בְּרִית עוֹלָם חַסְדֵי דָוִד הַנֶּאֱמָנִים` (ישעיהו נה:ג)
`כִּי כֹה אָמַר רָם וְנִשָּׂא שֹׁכֵן עַד וְקָדוֹשׁ שְׁמוֹ מָרוֹם וְקָדוֹשׁ אֶשְׁכּוֹן וְאֶת דַּכָּא וּשְׁפַל רוּחַ לְהַחֲיוֹת רוּחַ שְׁפָלִים וּלְהַחֲיוֹת לֵב נִדְכָּאִים` (ישעיהו נז:טו)
`הַיֹּשֵׁב בָּעִיר הַזֹּאת יָמוּת בַּחֶרֶב וּבָרָעָב וּבַדָּבֶר וְהַיּוֹצֵא וְנָפַל עַל הַכַּשְׂדִּים הַצָּרִים עֲלֵיכֶם יִחְיֶה וְהָיְתָה לּוֹ נַפְשׁוֹ לְשָׁלָל` (ירמיהו כא:ט)
`וְאֶל צִדְקִיָּה מֶלֶךְ יְהוּדָה דִּבַּרְתִּי כְּכָל הַדְּבָרִים הָאֵלֶּה לֵאמֹר הָבִיאוּ אֶת צַוְּארֵיכֶם בְּעֹל מֶלֶךְ בָּבֶל וְעִבְדוּ אֹתוֹ וְעַמּוֹ וִחְיוּ` (ירמיהו כז:יב)
`אַל תִּשְׁמְעוּ אֲלֵיהֶם עִבְדוּ אֶת מֶלֶךְ בָּבֶל וִחְיוּ לָמָּה תִהְיֶה הָעִיר הַזֹּאת חָרְבָּה` (ירמיהו כז:יז)
`וּבַיִת לֹא תִבְנוּ וְזֶרַע לֹא תִזְרָעוּ וְכֶרֶם לֹא תִטָּעוּ וְלֹא יִהְיֶה לָכֶם כִּי בָּאֳהָלִים תֵּשְׁבוּ כָּל יְמֵיכֶם לְמַעַן תִּחְיוּ יָמִים רַבִּים עַל פְּנֵי הָאֲדָמָה אֲשֶׁר אַתֶּם גָּרִים שָׁם` (ירמיהו לה:ז)
`כֹּה אָמַר יְהוָה הַיֹּשֵׁב בָּעִיר הַזֹּאת יָמוּת בַּחֶרֶב בָּרָעָב וּבַדָּבֶר וְהַיֹּצֵא אֶל הַכַּשְׂדִּים יִחְיֶה וְהָיְתָה לּוֹ נַפְשׁוֹ לְשָׁלָל וָחָי` (ירמיהו לח:ב)
`וַיֹּאמֶר יִרְמְיָהוּ אֶל צִדְקִיָּהוּ כֹּה אָמַר יְהוָה אֱלֹהֵי צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל אִם יָצֹא תֵצֵא אֶל שָׂרֵי מֶלֶךְ בָּבֶל וְחָיְתָה נַפְשֶׁךָ וְהָעִיר הַזֹּאת לֹא תִשָּׂרֵף בָּאֵשׁ וְחָיִתָה אַתָּה וּבֵיתֶךָ` (ירמיהו לח:יז)
`וַיֹּאמֶר יִרְמְיָהוּ לֹא יִתֵּנוּ שְׁמַע נָא בְּקוֹל יְהוָה לַאֲשֶׁר אֲנִי דֹּבֵר אֵלֶיךָ וְיִיטַב לְךָ וּתְחִי נַפְשֶׁךָ` (ירמיהו לח:כ)
`עָזְבָה יְתֹמֶיךָ אֲנִי אֲחַיֶּה וְאַלְמְנֹתֶיךָ עָלַי תִּבְטָחוּ` (ירמיהו מט:יא)
`בְּאָמְרִי לָרָשָׁע מוֹת תָּמוּת וְלֹא הִזְהַרְתּוֹ וְלֹא דִבַּרְתָּ לְהַזְהִיר רָשָׁע מִדַּרְכּוֹ הָרְשָׁעָה לְחַיֹּתוֹ הוּא רָשָׁע בַּעֲוֺנוֹ יָמוּת וְדָמוֹ מִיָּדְךָ אֲבַקֵּשׁ` (יחזקאל ג:יח)
`וְאַתָּה כִּי הִזְהַרְתּוֹ צַדִּיק לְבִלְתִּי חֲטֹא צַדִּיק וְהוּא לֹא חָטָא חָיוֹ יִחְיֶה כִּי נִזְהָר וְאַתָּה אֶת נַפְשְׁךָ הִצַּלְתָּ` (יחזקאל ג:כא)
`וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יְהוִה הוֹי לִמְתַפְּרוֹת כְּסָתוֹת עַל כָּל אַצִּילֵי יָדַי וְעֹשׂוֹת הַמִּסְפָּחוֹת עַל רֹאשׁ כָּל קוֹמָה לְצוֹדֵד נְפָשׁוֹת הַנְּפָשׁוֹת תְּצוֹדֵדְנָה לְעַמִּי וּנְפָשׁוֹת לָכֶנָה תְחַיֶּינָה` (יחזקאל יג:יח)
`וַתְּחַלֶּלְנָה אֹתִי אֶל עַמִּי בְּשַׁעֲלֵי שְׂעֹרִים וּבִפְתוֹתֵי לֶחֶם לְהָמִית נְפָשׁוֹת אֲשֶׁר לֹא תְמוּתֶנָה וּלְחַיּוֹת נְפָשׁוֹת אֲשֶׁר לֹא תִחְיֶינָה בְּכַזֶּבְכֶם לְעַמִּי שֹׁמְעֵי כָזָב` (יחזקאל יג:יט)
`יַעַן הַכְאוֹת לֵב צַדִּיק שֶׁקֶר וַאֲנִי לֹא הִכְאַבְתִּיו וּלְחַזֵּק יְדֵי רָשָׁע לְבִלְתִּי שׁוּב מִדַּרְכּוֹ הָרָע לְהַחֲיֹתוֹ` (יחזקאל יג:כב)
`וָאֶעֱבֹר עָלַיִךְ וָאֶרְאֵךְ מִתְבּוֹסֶסֶת בְּדָמָיִךְ וָאֹמַר לָךְ בְּדָמַיִךְ חֲיִי וָאֹמַר לָךְ בְּדָמַיִךְ חֲיִי` (יחזקאל טז:ו)
`בְּחֻקּוֹתַי יְהַלֵּךְ וּמִשְׁפָּטַי שָׁמַר לַעֲשׂוֹת אֱמֶת צַדִּיק הוּא חָיֹה יִחְיֶה נְאֻם אֲדֹנָי יְהוִה` (יחזקאל יח:ט)
`בַּנֶּשֶׁךְ נָתַן וְתַרְבִּית לָקַח וָחָי לֹא יִחְיֶה אֵת כָּל הַתּוֹעֵבוֹת הָאֵלֶּה עָשָׂה מוֹת יוּמָת דָּמָיו בּוֹ יִהְיֶה` (יחזקאל יח:יג)
`מֵעָנִי הֵשִׁיב יָדוֹ נֶשֶׁךְ וְתַרְבִּית לֹא לָקָח מִשְׁפָּטַי עָשָׂה בְּחֻקּוֹתַי הָלָךְ הוּא לֹא יָמוּת בַּעֲוֺן אָבִיו חָיֹה יִחְיֶה` (יחזקאל יח:יז)
`וַאֲמַרְתֶּם מַדֻּעַ לֹא נָשָׂא הַבֵּן בַּעֲוֺן הָאָב וְהַבֵּן מִשְׁפָּט וּצְדָקָה עָשָׂה אֵת כָּל חֻקּוֹתַי שָׁמַר וַיַּעֲשֶׂה אֹתָם חָיֹה יִחְיֶה` (יחזקאל יח:יט)
`וְהָרָשָׁע כִּי יָשׁוּב מִכָּל חַטָּאתוֹ אֲשֶׁר עָשָׂה וְשָׁמַר אֶת כָּל חֻקּוֹתַי וְעָשָׂה מִשְׁפָּט וּצְדָקָה חָיֹה יִחְיֶה לֹא יָמוּת` (יחזקאל יח:כא)
`כָּל פְּשָׁעָיו אֲשֶׁר עָשָׂה לֹא יִזָּכְרוּ לוֹ בְּצִדְקָתוֹ אֲשֶׁר עָשָׂה יִחְיֶה` (יחזקאל יח:כב)
`הֶחָפֹץ אֶחְפֹּץ מוֹת רָשָׁע נְאֻם אֲדֹנָי יְהוִה הֲלוֹא בְּשׁוּבוֹ מִדְּרָכָיו וְחָיָה` (יחזקאל יח:כג)
`וּבְשׁוּב צַדִּיק מִצִּדְקָתוֹ וְעָשָׂה עָוֶל כְּכֹל הַתּוֹעֵבוֹת אֲשֶׁר עָשָׂה הָרָשָׁע יַעֲשֶׂה וָחָי כָּל צִדְקָתוֹ אֲשֶׁר עָשָׂה לֹא תִזָּכַרְנָה בְּמַעֲלוֹ אֲשֶׁר מָעַל וּבְחַטָּאתוֹ אֲשֶׁר חָטָא בָּם יָמוּת` (יחזקאל יח:כד)
`וּבְשׁוּב רָשָׁע מֵרִשְׁעָתוֹ אֲשֶׁר עָשָׂה וַיַּעַשׂ מִשְׁפָּט וּצְדָקָה הוּא אֶת נַפְשׁוֹ יְחַיֶּה` (יחזקאל יח:כז)
`וַיִּרְאֶה וַיָּשׁוֹב מִכָּל פְּשָׁעָיו אֲשֶׁר עָשָׂה חָיוֹ יִחְיֶה לֹא יָמוּת` (יחזקאל יח:כח)
`כִּי לֹא אֶחְפֹּץ בְּמוֹת הַמֵּת נְאֻם אֲדֹנָי יְהוִה וְהָשִׁיבוּ וִחְיוּ` (יחזקאל יח:לב)
`וָאֶתֵּן לָהֶם אֶת חֻקּוֹתַי וְאֶת מִשְׁפָּטַי הוֹדַעְתִּי אוֹתָם אֲשֶׁר יַעֲשֶׂה אוֹתָם הָאָדָם וָחַי בָּהֶם` (יחזקאל כ:יא)
`וַיַּמְרוּ בִי בֵית יִשְׂרָאֵל בַּמִּדְבָּר בְּחֻקּוֹתַי לֹא הָלָכוּ וְאֶת מִשְׁפָּטַי מָאָסוּ אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם וָחַי בָּהֶם וְאֶת שַׁבְּתֹתַי חִלְּלוּ מְאֹד וָאֹמַר לִשְׁפֹּךְ חֲמָתִי עֲלֵיהֶם בַּמִּדְבָּר לְכַלּוֹתָם` (יחזקאל כ:יג)
`וַיַּמְרוּ בִי הַבָּנִים בְּחֻקּוֹתַי לֹא הָלָכוּ וְאֶת מִשְׁפָּטַי לֹא שָׁמְרוּ לַעֲשׂוֹת אוֹתָם אֲשֶׁר יַעֲשֶׂה אוֹתָם הָאָדָם וָחַי בָּהֶם אֶת שַׁבְּתוֹתַי חִלֵּלוּ וָאֹמַר לִשְׁפֹּךְ חֲמָתִי עֲלֵיהֶם לְכַלּוֹת אַפִּי בָּם בַּמִּדְבָּר` (יחזקאל כ:כא)
`וְגַם אֲנִי נָתַתִּי לָהֶם חֻקִּים לֹא טוֹבִים וּמִשְׁפָּטִים לֹא יִחְיוּ בָּהֶם` (יחזקאל כ:כה)
`וְאַתָּה בֶן אָדָם אֱמֹר אֶל בֵּית יִשְׂרָאֵל כֵּן אֲמַרְתֶּם לֵאמֹר כִּי פְשָׁעֵינוּ וְחַטֹּאתֵינוּ עָלֵינוּ וּבָם אֲנַחְנוּ נְמַקִּים וְאֵיךְ נִחְיֶה` (יחזקאל לג:י)
`אֱמֹר אֲלֵיהֶם חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם אֶחְפֹּץ בְּמוֹת הָרָשָׁע כִּי אִם בְּשׁוּב רָשָׁע מִדַּרְכּוֹ וְחָיָה שׁוּבוּ שׁוּבוּ מִדַּרְכֵיכֶם הָרָעִים וְלָמָּה תָמוּתוּ בֵּית יִשְׂרָאֵל` (יחזקאל לג:יא)
`וְאַתָּה בֶן אָדָם אֱמֹר אֶל בְּנֵי עַמְּךָ צִדְקַת הַצַּדִּיק לֹא תַצִּילֶנּוּ בְּיוֹם פִּשְׁעוֹ וְרִשְׁעַת הָרָשָׁע לֹא יִכָּשֶׁל בָּהּ בְּיוֹם שׁוּבוֹ מֵרִשְׁעוֹ וְצַדִּיק לֹא יוּכַל לִחְיוֹת בָּהּ בְּיוֹם חֲטֹאתוֹ` (יחזקאל לג:יב)
`בְּאָמְרִי לַצַּדִּיק חָיֹה יִחְיֶה וְהוּא בָטַח עַל צִדְקָתוֹ וְעָשָׂה עָוֶל כָּל צִדְקָתוֹ לֹא תִזָּכַרְנָה וּבְעַוְלוֹ אֲשֶׁר עָשָׂה בּוֹ יָמוּת` (יחזקאל לג:יג)
`חֲבֹל יָשִׁיב רָשָׁע גְּזֵלָה יְשַׁלֵּם בְּחֻקּוֹת הַחַיִּים הָלַךְ לְבִלְתִּי עֲשׂוֹת עָוֶל חָיוֹ יִחְיֶה לֹא יָמוּת` (יחזקאל לג:טו)
`כָּל חַטָּאתוֹ אֲשֶׁר חָטָא לֹא תִזָּכַרְנָה לוֹ מִשְׁפָּט וּצְדָקָה עָשָׂה חָיוֹ יִחְיֶה` (יחזקאל לג:טז)
`וּבְשׁוּב רָשָׁע מֵרִשְׁעָתוֹ וְעָשָׂה מִשְׁפָּט וּצְדָקָה עֲלֵיהֶם הוּא יִחְיֶה` (יחזקאל לג:יט)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם הֲתִחְיֶינָה הָעֲצָמוֹת הָאֵלֶּה וָאֹמַר אֲדֹנָי יְהוִה אַתָּה יָדָעְתָּ` (יחזקאל לז:ג)
`כֹּה אָמַר אֲדֹנָי יְהוִה לָעֲצָמוֹת הָאֵלֶּה הִנֵּה אֲנִי מֵבִיא בָכֶם רוּחַ וִחְיִיתֶם` (יחזקאל לז:ה)
`וְנָתַתִּי עֲלֵיכֶם גִּדִים וְהַעֲלֵתִי עֲלֵיכֶם בָּשָׂר וְקָרַמְתִּי עֲלֵיכֶם עוֹר וְנָתַתִּי בָכֶם רוּחַ וִחְיִיתֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה` (יחזקאל לז:ו)
`וַיֹּאמֶר אֵלַי הִנָּבֵא אֶל הָרוּחַ הִנָּבֵא בֶן אָדָם וְאָמַרְתָּ אֶל הָרוּחַ כֹּה אָמַר אֲדֹנָי יְהוִה מֵאַרְבַּע רוּחוֹת בֹּאִי הָרוּחַ וּפְחִי בַּהֲרוּגִים הָאֵלֶּה וְיִחְיוּ` (יחזקאל לז:ט)
`וְהִנַּבֵּאתִי כַּאֲשֶׁר צִוָּנִי וַתָּבוֹא בָהֶם הָרוּחַ וַיִּחְיוּ וַיַּעַמְדוּ עַל רַגְלֵיהֶם חַיִל גָּדוֹל מְאֹד מְאֹד` (יחזקאל לז:י)
`וְנָתַתִּי רוּחִי בָכֶם וִחְיִיתֶם וְהִנַּחְתִּי אֶתְכֶם עַל אַדְמַתְכֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה דִּבַּרְתִּי וְעָשִׂיתִי נְאֻם יְהוָה` (יחזקאל לז:יד)
`וְהָיָה כָל נֶפֶשׁ חַיָּה אֲ‍שֶׁר יִשְׁרֹץ אֶל כָּל אֲשֶׁר יָבוֹא שָׁם נַחֲלַיִם יִחְיֶה וְהָיָה הַדָּגָה רַבָּה מְאֹד כִּי בָאוּ שָׁמָּה הַמַּיִם הָאֵלֶּה וְיֵרָפְאוּ וָחָי כֹּל אֲשֶׁר יָבוֹא שָׁמָּה הַנָּחַל` (יחזקאל מז:ט)
`יְחַיֵּנוּ מִיֹּמָיִם בַּיּוֹם הַשְּׁלִישִׁי יְקִמֵנוּ וְנִחְיֶה לְפָנָיו` (הושע ו:ב)
`יָשֻׁבוּ יֹשְׁבֵי בְצִלּוֹ יְחַיּוּ דָגָן וְיִפְרְחוּ כַגָּפֶן זִכְרוֹ כְּיֵין לְבָנוֹן` (הושע יד:ז)
`כִּי כֹה אָמַר יְהוָה לְבֵית יִשְׂרָאֵל דִּרְשׁוּנִי וִחְיוּ` (עמוס ה:ד)
`דִּרְשׁוּ אֶת יְהוָה וִחְיוּ פֶּן יִצְלַח כָּאֵשׁ בֵּית יוֹסֵף וְאָכְלָה וְאֵין מְכַבֶּה לְבֵית אֵל` (עמוס ה:ו)
`דִּרְשׁוּ טוֹב וְאַל רָע לְמַעַן תִּחְיוּ וִיהִי כֵן יְהוָה אֱלֹהֵי צְבָאוֹת אִתְּכֶם כַּאֲשֶׁר אֲמַרְתֶּם` (עמוס ה:יד)
`הִנֵּה עֻפְּלָה לֹא יָשְׁרָה נַפְשׁוֹ בּוֹ וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה` (חבקוק ב:ד)
`יְהוָה שָׁמַעְתִּי שִׁמְעֲךָ יָרֵאתִי יְהוָה פָּעָלְךָ בְּקֶרֶב שָׁנִים חַיֵּיהוּ בְּקֶרֶב שָׁנִים תּוֹדִיעַ בְּרֹגֶז רַחֵם תִּזְכּוֹר` (חבקוק ג:ב)
`אֲבוֹתֵיכֶם אַיֵּה הֵם וְהַנְּבִאִים הַלְעוֹלָם יִחְיוּ` (זכריה א:ה)
`וְאֶזְרָעֵם בָּעַמִּים וּבַמֶּרְחַקִּים יִזְכְּרוּנִי וְחָיוּ אֶת בְּנֵיהֶם וָשָׁבוּ` (זכריה י:ט)
`וְהָיָה כִּי יִנָּבֵא אִישׁ עוֹד וְאָמְרוּ אֵלָיו אָבִיו וְאִמּוֹ יֹלְדָיו לֹא תִחְיֶה כִּי שֶׁקֶר דִּבַּרְתָּ בְּשֵׁם יְהוָה וּדְקָרֻהוּ אָבִיהוּ וְאִמּוֹ יֹלְדָיו בְּהִנָּבְאוֹ` (זכריה יג:ג)
`וַיִּבֶן הָעִיר מִסָּבִיב מִן הַמִּלּוֹא וְעַד הַסָּבִיב וְיוֹאָב יְחַיֶּה אֶת שְׁאָר הָעִיר` (דברי-הימים-א יא:ח)
`וַיּוֹצִיאוּ אֶת בֶּן הַמֶּלֶךְ וַיִּתְּנוּ עָלָיו אֶת הַנֵּזֶר וְאֶת הָעֵדוּת וַיַּמְלִיכוּ אֹתוֹ וַיִּמְשָׁחֻהוּ יְהוֹיָדָע וּבָנָיו וַיֹּאמְרוּ יְחִי הַמֶּלֶךְ` (דברי-הימים-ב כג:יא)
`וַיְחִי אֲמַצְיָהוּ בֶן יוֹאָשׁ מֶלֶךְ יְהוּדָה אַחֲרֵי מוֹת יוֹאָשׁ בֶּן יְהוֹאָחָז מֶלֶךְ יִשְׂרָאֵל חֲמֵשׁ עֶשְׂרֵה שָׁנָה` (דברי-הימים-ב כה:כה)
`יֹאכְלוּ עֲנָוִים וְיִשְׂבָּעוּ יְהַלְלוּ יְהוָה דֹּרְשָׁיו יְחִי לְבַבְכֶם לָעַד` (תהילים כב:כו)
`אָכְלוּ וַיִּשְׁתַּחֲוּוּ כָּל דִּשְׁנֵי אֶרֶץ לְפָנָיו יִכְרְעוּ כָּל יוֹרְדֵי עָפָר וְנַפְשׁוֹ לֹא חִיָּה` (תהילים כב:כט)
`יְהוָה הֶעֱלִיתָ מִן שְׁאוֹל נַפְשִׁי חִיִּיתַנִי מִיּוֹרְדֵי בוֹר` (תהילים ל:ג)
`לְהַצִּיל מִמָּוֶת נַפְשָׁם וּלְחַיּוֹתָם בָּרָעָב` (תהילים לג:יט)
`יְהוָה יִשְׁמְרֵהוּ וִיחַיֵּהוּ יֶאְשֹּׁר בָּאָרֶץ וְאַל תִּתְּנֵהוּ בְּנֶפֶשׁ אֹיְבָיו` (תהילים מא:ב)
`וִיחִי עוֹד לָנֶצַח לֹא יִרְאֶה הַשָּׁחַת` (תהילים מט:ט)
`רָאוּ עֲנָוִים יִשְׂמָחוּ דֹּרְשֵׁי אֱלֹהִים וִיחִי לְבַבְכֶם` (תהילים סט:לב)
`אֲשֶׁר הִרְאִיתַנוּ צָרוֹת רַבּוֹת וְרָעוֹת תָּשׁוּב תְּחַיֵּינוּ וּמִתְּהֹמוֹת הָאָרֶץ תָּשׁוּב תַּעֲלֵנִי` (תהילים עא:כ)
`וִיחִי וְיִתֶּן לוֹ מִזְּהַב שְׁבָא וְיִתְפַּלֵּל בַּעֲדוֹ תָמִיד כָּל הַיּוֹם יְבָרֲכֶנְהוּ` (תהילים עב:טו)
`וְלֹא נָסוֹג מִמֶּךָּ תְּחַיֵּנוּ וּבְשִׁמְךָ נִקְרָא` (תהילים פ:יח)
`הֲלֹא אַתָּה תָּשׁוּב תְּחַיֵּנוּ וְעַמְּךָ יִשְׂמְחוּ בָךְ` (תהילים פה:ו)
`מִי גֶבֶר יִחְיֶה וְלֹא יִרְאֶה מָּוֶת יְמַלֵּט נַפְשׁוֹ מִיַּד שְׁאוֹל סֶלָה` (תהילים פט:מח)
`לֹא אָמוּת כִּי אֶחְיֶה וַאֲסַפֵּר מַעֲשֵׂי יָהּ` (תהילים קיח:יז)
`גְּמֹל עַל עַבְדְּךָ אֶחְיֶה וְאֶשְׁמְרָה דְבָרֶךָ` (תהילים קיט:יז)
`דָּבְקָה לֶעָפָר נַפְשִׁי חַיֵּנִי כִּדְבָרֶךָ` (תהילים קיט:כה)
`הַעֲבֵר עֵינַי מֵרְאוֹת שָׁוְא בִּדְרָכֶךָ חַיֵּנִי` (תהילים קיט:לז)
`הִנֵּה תָּאַבְתִּי לְפִקֻּדֶיךָ בְּצִדְקָתְךָ חַיֵּנִי` (תהילים קיט:מ)
`זֹאת נֶחָמָתִי בְעָנְיִי כִּי אִמְרָתְךָ חִיָּתְנִי` (תהילים קיט:נ)
`יְבֹאוּנִי רַחֲמֶיךָ וְאֶחְיֶה כִּי תוֹרָתְךָ שַׁעֲשֻׁעָי` (תהילים קיט:עז)
`כְּחַסְדְּךָ חַיֵּנִי וְאֶשְׁמְרָה עֵדוּת פִּיךָ` (תהילים קיט:פח)
`לְעוֹלָם לֹא אֶשְׁכַּח פִּקּוּדֶיךָ כִּי בָם חִיִּיתָנִי` (תהילים קיט:צג)
`נַעֲנֵיתִי עַד מְאֹד יְהוָה חַיֵּנִי כִדְבָרֶךָ` (תהילים קיט:קז)
`סָמְכֵנִי כְאִמְרָתְךָ וְאֶחְיֶה וְאַל תְּבִישֵׁנִי מִשִּׂבְרִי` (תהילים קיט:קיו)
`צֶדֶק עֵדְוֺתֶיךָ לְעוֹלָם הֲבִינֵנִי וְאֶחְיֶה` (תהילים קיט:קמד)
`קוֹלִי שִׁמְעָה כְחַסְדֶּךָ יְהוָה כְּמִשְׁפָּטֶךָ חַיֵּנִי` (תהילים קיט:קמט)
`רִיבָה רִיבִי וּגְאָלֵנִי לְאִמְרָתְךָ חַיֵּנִי` (תהילים קיט:קנד)
`רַחֲמֶיךָ רַבִּים יְהוָה כְּמִשְׁפָּטֶיךָ חַיֵּנִי` (תהילים קיט:קנו)
`רְאֵה כִּי פִקּוּדֶיךָ אָהָבְתִּי יְהוָה כְּחַסְדְּךָ חַיֵּנִי` (תהילים קיט:קנט)
`תְּחִי נַפְשִׁי וּתְהַלְלֶךָּ וּמִשְׁפָּטֶךָ יַעֲזְרֻנִי` (תהילים קיט:קעה)
`אִם אֵלֵךְ בְּקֶרֶב צָרָה תְּחַיֵּנִי עַל אַף אֹיְבַי תִּשְׁלַח יָדֶךָ וְתוֹשִׁיעֵנִי יְמִינֶךָ` (תהילים קלח:ז)
`לְמַעַן שִׁמְךָ יְהוָה תְּחַיֵּנִי בְּצִדְקָתְךָ תוֹצִיא מִצָּרָה נַפְשִׁי` (תהילים קמג:יא)
`מָאַסְתִּי לֹא לְעֹלָם אֶחְיֶה חֲדַל מִמֶּנִּי כִּי הֶבֶל יָמָי` (איוב ז:טז)
`אִם יָמוּת גֶּבֶר הֲיִחְיֶה כָּל יְמֵי צְבָאִי אֲיַחֵל עַד בּוֹא חֲלִיפָתִי` (איוב יד:יד)
`מַדּוּעַ רְשָׁעִים יִחְיוּ עָתְקוּ גַּם גָּבְרוּ חָיִל` (איוב כא:ז)
`רוּחַ אֵל עָשָׂתְנִי וְנִשְׁמַת שַׁדַּי תְּחַיֵּנִי` (איוב לג:ד)
`לֹא יְחַיֶּה רָשָׁע וּמִשְׁפַּט עֲנִיִּים יִתֵּן` (איוב לו:ו)
`וַיְחִי אִיּוֹב אַחֲרֵי זֹאת מֵאָה וְאַרְבָּעִים שָׁנָה וַיַּרְא אֶת בָּנָיו וְאֶת בְּנֵי בָנָיו אַרְבָּעָה דֹּרוֹת` (איוב מב:טז)
`וַיֹּרֵנִי וַיֹּאמֶר לִי יִתְמָךְ דְּבָרַי לִבֶּךָ שְׁמֹר מִצְוֺתַי וֶחְיֵה` (משלי ד:ד)
`שְׁמֹר מִצְוֺתַי וֶחְיֵה וְתוֹרָתִי כְּאִישׁוֹן עֵינֶיךָ` (משלי ז:ב)
`עִזְבוּ פְתָאיִם וִחְיוּ וְאִשְׁרוּ בְּדֶרֶךְ בִּינָה` (משלי ט:ו)
`עֹכֵר בֵּיתוֹ בּוֹצֵעַ בָּצַע וְשׂוֹנֵא מַתָּנֹת יִחְיֶה` (משלי טו:כז)
`אִם יוֹלִיד אִישׁ מֵאָה וְשָׁנִים רַבּוֹת יִחְיֶה וְרַב שֶׁיִּהְיוּ יְמֵי שָׁנָיו וְנַפְשׁוֹ לֹא תִשְׂבַּע מִן הַטּוֹבָה וְגַם קְבוּרָה לֹא הָיְתָה לּוֹ אָמַרְתִּי טוֹב מִמֶּנּוּ הַנָּפֶל` (קהלת ו:ג)
`וְאִלּוּ חָיָה אֶלֶף שָׁנִים פַּעֲמַיִם וְטוֹבָה לֹא רָאָה הֲלֹא אֶל מָקוֹם אֶחָד הַכֹּל הוֹלֵךְ` (קהלת ו:ו)
`כִּי בְּצֵל הַחָכְמָה בְּצֵל הַכָּסֶף וְיִתְרוֹן דַּעַת הַחָכְמָה תְּחַיֶּה בְעָלֶיהָ` (קהלת ז:יב)
`כִּי אִם שָׁנִים הַרְבֵּה יִחְיֶה הָאָדָם בְּכֻלָּם יִשְׂמָח וְיִזְכֹּר אֶת יְמֵי הַחֹשֶׁךְ כִּי הַרְבֵּה יִהְיוּ כָּל שֶׁבָּא הָבֶל` (קהלת יא:ח)
`רוּחַ אַפֵּינוּ מְשִׁיחַ יְהוָה נִלְכַּד בִּשְׁחִיתוֹתָם אֲשֶׁר אָמַרְנוּ בְּצִלּוֹ נִחְיֶה בַגּוֹיִם` (איכה ד:כ)
`כָּל עַבְדֵי הַמֶּלֶךְ וְעַם מְדִינוֹת הַמֶּלֶךְ יוֹדְעִים אֲשֶׁר כָּל אִישׁ וְאִשָּׁה אֲשֶׁר יָבוֹא אֶל הַמֶּלֶךְ אֶל הֶחָצֵר הַפְּנִימִית אֲשֶׁר לֹא יִקָּרֵא אַחַת דָּתוֹ לְהָמִית לְבַד מֵאֲשֶׁר יוֹשִׁיט לוֹ הַמֶּלֶךְ אֶת שַׁרְבִיט הַזָּהָב וְחָיָה וַאֲנִי לֹא נִקְרֵאתי לָבוֹא אֶל הַמֶּלֶךְ זֶה שְׁלוֹשִׁים יוֹם` (אסתר ד:יא)
`וָאֹמַר לַמֶּלֶךְ הַמֶּלֶךְ לְעוֹלָם יִחְיֶה מַדּוּעַ לֹא יֵרְעוּ פָנַי אֲשֶׁר הָעִיר בֵּית קִבְרוֹת אֲבֹתַי חֲרֵבָה וּשְׁעָרֶיהָ אֻכְּלוּ בָאֵשׁ` (נחמיה ב:ג)
`וַיֹּאמֶר לִפְנֵי אֶחָיו וְחֵיל שֹׁמְרוֹן וַיֹּאמֶר מָה הַיְּהוּדִים הָאֲמֵלָלִים עֹשִׂים הֲיַעַזְבוּ לָהֶם הֲיִזְבָּחוּ הַיְכַלּוּ בַיּוֹם הַיְחַיּוּ אֶת הָאֲבָנִים מֵעֲרֵמוֹת הֶעָפָר וְהֵמָּה שְׂרוּפוֹת` (נחמיה ד:ב)
`וְיֵשׁ אֲשֶׁר אֹמְרִים בָּנֵינוּ וּבְנֹתֵינוּ אֲנַחְנוּ רַבִּים וְנִקְחָה דָגָן וְנֹאכְלָה וְנִחְיֶה` (נחמיה ה:ב)
`וָאֹמְרָה הַאִישׁ כָּמוֹנִי יִבְרָח וּמִי כָמוֹנִי אֲשֶׁר יָבוֹא אֶל הַהֵיכָל וָחָי לֹא אָבוֹא` (נחמיה ו:יא)
`אַתָּה הוּא יְהוָה לְבַדֶּךָ אַתְּ עָשִׂיתָ אֶת הַשָּׁמַיִם שְׁמֵי הַשָּׁמַיִם וְכָל צְבָאָם הָאָרֶץ וְכָל אֲשֶׁר עָלֶיהָ הַיַּמִּים וְכָל אֲשֶׁר בָּהֶם וְאַתָּה מְחַיֶּה אֶת כֻּלָּם וּצְבָא הַשָּׁמַיִם לְךָ מִשְׁתַּחֲוִים` (נחמיה ט:ו)
`וַתָּעַד בָּהֶם לַהֲשִׁיבָם אֶל תּוֹרָתֶךָ וְהֵמָּה הֵזִידוּ וְלֹא שָׁמְעוּ לְמִצְוֺתֶיךָ וּבְמִשְׁפָּטֶיךָ חָטְאוּ בָם אֲשֶׁר יַעֲשֶׂה אָדָם וְחָיָה בָהֶם וַיִּתְּנוּ כָתֵף סוֹרֶרֶת וְעָרְפָּם הִקְשׁוּ וְלֹא שָׁמֵעוּ` (נחמיה ט:כט)
