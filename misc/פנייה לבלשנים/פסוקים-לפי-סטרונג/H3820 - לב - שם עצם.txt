`וַיַּרְא יְהוָה כִּי רַבָּה רָעַת הָאָדָם בָּאָרֶץ וְכָל יֵצֶר מַחְשְׁבֹת לִבּוֹ רַק רַע כָּל הַיּוֹם` (בראשית ו:ה)
`וַיִּנָּחֶם יְהוָה כִּי עָשָׂה אֶת הָאָדָם בָּאָרֶץ וַיִּתְעַצֵּב אֶל לִבּוֹ` (בראשית ו:ו)
`וַיָּרַח יְהוָה אֶת רֵיחַ הַנִּיחֹחַ וַיֹּאמֶר יְהוָה אֶל לִבּוֹ לֹא אֹסִף לְקַלֵּל עוֹד אֶת הָאֲדָמָה בַּעֲבוּר הָאָדָם כִּי יֵצֶר לֵב הָאָדָם רַע מִנְּעֻרָיו וְלֹא אֹסִף עוֹד לְהַכּוֹת אֶת כָּל חַי כַּאֲשֶׁר עָשִׂיתִי` (בראשית ח:כא)
`וַיִּפֹּל אַבְרָהָם עַל פָּנָיו וַיִּצְחָק וַיֹּאמֶר בְּלִבּוֹ הַלְּבֶן מֵאָה שָׁנָה יִוָּלֵד וְאִם שָׂרָה הֲבַת תִּשְׁעִים שָׁנָה תֵּלֵד` (בראשית יז:יז)
`וְאֶקְחָה פַת לֶחֶם וְסַעֲדוּ לִבְּכֶם אַחַר תַּעֲבֹרוּ כִּי עַל כֵּן עֲבַרְתֶּם עַל עַבְדְּכֶם וַיֹּאמְרוּ כֵּן תַּעֲשֶׂה כַּאֲשֶׁר דִּבַּרְתָּ` (בראשית יח:ה)
`אֲנִי טֶרֶם אֲכַלֶּה לְדַבֵּר אֶל לִבִּי וְהִנֵּה רִבְקָה יֹצֵאת וְכַדָּהּ עַל שִׁכְמָהּ וַתֵּרֶד הָעַיְנָה וַתִּשְׁאָב וָאֹמַר אֵלֶיהָ הַשְׁקִינִי נָא` (בראשית כד:מה)
`וַיִּשְׂטֹם עֵשָׂו אֶת יַעֲקֹב עַל הַבְּרָכָה אֲשֶׁר בֵּרֲכוֹ אָבִיו וַיֹּאמֶר עֵשָׂו בְּלִבּוֹ יִקְרְבוּ יְמֵי אֵבֶל אָבִי וְאַהַרְגָה אֶת יַעֲקֹב אָחִי` (בראשית כז:מא)
`וַיִּגְנֹב יַעֲקֹב אֶת לֵב לָבָן הָאֲרַמִּי עַל בְּלִי הִגִּיד לוֹ כִּי בֹרֵחַ הוּא` (בראשית לא:כ)
`וַתִּדְבַּק נַפְשׁוֹ בְּדִינָה בַּת יַעֲקֹב וַיֶּאֱהַב אֶת הַנַּעֲרָ וַיְדַבֵּר עַל לֵב הַנַּעֲרָ` (בראשית לד:ג)
`וַיֹּאמֶר אֶל אֶחָיו הוּשַׁב כַּסְפִּי וְגַם הִנֵּה בְאַמְתַּחְתִּי וַיֵּצֵא לִבָּם וַיֶּחֶרְדוּ אִישׁ אֶל אָחִיו לֵאמֹר מַה זֹּאת עָשָׂה אֱלֹהִים לָנוּ` (בראשית מב:כח)
`וַיַּגִּדוּ לוֹ לֵאמֹר עוֹד יוֹסֵף חַי וְכִי הוּא מֹשֵׁל בְּכָל אֶרֶץ מִצְרָיִם וַיָּפָג לִבּוֹ כִּי לֹא הֶאֱמִין לָהֶם` (בראשית מה:כו)
`וְעַתָּה אַל תִּירָאוּ אָנֹכִי אֲכַלְכֵּל אֶתְכֶם וְאֶת טַפְּכֶם וַיְנַחֵם אוֹתָם וַיְדַבֵּר עַל לִבָּם` (בראשית נ:כא)
`וַיִּחַר אַף יְהוָה בְּמֹשֶׁה וַיֹּאמֶר הֲלֹא אַהֲרֹן אָחִיךָ הַלֵּוִי יָדַעְתִּי כִּי דַבֵּר יְדַבֵּר הוּא וְגַם הִנֵּה הוּא יֹצֵא לִקְרָאתֶךָ וְרָאֲךָ וְשָׂמַח בְּלִבּוֹ` (שמות ד:יד)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה בְּלֶכְתְּךָ לָשׁוּב מִצְרַיְמָה רְאֵה כָּל הַמֹּפְתִים אֲשֶׁר שַׂמְתִּי בְיָדֶךָ וַעֲשִׂיתָם לִפְנֵי פַרְעֹה וַאֲנִי אֲחַזֵּק אֶת לִבּוֹ וְלֹא יְשַׁלַּח אֶת הָעָם` (שמות ד:כא)
`וַאֲנִי אַקְשֶׁה אֶת לֵב פַּרְעֹה וְהִרְבֵּיתִי אֶת אֹתֹתַי וְאֶת מוֹפְתַי בְּאֶרֶץ מִצְרָיִם` (שמות ז:ג)
`וַיֶּחֱזַק לֵב פַּרְעֹה וְלֹא שָׁמַע אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר יְהוָה` (שמות ז:יג)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה כָּבֵד לֵב פַּרְעֹה מֵאֵן לְשַׁלַּח הָעָם` (שמות ז:יד)
`וַיַּעֲשׂוּ כֵן חַרְטֻמֵּי מִצְרַיִם בְּלָטֵיהֶם וַיֶּחֱזַק לֵב פַּרְעֹה וְלֹא שָׁמַע אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר יְהוָה` (שמות ז:כב)
`וַיִּפֶן פַּרְעֹה וַיָּבֹא אֶל בֵּיתוֹ וְלֹא שָׁת לִבּוֹ גַּם לָזֹאת` (שמות ז:כג)
`וַיַּרְא פַּרְעֹה כִּי הָיְתָה הָרְוָחָה וְהַכְבֵּד אֶת לִבּוֹ וְלֹא שָׁמַע אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר יְהוָה` (שמות ח:טו)
`וַיֹּאמְרוּ הַחַרְטֻמִּים אֶל פַּרְעֹה אֶצְבַּע אֱלֹהִים הִוא וַיֶּחֱזַק לֵב פַּרְעֹה וְלֹא שָׁמַע אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר יְהוָה` (שמות ח:יט)
`וַיַּכְבֵּד פַּרְעֹה אֶת לִבּוֹ גַּם בַּפַּעַם הַזֹּאת וְלֹא שִׁלַּח אֶת הָעָם` (שמות ח:לב)
`וַיִּשְׁלַח פַּרְעֹה וְהִנֵּה לֹא מֵת מִמִּקְנֵה יִשְׂרָאֵל עַד אֶחָד וַיִּכְבַּד לֵב פַּרְעֹה וְלֹא שִׁלַּח אֶת הָעָם` (שמות ט:ז)
`וַיְחַזֵּק יְהוָה אֶת לֵב פַּרְעֹה וְלֹא שָׁמַע אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר יְהוָה אֶל מֹשֶׁה` (שמות ט:יב)
`כִּי בַּפַּעַם הַזֹּאת אֲנִי שֹׁלֵחַ אֶת כָּל מַגֵּפֹתַי אֶל לִבְּךָ וּבַעֲבָדֶיךָ וּבְעַמֶּךָ בַּעֲבוּר תֵּדַע כִּי אֵין כָּמֹנִי בְּכָל הָאָרֶץ` (שמות ט:יד)
`וַאֲשֶׁר לֹא שָׂם לִבּוֹ אֶל דְּבַר יְהוָה וַיַּעֲזֹב אֶת עֲבָדָיו וְאֶת מִקְנֵהוּ בַּשָּׂדֶה` (שמות ט:כא)
`וַיַּרְא פַּרְעֹה כִּי חָדַל הַמָּטָר וְהַבָּרָד וְהַקֹּלֹת וַיֹּסֶף לַחֲטֹא וַיַּכְבֵּד לִבּוֹ הוּא וַעֲבָדָיו` (שמות ט:לד)
`וַיֶּחֱזַק לֵב פַּרְעֹה וְלֹא שִׁלַּח אֶת בְּנֵי יִשְׂרָאֵל כַּאֲשֶׁר דִּבֶּר יְהוָה בְּיַד מֹשֶׁה` (שמות ט:לה)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה בֹּא אֶל פַּרְעֹה כִּי אֲנִי הִכְבַּדְתִּי אֶת לִבּוֹ וְאֶת לֵב עֲבָדָיו לְמַעַן שִׁתִי אֹתֹתַי אֵלֶּה בְּקִרְבּוֹ` (שמות י:א)
`וַיְחַזֵּק יְהוָה אֶת לֵב פַּרְעֹה וְלֹא שִׁלַּח אֶת בְּנֵי יִשְׂרָאֵל` (שמות י:כ)
`וַיְחַזֵּק יְהוָה אֶת לֵב פַּרְעֹה וְלֹא אָבָה לְשַׁלְּחָם` (שמות י:כז)
`וּמֹשֶׁה וְאַהֲרֹן עָשׂוּ אֶת כָּל הַמֹּפְתִים הָאֵלֶּה לִפְנֵי פַרְעֹה וַיְחַזֵּק יְהוָה אֶת לֵב פַּרְעֹה וְלֹא שִׁלַּח אֶת בְּנֵי יִשְׂרָאֵל מֵאַרְצוֹ` (שמות יא:י)
`וְחִזַּקְתִּי אֶת לֵב פַּרְעֹה וְרָדַף אַחֲרֵיהֶם וְאִכָּבְדָה בְּפַרְעֹה וּבְכָל חֵילוֹ וְיָדְעוּ מִצְרַיִם כִּי אֲנִי יְהוָה וַיַּעֲשׂוּ כֵן` (שמות יד:ד)
`וַיְחַזֵּק יְהֹוָה אֶת לֵב פַּרְעֹה מֶלֶךְ מִצְרַיִם וַיִּרְדֹּף אַחֲרֵי בְּנֵי יִשְׂרָאֵל וּבְנֵי יִשְׂרָאֵל יֹצְאִים בְּיָד רָמָה` (שמות יד:ח)
`וַאֲנִי הִנְנִי מְחַזֵּק אֶת לֵב מִצְרַיִם וְיָבֹאוּ אַחֲרֵיהֶם וְאִכָּבְדָה בְּפַרְעֹה וּבְכָל חֵילוֹ בְּרִכְבּוֹ וּבְפָרָשָׁיו` (שמות יד:יז)
`וּבְרוּחַ אַפֶּיךָ נֶעֶרְמוּ מַיִם נִצְּבוּ כְמוֹ נֵד נֹזְלִים קָפְאוּ תְהֹמֹת בְּלֶב יָם` (שמות טו:ח)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל וְיִקְחוּ לִי תְּרוּמָה מֵאֵת כָּל אִישׁ אֲשֶׁר יִדְּבֶנּוּ לִבּוֹ תִּקְחוּ אֶת תְּרוּמָתִי` (שמות כה:ב)
`וְאַתָּה תְּדַבֵּר אֶל כָּל חַכְמֵי לֵב אֲשֶׁר מִלֵּאתִיו רוּחַ חָכְמָה וְעָשׂוּ אֶת בִּגְדֵי אַהֲרֹן לְקַדְּשׁוֹ לְכַהֲנוֹ לִי` (שמות כח:ג)
`וְנָשָׂא אַהֲרֹן אֶת שְׁמוֹת בְּנֵי יִשְׂרָאֵל בְּחֹשֶׁן הַמִּשְׁפָּט עַל לִבּוֹ בְּבֹאוֹ אֶל הַקֹּדֶשׁ לְזִכָּרֹן לִפְנֵי יְהוָה תָּמִיד` (שמות כח:כט)
`וְנָתַתָּ אֶל חֹשֶׁן הַמִּשְׁפָּט אֶת הָאוּרִים וְאֶת הַתֻּמִּים וְהָיוּ עַל לֵב אַהֲרֹן בְּבֹאוֹ לִפְנֵי יְהוָה וְנָשָׂא אַהֲרֹן אֶת מִשְׁפַּט בְּנֵי יִשְׂרָאֵל עַל לִבּוֹ לִפְנֵי יְהוָה תָּמִיד` (שמות כח:ל)
`וַאֲנִי הִנֵּה נָתַתִּי אִתּוֹ אֵת אָהֳלִיאָב בֶּן אֲחִיסָמָךְ לְמַטֵּה דָן וּבְלֵב כָּל חֲכַם לֵב נָתַתִּי חָכְמָה וְעָשׂוּ אֵת כָּל אֲשֶׁר צִוִּיתִךָ` (שמות לא:ו)
`קְחוּ מֵאִתְּכֶם תְּרוּמָה לַיהוָה כֹּל נְדִיב לִבּוֹ יְבִיאֶהָ אֵת תְּרוּמַת יְהוָה זָהָב וָכֶסֶף וּנְחֹשֶׁת` (שמות לה:ה)
`וְכָל חֲכַם לֵב בָּכֶם יָבֹאוּ וְיַעֲשׂוּ אֵת כָּל אֲשֶׁר צִוָּה יְהוָה` (שמות לה:י)
`וַיָּבֹאוּ כָּל אִישׁ אֲשֶׁר נְשָׂאוֹ לִבּוֹ וְכֹל אֲשֶׁר נָדְבָה רוּחוֹ אֹתוֹ הֵבִיאוּ אֶת תְּרוּמַת יְהוָה לִמְלֶאכֶת אֹהֶל מוֹעֵד וּלְכָל עֲבֹדָתוֹ וּלְבִגְדֵי הַקֹּדֶשׁ` (שמות לה:כא)
`וַיָּבֹאוּ הָאֲנָשִׁים עַל הַנָּשִׁים כֹּל נְדִיב לֵב הֵבִיאוּ חָח וָנֶזֶם וְטַבַּעַת וְכוּמָז כָּל כְּלִי זָהָב וְכָל אִישׁ אֲשֶׁר הֵנִיף תְּנוּפַת זָהָב לַיהוָה` (שמות לה:כב)
`וְכָל אִשָּׁה חַכְמַת לֵב בְּיָדֶיהָ טָווּ וַיָּבִיאוּ מַטְוֶה אֶת הַתְּכֵלֶת וְאֶת הָאַרְגָּמָן אֶת תּוֹלַעַת הַשָּׁנִי וְאֶת הַשֵּׁשׁ` (שמות לה:כה)
`וְכָל הַנָּשִׁים אֲשֶׁר נָשָׂא לִבָּן אֹתָנָה בְּחָכְמָה טָווּ אֶת הָעִזִּים` (שמות לה:כו)
`כָּל אִישׁ וְאִשָּׁה אֲשֶׁר נָדַב לִבָּם אֹתָם לְהָבִיא לְכָל הַמְּלָאכָה אֲשֶׁר צִוָּה יְהוָה לַעֲשׂוֹת בְּיַד מֹשֶׁה הֵבִיאוּ בְנֵי יִשְׂרָאֵל נְדָבָה לַיהוָה` (שמות לה:כט)
`וּלְהוֹרֹת נָתַן בְּלִבּוֹ הוּא וְאָהֳלִיאָב בֶּן אֲחִיסָמָךְ לְמַטֵּה דָן` (שמות לה:לד)
`מִלֵּא אֹתָם חָכְמַת לֵב לַעֲשׂוֹת כָּל מְלֶאכֶת חָרָשׁ וְחֹשֵׁב וְרֹקֵם בַּתְּכֵלֶת וּבָאַרְגָּמָן בְּתוֹלַעַת הַשָּׁנִי וּבַשֵּׁשׁ וְאֹרֵג עֹשֵׂי כָּל מְלָאכָה וְחֹשְׁבֵי מַחֲשָׁבֹת` (שמות לה:לה)
`וְעָשָׂה בְצַלְאֵל וְאָהֳלִיאָב וְכֹל אִישׁ חֲכַם לֵב אֲשֶׁר נָתַן יְהוָה חָכְמָה וּתְבוּנָה בָּהֵמָּה לָדַעַת לַעֲשֹׂת אֶת כָּל מְלֶאכֶת עֲבֹדַת הַקֹּדֶשׁ לְכֹל אֲשֶׁר צִוָּה יְהוָה` (שמות לו:א)
`וַיִּקְרָא מֹשֶׁה אֶל בְּצַלְאֵל וְאֶל אָהֳלִיאָב וְאֶל כָּל אִישׁ חֲכַם לֵב אֲשֶׁר נָתַן יְהוָה חָכְמָה בְּלִבּוֹ כֹּל אֲשֶׁר נְשָׂאוֹ לִבּוֹ לְקָרְבָה אֶל הַמְּלָאכָה לַעֲשֹׂת אֹתָהּ` (שמות לו:ב)
`וַיַּעֲשׂוּ כָל חֲכַם לֵב בְּעֹשֵׂי הַמְּלָאכָה אֶת הַמִּשְׁכָּן עֶשֶׂר יְרִיעֹת שֵׁשׁ מָשְׁזָר וּתְכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי כְּרֻבִים מַעֲשֵׂה חֹשֵׁב עָשָׂה אֹתָם` (שמות לו:ח)
`וַיֹּאמֶר מֹשֶׁה בְּזֹאת תֵּדְעוּן כִּי יְהוָה שְׁלָחַנִי לַעֲשׂוֹת אֵת כָּל הַמַּעֲשִׂים הָאֵלֶּה כִּי לֹא מִלִּבִּי` (במדבר טז:כח)
`אִם יִתֶּן לִי בָלָק מְלֹא בֵיתוֹ כֶּסֶף וְזָהָב לֹא אוּכַל לַעֲבֹר אֶת פִּי יְהוָה לַעֲשׂוֹת טוֹבָה אוֹ רָעָה מִלִּבִּי אֲשֶׁר יְדַבֵּר יְהוָה אֹתוֹ אֲדַבֵּר` (במדבר כד:יג)
`וְלָמָּה תְנוּאוּן אֶת לֵב בְּנֵי יִשְׂרָאֵל מֵעֲבֹר אֶל הָאָרֶץ אֲשֶׁר נָתַן לָהֶם יְהוָה` (במדבר לב:ז)
`וַיַּעֲלוּ עַד נַחַל אֶשְׁכּוֹל וַיִּרְאוּ אֶת הָאָרֶץ וַיָּנִיאוּ אֶת לֵב בְּנֵי יִשְׂרָאֵל לְבִלְתִּי בֹא אֶל הָאָרֶץ אֲשֶׁר נָתַן לָהֶם יְהוָה` (במדבר לב:ט)
`וַתִּקְרְבוּן וַתַּעַמְדוּן תַּחַת הָהָר וְהָהָר בֹּעֵר בָּאֵשׁ עַד לֵב הַשָּׁמַיִם חֹשֶׁךְ עָנָן וַעֲרָפֶל` (דברים ד:יא)
`וּבַגּוֹיִם הָהֵם לֹא תַרְגִּיעַ וְלֹא יִהְיֶה מָנוֹחַ לְכַף רַגְלֶךָ וְנָתַן יְהוָה לְךָ שָׁם לֵב רַגָּז וְכִלְיוֹן עֵינַיִם וְדַאֲבוֹן נָפֶשׁ` (דברים כח:סה)
`וְלֹא נָתַן יְהוָה לָכֶם לֵב לָדַעַת וְעֵינַיִם לִרְאוֹת וְאָזְנַיִם לִשְׁמֹעַ עַד הַיּוֹם הַזֶּה` (דברים כט:ד)
`וְהָיָה בְּשָׁמְעוֹ אֶת דִּבְרֵי הָאָלָה הַזֹּאת וְהִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לִּי כִּי בִּשְׁרִרוּת לִבִּי אֵלֵךְ לְמַעַן סְפוֹת הָרָוָה אֶת הַצְּמֵאָה` (דברים כט:יט)
`כִּי מֵאֵת יְהוָה הָיְתָה לְחַזֵּק אֶת לִבָּם לִקְרַאת הַמִּלְחָמָה אֶת יִשְׂרָאֵל לְמַעַן הַחֲרִימָם לְבִלְתִּי הֱיוֹת לָהֶם תְּחִנָּה כִּי לְמַעַן הַשְׁמִידָם כַּאֲשֶׁר צִוָּה יְהוָה אֶת מֹשֶׁה` (יהושע יא:כ)
`וְאַחַי אֲשֶׁר עָלוּ עִמִּי הִמְסִיו אֶת לֵב הָעָם וְאָנֹכִי מִלֵּאתִי אַחֲרֵי יְהוָה אֱלֹהָי` (יהושע יד:ח)
`לִבִּי לְחוֹקְקֵי יִשְׂרָאֵל הַמִּתְנַדְּבִים בָּעָם בָּרֲכוּ יְהוָה` (שופטים ה:ט)
`וְשָׂרַי בְּיִשָּׂשׂכָר עִם דְּבֹרָה וְיִשָּׂשׂכָר כֵּן בָּרָק בָּעֵמֶק שֻׁלַּח בְּרַגְלָיו בִּפְלַגּוֹת רְאוּבֵן גְּדֹלִים חִקְקֵי לֵב` (שופטים ה:טו)
`לָמָּה יָשַׁבְתָּ בֵּין הַמִּשְׁפְּתַיִם לִשְׁמֹעַ שְׁרִקוֹת עֲדָרִים לִפְלַגּוֹת רְאוּבֵן גְּדוֹלִים חִקְרֵי לֵב` (שופטים ה:טז)
`וַיְדַבְּרוּ אֲחֵי אִמּוֹ עָלָיו בְּאָזְנֵי כָּל בַּעֲלֵי שְׁכֶם אֵת כָּל הַדְּבָרִים הָאֵלֶּה וַיֵּט לִבָּם אַחֲרֵי אֲבִימֶלֶךְ כִּי אָמְרוּ אָחִינוּ הוּא` (שופטים ט:ג)
`וַתֹּאמֶר אֵלָיו אֵיךְ תֹּאמַר אֲהַבְתִּיךְ וְלִבְּךָ אֵין אִתִּי זֶה שָׁלֹשׁ פְּעָמִים הֵתַלְתָּ בִּי וְלֹא הִגַּדְתָּ לִּי בַּמֶּה כֹּחֲךָ גָדוֹל` (שופטים טז:טו)
`וַיַּגֶּד לָהּ אֶת כָּל לִבּוֹ וַיֹּאמֶר לָהּ מוֹרָה לֹא עָלָה עַל רֹאשִׁי כִּי נְזִיר אֱלֹהִים אֲנִי מִבֶּטֶן אִמִּי אִם גֻּלַּחְתִּי וְסָר מִמֶּנִּי כֹחִי וְחָלִיתִי וְהָיִיתִי כְּכָל הָאָדָם` (שופטים טז:יז)
`וַתֵּרֶא דְלִילָה כִּי הִגִּיד לָהּ אֶת כָּל לִבּוֹ וַתִּשְׁלַח וַתִּקְרָא לְסַרְנֵי פְלִשְׁתִּים לֵאמֹר עֲלוּ הַפַּעַם כִּי הִגִּיד לָהּ אֶת כָּל לִבּוֹ וְעָלוּ אֵלֶיהָ סַרְנֵי פְלִשְׁתִּים וַיַּעֲלוּ הַכֶּסֶף בְּיָדָם` (שופטים טז:יח)
`וַיְהִי כִּי טוֹב לִבָּם וַיֹּאמְרוּ קִרְאוּ לְשִׁמְשׁוֹן וִישַׂחֶק לָנוּ וַיִּקְרְאוּ לְשִׁמְשׁוֹן מִבֵּית האסירים וַיְצַחֵק לִפְנֵיהֶם וַיַּעֲמִידוּ אוֹתוֹ בֵּין הָעַמּוּדִים` (שופטים טז:כה)
`וַיִּיטַב לֵב הַכֹּהֵן וַיִּקַּח אֶת הָאֵפוֹד וְאֶת הַתְּרָפִים וְאֶת הַפָּסֶל וַיָּבֹא בְּקֶרֶב הָעָם` (שופטים יח:כ)
`וַיָּקָם אִישָׁהּ וַיֵּלֶךְ אַחֲרֶיהָ לְדַבֵּר עַל לִבָּהּ לַהֲשִׁיבוֹ וְנַעֲרוֹ עִמּוֹ וְצֶמֶד חֲמֹרִים וַתְּבִיאֵהוּ בֵּית אָבִיהָ וַיִּרְאֵהוּ אֲבִי הַנַּעֲרָה וַיִּשְׂמַח לִקְרָאתוֹ` (שופטים יט:ג)
`וַיְהִי בַּיּוֹם הָרְבִיעִי וַיַּשְׁכִּימוּ בַבֹּקֶר וַיָּקָם לָלֶכֶת וַיֹּאמֶר אֲבִי הַנַּעֲרָה אֶל חֲתָנוֹ סְעָד לִבְּךָ פַּת לֶחֶם וְאַחַר תֵּלֵכוּ` (שופטים יט:ה)
`וַיֵּשְׁבוּ וַיֹּאכְלוּ שְׁנֵיהֶם יַחְדָּו וַיִּשְׁתּוּ וַיֹּאמֶר אֲבִי הַנַּעֲרָה אֶל הָאִישׁ הוֹאֶל נָא וְלִין וְיִטַב לִבֶּךָ` (שופטים יט:ו)
`הֵמָּה מֵיטִיבִים אֶת לִבָּם וְהִנֵּה אַנְשֵׁי הָעִיר אַנְשֵׁי בְנֵי בְלִיַּעַל נָסַבּוּ אֶת הַבַּיִת מִתְדַּפְּקִים עַל הַדָּלֶת וַיֹּאמְרוּ אֶל הָאִישׁ בַּעַל הַבַּיִת הַזָּקֵן לֵאמֹר הוֹצֵא אֶת הָאִישׁ אֲשֶׁר בָּא אֶל בֵּיתְךָ וְנֵדָעֶנּוּ` (שופטים יט:כב)
`וְחַנָּה הִיא מְדַבֶּרֶת עַל לִבָּהּ רַק שְׂפָתֶיהָ נָּעוֹת וְקוֹלָהּ לֹא יִשָּׁמֵעַ וַיַּחְשְׁבֶהָ עֵלִי לְשִׁכֹּרָה` (שמואל-א א:יג)
`וַתִּתְפַּלֵּל חַנָּה וַתֹּאמַר עָלַץ לִבִּי בַּיהוָה רָמָה קַרְנִי בַּיהוָה רָחַב פִּי עַל אוֹיְבַי כִּי שָׂמַחְתִּי בִּישׁוּעָתֶךָ` (שמואל-א ב:א)
`וַיָּבוֹא וְהִנֵּה עֵלִי יֹשֵׁב עַל הַכִּסֵּא יַךְ דֶּרֶךְ מְצַפֶּה כִּי הָיָה לִבּוֹ חָרֵד עַל אֲרוֹן הָאֱלֹהִים וְהָאִישׁ בָּא לְהַגִּיד בָּעִיר וַתִּזְעַק כָּל הָעִיר` (שמואל-א ד:יג)
`וּכְעֵת מוּתָהּ וַתְּדַבֵּרְנָה הַנִּצָּבוֹת עָלֶיהָ אַל תִּירְאִי כִּי בֵן יָלָדְתְּ וְלֹא עָנְתָה וְלֹא שָׁתָה לִבָּהּ` (שמואל-א ד:כ)
`וְלָמָּה תְכַבְּדוּ אֶת לְבַבְכֶם כַּאֲשֶׁר כִּבְּדוּ מִצְרַיִם וּפַרְעֹה אֶת לִבָּם הֲלוֹא כַּאֲשֶׁר הִתְעַלֵּל בָּהֶם וַיְשַׁלְּחוּם וַיֵּלֵכוּ` (שמואל-א ו:ו)
`וְלָאֲתֹנוֹת הָאֹבְדוֹת לְךָ הַיּוֹם שְׁלֹשֶׁת הַיָּמִים אַל תָּשֶׂם אֶת לִבְּךָ לָהֶם כִּי נִמְצָאוּ וּלְמִי כָּל חֶמְדַּת יִשְׂרָאֵל הֲלוֹא לְךָ וּלְכֹל בֵּית אָבִיךָ` (שמואל-א ט:כ)
`וְהָיָה כְּהַפְנֹתוֹ שִׁכְמוֹ לָלֶכֶת מֵעִם שְׁמוּאֵל וַיַּהֲפָךְ לוֹ אֱלֹהִים לֵב אַחֵר וַיָּבֹאוּ כָּל הָאֹתוֹת הָאֵלֶּה בַּיּוֹם הַהוּא` (שמואל-א י:ט)
`וְגַם שָׁאוּל הָלַךְ לְבֵיתוֹ גִּבְעָתָה וַיֵּלְכוּ עִמּוֹ הַחַיִל אֲשֶׁר נָגַע אֱלֹהִים בְּלִבָּם` (שמואל-א י:כו)
`וַיֹּאמֶר דָּוִד אֶל שָׁאוּל אַל יִפֹּל לֵב אָדָם עָלָיו עַבְדְּךָ יֵלֵךְ וְנִלְחַם עִם הַפְּלִשְׁתִּי הַזֶּה` (שמואל-א יז:לב)
`וַיְהִי אַחֲרֵי כֵן וַיַּךְ לֵב דָּוִד אֹתוֹ עַל אֲשֶׁר כָּרַת אֶת כָּנָף אֲשֶׁר לְשָׁאוּל` (שמואל-א כד:ה)
`וְשֵׁם הָאִישׁ נָבָל וְשֵׁם אִשְׁתּוֹ אֲבִגָיִל וְהָאִשָּׁה טוֹבַת שֶׂכֶל וִיפַת תֹּאַר וְהָאִישׁ קָשֶׁה וְרַע מַעֲלָלִים וְהוּא כְלִבּוֹ` (שמואל-א כה:ג)
`אַל נָא יָשִׂים אֲדֹנִי אֶת לִבּוֹ אֶל אִישׁ הַבְּלִיַּעַל הַזֶּה עַל נָבָל כִּי כִשְׁמוֹ כֶּן הוּא נָבָל שְׁמוֹ וּנְבָלָה עִמּוֹ וַאֲנִי אֲמָתְךָ לֹא רָאִיתִי אֶת נַעֲרֵי אֲדֹנִי אֲשֶׁר שָׁלָחְתָּ` (שמואל-א כה:כה)
`וְלֹא תִהְיֶה זֹאת לְךָ לְפוּקָה וּלְמִכְשׁוֹל לֵב לַאדֹנִי וְלִשְׁפָּךְ דָּם חִנָּם וּלְהוֹשִׁיעַ אֲדֹנִי לוֹ וְהֵיטִב יְהוָה לַאדֹנִי וְזָכַרְתָּ אֶת אֲמָתֶךָ` (שמואל-א כה:לא)
`וַתָּבֹא אֲבִיגַיִל אֶל נָבָל וְהִנֵּה לוֹ מִשְׁתֶּה בְּבֵיתוֹ כְּמִשְׁתֵּה הַמֶּלֶךְ וְלֵב נָבָל טוֹב עָלָיו וְהוּא שִׁכֹּר עַד מְאֹד וְלֹא הִגִּידָה לּוֹ דָּבָר קָטֹן וְגָדוֹל עַד אוֹר הַבֹּקֶר` (שמואל-א כה:לו)
`וַיְהִי בַבֹּקֶר בְּצֵאת הַיַּיִן מִנָּבָל וַתַּגֶּד לוֹ אִשְׁתּוֹ אֶת הַדְּבָרִים הָאֵלֶּה וַיָּמָת לִבּוֹ בְּקִרְבּוֹ וְהוּא הָיָה לְאָבֶן` (שמואל-א כה:לז)
`וַיֹּאמֶר דָּוִד אֶל לִבּוֹ עַתָּה אֶסָּפֶה יוֹם אֶחָד בְּיַד שָׁאוּל אֵין לִי טוֹב כִּי הִמָּלֵט אִמָּלֵט אֶל אֶרֶץ פְּלִשְׁתִּים וְנוֹאַשׁ מִמֶּנִּי שָׁאוּל לְבַקְשֵׁנִי עוֹד בְּכָל גְּבוּל יִשְׂרָאֵל וְנִמְלַטְתִּי מִיָּדוֹ` (שמואל-א כז:א)
`וַיַּרְא שָׁאוּל אֶת מַחֲנֵה פְלִשְׁתִּים וַיִּרָא וַיֶּחֱרַד לִבּוֹ מְאֹד` (שמואל-א כח:ה)
`וְהָיָה אֲרוֹן יְהוָה בָּא עִיר דָּוִד וּמִיכַל בַּת שָׁאוּל נִשְׁקְפָה בְּעַד הַחַלּוֹן וַתֵּרֶא אֶת הַמֶּלֶךְ דָּוִד מְפַזֵּז וּמְכַרְכֵּר לִפְנֵי יְהוָה וַתִּבֶז לוֹ בְּלִבָּהּ` (שמואל-ב ו:טז)
`בַּעֲבוּר דְּבָרְךָ וּכְלִבְּךָ עָשִׂיתָ אֵת כָּל הַגְּדוּלָּה הַזֹּאת לְהוֹדִיעַ אֶת עַבְדֶּךָ` (שמואל-ב ז:כא)
`כִּי אַתָּה יְהוָה צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל גָּלִיתָה אֶת אֹזֶן עַבְדְּךָ לֵאמֹר בַּיִת אֶבְנֶה לָּךְ עַל כֵּן מָצָא עַבְדְּךָ אֶת לִבּוֹ לְהִתְפַּלֵּל אֵלֶיךָ אֶת הַתְּפִלָּה הַזֹּאת` (שמואל-ב ז:כז)
`וַיֹּאמֶר אֵלֶיהָ אַבְשָׁלוֹם אָחִיהָ הַאֲמִינוֹן אָחִיךְ הָיָה עִמָּךְ וְעַתָּה אֲחוֹתִי הַחֲרִישִׁי אָחִיךְ הוּא אַל תָּשִׁיתִי אֶת לִבֵּךְ לַדָּבָר הַזֶּה וַתֵּשֶׁב תָּמָר וְשֹׁמֵמָה בֵּית אַבְשָׁלוֹם אָחִיהָ` (שמואל-ב יג:כ)
`וַיְצַו אַבְשָׁלוֹם אֶת נְעָרָיו לֵאמֹר רְאוּ נָא כְּטוֹב לֵב אַמְנוֹן בַּיַּיִן וְאָמַרְתִּי אֲלֵיכֶם הַכּוּ אֶת אַמְנוֹן וַהֲמִתֶּם אֹתוֹ אַל תִּירָאוּ הֲלוֹא כִּי אָנֹכִי צִוִּיתִי אֶתְכֶם חִזְקוּ וִהְיוּ לִבְנֵי חָיִל` (שמואל-ב יג:כח)
`וְעַתָּה אַל יָשֵׂם אֲדֹנִי הַמֶּלֶךְ אֶל לִבּוֹ דָּבָר לֵאמֹר כָּל בְּנֵי הַמֶּלֶךְ מֵתוּ כִּי אִם אַמְנוֹן לְבַדּוֹ מֵת` (שמואל-ב יג:לג)
`וַיֵּדַע יוֹאָב בֶּן צְרֻיָה כִּי לֵב הַמֶּלֶךְ עַל אַבְשָׁלוֹם` (שמואל-ב יד:א)
`וַיַּעַשׂ אַבְשָׁלוֹם כַּדָּבָר הַזֶּה לְכָל יִשְׂרָאֵל אֲשֶׁר יָבֹאוּ לַמִּשְׁפָּט אֶל הַמֶּלֶךְ וַיְגַנֵּב אַבְשָׁלוֹם אֶת לֵב אַנְשֵׁי יִשְׂרָאֵל` (שמואל-ב טו:ו)
`וַיָּבֹא הַמַּגִּיד אֶל דָּוִד לֵאמֹר הָיָה לֶב אִישׁ יִשְׂרָאֵל אַחֲרֵי אַבְשָׁלוֹם` (שמואל-ב טו:יג)
`וְהוּא גַם בֶּן חַיִל אֲשֶׁר לִבּוֹ כְּלֵב הָאַרְיֵה הִמֵּס יִמָּס כִּי יֹדֵעַ כָּל יִשְׂרָאֵל כִּי גִבּוֹר אָבִיךָ וּבְנֵי חַיִל אֲשֶׁר אִתּוֹ` (שמואל-ב יז:י)
`וַיֹּאמֶר הָעָם לֹא תֵצֵא כִּי אִם נֹס נָנוּס לֹא יָשִׂימוּ אֵלֵינוּ לֵב וְאִם יָמֻתוּ חֶצְיֵנוּ לֹא יָשִׂימוּ אֵלֵינוּ לֵב כִּי עַתָּה כָמֹנוּ עֲשָׂרָה אֲלָפִים וְעַתָּה טוֹב כִּי תִהְיֶה לָּנוּ מֵעִיר לַעְזִיר` (שמואל-ב יח:ג)
`וַיֹּאמֶר יוֹאָב לֹא כֵן אֹחִילָה לְפָנֶיךָ וַיִּקַּח שְׁלֹשָׁה שְׁבָטִים בְּכַפּוֹ וַיִּתְקָעֵם בְּלֵב אַבְשָׁלוֹם עוֹדֶנּוּ חַי בְּלֵב הָאֵלָה` (שמואל-ב יח:יד)
`וְעַתָּה קוּם צֵא וְדַבֵּר עַל לֵב עֲבָדֶיךָ כִּי בַיהוָה נִשְׁבַּעְתִּי כִּי אֵינְךָ יוֹצֵא אִם יָלִין אִישׁ אִתְּךָ הַלַּיְלָה וְרָעָה לְךָ זֹאת מִכָּל הָרָעָה אֲשֶׁר בָּאָה עָלֶיךָ מִנְּעֻרֶיךָ עַד עָתָּה` (שמואל-ב יט:ז)
`וַיֹּאמֶר אֶל הַמֶּלֶךְ אַל יַחֲשָׁב לִי אֲדֹנִי עָוֺן וְאַל תִּזְכֹּר אֵת אֲשֶׁר הֶעֱוָה עַבְדְּךָ בַּיּוֹם אֲשֶׁר יָׄצָׄאׄ אֲדֹנִי הַמֶּלֶךְ מִירוּשָׁלִָם לָשׂוּם הַמֶּלֶךְ אֶל לִבּוֹ` (שמואל-ב יט:יט)
`וַיַּךְ לֵב דָּוִד אֹתוֹ אַחֲרֵי כֵן סָפַר אֶת הָעָם וַיֹּאמֶר דָּוִד אֶל יְהוָה חָטָאתִי מְאֹד אֲשֶׁר עָשִׂיתִי וְעַתָּה יְהוָה הַעֲבֶר נָא אֶת עֲוֺן עַבְדְּךָ כִּי נִסְכַּלְתִּי מְאֹד` (שמואל-ב כד:י)
`וְנָתַתָּ לְעַבְדְּךָ לֵב שֹׁמֵעַ לִשְׁפֹּט אֶת עַמְּךָ לְהָבִין בֵּין טוֹב לְרָע כִּי מִי יוּכַל לִשְׁפֹּט אֶת עַמְּךָ הַכָּבֵד הַזֶּה` (מלכים-א ג:ט)
`הִנֵּה עָשִׂיתִי כִּדְבָרֶיךָ הִנֵּה נָתַתִּי לְךָ לֵב חָכָם וְנָבוֹן אֲשֶׁר כָּמוֹךָ לֹא הָיָה לְפָנֶיךָ וְאַחֲרֶיךָ לֹא יָקוּם כָּמוֹךָ` (מלכים-א ג:יב)
`וַיִּתֵּן אֱלֹהִים חָכְמָה לִשְׁלֹמֹה וּתְבוּנָה הַרְבֵּה מְאֹד וְרֹחַב לֵב כַּחוֹל אֲשֶׁר עַל שְׂפַת הַיָּם` (מלכים-א ד:כט)
`וַיֹּאמַר יְהוָה אֱלֹהֵי יִשְׂרָאֵל אֵין כָּמוֹךָ אֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת שֹׁמֵר הַבְּרִית וְהַחֶסֶד לַעֲבָדֶיךָ הַהֹלְכִים לְפָנֶיךָ בְּכָל לִבָּם` (מלכים-א ח:כג)
`וְהֵשִׁיבוּ אֶל לִבָּם בָּאָרֶץ אֲשֶׁר נִשְׁבּוּ שָׁם וְשָׁבוּ וְהִתְחַנְּנוּ אֵלֶיךָ בְּאֶרֶץ שֹׁבֵיהֶם לֵאמֹר חָטָאנוּ וְהֶעֱוִינוּ רָשָׁעְנוּ` (מלכים-א ח:מז)
`בַּיּוֹם הַשְּׁמִינִי שִׁלַּח אֶת הָעָם וַיְבָרֲכוּ אֶת הַמֶּלֶךְ וַיֵּלְכוּ לְאָהֳלֵיהֶם שְׂמֵחִים וְטוֹבֵי לֵב עַל כָּל הַטּוֹבָה אֲשֶׁר עָשָׂה יְהוָה לְדָוִד עַבְדּוֹ וּלְיִשְׂרָאֵל עַמּוֹ` (מלכים-א ח:סו)
`וַיֹּאמֶר יְהוָה אֵלָיו שָׁמַעְתִּי אֶת תְּפִלָּתְךָ וְאֶת תְּחִנָּתְךָ אֲשֶׁר הִתְחַנַּנְתָּה לְפָנַי הִקְדַּשְׁתִּי אֶת הַבַּיִת הַזֶּה אֲשֶׁר בָּנִתָה לָשׂוּם שְׁמִי שָׁם עַד עוֹלָם וְהָיוּ עֵינַי וְלִבִּי שָׁם כָּל הַיָּמִים` (מלכים-א ט:ג)
`וְכָל הָאָרֶץ מְבַקְשִׁים אֶת פְּנֵי שְׁלֹמֹה לִשְׁמֹעַ אֶת חָכְמָתוֹ אֲשֶׁר נָתַן אֱלֹהִים בְּלִבּוֹ` (מלכים-א י:כד)
`וַיְהִי לוֹ נָשִׁים שָׂרוֹת שְׁבַע מֵאוֹת וּפִלַגְשִׁים שְׁלֹשׁ מֵאוֹת וַיַּטּוּ נָשָׁיו אֶת לִבּוֹ` (מלכים-א יא:ג)
`וַיֹּאמֶר יָרָבְעָם בְּלִבּוֹ עַתָּה תָּשׁוּב הַמַּמְלָכָה לְבֵית דָּוִד` (מלכים-א יב:כו)
`אִם יַעֲלֶה הָעָם הַזֶּה לַעֲשׂוֹת זְבָחִים בְּבֵית יְהוָה בִּירוּשָׁלִַם וְשָׁב לֵב הָעָם הַזֶּה אֶל אֲדֹנֵיהֶם אֶל רְחַבְעָם מֶלֶךְ יְהוּדָה וַהֲרָגֻנִי וְשָׁבוּ אֶל רְחַבְעָם מֶלֶךְ יְהוּדָה` (מלכים-א יב:כז)
`עֲנֵנִי יְהוָה עֲנֵנִי וְיֵדְעוּ הָעָם הַזֶּה כִּי אַתָּה יְהוָה הָאֱלֹהִים וְאַתָּה הֲסִבֹּתָ אֶת לִבָּם אֲחֹרַנִּית` (מלכים-א יח:לז)
`וַתֹּאמֶר אֵלָיו אִיזֶבֶל אִשְׁתּוֹ אַתָּה עַתָּה תַּעֲשֶׂה מְלוּכָה עַל יִשְׂרָאֵל קוּם אֱכָל לֶחֶם וְיִטַב לִבֶּךָ אֲנִי אֶתֵּן לְךָ אֶת כֶּרֶם נָבוֹת הַיִּזְרְעֵאלִי` (מלכים-א כא:ז)
`וַיֹּאמֶר אֵלָיו לֹא לִבִּי הָלַךְ כַּאֲשֶׁר הָפַךְ אִישׁ מֵעַל מֶרְכַּבְתּוֹ לִקְרָאתֶךָ הַעֵת לָקַחַת אֶת הַכֶּסֶף וְלָקַחַת בְּגָדִים וְזֵיתִים וּכְרָמִים וְצֹאן וּבָקָר וַעֲבָדִים וּשְׁפָחוֹת` (מלכים-ב ה:כו)
`וַיִּסָּעֵר לֵב מֶלֶךְ אֲרָם עַל הַדָּבָר הַזֶּה וַיִּקְרָא אֶל עֲבָדָיו וַיֹּאמֶר אֲלֵיהֶם הֲלוֹא תַּגִּידוּ לִי מִי מִשֶּׁלָּנוּ אֶל מֶלֶךְ יִשְׂרָאֵל` (מלכים-ב ו:יא)
`וְיֵהוּא מִלֵּא יָדוֹ בַקֶּשֶׁת וַיַּךְ אֶת יְהוֹרָם בֵּין זְרֹעָיו וַיֵּצֵא הַחֵצִי מִלִּבּוֹ וַיִּכְרַע בְּרִכְבּוֹ` (מלכים-ב ט:כד)
`וַיֹּאמֶר יְהוֹאָשׁ אֶל הַכֹּהֲנִים כֹּל כֶּסֶף הַקֳּדָשִׁים אֲשֶׁר יוּבָא בֵית יְהוָה כֶּסֶף עוֹבֵר אִישׁ כֶּסֶף נַפְשׁוֹת עֶרְכּוֹ כָּל כֶּסֶף אֲשֶׁר יַעֲלֶה עַל לֶב אִישׁ לְהָבִיא בֵּית יְהוָה` (מלכים-ב יב:ד)
`הַכֵּה הִכִּיתָ אֶת אֱדוֹם וּנְשָׂאֲךָ לִבֶּךָ הִכָּבֵד וְשֵׁב בְּבֵיתֶךָ וְלָמָּה תִתְגָּרֶה בְּרָעָה וְנָפַלְתָּה אַתָּה וִיהוּדָה עִמָּךְ` (מלכים-ב יד:י)
`וַיַּעֲמֹד הַמֶּלֶךְ עַל הָעַמּוּד וַיִּכְרֹת אֶת הַבְּרִית לִפְנֵי יְהוָה לָלֶכֶת אַחַר יְהוָה וְלִשְׁמֹר מִצְוֺתָיו וְאֶת עֵדְוֺתָיו וְאֶת חֻקֹּתָיו בְּכָל לֵב וּבְכָל נֶפֶשׁ לְהָקִים אֶת דִּבְרֵי הַבְּרִית הַזֹּאת הַכְּתֻבִים עַל הַסֵּפֶר הַזֶּה וַיַּעֲמֹד כָּל הָעָם בַּבְּרִית` (מלכים-ב כג:ג)
`הַשְׁמֵן לֵב הָעָם הַזֶּה וְאָזְנָיו הַכְבֵּד וְעֵינָיו הָשַׁע פֶּן יִרְאֶה בְעֵינָיו וּבְאָזְנָיו יִשְׁמָע וּלְבָבוֹ יָבִין וָשָׁב וְרָפָא לוֹ` (ישעיהו ו:י)
`לִבִּי לְמוֹאָב יִזְעָק בְּרִיחֶהָ עַד צֹעַר עֶגְלַת שְׁלִשִׁיָּה כִּי מַעֲלֵה הַלּוּחִית בִּבְכִי יַעֲלֶה בּוֹ כִּי דֶּרֶךְ חוֹרֹנַיִם זַעֲקַת שֶׁבֶר יְעֹעֵרוּ` (ישעיהו טו:ה)
`אָבַל תִּירוֹשׁ אֻמְלְלָה גָפֶן נֶאֶנְחוּ כָּל שִׂמְחֵי לֵב` (ישעיהו כד:ז)
`וַיֹּאמֶר אֲדֹנָי יַעַן כִּי נִגַּשׁ הָעָם הַזֶּה בְּפִיו וּבִשְׂפָתָיו כִּבְּדוּנִי וְלִבּוֹ רִחַק מִמֶּנִּי וַתְּהִי יִרְאָתָם אֹתִי מִצְוַת אֲנָשִׁים מְלֻמָּדָה` (ישעיהו כט:יג)
`כִּי נָבָל נְבָלָה יְדַבֵּר וְלִבּוֹ יַעֲשֶׂה אָוֶן לַעֲשׂוֹת חֹנֶף וּלְדַבֵּר אֶל יְהוָה תּוֹעָה לְהָרִיק נֶפֶשׁ רָעֵב וּמַשְׁקֶה צָמֵא יַחְסִיר` (ישעיהו לב:ו)
`לִבְּךָ יֶהְגֶּה אֵימָה אַיֵּה סֹפֵר אַיֵּה שֹׁקֵל אַיֵּה סֹפֵר אֶת הַמִּגְדָּלִים` (ישעיהו לג:יח)
`אִמְרוּ לְנִמְהֲרֵי לֵב חִזְקוּ אַל תִּירָאוּ הִנֵּה אֱלֹהֵיכֶם נָקָם יָבוֹא גְּמוּל אֱלֹהִים הוּא יָבוֹא וְיֹשַׁעֲכֶם` (ישעיהו לה:ד)
`וַיֹּאמַר אָנָּה יְהוָה זְכָר נָא אֵת אֲשֶׁר הִתְהַלַּכְתִּי לְפָנֶיךָ בֶּאֱמֶת וּבְלֵב שָׁלֵם וְהַטּוֹב בְּעֵינֶיךָ עָשִׂיתִי וַיֵּבְךְּ חִזְקִיָּהוּ בְּכִי גָדוֹל` (ישעיהו לח:ג)
`דַּבְּרוּ עַל לֵב יְרוּשָׁלִַם וְקִרְאוּ אֵלֶיהָ כִּי מָלְאָה צְבָאָהּ כִּי נִרְצָה עֲוֺנָהּ כִּי לָקְחָה מִיַּד יְהוָה כִּפְלַיִם בְּכָל חַטֹּאתֶיהָ` (ישעיהו מ:ב)
`יַגִּישׁוּ וְיַגִּידוּ לָנוּ אֵת אֲשֶׁר תִּקְרֶינָה הָרִאשֹׁנוֹת מָה הֵנָּה הַגִּידוּ וְנָשִׂימָה לִבֵּנוּ וְנֵדְעָה אַחֲרִיתָן אוֹ הַבָּאוֹת הַשְׁמִיעֻנוּ` (ישעיהו מא:כב)
`וַיִּשְׁפֹּךְ עָלָיו חֵמָה אַפּוֹ וֶעֱזוּז מִלְחָמָה וַתְּלַהֲטֵהוּ מִסָּבִיב וְלֹא יָדָע וַתִּבְעַר בּוֹ וְלֹא יָשִׂים עַל לֵב` (ישעיהו מב:כה)
`לֹא יָדְעוּ וְלֹא יָבִינוּ כִּי טַח מֵרְאוֹת עֵינֵיהֶם מֵהַשְׂכִּיל לִבֹּתָם` (ישעיהו מד:יח)
`וְלֹא יָשִׁיב אֶל לִבּוֹ וְלֹא דַעַת וְלֹא תְבוּנָה לֵאמֹר חֶצְיוֹ שָׂרַפְתִּי בְמוֹ אֵשׁ וְאַף אָפִיתִי עַל גֶּחָלָיו לֶחֶם אֶצְלֶה בָשָׂר וְאֹכֵל וְיִתְרוֹ לְתוֹעֵבָה אֶעֱשֶׂה לְבוּל עֵץ אֶסְגּוֹד` (ישעיהו מד:יט)
`רֹעֶה אֵפֶר לֵב הוּתַל הִטָּהוּ וְלֹא יַצִּיל אֶת נַפְשׁוֹ וְלֹא יֹאמַר הֲלוֹא שֶׁקֶר בִּימִינִי` (ישעיהו מד:כ)
`זִכְרוּ זֹאת וְהִתְאֹשָׁשׁוּ הָשִׁיבוּ פוֹשְׁעִים עַל לֵב` (ישעיהו מו:ח)
`שִׁמְעוּ אֵלַי אַבִּירֵי לֵב הָרְחוֹקִים מִצְּדָקָה` (ישעיהו מו:יב)
`וַתֹּאמְרִי לְעוֹלָם אֶהְיֶה גְבָרֶת עַד לֹא שַׂמְתְּ אֵלֶּה עַל לִבֵּךְ לֹא זָכַרְתְּ אַחֲרִיתָהּ` (ישעיהו מז:ז)
`וַתִּבְטְחִי בְרָעָתֵךְ אָמַרְתְּ אֵין רֹאָנִי חָכְמָתֵךְ וְדַעְתֵּךְ הִיא שׁוֹבְבָתֶךְ וַתֹּאמְרִי בְלִבֵּךְ אֲנִי וְאַפְסִי עוֹד` (ישעיהו מז:י)
`שִׁמְעוּ אֵלַי יֹדְעֵי צֶדֶק עַם תּוֹרָתִי בְלִבָּם אַל תִּירְאוּ חֶרְפַּת אֱנוֹשׁ וּמִגִּדֻּפֹתָם אַל תֵּחָתּוּ` (ישעיהו נא:ז)
`הַצַּדִּיק אָבָד וְאֵין אִישׁ שָׂם עַל לֵב וְאַנְשֵׁי חֶסֶד נֶאֱסָפִים בְּאֵין מֵבִין כִּי מִפְּנֵי הָרָעָה נֶאֱסַף הַצַּדִּיק` (ישעיהו נז:א)
`וְאֶת מִי דָּאַגְתְּ וַתִּירְאִי כִּי תְכַזֵּבִי וְאוֹתִי לֹא זָכַרְתְּ לֹא שַׂמְתְּ עַל לִבֵּךְ הֲלֹא אֲנִי מַחְשֶׁה וּמֵעֹלָם וְאוֹתִי לֹא תִירָאִי` (ישעיהו נז:יא)
`כִּי כֹה אָמַר רָם וְנִשָּׂא שֹׁכֵן עַד וְקָדוֹשׁ שְׁמוֹ מָרוֹם וְקָדוֹשׁ אֶשְׁכּוֹן וְאֶת דַּכָּא וּשְׁפַל רוּחַ לְהַחֲיוֹת רוּחַ שְׁפָלִים וּלְהַחֲיוֹת לֵב נִדְכָּאִים` (ישעיהו נז:טו)
`בַּעֲוֺן בִּצְעוֹ קָצַפְתִּי וְאַכֵּהוּ הַסְתֵּר וְאֶקְצֹף וַיֵּלֶךְ שׁוֹבָב בְּדֶרֶךְ לִבּוֹ` (ישעיהו נז:יז)
`פָּשֹׁעַ וְכַחֵשׁ בַּיהוָה וְנָסוֹג מֵאַחַר אֱלֹהֵינוּ דַּבֶּר עֹשֶׁק וְסָרָה הֹרוֹ וְהֹגוֹ מִלֵּב דִּבְרֵי שָׁקֶר` (ישעיהו נט:יג)
`רוּחַ אֲדֹנָי יְהוִה עָלָי יַעַן מָשַׁח יְהוָה אֹתִי לְבַשֵּׂר עֲנָוִים שְׁלָחַנִי לַחֲבֹשׁ לְנִשְׁבְּרֵי לֵב לִקְרֹא לִשְׁבוּיִם דְּרוֹר וְלַאֲסוּרִים פְּקַח קוֹחַ` (ישעיהו סא:א)
`כִּי יוֹם נָקָם בְּלִבִּי וּשְׁנַת גְּאוּלַי בָּאָה` (ישעיהו סג:ד)
`לָמָּה תַתְעֵנוּ יְהוָה מִדְּרָכֶיךָ תַּקְשִׁיחַ לִבֵּנוּ מִיִּרְאָתֶךָ שׁוּב לְמַעַן עֲבָדֶיךָ שִׁבְטֵי נַחֲלָתֶךָ` (ישעיהו סג:יז)
`הִנֵּה עֲבָדַי יָרֹנּוּ מִטּוּב לֵב וְאַתֶּם תִּצְעֲקוּ מִכְּאֵב לֵב וּמִשֵּׁבֶר רוּחַ תְּיֵלִילוּ` (ישעיהו סה:יד)
`כִּי הִנְנִי בוֹרֵא שָׁמַיִם חֲדָשִׁים וָאָרֶץ חֲדָשָׁה וְלֹא תִזָּכַרְנָה הָרִאשֹׁנוֹת וְלֹא תַעֲלֶינָה עַל לֵב` (ישעיהו סה:יז)
`וּרְאִיתֶם וְשָׂשׂ לִבְּכֶם וְעַצְמוֹתֵיכֶם כַּדֶּשֶׁא תִפְרַחְנָה וְנוֹדְעָה יַד יְהוָה אֶת עֲבָדָיו וְזָעַם אֶת אֹיְבָיו` (ישעיהו סו:יד)
`וְגַם בְּכָל זֹאת לֹא שָׁבָה אֵלַי בָּגוֹדָה אֲחוֹתָהּ יְהוּדָה בְּכָל לִבָּהּ כִּי אִם בְּשֶׁקֶר נְאֻם יְהוָה` (ירמיהו ג:י)
`וְנָתַתִּי לָכֶם רֹעִים כְּלִבִּי וְרָעוּ אֶתְכֶם דֵּעָה וְהַשְׂכֵּיל` (ירמיהו ג:טו)
`וְהָיָה כִּי תִרְבּוּ וּפְרִיתֶם בָּאָרֶץ בַּיָּמִים הָהֵמָּה נְאֻם יְהוָה לֹא יֹאמְרוּ עוֹד אֲרוֹן בְּרִית יְהוָה וְלֹא יַעֲלֶה עַל לֵב וְלֹא יִזְכְּרוּ בוֹ וְלֹא יִפְקֹדוּ וְלֹא יֵעָשֶׂה עוֹד` (ירמיהו ג:טז)
`בָּעֵת הַהִיא יִקְרְאוּ לִירוּשָׁלִַם כִּסֵּא יְהוָה וְנִקְוּוּ אֵלֶיהָ כָל הַגּוֹיִם לְשֵׁם יְהוָה לִירוּשָׁלִָם וְלֹא יֵלְכוּ עוֹד אַחֲרֵי שְׁרִרוּת לִבָּם הָרָע` (ירמיהו ג:יז)
`וְהָיָה בַיּוֹם הַהוּא נְאֻם יְהוָה יֹאבַד לֵב הַמֶּלֶךְ וְלֵב הַשָּׂרִים וְנָשַׁמּוּ הַכֹּהֲנִים וְהַנְּבִיאִים יִתְמָהוּ` (ירמיהו ד:ט)
`כַּבְּסִי מֵרָעָה לִבֵּךְ יְרוּשָׁלִַם לְמַעַן תִּוָּשֵׁעִי עַד מָתַי תָּלִין בְּקִרְבֵּךְ מַחְשְׁבוֹת אוֹנֵךְ` (ירמיהו ד:יד)
`דַּרְכֵּךְ וּמַעֲלָלַיִךְ עָשׂוֹ אֵלֶּה לָךְ זֹאת רָעָתֵךְ כִּי מָר כִּי נָגַע עַד לִבֵּךְ` (ירמיהו ד:יח)
`מֵעַי מֵעַי אָחוּלָה קִירוֹת לִבִּי הֹמֶה לִּי לִבִּי לֹא אַחֲרִישׁ כִּי קוֹל שׁוֹפָר שָׁמַעְתִּי נַפְשִׁי תְּרוּעַת מִלְחָמָה` (ירמיהו ד:יט)
`שִׁמְעוּ נָא זֹאת עַם סָכָל וְאֵין לֵב עֵינַיִם לָהֶם וְלֹא יִרְאוּ אָזְנַיִם לָהֶם וְלֹא יִשְׁמָעוּ` (ירמיהו ה:כא)
`וְלָעָם הַזֶּה הָיָה לֵב סוֹרֵר וּמוֹרֶה סָרוּ וַיֵּלֵכוּ` (ירמיהו ה:כג)
`וְלֹא שָׁמְעוּ וְלֹא הִטּוּ אֶת אָזְנָם וַיֵּלְכוּ בְּמֹעֵצוֹת בִּשְׁרִרוּת לִבָּם הָרָע וַיִּהְיוּ לְאָחוֹר וְלֹא לְפָנִים` (ירמיהו ז:כד)
`וּבָנוּ בָּמוֹת הַתֹּפֶת אֲשֶׁר בְּגֵיא בֶן הִנֹּם לִשְׂרֹף אֶת בְּנֵיהֶם וְאֶת בְּנֹתֵיהֶם בָּאֵשׁ אֲשֶׁר לֹא צִוִּיתִי וְלֹא עָלְתָה עַל לִבִּי` (ירמיהו ז:לא)
`מַבְלִיגִיתִי עֲלֵי יָגוֹן עָלַי לִבִּי דַוָּי` (ירמיהו ח:יח)
`וַיֵּלְכוּ אַחֲרֵי שְׁרִרוּת לִבָּם וְאַחֲרֵי הַבְּעָלִים אֲשֶׁר לִמְּדוּם אֲבוֹתָם` (ירמיהו ט:יד)
`עַל מִצְרַיִם וְעַל יְהוּדָה וְעַל אֱדוֹם וְעַל בְּנֵי עַמּוֹן וְעַל מוֹאָב וְעַל כָּל קְצוּצֵי פֵאָה הַיֹּשְׁבִים בַּמִּדְבָּר כִּי כָל הַגּוֹיִם עֲרֵלִים וְכָל בֵּית יִשְׂרָאֵל עַרְלֵי לֵב` (ירמיהו ט:כו)
`וְלֹא שָׁמְעוּ וְלֹא הִטּוּ אֶת אָזְנָם וַיֵּלְכוּ אִישׁ בִּשְׁרִירוּת לִבָּם הָרָע וָאָבִיא עֲלֵיהֶם אֶת כָּל דִּבְרֵי הַבְּרִית הַזֹּאת אֲשֶׁר צִוִּיתִי לַעֲשׂוֹת וְלֹא עָשׂוּ` (ירמיהו יא:ח)
`וַיהוָה צְבָאוֹת שֹׁפֵט צֶדֶק בֹּחֵן כְּלָיוֹת וָלֵב אֶרְאֶה נִקְמָתְךָ מֵהֶם כִּי אֵלֶיךָ גִּלִּיתִי אֶת רִיבִי` (ירמיהו יא:כ)
`וְאַתָּה יְהוָה יְדַעְתָּנִי תִּרְאֵנִי וּבָחַנְתָּ לִבִּי אִתָּךְ הַתִּקֵם כְּצֹאן לְטִבְחָה וְהַקְדִּשֵׁם לְיוֹם הֲרֵגָה` (ירמיהו יב:ג)
`שָׂמָהּ לִשְׁמָמָה אָבְלָה עָלַי שְׁמֵמָה נָשַׁמָּה כָּל הָאָרֶץ כִּי אֵין אִישׁ שָׂם עַל לֵב` (ירמיהו יב:יא)
`הָעָם הַזֶּה הָרָע הַמֵּאֲנִים לִשְׁמוֹעַ אֶת דְּבָרַי הַהֹלְכִים בִּשְׁרִרוּת לִבָּם וַיֵּלְכוּ אַחֲרֵי אֱלֹהִים אֲחֵרִים לְעָבְדָם וּלְהִשְׁתַּחֲוֺת לָהֶם וִיהִי כָּאֵזוֹר הַזֶּה אֲשֶׁר לֹא יִצְלַח לַכֹּל` (ירמיהו יג:י)
`וַיֹּאמֶר יְהוָה אֵלַי שֶׁקֶר הַנְּבִאִים נִבְּאִים בִּשְׁמִי לֹא שְׁלַחְתִּים וְלֹא צִוִּיתִים וְלֹא דִבַּרְתִּי אֲלֵיהֶם חֲזוֹן שֶׁקֶר וְקֶסֶם וֶאֱלוּל וְתַרְמוּת לִבָּם הֵמָּה מִתְנַבְּאִים לָכֶם` (ירמיהו יד:יד)
`וְאַתֶּם הֲרֵעֹתֶם לַעֲשׂוֹת מֵאֲבוֹתֵיכֶם וְהִנְּכֶם הֹלְכִים אִישׁ אַחֲרֵי שְׁרִרוּת לִבּוֹ הָרָע לְבִלְתִּי שְׁמֹעַ אֵלָי` (ירמיהו טז:יב)
`חַטַּאת יְהוּדָה כְּתוּבָה בְּעֵט בַּרְזֶל בְּצִפֹּרֶן שָׁמִיר חֲרוּשָׁה עַל לוּחַ לִבָּם וּלְקַרְנוֹת מִזְבְּחוֹתֵיכֶם` (ירמיהו יז:א)
`כֹּה אָמַר יְהוָה אָרוּר הַגֶּבֶר אֲשֶׁר יִבְטַח בָּאָדָם וְשָׂם בָּשָׂר זְרֹעוֹ וּמִן יְהוָה יָסוּר לִבּוֹ` (ירמיהו יז:ה)
`עָקֹב הַלֵּב מִכֹּל וְאָנֻשׁ הוּא מִי יֵדָעֶנּוּ` (ירמיהו יז:ט)
`אֲנִי יְהוָה חֹקֵר לֵב בֹּחֵן כְּלָיוֹת וְלָתֵת לְאִישׁ כְּדַרְכּוֹ כִּפְרִי מַעֲלָלָיו` (ירמיהו יז:י)
`וְאָמְרוּ נוֹאָשׁ כִּי אַחֲרֵי מַחְשְׁבוֹתֵינוּ נֵלֵךְ וְאִישׁ שְׁרִרוּת לִבּוֹ הָרָע נַעֲשֶׂה` (ירמיהו יח:יב)
`וּבָנוּ אֶת בָּמוֹת הַבַּעַל לִשְׂרֹף אֶת בְּנֵיהֶם בָּאֵשׁ עֹלוֹת לַבָּעַל אֲשֶׁר לֹא צִוִּיתִי וְלֹא דִבַּרְתִּי וְלֹא עָלְתָה עַל לִבִּי` (ירמיהו יט:ה)
`וְאָמַרְתִּי לֹא אֶזְכְּרֶנּוּ וְלֹא אֲדַבֵּר עוֹד בִּשְׁמוֹ וְהָיָה בְלִבִּי כְּאֵשׁ בֹּעֶרֶת עָצֻר בְּעַצְמֹתָי וְנִלְאֵיתִי כַּלְכֵל וְלֹא אוּכָל` (ירמיהו כ:ט)
`וַיהוָה צְבָאוֹת בֹּחֵן צַדִּיק רֹאֶה כְלָיוֹת וָלֵב אֶרְאֶה נִקְמָתְךָ מֵהֶם כִּי אֵלֶיךָ גִּלִּיתִי אֶת רִיבִי` (ירמיהו כ:יב)
`כִּי אֵין עֵינֶיךָ וְלִבְּךָ כִּי אִם עַל בִּצְעֶךָ וְעַל דַּם הַנָּקִי לִשְׁפּוֹךְ וְעַל הָעֹשֶׁק וְעַל הַמְּרוּצָה לַעֲשׂוֹת` (ירמיהו כב:יז)
`לַנְּבִאִים נִשְׁבַּר לִבִּי בְקִרְבִּי רָחֲפוּ כָּל עַצְמוֹתַי הָיִיתִי כְּאִישׁ שִׁכּוֹר וּכְגֶבֶר עֲבָרוֹ יָיִן מִפְּנֵי יְהוָה וּמִפְּנֵי דִּבְרֵי קָדְשׁוֹ` (ירמיהו כג:ט)
`כֹּה אָמַר יְהוָה צְבָאוֹת אַל תִּשְׁמְעוּ עַל דִּבְרֵי הַנְּבִאִים הַנִּבְּאִים לָכֶם מַהְבִּלִים הֵמָּה אֶתְכֶם חֲזוֹן לִבָּם יְדַבֵּרוּ לֹא מִפִּי יְהוָה` (ירמיהו כג:טז)
`אֹמְרִים אָמוֹר לִמְנַאֲצַי דִּבֶּר יְהוָה שָׁלוֹם יִהְיֶה לָכֶם וְכֹל הֹלֵךְ בִּשְׁרִרוּת לִבּוֹ אָמְרוּ לֹא תָבוֹא עֲלֵיכֶם רָעָה` (ירמיהו כג:יז)
`לֹא יָשׁוּב אַף יְהוָה עַד עֲשֹׂתוֹ וְעַד הֲקִימוֹ מְזִמּוֹת לִבּוֹ בְּאַחֲרִית הַיָּמִים תִּתְבּוֹנְנוּ בָהּ בִּינָה` (ירמיהו כג:כ)
`עַד מָתַי הֲיֵשׁ בְּלֵב הַנְּבִאִים נִבְּאֵי הַשָּׁקֶר וּנְבִיאֵי תַּרְמִת לִבָּם` (ירמיהו כג:כו)
`וְנָתַתִּי לָהֶם לֵב לָדַעַת אֹתִי כִּי אֲנִי יְהוָה וְהָיוּ לִי לְעָם וְאָנֹכִי אֶהְיֶה לָהֶם לֵאלֹהִים כִּי יָשֻׁבוּ אֵלַי בְּכָל לִבָּם` (ירמיהו כד:ז)
`וְהָיָה אַדִּירוֹ מִמֶּנּוּ וּמֹשְׁלוֹ מִקִּרְבּוֹ יֵצֵא וְהִקְרַבְתִּיו וְנִגַּשׁ אֵלָי כִּי מִי הוּא זֶה עָרַב אֶת לִבּוֹ לָגֶשֶׁת אֵלַי נְאֻם יְהוָה` (ירמיהו ל:כא)
`לֹא יָשׁוּב חֲרוֹן אַף יְהוָה עַד עֲשֹׂתוֹ וְעַד הֲקִימוֹ מְזִמּוֹת לִבּוֹ בְּאַחֲרִית הַיָּמִים תִּתְבּוֹנְנוּ בָהּ` (ירמיהו ל:כד)
`הַצִּיבִי לָךְ צִיֻּנִים שִׂמִי לָךְ תַּמְרוּרִים שִׁתִי לִבֵּךְ לַמְסִלָּה דֶּרֶךְ הָלָכְתִּי שׁוּבִי בְּתוּלַת יִשְׂרָאֵל שֻׁבִי אֶל עָרַיִךְ אֵלֶּה` (ירמיהו לא:כא)
`כִּי זֹאת הַבְּרִית אֲשֶׁר אֶכְרֹת אֶת בֵּית יִשְׂרָאֵל אַחֲרֵי הַיָּמִים הָהֵם נְאֻם יְהוָה נָתַתִּי אֶת תּוֹרָתִי בְּקִרְבָּם וְעַל לִבָּם אֶכְתֲּבֶנָּה וְהָיִיתִי לָהֶם לֵאלֹהִים וְהֵמָּה יִהְיוּ לִי לְעָם` (ירמיהו לא:לג)
`וַיִּבְנוּ אֶת בָּמוֹת הַבַּעַל אֲשֶׁר בְּגֵיא בֶן הִנֹּם לְהַעֲבִיר אֶת בְּנֵיהֶם וְאֶת בְּנוֹתֵיהֶם לַמֹּלֶךְ אֲשֶׁר לֹא צִוִּיתִים וְלֹא עָלְתָה עַל לִבִּי לַעֲשׂוֹת הַתּוֹעֵבָה הַזֹּאת לְמַעַן הַחֲטִי אֶת יְהוּדָה` (ירמיהו לב:לה)
`וְנָתַתִּי לָהֶם לֵב אֶחָד וְדֶרֶךְ אֶחָד לְיִרְאָה אוֹתִי כָּל הַיָּמִים לְטוֹב לָהֶם וְלִבְנֵיהֶם אַחֲרֵיהֶם` (ירמיהו לב:לט)
`וְשַׂשְׂתִּי עֲלֵיהֶם לְהֵטִיב אוֹתָם וּנְטַעְתִּים בָּאָרֶץ הַזֹּאת בֶּאֱמֶת בְּכָל לִבִּי וּבְכָל נַפְשִׁי` (ירמיהו לב:מא)
`הֲלוֹא אֶת הַקִּטֵּר אֲשֶׁר קִטַּרְתֶּם בְּעָרֵי יְהוּדָה וּבְחֻצוֹת יְרוּשָׁלִַם אַתֶּם וַאֲבוֹתֵיכֶם מַלְכֵיכֶם וְשָׂרֵיכֶם וְעַם הָאָרֶץ אֹתָם זָכַר יְהוָה וַתַּעֲלֶה עַל לִבּוֹ` (ירמיהו מד:כא)
`שָׁמַעְנוּ גְאוֹן מוֹאָב גֵּאֶה מְאֹד גָּבְהוֹ וּגְאוֹנוֹ וְגַאֲוָתוֹ וְרֻם לִבּוֹ` (ירמיהו מח:כט)
`עַל כֵּן לִבִּי לְמוֹאָב כַּחֲלִלִים יֶהֱמֶה וְלִבִּי אֶל אַנְשֵׁי קִיר חֶרֶשׂ כַּחֲלִילִים יֶהֱמֶה עַל כֵּן יִתְרַת עָשָׂה אָבָדוּ` (ירמיהו מח:לו)
`נִלְכְּדָה הַקְּרִיּוֹת וְהַמְּצָדוֹת נִתְפָּשָׂה וְהָיָה לֵב גִּבּוֹרֵי מוֹאָב בַּיּוֹם הַהוּא כְּלֵב אִשָּׁה מְצֵרָה` (ירמיהו מח:מא)
`תִּפְלַצְתְּךָ הִשִּׁיא אֹתָךְ זְדוֹן לִבֶּךָ שֹׁכְנִי בְּחַגְוֵי הַסֶּלַע תֹּפְשִׂי מְרוֹם גִּבְעָה כִּי תַגְבִּיהַ כַּנֶּשֶׁר קִנֶּךָ מִשָּׁם אוֹרִידְךָ נְאֻם יְהוָה` (ירמיהו מט:טז)
`הִנֵּה כַנֶּשֶׁר יַעֲלֶה וְיִדְאֶה וְיִפְרֹשׂ כְּנָפָיו עַל בָּצְרָה וְהָיָה לֵב גִּבּוֹרֵי אֱדוֹם בַּיּוֹם הַהוּא כְּלֵב אִשָּׁה מְצֵרָה` (ירמיהו מט:כב)
`כֹּה אָמַר יְהוָה הִנְנִי מֵעִיר עַל בָּבֶל וְאֶל יֹשְׁבֵי לֵב קָמָי רוּחַ מַשְׁחִית` (ירמיהו נא:א)
`וְהַבָּנִים קְשֵׁי פָנִים וְחִזְקֵי לֵב אֲנִי שׁוֹלֵחַ אוֹתְךָ אֲלֵיהֶם וְאָמַרְתָּ אֲלֵיהֶם כֹּה אָמַר אֲדֹנָי יְהֹוִה` (יחזקאל ב:ד)
`וּבֵית יִשְׂרָאֵל לֹא יֹאבוּ לִשְׁמֹעַ אֵלֶיךָ כִּי אֵינָם אֹבִים לִשְׁמֹעַ אֵלָי כִּי כָּל בֵּית יִשְׂרָאֵל חִזְקֵי מֵצַח וּקְשֵׁי לֵב הֵמָּה` (יחזקאל ג:ז)
`וְזָכְרוּ פְלִיטֵיכֶם אוֹתִי בַּגּוֹיִם אֲשֶׁר נִשְׁבּוּ שָׁם אֲשֶׁר נִשְׁבַּרְתִּי אֶת לִבָּם הַזּוֹנֶה אֲשֶׁר סָר מֵעָלַי וְאֵת עֵינֵיהֶם הַזֹּנוֹת אַחֲרֵי גִּלּוּלֵיהֶם וְנָקֹטּוּ בִּפְנֵיהֶם אֶל הָרָעוֹת אֲשֶׁר עָשׂוּ לְכֹל תּוֹעֲבֹתֵיהֶם` (יחזקאל ו:ט)
`וְנָתַתִּי לָהֶם לֵב אֶחָד וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי לֵב הָאֶבֶן מִבְּשָׂרָם וְנָתַתִּי לָהֶם לֵב בָּשָׂר` (יחזקאל יא:יט)
`וְאֶל לֵב שִׁקּוּצֵיהֶם וְתוֹעֲבוֹתֵיהֶם לִבָּם הֹלֵךְ דַּרְכָּם בְּרֹאשָׁם נָתַתִּי נְאֻם אֲדֹנָי יְהוִה` (יחזקאל יא:כא)
`בֶּן אָדָם הִנָּבֵא אֶל נְבִיאֵי יִשְׂרָאֵל הַנִּבָּאִים וְאָמַרְתָּ לִנְבִיאֵי מִלִּבָּם שִׁמְעוּ דְּבַר יְהוָה` (יחזקאל יג:ב)
`וְאַתָּה בֶן אָדָם שִׂים פָּנֶיךָ אֶל בְּנוֹת עַמְּךָ הַמִּתְנַבְּאוֹת מִלִּבְּהֶן וְהִנָּבֵא עֲלֵיהֶן` (יחזקאל יג:יז)
`יַעַן הַכְאוֹת לֵב צַדִּיק שֶׁקֶר וַאֲנִי לֹא הִכְאַבְתִּיו וּלְחַזֵּק יְדֵי רָשָׁע לְבִלְתִּי שׁוּב מִדַּרְכּוֹ הָרָע לְהַחֲיֹתוֹ` (יחזקאל יג:כב)
`בֶּן אָדָם הָאֲנָשִׁים הָאֵלֶּה הֶעֱלוּ גִלּוּלֵיהֶם עַל לִבָּם וּמִכְשׁוֹל עֲוֺנָם נָתְנוּ נֹכַח פְּנֵיהֶם הַאִדָּרֹשׁ אִדָּרֵשׁ לָהֶם` (יחזקאל יד:ג)
`לָכֵן דַּבֵּר אוֹתָם וְאָמַרְתָּ אֲלֵיהֶם כֹּה אָמַר אֲדֹנָי יְהוִה אִישׁ אִישׁ מִבֵּית יִשְׂרָאֵל אֲשֶׁר יַעֲלֶה אֶת גִּלּוּלָיו אֶל לִבּוֹ וּמִכְשׁוֹל עֲוֺנוֹ יָשִׂים נֹכַח פָּנָיו וּבָא אֶל הַנָּבִיא אֲנִי יְהוָה נַעֲנֵיתִי לוֹ בָהּ בְּרֹב גִּלּוּלָיו` (יחזקאל יד:ד)
`לְמַעַן תְּפֹשׂ אֶת בֵּית יִשְׂרָאֵל בְּלִבָּם אֲשֶׁר נָזֹרוּ מֵעָלַי בְּגִלּוּלֵיהֶם כֻּלָּם` (יחזקאל יד:ה)
`כִּי אִישׁ אִישׁ מִבֵּית יִשְׂרָאֵל וּמֵהַגֵּר אֲשֶׁר יָגוּר בְּיִשְׂרָאֵל וְיִנָּזֵר מֵאַחֲרַי וְיַעַל גִּלּוּלָיו אֶל לִבּוֹ וּמִכְשׁוֹל עֲוֺנוֹ יָשִׂים נֹכַח פָּנָיו וּבָא אֶל הַנָּבִיא לִדְרָשׁ לוֹ בִי אֲנִי יְהוָה נַעֲנֶה לּוֹ בִּי` (יחזקאל יד:ז)
`הַשְׁלִיכוּ מֵעֲלֵיכֶם אֶת כָּל פִּשְׁעֵיכֶם אֲשֶׁר פְּשַׁעְתֶּם בָּם וַעֲשׂוּ לָכֶם לֵב חָדָשׁ וְרוּחַ חֲדָשָׁה וְלָמָּה תָמֻתוּ בֵּית יִשְׂרָאֵל` (יחזקאל יח:לא)
`יַעַן בְּמִשְׁפָּטַי מָאָסוּ וְאֶת חֻקּוֹתַי לֹא הָלְכוּ בָהֶם וְאֶת שַׁבְּתוֹתַי חִלֵּלוּ כִּי אַחֲרֵי גִלּוּלֵיהֶם לִבָּם הֹלֵךְ` (יחזקאל כ:טז)
`וְהָיָה כִּי יֹאמְרוּ אֵלֶיךָ עַל מָה אַתָּה נֶאֱנָח וְאָמַרְתָּ אֶל שְׁמוּעָה כִי בָאָה וְנָמֵס כָּל לֵב וְרָפוּ כָל יָדַיִם וְכִהֲתָה כָל רוּחַ וְכָל בִּרְכַּיִם תֵּלַכְנָה מַּיִם הִנֵּה בָאָה וְנִהְיָתָה נְאֻם אֲדֹנָי יְהוִה` (יחזקאל כא:ז)
`לְמַעַן לָמוּג לֵב וְהַרְבֵּה הַמִּכְשֹׁלִים עַל כָּל שַׁעֲרֵיהֶם נָתַתִּי אִבְחַת חָרֶב אָח עֲשׂוּיָה לְבָרָק מְעֻטָּה לְטָבַח` (יחזקאל כא:טו)
`הֲיַעֲמֹד לִבֵּךְ אִם תֶּחֱזַקְנָה יָדַיִךְ לַיָּמִים אֲשֶׁר אֲנִי עֹשֶׂה אוֹתָךְ אֲנִי יְהוָה דִּבַּרְתִּי וְעָשִׂיתִי` (יחזקאל כב:יד)
`בְּלֵב יַמִּים גְּבוּלָיִךְ בֹּנַיִךְ כָּלְלוּ יָפְיֵךְ` (יחזקאל כז:ד)
`אֳנִיּוֹת תַּרְשִׁישׁ שָׁרוֹתַיִךְ מַעֲרָבֵךְ וַתִּמָּלְאִי וַתִּכְבְּדִי מְאֹד בְּלֵב יַמִּים` (יחזקאל כז:כה)
`בְּמַיִם רַבִּים הֱבִיאוּךְ הַשָּׁטִים אֹתָךְ רוּחַ הַקָּדִים שְׁבָרֵךְ בְּלֵב יַמִּים` (יחזקאל כז:כו)
`הוֹנֵךְ וְעִזְבוֹנַיִךְ מַעֲרָבֵךְ מַלָּחַיִךְ וְחֹבְלָיִךְ מַחֲזִיקֵי בִדְקֵך וְעֹרְבֵי מַעֲרָבֵךְ וְכָל אַנְשֵׁי מִלְחַמְתֵּךְ אֲשֶׁר בָּךְ וּבְכָל קְהָלֵךְ אֲשֶׁר בְּתוֹכֵךְ יִפְּלוּ בְּלֵב יַמִּים בְּיוֹם מַפַּלְתֵּךְ` (יחזקאל כז:כז)
`בֶּן אָדָם אֱמֹר לִנְגִיד צֹר כֹּה אָמַר אֲדֹנָי יְהֹוִה יַעַן גָּבַהּ לִבְּךָ וַתֹּאמֶר אֵל אָנִי מוֹשַׁב אֱלֹהִים יָשַׁבְתִּי בְּלֵב יַמִּים וְאַתָּה אָדָם וְלֹא אֵל וַתִּתֵּן לִבְּךָ כְּלֵב אֱלֹהִים` (יחזקאל כח:ב)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה יַעַן תִּתְּךָ אֶת לְבָבְךָ כְּלֵב אֱלֹהִים` (יחזקאל כח:ו)
`לַשַּׁחַת יוֹרִדוּךָ וָמַתָּה מְמוֹתֵי חָלָל בְּלֵב יַמִּים` (יחזקאל כח:ח)
`גָּבַהּ לִבְּךָ בְּיָפְיֶךָ שִׁחַתָּ חָכְמָתְךָ עַל יִפְעָתֶךָ עַל אֶרֶץ הִשְׁלַכְתִּיךָ לִפְנֵי מְלָכִים נְתַתִּיךָ לְרַאֲוָה בָךְ` (יחזקאל כח:יז)
`וְהִכְעַסְתִּי לֵב עַמִּים רַבִּים בַּהֲבִיאִי שִׁבְרְךָ בַּגּוֹיִם עַל אֲרָצוֹת אֲשֶׁר לֹא יְדַעְתָּם` (יחזקאל לב:ט)
`וְיָבוֹאוּ אֵלֶיךָ כִּמְבוֹא עָם וְיֵשְׁבוּ לְפָנֶיךָ עַמִּי וְשָׁמְעוּ אֶת דְּבָרֶיךָ וְאוֹתָם לֹא יַעֲשׂוּ כִּי עֲגָבִים בְּפִיהֶם הֵמָּה עֹשִׂים אַחֲרֵי בִצְעָם לִבָּם הֹלֵךְ` (יחזקאל לג:לא)
`וְנָתַתִּי לָכֶם לֵב חָדָשׁ וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי אֶת לֵב הָאֶבֶן מִבְּשַׂרְכֶם וְנָתַתִּי לָכֶם לֵב בָּשָׂר` (יחזקאל לו:כו)
`וַיְדַבֵּר אֵלַי הָאִישׁ בֶּן אָדָם רְאֵה בְעֵינֶיךָ וּבְאָזְנֶיךָ שְּׁמָע וְשִׂים לִבְּךָ לְכֹל אֲשֶׁר אֲנִי מַרְאֶה אוֹתָךְ כִּי לְמַעַן הַרְאוֹתְכָה הֻבָאתָה הֵנָּה הַגֵּד אֶת כָּל אֲשֶׁר אַתָּה רֹאֶה לְבֵית יִשְׂרָאֵל` (יחזקאל מ:ד)
`וַיֹּאמֶר אֵלַי יְהֹוָה בֶּן אָדָם שִׂים לִבְּךָ וּרְאֵה בְעֵינֶיךָ וּבְאָזְנֶיךָ שְּׁמָע אֵת כָּל אֲשֶׁר אֲנִי מְדַבֵּר אֹתָךְ לְכָל חֻקּוֹת בֵּית יְהוָה וּלְכָל תּוֹרֹתוֹ וְשַׂמְתָּ לִבְּךָ לִמְבוֹא הַבַּיִת בְּכֹל מוֹצָאֵי הַמִּקְדָּשׁ` (יחזקאל מד:ה)
`בַּהֲבִיאֲכֶם בְּנֵי נֵכָר עַרְלֵי לֵב וְעַרְלֵי בָשָׂר לִהְיוֹת בְּמִקְדָּשִׁי לְחַלְּלוֹ אֶת בֵּיתִי בְּהַקְרִיבְכֶם אֶת לַחְמִי חֵלֶב וָדָם וַיָּפֵרוּ אֶת בְּרִיתִי אֶל כָּל תּוֹעֲבוֹתֵיכֶם` (יחזקאל מד:ז)
`כֹּה אָמַר אֲדֹנָי יְהוִה כָּל בֶּן נֵכָר עֶרֶל לֵב וְעֶרֶל בָּשָׂר לֹא יָבוֹא אֶל מִקְדָּשִׁי לְכָל בֶּן נֵכָר אֲשֶׁר בְּתוֹךְ בְּנֵי יִשְׂרָאֵל` (יחזקאל מד:ט)
`לָכֵן הִנֵּה אָנֹכִי מְפַתֶּיהָ וְהֹלַכְתִּיהָ הַמִּדְבָּר וְדִבַּרְתִּי עַל לִבָּהּ` (הושע ב:יד)
`זְנוּת וְיַיִן וְתִירוֹשׁ יִקַּח לֵב` (הושע ד:יא)
`כִּי קֵרְבוּ כַתַּנּוּר לִבָּם בְּאָרְבָּם כָּל הַלַּיְלָה יָשֵׁן אֹפֵהֶם בֹּקֶר הוּא בֹעֵר כְּאֵשׁ לֶהָבָה` (הושע ז:ו)
`וַיְהִי אֶפְרַיִם כְּיוֹנָה פוֹתָה אֵין לֵב מִצְרַיִם קָרָאוּ אַשּׁוּר הָלָכוּ` (הושע ז:יא)
`וְלֹא זָעֲקוּ אֵלַי בְּלִבָּם כִּי יְיֵלִילוּ עַל מִשְׁכְּבוֹתָם עַל דָּגָן וְתִירוֹשׁ יִתְגּוֹרָרוּ יָסוּרוּ בִי` (הושע ז:יד)
`חָלַק לִבָּם עַתָּה יֶאְשָׁמוּ הוּא יַעֲרֹף מִזְבְּחוֹתָם יְשֹׁדֵד מַצֵּבוֹתָם` (הושע י:ב)
`אֵיךְ אֶתֶּנְךָ אֶפְרַיִם אֲמַגֶּנְךָ יִשְׂרָאֵל אֵיךְ אֶתֶּנְךָ כְאַדְמָה אֲשִׂימְךָ כִּצְבֹאיִם נֶהְפַּךְ עָלַי לִבִּי יַחַד נִכְמְרוּ נִחוּמָי` (הושע יא:ח)
`כְּמַרְעִיתָם וַיִּשְׂבָּעוּ שָׂבְעוּ וַיָּרָם לִבָּם עַל כֵּן שְׁכֵחוּנִי` (הושע יג:ו)
`אֶפְגְּשֵׁם כְּדֹב שַׁכּוּל וְאֶקְרַע סְגוֹר לִבָּם וְאֹכְלֵם שָׁם כְּלָבִיא חַיַּת הַשָּׂדֶה תְּבַקְּעֵם` (הושע יג:ח)
`וְאַמִּיץ לִבּוֹ בַּגִּבּוֹרִים עָרוֹם יָנוּס בַּיּוֹם הַהוּא נְאֻם יְהוָה` (עמוס ב:טז)
`זְדוֹן לִבְּךָ הִשִּׁיאֶךָ שֹׁכְנִי בְחַגְוֵי סֶּלַע מְרוֹם שִׁבְתּוֹ אֹמֵר בְּלִבּוֹ מִי יוֹרִדֵנִי אָרֶץ` (עובדיה א:ג)
`בּוּקָה וּמְבוּקָה וּמְבֻלָּקָה וְלֵב נָמֵס וּפִק בִּרְכַּיִם וְחַלְחָלָה בְּכָל מָתְנַיִם וּפְנֵי כֻלָּם קִבְּצוּ פָארוּר` (נחום ב:י)
`רָנִּי בַּת צִיּוֹן הָרִיעוּ יִשְׂרָאֵל שִׂמְחִי וְעָלְזִי בְּכָל לֵב בַּת יְרוּשָׁלִָם` (צפניה ג:יד)
`וְלִבָּם שָׂמוּ שָׁמִיר מִשְּׁמוֹעַ אֶת הַתּוֹרָה וְאֶת הַדְּבָרִים אֲשֶׁר שָׁלַח יְהוָה צְבָאוֹת בְּרוּחוֹ בְּיַד הַנְּבִיאִים הָרִאשֹׁנִים וַיְהִי קֶצֶף גָּדוֹל מֵאֵת יְהוָה צְבָאוֹת` (זכריה ז:יב)
`וְהָיוּ כְגִבּוֹר אֶפְרַיִם וְשָׂמַח לִבָּם כְּמוֹ יָיִן וּבְנֵיהֶם יִרְאוּ וְשָׂמֵחוּ יָגֵל לִבָּם בַּיהוָה` (זכריה י:ז)
`וְאָמְרוּ אַלֻּפֵי יְהוּדָה בְּלִבָּם אַמְצָה לִי יֹשְׁבֵי יְרוּשָׁלִַם בַּיהוָה צְבָאוֹת אֱלֹהֵיהֶם` (זכריה יב:ה)
`אִם לֹא תִשְׁמְעוּ וְאִם לֹא תָשִׂימוּ עַל לֵב לָתֵת כָּבוֹד לִשְׁמִי אָמַר יְהוָה צְבָאוֹת וְשִׁלַּחְתִּי בָכֶם אֶת הַמְּאֵרָה וְאָרוֹתִי אֶת בִּרְכוֹתֵיכֶם וְגַם אָרוֹתִיהָ כִּי אֵינְכֶם שָׂמִים עַל לֵב` (מלאכי ב:ב)
`וְהֵשִׁיב לֵב אָבוֹת עַל בָּנִים וְלֵב בָּנִים עַל אֲבוֹתָם פֶּן אָבוֹא וְהִכֵּיתִי אֶת הָאָרֶץ חֵרֶם` (מלאכי ד:ו)
`מִזְּבֻלוּן יוֹצְאֵי צָבָא עֹרְכֵי מִלְחָמָה בְּכָל כְּלֵי מִלְחָמָה חֲמִשִּׁים אָלֶף וְלַעֲדֹר בְּלֹא לֵב וָלֵב` (דברי-הימים-א יב:לג)
`כָּל אֵלֶּה אַנְשֵׁי מִלְחָמָה עֹדְרֵי מַעֲרָכָה בְּלֵבָב שָׁלֵם בָּאוּ חֶבְרוֹנָה לְהַמְלִיךְ אֶת דָּוִיד עַל כָּל יִשְׂרָאֵל וְגַם כָּל שֵׁרִית יִשְׂרָאֵל לֵב אֶחָד לְהַמְלִיךְ אֶת דָּוִיד` (דברי-הימים-א יב:לח)
`וַיְהִי אֲרוֹן בְּרִית יְהוָה בָּא עַד עִיר דָּוִיד וּמִיכַל בַּת שָׁאוּל נִשְׁקְפָה בְּעַד הַחַלּוֹן וַתֵּרֶא אֶת הַמֶּלֶךְ דָּוִיד מְרַקֵּד וּמְשַׂחֵק וַתִּבֶז לוֹ בְּלִבָּהּ` (דברי-הימים-א טו:כט)
`הִתְהַלְלוּ בְּשֵׁם קָדְשׁוֹ יִשְׂמַח לֵב מְבַקְשֵׁי יְהוָה` (דברי-הימים-א טז:י)
`יְהוָה בַּעֲבוּר עַבְדְּךָ וּכְלִבְּךָ עָשִׂיתָ אֵת כָּל הַגְּדוּלָּה הַזֹּאת לְהֹדִיעַ אֶת כָּל הַגְּדֻלּוֹת` (דברי-הימים-א יז:יט)
`וְאַתָּה שְׁלֹמֹה בְנִי דַּע אֶת אֱלֹהֵי אָבִיךָ וְעָבְדֵהוּ בְּלֵב שָׁלֵם וּבְנֶפֶשׁ חֲפֵצָה כִּי כָל לְבָבוֹת דּוֹרֵשׁ יְהוָה וְכָל יֵצֶר מַחֲשָׁבוֹת מֵבִין אִם תִּדְרְשֶׁנּוּ יִמָּצֵא לָךְ וְאִם תַּעַזְבֶנּוּ יַזְנִיחֲךָ לָעַד` (דברי-הימים-א כח:ט)
`וַיִּשְׂמְחוּ הָעָם עַל הִתְנַדְּבָם כִּי בְּלֵב שָׁלֵם הִתְנַדְּבוּ לַיהוָה וְגַם דָּוִיד הַמֶּלֶךְ שָׂמַח שִׂמְחָה גְדוֹלָה` (דברי-הימים-א כט:ט)
`וַיֹּאמַר יְהוָה אֱלֹהֵי יִשְׂרָאֵל אֵין כָּמוֹךָ אֱלֹהִים בַּשָּׁמַיִם וּבָאָרֶץ שֹׁמֵר הַבְּרִית וְהַחֶסֶד לַעֲבָדֶיךָ הַהֹלְכִים לְפָנֶיךָ בְּכָל לִבָּם` (דברי-הימים-ב ו:יד)
`וְשָׁבוּ אֵלֶיךָ בְּכָל לִבָּם וּבְכָל נַפְשָׁם בְּאֶרֶץ שִׁבְיָם אֲשֶׁר שָׁבוּ אֹתָם וְהִתְפַּלְלוּ דֶּרֶךְ אַרְצָם אֲשֶׁר נָתַתָּה לַאֲבוֹתָם וְהָעִיר אֲשֶׁר בָּחַרְתָּ וְלַבַּיִת אֲשֶׁר בָּנִיתִי לִשְׁמֶךָ` (דברי-הימים-ב ו:לח)
`וּבְיוֹם עֶשְׂרִים וּשְׁלֹשָׁה לַחֹדֶשׁ הַשְּׁבִיעִי שִׁלַּח אֶת הָעָם לְאָהֳלֵיהֶם שְׂמֵחִים וְטוֹבֵי לֵב עַל הַטּוֹבָה אֲשֶׁר עָשָׂה יְהוָה לְדָוִיד וְלִשְׁלֹמֹה וּלְיִשְׂרָאֵל עַמּוֹ` (דברי-הימים-ב ז:י)
`וַיְכַל שְׁלֹמֹה אֶת בֵּית יְהוָה וְאֶת בֵּית הַמֶּלֶךְ וְאֵת כָּל הַבָּא עַל לֵב שְׁלֹמֹה לַעֲשׂוֹת בְּבֵית יְהוָה וּבְבֵיתוֹ הִצְלִיחַ` (דברי-הימים-ב ז:יא)
`וְעַתָּה בָּחַרְתִּי וְהִקְדַּשְׁתִּי אֶת הַבַּיִת הַזֶּה לִהְיוֹת שְׁמִי שָׁם עַד עוֹלָם וְהָיוּ עֵינַי וְלִבִּי שָׁם כָּל הַיָּמִים` (דברי-הימים-ב ז:טז)
`וְכֹל מַלְכֵי הָאָרֶץ מְבַקְשִׁים אֶת פְּנֵי שְׁלֹמֹה לִשְׁמֹעַ אֶת חָכְמָתוֹ אֲשֶׁר נָתַן הָאֱלֹהִים בְּלִבּוֹ` (דברי-הימים-ב ט:כג)
`וַיַּעַשׂ הָרָע כִּי לֹא הֵכִין לִבּוֹ לִדְרוֹשׁ אֶת יְהוָה` (דברי-הימים-ב יב:יד)
`וַיִּגְבַּהּ לִבּוֹ בְּדַרְכֵי יְהוָה וְעוֹד הֵסִיר אֶת הַבָּמוֹת וְאֶת הָאֲשֵׁרִים מִיהוּדָה` (דברי-הימים-ב יז:ו)
`וַיְהִי אַחֲרֵיכֵן הָיָה עִם לֵב יוֹאָשׁ לְחַדֵּשׁ אֶת בֵּית יְהוָה` (דברי-הימים-ב כד:ד)
`אָמַרְתָּ הִנֵּה הִכִּיתָ אֶת אֱדוֹם וּנְשָׂאֲךָ לִבְּךָ לְהַכְבִּיד עַתָּה שְׁבָה בְּבֵיתֶךָ לָמָּה תִתְגָּרֶה בְּרָעָה וְנָפַלְתָּ אַתָּה וִיהוּדָה עִמָּךְ` (דברי-הימים-ב כה:יט)
`וּכְחֶזְקָתוֹ גָּבַהּ לִבּוֹ עַד לְהַשְׁחִית וַיִּמְעַל בַּיהוָה אֱלֹהָיו וַיָּבֹא אֶל הֵיכַל יְהוָה לְהַקְטִיר עַל מִזְבַּח הַקְּטֹרֶת` (דברי-הימים-ב כו:טז)
`וַיַּעַן יְחִזְקִיָּהוּ וַיֹּאמֶר עַתָּה מִלֵּאתֶם יֶדְכֶם לַיהוָה גֹּשׁוּ וְהָבִיאוּ זְבָחִים וְתוֹדוֹת לְבֵית יְהוָה וַיָּבִיאוּ הַקָּהָל זְבָחִים וְתוֹדוֹת וְכָל נְדִיב לֵב עֹלוֹת` (דברי-הימים-ב כט:לא)
`גַּם בִּיהוּדָה הָיְתָה יַד הָאֱלֹהִים לָתֵת לָהֶם לֵב אֶחָד לַעֲשׂוֹת מִצְוַת הַמֶּלֶךְ וְהַשָּׂרִים בִּדְבַר יְהוָה` (דברי-הימים-ב ל:יב)
`וַיְדַבֵּר יְחִזְקִיָּהוּ עַל לֵב כָּל הַלְוִיִּם הַמַּשְׂכִּילִים שֵׂכֶל טוֹב לַיהוָה וַיֹּאכְלוּ אֶת הַמּוֹעֵד שִׁבְעַת הַיָּמִים מְזַבְּחִים זִבְחֵי שְׁלָמִים וּמִתְוַדִּים לַיהוָה אֱלֹהֵי אֲבוֹתֵיהֶם` (דברי-הימים-ב ל:כב)
`וְלֹא כִגְמֻל עָלָיו הֵשִׁיב יְחִזְקִיָּהוּ כִּי גָבַהּ לִבּוֹ וַיְהִי עָלָיו קֶצֶף וְעַל יְהוּדָה וִירוּשָׁלִָם` (דברי-הימים-ב לב:כה)
`וַיִּכָּנַע יְחִזְקִיָּהוּ בְּגֹבַהּ לִבּוֹ הוּא וְיֹשְׁבֵי יְרוּשָׁלִָם וְלֹא בָא עֲלֵיהֶם קֶצֶף יְהוָה בִּימֵי יְחִזְקִיָּהוּ` (דברי-הימים-ב לב:כו)
`נָתַתָּה שִׂמְחָה בְלִבִּי מֵעֵת דְּגָנָם וְתִירוֹשָׁם רָבּוּ` (תהילים ד:ז)
`מָגִנִּי עַל אֱלֹהִים מוֹשִׁיעַ יִשְׁרֵי לֵב` (תהילים ז:י)
`לַמְנַצֵּחַ עַלְמוּת לַבֵּן מִזְמוֹר לְדָוִד אוֹדֶה יְהוָה בְּכָל לִבִּי אֲסַפְּרָה כָּל נִפְלְאוֹתֶיךָ` (תהילים ט:א)
`אָמַר בְּלִבּוֹ בַּל אֶמּוֹט לְדֹר וָדֹר אֲשֶׁר לֹא בְרָע` (תהילים י:ו)
`אָמַר בְּלִבּוֹ שָׁכַח אֵל הִסְתִּיר פָּנָיו בַּל רָאָה לָנֶצַח` (תהילים י:יא)
`עַל מֶה נִאֵץ רָשָׁע אֱלֹהִים אָמַר בְּלִבּוֹ לֹא תִּדְרֹשׁ` (תהילים י:יג)
`תַּאֲוַת עֲנָוִים שָׁמַעְתָּ יְהוָה תָּכִין לִבָּם תַּקְשִׁיב אָזְנֶךָ` (תהילים י:יז)
`כִּי הִנֵּה הָרְשָׁעִים יִדְרְכוּן קֶשֶׁת כּוֹנְנוּ חִצָּם עַל יֶתֶר לִירוֹת בְּמוֹ אֹפֶל לְיִשְׁרֵי לֵב` (תהילים יא:ב)
`שָׁוְא יְדַבְּרוּ אִישׁ אֶת רֵעֵהוּ שְׂפַת חֲלָקוֹת בְּלֵב וָלֵב יְדַבֵּרוּ` (תהילים יב:ב)
`וַאֲנִי בְּחַסְדְּךָ בָטַחְתִּי יָגֵל לִבִּי בִּישׁוּעָתֶךָ` (תהילים יג:ה)
`לַמְנַצֵּחַ לְדָוִד אָמַר נָבָל בְּלִבּוֹ אֵין אֱלֹהִים הִשְׁחִיתוּ הִתְעִיבוּ עֲלִילָה אֵין עֹשֵׂה טוֹב` (תהילים יד:א)
`לָכֵן שָׂמַח לִבִּי וַיָּגֶל כְּבוֹדִי אַף בְּשָׂרִי יִשְׁכֹּן לָבֶטַח` (תהילים טז:ט)
`בָּחַנְתָּ לִבִּי פָּקַדְתָּ לַּיְלָה צְרַפְתַּנִי בַל תִּמְצָא זַמֹּתִי בַּל יַעֲבָר פִּי` (תהילים יז:ג)
`פִּקּוּדֵי יְהוָה יְשָׁרִים מְשַׂמְּחֵי לֵב מִצְוַת יְהוָה בָּרָה מְאִירַת עֵינָיִם` (תהילים יט:ח)
`יִהְיוּ לְרָצוֹן אִמְרֵי פִי וְהֶגְיוֹן לִבִּי לְפָנֶיךָ יְהוָה צוּרִי וְגֹאֲלִי` (תהילים יט:יד)
`תַּאֲוַת לִבּוֹ נָתַתָּה לּוֹ וַאֲרֶשֶׁת שְׂפָתָיו בַּל מָנַעְתָּ סֶּלָה` (תהילים כא:ב)
`כַּמַּיִם נִשְׁפַּכְתִּי וְהִתְפָּרְדוּ כָּל עַצְמוֹתָי הָיָה לִבִּי כַּדּוֹנָג נָמֵס בְּתוֹךְ מֵעָי` (תהילים כב:יד)
`בְּחָנֵנִי יְהוָה וְנַסֵּנִי צְרוֹפָה כִלְיוֹתַי וְלִבִּי` (תהילים כו:ב)
`אִם תַּחֲנֶה עָלַי מַחֲנֶה לֹא יִירָא לִבִּי אִם תָּקוּם עָלַי מִלְחָמָה בְּזֹאת אֲנִי בוֹטֵחַ` (תהילים כז:ג)
`לְךָ אָמַר לִבִּי בַּקְּשׁוּ פָנָי אֶת פָּנֶיךָ יְהוָה אֲבַקֵּשׁ` (תהילים כז:ח)
`קַוֵּה אֶל יְהוָה חֲזַק וְיַאֲמֵץ לִבֶּךָ וְקַוֵּה אֶל יְהוָה` (תהילים כז:יד)
`יְהוָה עֻזִּי וּמָגִנִּי בּוֹ בָטַח לִבִּי וְנֶעֱזָרְתִּי וַיַּעֲלֹז לִבִּי וּמִשִּׁירִי אֲהוֹדֶנּוּ` (תהילים כח:ז)
`נִשְׁכַּחְתִּי כְּמֵת מִלֵּב הָיִיתִי כִּכְלִי אֹבֵד` (תהילים לא:יב)
`שִׂמְחוּ בַיהוָה וְגִילוּ צַדִּיקִים וְהַרְנִינוּ כָּל יִשְׁרֵי לֵב` (תהילים לב:יא)
`עֲצַת יְהוָה לְעוֹלָם תַּעֲמֹד מַחְשְׁבוֹת לִבּוֹ לְדֹר וָדֹר` (תהילים לג:יא)
`הַיֹּצֵר יַחַד לִבָּם הַמֵּבִין אֶל כָּל מַעֲשֵׂיהֶם` (תהילים לג:טו)
`כִּי בוֹ יִשְׂמַח לִבֵּנוּ כִּי בְשֵׁם קָדְשׁוֹ בָטָחְנוּ` (תהילים לג:כא)
`קָרוֹב יְהוָה לְנִשְׁבְּרֵי לֵב וְאֶת דַּכְּאֵי רוּחַ יוֹשִׁיעַ` (תהילים לד:יח)
`אַל יֹאמְרוּ בְלִבָּם הֶאָח נַפְשֵׁנוּ אַל יֹאמְרוּ בִּלַּעֲנוּהוּ` (תהילים לה:כה)
`לַמְנַצֵּחַ לְעֶבֶד יְהוָה לְדָוִד נְאֻם פֶּשַׁע לָרָשָׁע בְּקֶרֶב לִבִּי אֵין פַּחַד אֱלֹהִים לְנֶגֶד עֵינָיו` (תהילים לו:א)
`מְשֹׁךְ חַסְדְּךָ לְיֹדְעֶיךָ וְצִדְקָתְךָ לְיִשְׁרֵי לֵב` (תהילים לו:י)
`וְהִתְעַנַּג עַל יְהוָה וְיִתֶּן לְךָ מִשְׁאֲלֹת לִבֶּךָ` (תהילים לז:ד)
`חַרְבָּם תָּבוֹא בְלִבָּם וְקַשְּׁתוֹתָם תִּשָּׁבַרְנָה` (תהילים לז:טו)
`תּוֹרַת אֱלֹהָיו בְּלִבּוֹ לֹא תִמְעַד אֲשֻׁרָיו` (תהילים לז:לא)
`נְפוּגוֹתִי וְנִדְכֵּיתִי עַד מְאֹד שָׁאַגְתִּי מִנַּהֲמַת לִבִּי` (תהילים לח:ח)
`לִבִּי סְחַרְחַר עֲזָבַנִי כֹחִי וְאוֹר עֵינַי גַּם הֵם אֵין אִתִּי` (תהילים לח:י)
`חַם לִבִּי בְּקִרְבִּי בַּהֲגִיגִי תִבְעַר אֵשׁ דִּבַּרְתִּי בִּלְשׁוֹנִי` (תהילים לט:ג)
`צִדְקָתְךָ לֹא כִסִּיתִי בְּתוֹךְ לִבִּי אֱמוּנָתְךָ וּתְשׁוּעָתְךָ אָמָרְתִּי לֹא כִחַדְתִּי חַסְדְּךָ וַאֲמִתְּךָ לְקָהָל רָב` (תהילים מ:י)
`כִּי אָפְפוּ עָלַי רָעוֹת עַד אֵין מִסְפָּר הִשִּׂיגוּנִי עֲוֺנֹתַי וְלֹא יָכֹלְתִּי לִרְאוֹת עָצְמוּ מִשַּׂעֲרוֹת רֹאשִׁי וְלִבִּי עֲזָבָנִי` (תהילים מ:יב)
`וְאִם בָּא לִרְאוֹת שָׁוְא יְדַבֵּר לִבּוֹ יִקְבָּץ אָוֶן לוֹ יֵצֵא לַחוּץ יְדַבֵּר` (תהילים מא:ו)
`לֹא נָסוֹג אָחוֹר לִבֵּנוּ וַתֵּט אֲשֻׁרֵינוּ מִנִּי אָרְחֶךָ` (תהילים מד:יח)
`הֲלֹא אֱלֹהִים יַחֲקָר זֹאת כִּי הוּא יֹדֵעַ תַּעֲלֻמוֹת לֵב` (תהילים מד:כא)
`לַמְנַצֵּחַ עַל שֹׁשַׁנִּים לִבְנֵי קֹרַח מַשְׂכִּיל שִׁיר יְדִידֹת רָחַשׁ לִבִּי דָּבָר טוֹב אֹמֵר אָנִי מַעֲשַׂי לְמֶלֶךְ לְשׁוֹנִי עֵט סוֹפֵר מָהִיר` (תהילים מה:א)
`חִצֶּיךָ שְׁנוּנִים עַמִּים תַּחְתֶּיךָ יִפְּלוּ בְּלֵב אוֹיְבֵי הַמֶּלֶךְ` (תהילים מה:ה)
`עַל כֵּן לֹא נִירָא בְּהָמִיר אָרֶץ וּבְמוֹט הָרִים בְּלֵב יַמִּים` (תהילים מו:ב)
`שִׁיתוּ לִבְּכֶם לְחֵילָה פַּסְּגוּ אַרְמְנוֹתֶיהָ לְמַעַן תְּסַפְּרוּ לְדוֹר אַחֲרוֹן` (תהילים מח:יג)
`פִּי יְדַבֵּר חָכְמוֹת וְהָגוּת לִבִּי תְבוּנוֹת` (תהילים מט:ג)
`לֵב טָהוֹר בְּרָא לִי אֱלֹהִים וְרוּחַ נָכוֹן חַדֵּשׁ בְּקִרְבִּי` (תהילים נא:י)
`זִבְחֵי אֱלֹהִים רוּחַ נִשְׁבָּרָה לֵב נִשְׁבָּר וְנִדְכֶּה אֱלֹהִים לֹא תִבְזֶה` (תהילים נא:יז)
`לַמְנַצֵּחַ עַל מָחֲלַת מַשְׂכִּיל לְדָוִד אָמַר נָבָל בְּלִבּוֹ אֵין אֱלֹהִים הִשְׁחִיתוּ וְהִתְעִיבוּ עָוֶל אֵין עֹשֵׂה טוֹב` (תהילים נג:א)
`לִבִּי יָחִיל בְּקִרְבִּי וְאֵימוֹת מָוֶת נָפְלוּ עָלָי` (תהילים נה:ד)
`חָלְקוּ מַחְמָאֹת פִּיו וּקֲרָב לִבּוֹ רַכּוּ דְבָרָיו מִשֶּׁמֶן וְהֵמָּה פְתִחוֹת` (תהילים נה:כא)
`נָכוֹן לִבִּי אֱלֹהִים נָכוֹן לִבִּי אָשִׁירָה וַאֲזַמֵּרָה` (תהילים נז:ז)
`אַף בְּלֵב עוֹלֹת תִּפְעָלוּן בָּאָרֶץ חֲמַס יְדֵיכֶם תְּפַלֵּסוּן` (תהילים נח:ב)
`מִקְצֵה הָאָרֶץ אֵלֶיךָ אֶקְרָא בַּעֲטֹף לִבִּי בְּצוּר יָרוּם מִמֶּנִּי תַנְחֵנִי` (תהילים סא:ב)
`אַל תִּבְטְחוּ בְעֹשֶׁק וּבְגָזֵל אַל תֶּהְבָּלוּ חַיִל כִּי יָנוּב אַל תָּשִׁיתוּ לֵב` (תהילים סב:י)
`יַחְפְּשׂוּ עוֹלֹת תַּמְנוּ חֵפֶשׂ מְחֻפָּשׂ וְקֶרֶב אִישׁ וְלֵב עָמֹק` (תהילים סד:ו)
`יִשְׂמַח צַדִּיק בַּיהוָה וְחָסָה בוֹ וְיִתְהַלְלוּ כָּל יִשְׁרֵי לֵב` (תהילים סד:י)
`אָוֶן אִם רָאִיתִי בְלִבִּי לֹא יִשְׁמַע אֲדֹנָי` (תהילים סו:יח)
`חֶרְפָּה שָׁבְרָה לִבִּי וָאָנוּשָׁה וָאֲקַוֶּה לָנוּד וָאַיִן וְלַמְנַחֲמִים וְלֹא מָצָאתִי` (תהילים סט:כ)
`אָמְרוּ בְלִבָּם נִינָם יָחַד שָׂרְפוּ כָל מוֹעֲדֵי אֵל בָּאָרֶץ` (תהילים עד:ח)
`אֶשְׁתּוֹלְלוּ אַבִּירֵי לֵב נָמוּ שְׁנָתָם וְלֹא מָצְאוּ כָל אַנְשֵׁי חַיִל יְדֵיהֶם` (תהילים עו:ה)
`וְלֹא יִהְיוּ כַּאֲבוֹתָם דּוֹר סוֹרֵר וּמֹרֶה דּוֹר לֹא הֵכִין לִבּוֹ וְלֹא נֶאֶמְנָה אֶת אֵל רוּחוֹ` (תהילים עח:ח)
`וְלִבָּם לֹא נָכוֹן עִמּוֹ וְלֹא נֶאֶמְנוּ בִּבְרִיתוֹ` (תהילים עח:לז)
`וָאֲשַׁלְּחֵהוּ בִּשְׁרִירוּת לִבָּם יֵלְכוּ בְּמוֹעֲצוֹתֵיהֶם` (תהילים פא:יב)
`כִּי נוֹעֲצוּ לֵב יַחְדָּו עָלֶיךָ בְּרִית יִכְרֹתוּ` (תהילים פג:ה)
`נִכְסְפָה וְגַם כָּלְתָה נַפְשִׁי לְחַצְרוֹת יְהוָה לִבִּי וּבְשָׂרִי יְרַנְּנוּ אֶל אֵל חָי` (תהילים פד:ב)
`כִּי עַד צֶדֶק יָשׁוּב מִשְׁפָּט וְאַחֲרָיו כָּל יִשְׁרֵי לֵב` (תהילים צד:טו)
`אוֹר זָרֻעַ לַצַּדִּיק וּלְיִשְׁרֵי לֵב שִׂמְחָה` (תהילים צז:יא)
`הוּכָּה כָעֵשֶׂב וַיִּבַשׁ לִבִּי כִּי שָׁכַחְתִּי מֵאֲכֹל לַחְמִי` (תהילים קב:ד)
`הִתְהַלְלוּ בְּשֵׁם קָדְשׁוֹ יִשְׂמַח לֵב מְבַקְשֵׁי יְהוָה` (תהילים קה:ג)
`הָפַךְ לִבָּם לִשְׂנֹא עַמּוֹ לְהִתְנַכֵּל בַּעֲבָדָיו` (תהילים קה:כה)
`וַיַּכְנַע בֶּעָמָל לִבָּם כָּשְׁלוּ וְאֵין עֹזֵר` (תהילים קז:יב)
`שִׁיר מִזְמוֹר לְדָוִד נָכוֹן לִבִּי אֱלֹהִים אָשִׁירָה וַאֲזַמְּרָה אַף כְּבוֹדִי` (תהילים קח:א)
`כִּי עָנִי וְאֶבְיוֹן אָנֹכִי וְלִבִּי חָלַל בְּקִרְבִּי` (תהילים קט:כב)
`מִשְּׁמוּעָה רָעָה לֹא יִירָא נָכוֹן לִבּוֹ בָּטֻחַ בַּיהוָה` (תהילים קיב:ז)
`סָמוּךְ לִבּוֹ לֹא יִירָא עַד אֲשֶׁר יִרְאֶה בְצָרָיו` (תהילים קיב:ח)
`אַשְׁרֵי נֹצְרֵי עֵדֹתָיו בְּכָל לֵב יִדְרְשׁוּהוּ` (תהילים קיט:ב)
`בְּכָל לִבִּי דְרַשְׁתִּיךָ אַל תַּשְׁגֵּנִי מִמִּצְוֺתֶיךָ` (תהילים קיט:י)
`בְּלִבִּי צָפַנְתִּי אִמְרָתֶךָ לְמַעַן לֹא אֶחֱטָא לָךְ` (תהילים קיט:יא)
`דֶּרֶךְ מִצְוֺתֶיךָ אָרוּץ כִּי תַרְחִיב לִבִּי` (תהילים קיט:לב)
`הֲבִינֵנִי וְאֶצְּרָה תוֹרָתֶךָ וְאֶשְׁמְרֶנָּה בְכָל לֵב` (תהילים קיט:לד)
`הַט לִבִּי אֶל עֵדְוֺתֶיךָ וְאַל אֶל בָּצַע` (תהילים קיט:לו)
`חִלִּיתִי פָנֶיךָ בְכָל לֵב חָנֵּנִי כְּאִמְרָתֶךָ` (תהילים קיט:נח)
`טָפְלוּ עָלַי שֶׁקֶר זֵדִים אֲנִי בְּכָל לֵב אֱצֹּר פִּקּוּדֶיךָ` (תהילים קיט:סט)
`טָפַשׁ כַּחֵלֶב לִבָּם אֲנִי תּוֹרָתְךָ שִׁעֲשָׁעְתִּי` (תהילים קיט:ע)
`יְהִי לִבִּי תָמִים בְּחֻקֶּיךָ לְמַעַן לֹא אֵבוֹשׁ` (תהילים קיט:פ)
`נָחַלְתִּי עֵדְוֺתֶיךָ לְעוֹלָם כִּי שְׂשׂוֹן לִבִּי הֵמָּה` (תהילים קיט:קיא)
`נָטִיתִי לִבִּי לַעֲשׂוֹת חֻקֶּיךָ לְעוֹלָם עֵקֶב` (תהילים קיט:קיב)
`קָרָאתִי בְכָל לֵב עֲנֵנִי יְהוָה חֻקֶּיךָ אֶצֹּרָה` (תהילים קיט:קמה)
`שָׂרִים רְדָפוּנִי חִנָּם וּמִדְּבָרֶיךָ פָּחַד לִבִּי` (תהילים קיט:קסא)
`שִׁיר הַמַּעֲלוֹת לְדָוִד יְהוָה לֹא גָבַהּ לִבִּי וְלֹא רָמוּ עֵינַי וְלֹא הִלַּכְתִּי בִּגְדֹלוֹת וּבְנִפְלָאוֹת מִמֶּנִּי` (תהילים קלא:א)
`לְדָוִד אוֹדְךָ בְכָל לִבִּי נֶגֶד אֱלֹהִים אֲזַמְּרֶךָּ` (תהילים קלח:א)
`אֲשֶׁר חָשְׁבוּ רָעוֹת בְּלֵב כָּל יוֹם יָגוּרוּ מִלְחָמוֹת` (תהילים קמ:ב)
`אַל תַּט לִבִּי לְדָבָר רָע לְהִתְעוֹלֵל עֲלִלוֹת בְּרֶשַׁע אֶת אִישִׁים פֹּעֲלֵי אָוֶן וּבַל אֶלְחַם בְּמַנְעַמֵּיהֶם` (תהילים קמא:ד)
`וַתִּתְעַטֵּף עָלַי רוּחִי בְּתוֹכִי יִשְׁתּוֹמֵם לִבִּי` (תהילים קמג:ד)
`הָרֹפֵא לִשְׁבוּרֵי לֵב וּמְחַבֵּשׁ לְעַצְּבוֹתָם` (תהילים קמז:ג)
`וַיֹּאמֶר יְהוָה אֶל הַשָּׂטָן הֲשַׂמְתָּ לִבְּךָ עַל עַבְדִּי אִיּוֹב כִּי אֵין כָּמֹהוּ בָּאָרֶץ אִישׁ תָּם וְיָשָׁר יְרֵא אֱלֹהִים וְסָר מֵרָע` (איוב א:ח)
`וַיֹּאמֶר יְהוָה אֶל הַשָּׂטָן הֲשַׂמְתָּ לִבְּךָ אֶל עַבְדִּי אִיּוֹב כִּי אֵין כָּמֹהוּ בָּאָרֶץ אִישׁ תָּם וְיָשָׁר יְרֵא אֱלֹהִים וְסָר מֵרָע וְעֹדֶנּוּ מַחֲזִיק בְּתֻמָּתוֹ וַתְּסִיתֵנִי בוֹ לְבַלְּעוֹ חִנָּם` (איוב ב:ג)
`מָה אֱנוֹשׁ כִּי תְגַדְּלֶנּוּ וְכִי תָשִׁית אֵלָיו לִבֶּךָ` (איוב ז:יז)
`הֲלֹא הֵם יוֹרוּךָ יֹאמְרוּ לָךְ וּמִלִּבָּם יוֹצִאוּ מִלִּים` (איוב ח:י)
`אִם אַתָּה הֲכִינוֹתָ לִבֶּךָ וּפָרַשְׂתָּ אֵלָיו כַּפֶּךָ` (איוב יא:יג)
`מֵסִיר לֵב רָאשֵׁי עַם הָאָרֶץ וַיַּתְעֵם בְּתֹהוּ לֹא דָרֶךְ` (איוב יב:כד)
`מַה יִּקָּחֲךָ לִבֶּךָ וּמַה יִּרְזְמוּן עֵינֶיךָ` (איוב טו:יב)
`כִּי לִבָּם צָפַנְתָּ מִּשָּׂכֶל עַל כֵּן לֹא תְרֹמֵם` (איוב יז:ד)
`וְאֵל הֵרַךְ לִבִּי וְשַׁדַּי הִבְהִילָנִי` (איוב כג:טז)
`בִּרְכַּת אֹבֵד עָלַי תָּבֹא וְלֵב אַלְמָנָה אַרְנִן` (איוב כט:יג)
`אִם תִּטֶּה אַשֻּׁרִי מִנִּי הַדָּרֶךְ וְאַחַר עֵינַי הָלַךְ לִבִּי וּבְכַפַּי דָּבַק מֻאוּם` (איוב לא:ז)
`אִם נִפְתָּה לִבִּי עַל אִשָּׁה וְעַל פֶּתַח רֵעִי אָרָבְתִּי` (איוב לא:ט)
`וַיִּפְתְּ בַּסֵּתֶר לִבִּי וַתִּשַּׁק יָדִי לְפִי` (איוב לא:כז)
`יֹשֶׁר לִבִּי אֲמָרָי וְדַעַת שְׂפָתַי בָּרוּר מִלֵּלוּ` (איוב לג:ג)
`אִם יָשִׂים אֵלָיו לִבּוֹ רוּחוֹ וְנִשְׁמָתוֹ אֵלָיו יֶאֱסֹף` (איוב לד:יד)
`הֶן אֵל כַּבִּיר וְלֹא יִמְאָס כַּבִּיר כֹּחַ לֵב` (איוב לו:ה)
`וְחַנְפֵי לֵב יָשִׂימוּ אָף לֹא יְשַׁוְּעוּ כִּי אֲסָרָם` (איוב לו:יג)
`אַף לְזֹאת יֶחֱרַד לִבִּי וְיִתַּר מִמְּקוֹמוֹ` (איוב לז:א)
`לָכֵן יְרֵאוּהוּ אֲנָשִׁים לֹא יִרְאֶה כָּל חַכְמֵי לֵב` (איוב לז:כד)
`לִבּוֹ יָצוּק כְּמוֹ אָבֶן וְיָצוּק כְּפֶלַח תַּחְתִּית` (איוב מא:כד)
`לְהַקְשִׁיב לַחָכְמָה אָזְנֶךָ תַּטֶּה לִבְּךָ לַתְּבוּנָה` (משלי ב:ב)
`כִּי תָבוֹא חָכְמָה בְלִבֶּךָ וְדַעַת לְנַפְשְׁךָ יִנְעָם` (משלי ב:י)
`בְּנִי תּוֹרָתִי אַל תִּשְׁכָּח וּמִצְוֺתַי יִצֹּר לִבֶּךָ` (משלי ג:א)
`חֶסֶד וֶאֱמֶת אַל יַעַזְבֻךָ קָשְׁרֵם עַל גַּרְגְּרוֹתֶיךָ כָּתְבֵם עַל לוּחַ לִבֶּךָ` (משלי ג:ג)
`בְּטַח אֶל יְהוָה בְּכָל לִבֶּךָ וְאֶל בִּינָתְךָ אַל תִּשָּׁעֵן` (משלי ג:ה)
`וַיֹּרֵנִי וַיֹּאמֶר לִי יִתְמָךְ דְּבָרַי לִבֶּךָ שְׁמֹר מִצְוֺתַי וֶחְיֵה` (משלי ד:ד)
`מִכָּל מִשְׁמָר נְצֹר לִבֶּךָ כִּי מִמֶּנּוּ תּוֹצְאוֹת חַיִּים` (משלי ד:כג)
`וְאָמַרְתָּ אֵיךְ שָׂנֵאתִי מוּסָר וְתוֹכַחַת נָאַץ לִבִּי` (משלי ה:יב)
`תַּהְפֻּכוֹת בְּלִבּוֹ חֹרֵשׁ רָע בְּכָל עֵת מְדָנִים יְשַׁלֵּחַ` (משלי ו:יד)
`לֵב חֹרֵשׁ מַחְשְׁבוֹת אָוֶן רַגְלַיִם מְמַהֲרוֹת לָרוּץ לָרָעָה` (משלי ו:יח)
`קָשְׁרֵם עַל לִבְּךָ תָמִיד עָנְדֵם עַל גַּרְגְּרֹתֶךָ` (משלי ו:כא)
`נֹאֵף אִשָּׁה חֲסַר לֵב מַשְׁחִית נַפְשׁוֹ הוּא יַעֲשֶׂנָּה` (משלי ו:לב)
`קָשְׁרֵם עַל אֶצְבְּעֹתֶיךָ כָּתְבֵם עַל לוּחַ לִבֶּךָ` (משלי ז:ג)
`וָאֵרֶא בַפְּתָאיִם אָבִינָה בַבָּנִים נַעַר חֲסַר לֵב` (משלי ז:ז)
`וְהִנֵּה אִשָּׁה לִקְרָאתוֹ שִׁית זוֹנָה וּנְצֻרַת לֵב` (משלי ז:י)
`אַל יֵשְׂטְ אֶל דְּרָכֶיהָ לִבֶּךָ אַל תֵּתַע בִּנְתִיבוֹתֶיהָ` (משלי ז:כה)
`הָבִינוּ פְתָאיִם עָרְמָה וּכְסִילִים הָבִינוּ לֵב` (משלי ח:ה)
`מִי פֶתִי יָסֻר הֵנָּה חֲסַר לֵב אָמְרָה לּוֹ` (משלי ט:ד)
`מִי פֶתִי יָסֻר הֵנָּה וַחֲסַר לֵב וְאָמְרָה לּוֹ` (משלי ט:טז)
`חֲכַם לֵב יִקַּח מִצְוֺת וֶאֱוִיל שְׂפָתַיִם יִלָּבֵט` (משלי י:ח)
`בְּשִׂפְתֵי נָבוֹן תִּמָּצֵא חָכְמָה וְשֵׁבֶט לְגֵו חֲסַר לֵב` (משלי י:יג)
`כֶּסֶף נִבְחָר לְשׁוֹן צַדִּיק לֵב רְשָׁעִים כִּמְעָט` (משלי י:כ)
`שִׂפְתֵי צַדִּיק יִרְעוּ רַבִּים וֶאֱוִילִים בַּחֲסַר לֵב יָמוּתוּ` (משלי י:כא)
`בָּז לְרֵעֵהוּ חֲסַר לֵב וְאִישׁ תְּבוּנוֹת יַחֲרִישׁ` (משלי יא:יב)
`תּוֹעֲבַת יְהוָה עִקְּשֵׁי לֵב וּרְצוֹנוֹ תְּמִימֵי דָרֶךְ` (משלי יא:כ)
`עוֹכֵר בֵּיתוֹ יִנְחַל רוּחַ וְעֶבֶד אֱוִיל לַחֲכַם לֵב` (משלי יא:כט)
`לְפִי שִׂכְלוֹ יְהֻלַּל אִישׁ וְנַעֲוֵה לֵב יִהְיֶה לָבוּז` (משלי יב:ח)
`עֹבֵד אַדְמָתוֹ יִשְׂבַּע לָחֶם וּמְרַדֵּף רֵיקִים חֲסַר לֵב` (משלי יב:יא)
`מִרְמָה בְּלֶב חֹרְשֵׁי רָע וּלְיֹעֲצֵי שָׁלוֹם שִׂמְחָה` (משלי יב:כ)
`אָדָם עָרוּם כֹּסֶה דָּעַת וְלֵב כְּסִילִים יִקְרָא אִוֶּלֶת` (משלי יב:כג)
`דְּאָגָה בְלֶב אִישׁ יַשְׁחֶנָּה וְדָבָר טוֹב יְשַׂמְּחֶנָּה` (משלי יב:כה)
`תּוֹחֶלֶת מְמֻשָּׁכָה מַחֲלָה לֵב וְעֵץ חַיִּים תַּאֲוָה בָאָה` (משלי יג:יב)
`לֵב יוֹדֵעַ מָרַּת נַפְשׁוֹ וּבְשִׂמְחָתוֹ לֹא יִתְעָרַב זָר` (משלי יד:י)
`גַּם בִּשְׂחוֹק יִכְאַב לֵב וְאַחֲרִיתָהּ שִׂמְחָה תוּגָה` (משלי יד:יג)
`מִדְּרָכָיו יִשְׂבַּע סוּג לֵב וּמֵעָלָיו אִישׁ טוֹב` (משלי יד:יד)
`חַיֵּי בְשָׂרִים לֵב מַרְפֵּא וּרְקַב עֲצָמוֹת קִנְאָה` (משלי יד:ל)
`בְּלֵב נָבוֹן תָּנוּחַ חָכְמָה וּבְקֶרֶב כְּסִילִים תִּוָּדֵעַ` (משלי יד:לג)
`שִׂפְתֵי חֲכָמִים יְזָרוּ דָעַת וְלֵב כְּסִילִים לֹא כֵן` (משלי טו:ז)
`לֵב שָׂמֵחַ יֵיטִב פָּנִים וּבְעַצְּבַת לֵב רוּחַ נְכֵאָה` (משלי טו:יג)
`לֵב נָבוֹן יְבַקֶּשׁ דָּעַת וּפְנֵי כְסִילִים יִרְעֶה אִוֶּלֶת` (משלי טו:יד)
`כָּל יְמֵי עָנִי רָעִים וְטוֹב לֵב מִשְׁתֶּה תָמִיד` (משלי טו:טו)
`אִוֶּלֶת שִׂמְחָה לַחֲסַר לֵב וְאִישׁ תְּבוּנָה יְיַשֶׁר לָכֶת` (משלי טו:כא)
`לֵב צַדִּיק יֶהְגֶּה לַעֲנוֹת וּפִי רְשָׁעִים יַבִּיעַ רָעוֹת` (משלי טו:כח)
`מְאוֹר עֵינַיִם יְשַׂמַּח לֵב שְׁמוּעָה טוֹבָה תְּדַשֶּׁן עָצֶם` (משלי טו:ל)
`פּוֹרֵעַ מוּסָר מוֹאֵס נַפְשׁוֹ וְשׁוֹמֵעַ תּוֹכַחַת קוֹנֶה לֵּב` (משלי טו:לב)
`לְאָדָם מַעַרְכֵי לֵב וּמֵיְהוָה מַעֲנֵה לָשׁוֹן` (משלי טז:א)
`תּוֹעֲבַת יְהוָה כָּל גְּבַהּ לֵב יָד לְיָד לֹא יִנָּקֶה` (משלי טז:ה)
`לֵב אָדָם יְחַשֵּׁב דַּרְכּוֹ וַיהוָה יָכִין צַעֲדוֹ` (משלי טז:ט)
`לַחֲכַם לֵב יִקָּרֵא נָבוֹן וּמֶתֶק שְׂפָתַיִם יֹסִיף לֶקַח` (משלי טז:כא)
`לֵב חָכָם יַשְׂכִּיל פִּיהוּ וְעַל שְׂפָתָיו יֹסִיף לֶקַח` (משלי טז:כג)
`לָמָּה זֶּה מְחִיר בְּיַד כְּסִיל לִקְנוֹת חָכְמָה וְלֶב אָיִן` (משלי יז:טז)
`אָדָם חֲסַר לֵב תּוֹקֵעַ כָּף עֹרֵב עֲרֻבָּה לִפְנֵי רֵעֵהוּ` (משלי יז:יח)
`עִקֶּשׁ לֵב לֹא יִמְצָא טוֹב וְנֶהְפָּךְ בִּלְשׁוֹנוֹ יִפּוֹל בְּרָעָה` (משלי יז:כ)
`לֵב שָׂמֵחַ יֵיטִב גֵּהָה וְרוּחַ נְכֵאָה תְּיַבֶּשׁ גָּרֶם` (משלי יז:כב)
`לֹא יַחְפֹּץ כְּסִיל בִּתְבוּנָה כִּי אִם בְּהִתְגַּלּוֹת לִבּוֹ` (משלי יח:ב)
`לִפְנֵי שֶׁבֶר יִגְבַּהּ לֵב אִישׁ וְלִפְנֵי כָבוֹד עֲנָוָה` (משלי יח:יב)
`לֵב נָבוֹן יִקְנֶה דָּעַת וְאֹזֶן חֲכָמִים תְּבַקֶּשׁ דָּעַת` (משלי יח:טו)
`אִוֶּלֶת אָדָם תְּסַלֵּף דַּרְכּוֹ וְעַל יְהוָה יִזְעַף לִבּוֹ` (משלי יט:ג)
`קֹנֶה לֵּב אֹהֵב נַפְשׁוֹ שֹׁמֵר תְּבוּנָה לִמְצֹא טוֹב` (משלי יט:ח)
`רַבּוֹת מַחֲשָׁבוֹת בְּלֶב אִישׁ וַעֲצַת יְהוָה הִיא תָקוּם` (משלי יט:כא)
`מַיִם עֲמֻקִּים עֵצָה בְלֶב אִישׁ וְאִישׁ תְּבוּנָה יִדְלֶנָּה` (משלי כ:ה)
`מִי יֹאמַר זִכִּיתִי לִבִּי טָהַרְתִּי מֵחַטָּאתִי` (משלי כ:ט)
`פַּלְגֵי מַיִם לֶב מֶלֶךְ בְּיַד יְהוָה עַל כָּל אֲשֶׁר יַחְפֹּץ יַטֶּנּוּ` (משלי כא:א)
`רוּם עֵינַיִם וּרְחַב לֵב נִר רְשָׁעִים חַטָּאת` (משלי כא:ד)
`אֹהֵב טְהוֹר לֵב חֵן שְׂפָתָיו רֵעֵהוּ מֶלֶךְ` (משלי כב:יא)
`אִוֶּלֶת קְשׁוּרָה בְלֶב נָעַר שֵׁבֶט מוּסָר יַרְחִיקֶנָּה מִמֶּנּוּ` (משלי כב:טו)
`הַט אָזְנְךָ וּשְׁמַע דִּבְרֵי חֲכָמִים וְלִבְּךָ תָּשִׁית לְדַעְתִּי` (משלי כב:יז)
`כִּי כְּמוֹ שָׁעַר בְּנַפְשׁוֹ כֶּן הוּא אֱכֹל וּשְׁתֵה יֹאמַר לָךְ וְלִבּוֹ בַּל עִמָּךְ` (משלי כג:ז)
`הָבִיאָה לַמּוּסָר לִבֶּךָ וְאָזְנֶךָ לְאִמְרֵי דָעַת` (משלי כג:יב)
`בְּנִי אִם חָכַם לִבֶּךָ יִשְׂמַח לִבִּי גַם אָנִי` (משלי כג:טו)
`אַל יְקַנֵּא לִבְּךָ בַּחַטָּאִים כִּי אִם בְּיִרְאַת יְהוָה כָּל הַיּוֹם` (משלי כג:יז)
`שְׁמַע אַתָּה בְנִי וַחֲכָם וְאַשֵּׁר בַּדֶּרֶךְ לִבֶּךָ` (משלי כג:יט)
`תְּנָה בְנִי לִבְּךָ לִי וְעֵינֶיךָ דְּרָכַי תִּרְצֶנָה` (משלי כג:כו)
`עֵינֶיךָ יִרְאוּ זָרוֹת וְלִבְּךָ יְדַבֵּר תַּהְפֻּכוֹת` (משלי כג:לג)
`וְהָיִיתָ כְּשֹׁכֵב בְּלֶב יָם וּכְשֹׁכֵב בְּרֹאשׁ חִבֵּל` (משלי כג:לד)
`כִּי שֹׁד יֶהְגֶּה לִבָּם וְעָמָל שִׂפְתֵיהֶם תְּדַבֵּרְנָה` (משלי כד:ב)
`בִּנְפֹל אוֹיְבֶיךָ אַל תִּשְׂמָח וּבִכָּשְׁלוֹ אַל יָגֵל לִבֶּךָ` (משלי כד:יז)
`עַל שְׂדֵה אִישׁ עָצֵל עָבַרְתִּי וְעַל כֶּרֶם אָדָם חֲסַר לֵב` (משלי כד:ל)
`וָאֶחֱזֶה אָנֹכִי אָשִׁית לִבִּי רָאִיתִי לָקַחְתִּי מוּסָר` (משלי כד:לב)
`שָׁמַיִם לָרוּם וָאָרֶץ לָעֹמֶק וְלֵב מְלָכִים אֵין חֵקֶר` (משלי כה:ג)
`מַעֲדֶה בֶּגֶד בְּיוֹם קָרָה חֹמֶץ עַל נָתֶר וְשָׁר בַּשִּׁרִים עַל לֶב רָע` (משלי כה:כ)
`כֶּסֶף סִיגִים מְצֻפֶּה עַל חָרֶשׂ שְׂפָתַיִם דֹּלְקִים וְלֶב רָע` (משלי כו:כג)
`כִּי יְחַנֵּן קוֹלוֹ אַל תַּאֲמֶן בּוֹ כִּי שֶׁבַע תּוֹעֵבוֹת בְּלִבּוֹ` (משלי כו:כה)
`שֶׁמֶן וּקְטֹרֶת יְשַׂמַּח לֵב וּמֶתֶק רֵעֵהוּ מֵעֲצַת נָפֶשׁ` (משלי כז:ט)
`חֲכַם בְּנִי וְשַׂמַּח לִבִּי וְאָשִׁיבָה חֹרְפִי דָבָר` (משלי כז:יא)
`כַּמַּיִם הַפָּנִים לַפָּנִים כֵּן לֵב הָאָדָם לָאָדָם` (משלי כז:יט)
`יָדֹעַ תֵּדַע פְּנֵי צֹאנֶךָ שִׁית לִבְּךָ לַעֲדָרִים` (משלי כז:כג)
`אַשְׁרֵי אָדָם מְפַחֵד תָּמִיד וּמַקְשֶׁה לִבּוֹ יִפּוֹל בְּרָעָה` (משלי כח:יד)
`בּוֹטֵחַ בְּלִבּוֹ הוּא כְסִיל וְהוֹלֵךְ בְּחָכְמָה הוּא יִמָּלֵט` (משלי כח:כו)
`דֶּרֶךְ הַנֶּשֶׁר בַּשָּׁמַיִם דֶּרֶךְ נָחָשׁ עֲלֵי צוּר דֶּרֶךְ אֳנִיָּה בְלֶב יָם וְדֶרֶךְ גֶּבֶר בְּעַלְמָה` (משלי ל:יט)
`בָּטַח בָּהּ לֵב בַּעְלָהּ וְשָׁלָל לֹא יֶחְסָר` (משלי לא:יא)
`וַתֹּאמֶר אֶמְצָא חֵן בְּעֵינֶיךָ אֲדֹנִי כִּי נִחַמְתָּנִי וְכִי דִבַּרְתָּ עַל לֵב שִׁפְחָתֶךָ וְאָנֹכִי לֹא אֶהְיֶה כְּאַחַת שִׁפְחֹתֶיךָ` (רות ב:יג)
`וַיֹּאכַל בֹּעַז וַיֵּשְׁתְּ וַיִּיטַב לִבּוֹ וַיָּבֹא לִשְׁכַּב בִּקְצֵה הָעֲרֵמָה וַתָּבֹא בַלָּט וַתְּגַל מַרְגְּלֹתָיו וַתִּשְׁכָּב` (רות ג:ז)
`צְאֶינָה וּרְאֶינָה בְּנוֹת צִיּוֹן בַּמֶּלֶךְ שְׁלֹמֹה בָּעֲטָרָה שֶׁעִטְּרָה לּוֹ אִמּוֹ בְּיוֹם חֲתֻנָּתוֹ וּבְיוֹם שִׂמְחַת לִבּוֹ` (שיר-השירים ג:יא)
`אֲנִי יְשֵׁנָה וְלִבִּי עֵר קוֹל דּוֹדִי דוֹפֵק פִּתְחִי לִי אֲחֹתִי רַעְיָתִי יוֹנָתִי תַמָּתִי שֶׁרֹּאשִׁי נִמְלָא טָל קְוֻּצּוֹתַי רְסִיסֵי לָיְלָה` (שיר-השירים ה:ב)
`שִׂימֵנִי כַחוֹתָם עַל לִבֶּךָ כַּחוֹתָם עַל זְרוֹעֶךָ כִּי עַזָּה כַמָּוֶת אַהֲבָה קָשָׁה כִשְׁאוֹל קִנְאָה רְשָׁפֶיהָ רִשְׁפֵּי אֵשׁ שַׁלְהֶבֶתְיָה` (שיר-השירים ח:ו)
`וְנָתַתִּי אֶת לִבִּי לִדְרוֹשׁ וְלָתוּר בַּחָכְמָה עַל כָּל אֲשֶׁר נַעֲשָׂה תַּחַת הַשָּׁמָיִם הוּא עִנְיַן רָע נָתַן אֱלֹהִים לִבְנֵי הָאָדָם לַעֲנוֹת בּוֹ` (קהלת א:יג)
`דִּבַּרְתִּי אֲנִי עִם לִבִּי לֵאמֹר אֲנִי הִנֵּה הִגְדַּלְתִּי וְהוֹסַפְתִּי חָכְמָה עַל כָּל אֲשֶׁר הָיָה לְפָנַי עַל יְרוּשָׁלִָם וְלִבִּי רָאָה הַרְבֵּה חָכְמָה וָדָעַת` (קהלת א:טז)
`וָאֶתְּנָה לִבִּי לָדַעַת חָכְמָה וְדַעַת הוֹלֵלוֹת וְשִׂכְלוּת יָדַעְתִּי שֶׁגַּם זֶה הוּא רַעְיוֹן רוּחַ` (קהלת א:יז)
`אָמַרְתִּי אֲנִי בְּלִבִּי לְכָה נָּא אֲנַסְּכָה בְשִׂמְחָה וּרְאֵה בְטוֹב וְהִנֵּה גַם הוּא הָבֶל` (קהלת ב:א)
`תַּרְתִּי בְלִבִּי לִמְשׁוֹךְ בַּיַּיִן אֶת בְּשָׂרִי וְלִבִּי נֹהֵג בַּחָכְמָה וְלֶאֱחֹז בְּסִכְלוּת עַד אֲשֶׁר אֶרְאֶה אֵי זֶה טוֹב לִבְנֵי הָאָדָם אֲשֶׁר יַעֲשׂוּ תַּחַת הַשָּׁמַיִם מִסְפַּר יְמֵי חַיֵּיהֶם` (קהלת ב:ג)
`וְכֹל אֲשֶׁר שָׁאֲלוּ עֵינַי לֹא אָצַלְתִּי מֵהֶם לֹא מָנַעְתִּי אֶת לִבִּי מִכָּל שִׂמְחָה כִּי לִבִּי שָׂמֵחַ מִכָּל עֲמָלִי וְזֶה הָיָה חֶלְקִי מִכָּל עֲמָלִי` (קהלת ב:י)
`וְאָמַרְתִּי אֲנִי בְּלִבִּי כְּמִקְרֵה הַכְּסִיל גַּם אֲנִי יִקְרֵנִי וְלָמָּה חָכַמְתִּי אֲנִי אָז יוֹתֵר וְדִבַּרְתִּי בְלִבִּי שֶׁגַּם זֶה הָבֶל` (קהלת ב:טו)
`וְסַבּוֹתִי אֲנִי לְיַאֵשׁ אֶת לִבִּי עַל כָּל הֶעָמָל שֶׁעָמַלְתִּי תַּחַת הַשָּׁמֶשׁ` (קהלת ב:כ)
`כִּי מֶה הֹוֶה לָאָדָם בְּכָל עֲמָלוֹ וּבְרַעְיוֹן לִבּוֹ שֶׁהוּא עָמֵל תַּחַת הַשָּׁמֶשׁ` (קהלת ב:כב)
`כִּי כָל יָמָיו מַכְאֹבִים וָכַעַס עִנְיָנוֹ גַּם בַּלַּיְלָה לֹא שָׁכַב לִבּוֹ גַּם זֶה הֶבֶל הוּא` (קהלת ב:כג)
`אֶת הַכֹּל עָשָׂה יָפֶה בְעִתּוֹ גַּם אֶת הָעֹלָם נָתַן בְּלִבָּם מִבְּלִי אֲשֶׁר לֹא יִמְצָא הָאָדָם אֶת הַמַּעֲשֶׂה אֲשֶׁר עָשָׂה הָאֱלֹהִים מֵרֹאשׁ וְעַד סוֹף` (קהלת ג:יא)
`אָמַרְתִּי אֲנִי בְּלִבִּי אֶת הַצַּדִּיק וְאֶת הָרָשָׁע יִשְׁפֹּט הָאֱלֹהִים כִּי עֵת לְכָל חֵפֶץ וְעַל כָּל הַמַּעֲשֶׂה שָׁם` (קהלת ג:יז)
`אָמַרְתִּי אֲנִי בְּלִבִּי עַל דִּבְרַת בְּנֵי הָאָדָם לְבָרָם הָאֱלֹהִים וְלִרְאוֹת שְׁהֶם בְּהֵמָה הֵמָּה לָהֶם` (קהלת ג:יח)
`אַל תְּבַהֵל עַל פִּיךָ וְלִבְּךָ אַל יְמַהֵר לְהוֹצִיא דָבָר לִפְנֵי הָאֱלֹהִים כִּי הָאֱלֹהִים בַּשָּׁמַיִם וְאַתָּה עַל הָאָרֶץ עַל כֵּן יִהְיוּ דְבָרֶיךָ מְעַטִּים` (קהלת ה:ב)
`כִּי לֹא הַרְבֵּה יִזְכֹּר אֶת יְמֵי חַיָּיו כִּי הָאֱלֹהִים מַעֲנֶה בְּשִׂמְחַת לִבּוֹ` (קהלת ה:כ)
`טוֹב לָלֶכֶת אֶל בֵּית אֵבֶל מִלֶּכֶת אֶל בֵּית מִשְׁתֶּה בַּאֲשֶׁר הוּא סוֹף כָּל הָאָדָם וְהַחַי יִתֵּן אֶל לִבּוֹ` (קהלת ז:ב)
`טוֹב כַּעַס מִשְּׂחֹק כִּי בְרֹעַ פָּנִים יִיטַב לֵב` (קהלת ז:ג)
`לֵב חֲכָמִים בְּבֵית אֵבֶל וְלֵב כְּסִילִים בְּבֵית שִׂמְחָה` (קהלת ז:ד)
`כִּי הָעֹשֶׁק יְהוֹלֵל חָכָם וִיאַבֵּד אֶת לֵב מַתָּנָה` (קהלת ז:ז)
`גַּם לְכָל הַדְּבָרִים אֲשֶׁר יְדַבֵּרוּ אַל תִּתֵּן לִבֶּךָ אֲשֶׁר לֹא תִשְׁמַע אֶת עַבְדְּךָ מְקַלְלֶךָ` (קהלת ז:כא)
`כִּי גַּם פְּעָמִים רַבּוֹת יָדַע לִבֶּךָ אֲשֶׁר גַּם אַתְּ קִלַּלְתָּ אֲחֵרִים` (קהלת ז:כב)
`סַבּוֹתִי אֲנִי וְלִבִּי לָדַעַת וְלָתוּר וּבַקֵּשׁ חָכְמָה וְחֶשְׁבּוֹן וְלָדַעַת רֶשַׁע כֶּסֶל וְהַסִּכְלוּת הוֹלֵלוֹת` (קהלת ז:כה)
`וּמוֹצֶא אֲנִי מַר מִמָּוֶת אֶת הָאִשָּׁה אֲשֶׁר הִיא מְצוֹדִים וַחֲרָמִים לִבָּהּ אֲסוּרִים יָדֶיהָ טוֹב לִפְנֵי הָאֱלֹהִים יִמָּלֵט מִמֶּנָּה וְחוֹטֵא יִלָּכֶד בָּהּ` (קהלת ז:כו)
`שׁוֹמֵר מִצְוָה לֹא יֵדַע דָּבָר רָע וְעֵת וּמִשְׁפָּט יֵדַע לֵב חָכָם` (קהלת ח:ה)
`אֶת כָּל זֶה רָאִיתִי וְנָתוֹן אֶת לִבִּי לְכָל מַעֲשֶׂה אֲשֶׁר נַעֲשָׂה תַּחַת הַשָּׁמֶשׁ עֵת אֲשֶׁר שָׁלַט הָאָדָם בְּאָדָם לְרַע לוֹ` (קהלת ח:ט)
`אֲשֶׁר אֵין נַעֲשָׂה פִתְגָם מַעֲשֵׂה הָרָעָה מְהֵרָה עַל כֵּן מָלֵא לֵב בְּנֵי הָאָדָם בָּהֶם לַעֲשׂוֹת רָע` (קהלת ח:יא)
`כַּאֲשֶׁר נָתַתִּי אֶת לִבִּי לָדַעַת חָכְמָה וְלִרְאוֹת אֶת הָעִנְיָן אֲשֶׁר נַעֲשָׂה עַל הָאָרֶץ כִּי גַם בַּיּוֹם וּבַלַּיְלָה שֵׁנָה בְּעֵינָיו אֵינֶנּוּ רֹאֶה` (קהלת ח:טז)
`כִּי אֶת כָּל זֶה נָתַתִּי אֶל לִבִּי וְלָבוּר אֶת כָּל זֶה אֲשֶׁר הַצַּדִּיקִים וְהַחֲכָמִים וַעֲבָדֵיהֶם בְּיַד הָאֱלֹהִים גַּם אַהֲבָה גַם שִׂנְאָה אֵין יוֹדֵעַ הָאָדָם הַכֹּל לִפְנֵיהֶם` (קהלת ט:א)
`זֶה רָע בְּכֹל אֲשֶׁר נַעֲשָׂה תַּחַת הַשֶּׁמֶשׁ כִּי מִקְרֶה אֶחָד לַכֹּל וְגַם לֵב בְּנֵי הָאָדָם מָלֵא רָע וְהוֹלֵלוֹת בִּלְבָבָם בְּחַיֵּיהֶם וְאַחֲרָיו אֶל הַמֵּתִים` (קהלת ט:ג)
`לֵךְ אֱכֹל בְּשִׂמְחָה לַחְמֶךָ וּשֲׁתֵה בְלֶב טוֹב יֵינֶךָ כִּי כְבָר רָצָה הָאֱלֹהִים אֶת מַעֲשֶׂיךָ` (קהלת ט:ז)
`לֵב חָכָם לִימִינוֹ וְלֵב כְּסִיל לִשְׂמֹאלוֹ` (קהלת י:ב)
`וְגַם בַּדֶּרֶךְ כְּשֶׁהַסָּכָל הֹלֵךְ לִבּוֹ חָסֵר וְאָמַר לַכֹּל סָכָל הוּא` (קהלת י:ג)
`שְׂמַח בָּחוּר בְּיַלְדוּתֶיךָ וִיטִיבְךָ לִבְּךָ בִּימֵי בְחוּרוֹתֶךָ וְהַלֵּךְ בְּדַרְכֵי לִבְּךָ וּבְמַרְאֵי עֵינֶיךָ וְדָע כִּי עַל כָּל אֵלֶּה יְבִיאֲךָ הָאֱלֹהִים בַּמִּשְׁפָּט` (קהלת יא:ט)
`וְהָסֵר כַּעַס מִלִּבֶּךָ וְהַעֲבֵר רָעָה מִבְּשָׂרֶךָ כִּי הַיַּלְדוּת וְהַשַּׁחֲרוּת הָבֶל` (קהלת יא:י)
`רְאֵה יְהוָה כִּי צַר לִי מֵעַי חֳמַרְמָרוּ נֶהְפַּךְ לִבִּי בְּקִרְבִּי כִּי מָרוֹ מָרִיתִי מִחוּץ שִׁכְּלָה חֶרֶב בַּבַּיִת כַּמָּוֶת` (איכה א:כ)
`תָּבֹא כָל רָעָתָם לְפָנֶיךָ וְעוֹלֵל לָמוֹ כַּאֲשֶׁר עוֹלַלְתָּ לִי עַל כָּל פְּשָׁעָי כִּי רַבּוֹת אַנְחֹתַי וְלִבִּי דַוָּי` (איכה א:כב)
`צָעַק לִבָּם אֶל אֲדֹנָי חוֹמַת בַּת צִיּוֹן הוֹרִידִי כַנַּחַל דִּמְעָה יוֹמָם וָלַיְלָה אַל תִּתְּנִי פוּגַת לָךְ אַל תִּדֹּם בַּת עֵינֵךְ` (איכה ב:יח)
`קוּמִי רֹנִּי בַלַּיִל לְרֹאשׁ אַשְׁמֻרוֹת שִׁפְכִי כַמַּיִם לִבֵּךְ נֹכַח פְּנֵי אֲדֹנָי שְׂאִי אֵלָיו כַּפַּיִךְ עַל נֶפֶשׁ עוֹלָלַיִךְ הָעֲטוּפִים בְּרָעָב בְּרֹאשׁ כָּל חוּצוֹת` (איכה ב:יט)
`זֹאת אָשִׁיב אֶל לִבִּי עַל כֵּן אוֹחִיל` (איכה ג:כא)
`כִּי לֹא עִנָּה מִלִּבּוֹ וַיַּגֶּה בְנֵי אִישׁ` (איכה ג:לג)
`תִּתֵּן לָהֶם מְגִנַּת לֵב תַּאֲלָתְךָ לָהֶם` (איכה ג:סה)
`שָׁבַת מְשׂוֹשׂ לִבֵּנוּ נֶהְפַּךְ לְאֵבֶל מְחֹלֵנוּ` (איכה ה:טו)
`עַל זֶה הָיָה דָוֶה לִבֵּנוּ עַל אֵלֶּה חָשְׁכוּ עֵינֵינוּ` (איכה ה:יז)
`בַּיּוֹם הַשְּׁבִיעִי כְּטוֹב לֵב הַמֶּלֶךְ בַּיָּיִן אָמַר לִמְהוּמָן בִּזְּתָא חַרְבוֹנָא בִּגְתָא וַאֲבַגְתָא זֵתַר וְכַרְכַּס שִׁבְעַת הַסָּרִיסִים הַמְשָׁרְתִים אֶת פְּנֵי הַמֶּלֶךְ אֲחַשְׁוֵרוֹשׁ` (אסתר א:י)
`וַיֵּצֵא הָמָן בַּיּוֹם הַהוּא שָׂמֵחַ וְטוֹב לֵב וְכִרְאוֹת הָמָן אֶת מָרְדֳּכַי בְּשַׁעַר הַמֶּלֶךְ וְלֹא קָם וְלֹא זָע מִמֶּנּוּ וַיִּמָּלֵא הָמָן עַל מָרְדֳּכַי חֵמָה` (אסתר ה:ט)
`וַיָּבוֹא הָמָן וַיֹּאמֶר לוֹ הַמֶּלֶךְ מַה לַעֲשׂוֹת בָּאִישׁ אֲשֶׁר הַמֶּלֶךְ חָפֵץ בִּיקָרוֹ וַיֹּאמֶר הָמָן בְּלִבּוֹ לְמִי יַחְפֹּץ הַמֶּלֶךְ לַעֲשׂוֹת יְקָר יוֹתֵר מִמֶּנִּי` (אסתר ו:ו)
`וַיֹּאמֶר הַמֶּלֶךְ אֲחַשְׁוֵרוֹשׁ וַיֹּאמֶר לְאֶסְתֵּר הַמַּלְכָּה מִי הוּא זֶה וְאֵי זֶה הוּא אֲשֶׁר מְלָאוֹ לִבּוֹ לַעֲשׂוֹת כֵּן` (אסתר ז:ה)
`וַיָּשֶׂם דָּנִיֵּאל עַל לִבּוֹ אֲשֶׁר לֹא יִתְגָּאַל בְּפַתְבַּג הַמֶּלֶךְ וּבְיֵין מִשְׁתָּיו וַיְבַקֵּשׁ מִשַּׂר הַסָּרִיסִים אֲשֶׁר לֹא יִתְגָּאָל` (דניאל א:ח)
`וַיֹּאמֶר אֵלַי אַל תִּירָא דָנִיֵּאל כִּי מִן הַיּוֹם הָרִאשׁוֹן אֲשֶׁר נָתַתָּ אֶת לִבְּךָ לְהָבִין וּלְהִתְעַנּוֹת לִפְנֵי אֱלֹהֶיךָ נִשְׁמְעוּ דְבָרֶיךָ וַאֲנִי בָאתִי בִּדְבָרֶיךָ` (דניאל י:יב)
`וַיַּעֲשׂוּ חַג מַצּוֹת שִׁבְעַת יָמִים בְּשִׂמְחָה כִּי שִׂמְּחָם יְהוָה וְהֵסֵב לֵב מֶלֶךְ אַשּׁוּר עֲלֵיהֶם לְחַזֵּק יְדֵיהֶם בִּמְלֶאכֶת בֵּית הָאֱלֹהִים אֱלֹהֵי יִשְׂרָאֵל` (עזרא ו:כב)
`בָּרוּךְ יְהוָה אֱלֹהֵי אֲבוֹתֵינוּ אֲשֶׁר נָתַן כָּזֹאת בְּלֵב הַמֶּלֶךְ לְפָאֵר אֶת בֵּית יְהוָה אֲשֶׁר בִּירוּשָׁלִָם` (עזרא ז:כז)
`וַיֹּאמֶר לִי הַמֶּלֶךְ מַדּוּעַ פָּנֶיךָ רָעִים וְאַתָּה אֵינְךָ חוֹלֶה אֵין זֶה כִּי אִם רֹעַ לֵב וָאִירָא הַרְבֵּה מְאֹד` (נחמיה ב:ב)
`וָאָקוּם לַיְלָה אֲנִי וַאֲנָשִׁים מְעַט עִמִּי וְלֹא הִגַּדְתִּי לְאָדָם מָה אֱלֹהַי נֹתֵן אֶל לִבִּי לַעֲשׂוֹת לִירוּשָׁלִָם וּבְהֵמָה אֵין עִמִּי כִּי אִם הַבְּהֵמָה אֲשֶׁר אֲנִי רֹכֵב בָּהּ` (נחמיה ב:יב)
`וַנִּבְנֶה אֶת הַחוֹמָה וַתִּקָּשֵׁר כָּל הַחוֹמָה עַד חֶצְיָהּ וַיְהִי לֵב לָעָם לַעֲשׂוֹת` (נחמיה ד:ו)
`וַיִּמָּלֵךְ לִבִּי עָלַי וָאָרִיבָה אֶת הַחֹרִים וְאֶת הַסְּגָנִים וָאֹמְרָה לָהֶם מַשָּׁא אִישׁ בְּאָחִיו אַתֶּם נֹשִׁאים וָאֶתֵּן עֲלֵיהֶם קְהִלָּה גְדוֹלָה` (נחמיה ה:ז)
`וָאֶשְׁלְחָה אֵלָיו לֵאמֹר לֹא נִהְיָה כַּדְּבָרִים הָאֵלֶּה אֲשֶׁר אַתָּה אוֹמֵר כִּי מִלִּבְּךָ אַתָּה בוֹדָאם` (נחמיה ו:ח)
`וַיִּתֵּן אֱלֹהַי אֶל לִבִּי וָאֶקְבְּצָה אֶת הַחֹרִים וְאֶת הַסְּגָנִים וְאֶת הָעָם לְהִתְיַחֵשׂ וָאֶמְצָא סֵפֶר הַיַּחַשׂ הָעוֹלִים בָּרִאשׁוֹנָה וָאֶמְצָא כָּתוּב בּוֹ` (נחמיה ז:ה)
