`וְהָאָרֶץ הָיְתָה תֹהוּ וָבֹהוּ וְחֹשֶׁךְ עַל פְּנֵי תְהוֹם וְרוּחַ אֱלֹהִים מְרַחֶפֶת עַל פְּנֵי הַמָּיִם` (בראשית א:ב)
`וַיִּשְׁמְעוּ אֶת קוֹל יְהוָה אֱלֹהִים מִתְהַלֵּךְ בַּגָּן לְרוּחַ הַיּוֹם וַיִּתְחַבֵּא הָאָדָם וְאִשְׁתּוֹ מִפְּנֵי יְהוָה אֱלֹהִים בְּתוֹךְ עֵץ הַגָּן` (בראשית ג:ח)
`וַיֹּאמֶר יְהוָה לֹא יָדוֹן רוּחִי בָאָדָם לְעֹלָם בְּשַׁגַּם הוּא בָשָׂר וְהָיוּ יָמָיו מֵאָה וְעֶשְׂרִים שָׁנָה` (בראשית ו:ג)
`וַאֲנִי הִנְנִי מֵבִיא אֶת הַמַּבּוּל מַיִם עַל הָאָרֶץ לְשַׁחֵת כָּל בָּשָׂר אֲשֶׁר בּוֹ רוּחַ חַיִּים מִתַּחַת הַשָּׁמָיִם כֹּל אֲשֶׁר בָּאָרֶץ יִגְוָע` (בראשית ו:יז)
`וַיָּבֹאוּ אֶל נֹחַ אֶל הַתֵּבָה שְׁנַיִם שְׁנַיִם מִכָּל הַבָּשָׂר אֲשֶׁר בּוֹ רוּחַ חַיִּים` (בראשית ז:טו)
`כֹּל אֲשֶׁר נִשְׁמַת רוּחַ חַיִּים בְּאַפָּיו מִכֹּל אֲשֶׁר בֶּחָרָבָה מֵתוּ` (בראשית ז:כב)
`וַיִּזְכֹּר אֱלֹהִים אֶת נֹחַ וְאֵת כָּל הַחַיָּה וְאֶת כָּל הַבְּהֵמָה אֲשֶׁר אִתּוֹ בַּתֵּבָה וַיַּעֲבֵר אֱלֹהִים רוּחַ עַל הָאָרֶץ וַיָּשֹׁכּוּ הַמָּיִם` (בראשית ח:א)
`וַתִּהְיֶיןָ מֹרַת רוּחַ לְיִצְחָק וּלְרִבְקָה` (בראשית כו:לה)
`וַיְהִי בַבֹּקֶר וַתִּפָּעֶם רוּחוֹ וַיִּשְׁלַח וַיִּקְרָא אֶת כָּל חַרְטֻמֵּי מִצְרַיִם וְאֶת כָּל חֲכָמֶיהָ וַיְסַפֵּר פַּרְעֹה לָהֶם אֶת חֲלֹמוֹ וְאֵין פּוֹתֵר אוֹתָם לְפַרְעֹה` (בראשית מא:ח)
`וַיֹּאמֶר פַּרְעֹה אֶל עֲבָדָיו הֲנִמְצָא כָזֶה אִישׁ אֲשֶׁר רוּחַ אֱלֹהִים בּוֹ` (בראשית מא:לח)
`וַיְדַבְּרוּ אֵלָיו אֵת כָּל דִּבְרֵי יוֹסֵף אֲשֶׁר דִּבֶּר אֲלֵהֶם וַיַּרְא אֶת הָעֲגָלוֹת אֲשֶׁר שָׁלַח יוֹסֵף לָשֵׂאת אֹתוֹ וַתְּחִי רוּחַ יַעֲקֹב אֲבִיהֶם` (בראשית מה:כז)
`וַיְדַבֵּר מֹשֶׁה כֵּן אֶל בְּנֵי יִשְׂרָאֵל וְלֹא שָׁמְעוּ אֶל מֹשֶׁה מִקֹּצֶר רוּחַ וּמֵעֲבֹדָה קָשָׁה` (שמות ו:ט)
`וַיֵּט מֹשֶׁה אֶת מַטֵּהוּ עַל אֶרֶץ מִצְרַיִם וַיהוָה נִהַג רוּחַ קָדִים בָּאָרֶץ כָּל הַיּוֹם הַהוּא וְכָל הַלָּיְלָה הַבֹּקֶר הָיָה וְרוּחַ הַקָּדִים נָשָׂא אֶת הָאַרְבֶּה` (שמות י:יג)
`וַיַּהֲפֹךְ יְהוָה רוּחַ יָם חָזָק מְאֹד וַיִּשָּׂא אֶת הָאַרְבֶּה וַיִּתְקָעֵהוּ יָמָּה סּוּף לֹא נִשְׁאַר אַרְבֶּה אֶחָד בְּכֹל גְּבוּל מִצְרָיִם` (שמות י:יט)
`וַיֵּט מֹשֶׁה אֶת יָדוֹ עַל הַיָּם וַיּוֹלֶךְ יְהוָה אֶת הַיָּם בְּרוּחַ קָדִים עַזָּה כָּל הַלַּיְלָה וַיָּשֶׂם אֶת הַיָּם לֶחָרָבָה וַיִּבָּקְעוּ הַמָּיִם` (שמות יד:כא)
`וּבְרוּחַ אַפֶּיךָ נֶעֶרְמוּ מַיִם נִצְּבוּ כְמוֹ נֵד נֹזְלִים קָפְאוּ תְהֹמֹת בְּלֶב יָם` (שמות טו:ח)
`נָשַׁפְתָּ בְרוּחֲךָ כִּסָּמוֹ יָם צָלֲלוּ כַּעוֹפֶרֶת בְּמַיִם אַדִּירִים` (שמות טו:י)
`וְאַתָּה תְּדַבֵּר אֶל כָּל חַכְמֵי לֵב אֲשֶׁר מִלֵּאתִיו רוּחַ חָכְמָה וְעָשׂוּ אֶת בִּגְדֵי אַהֲרֹן לְקַדְּשׁוֹ לְכַהֲנוֹ לִי` (שמות כח:ג)
`וָאֲמַלֵּא אֹתוֹ רוּחַ אֱלֹהִים בְּחָכְמָה וּבִתְבוּנָה וּבְדַעַת וּבְכָל מְלָאכָה` (שמות לא:ג)
`וַיָּבֹאוּ כָּל אִישׁ אֲשֶׁר נְשָׂאוֹ לִבּוֹ וְכֹל אֲשֶׁר נָדְבָה רוּחוֹ אֹתוֹ הֵבִיאוּ אֶת תְּרוּמַת יְהוָה לִמְלֶאכֶת אֹהֶל מוֹעֵד וּלְכָל עֲבֹדָתוֹ וּלְבִגְדֵי הַקֹּדֶשׁ` (שמות לה:כא)
`וַיְמַלֵּא אֹתוֹ רוּחַ אֱלֹהִים בְּחָכְמָה בִּתְבוּנָה וּבְדַעַת וּבְכָל מְלָאכָה` (שמות לה:לא)
`וְעָבַר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת אִשְׁתּוֹ וְהִוא נִטְמָאָה אוֹ עָבַר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת אִשְׁתּוֹ וְהִיא לֹא נִטְמָאָה` (במדבר ה:יד)
`אוֹ אִישׁ אֲשֶׁר תַּעֲבֹר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת אִשְׁתּוֹ וְהֶעֱמִיד אֶת הָאִשָּׁה לִפְנֵי יְהוָה וְעָשָׂה לָהּ הַכֹּהֵן אֵת כָּל הַתּוֹרָה הַזֹּאת` (במדבר ה:ל)
`וְיָרַדְתִּי וְדִבַּרְתִּי עִמְּךָ שָׁם וְאָצַלְתִּי מִן הָרוּחַ אֲשֶׁר עָלֶיךָ וְשַׂמְתִּי עֲלֵיהֶם וְנָשְׂאוּ אִתְּךָ בְּמַשָּׂא הָעָם וְלֹא תִשָּׂא אַתָּה לְבַדֶּךָ` (במדבר יא:יז)
`וַיֵּרֶד יְהוָה בֶּעָנָן וַיְדַבֵּר אֵלָיו וַיָּאצֶל מִן הָרוּחַ אֲשֶׁר עָלָיו וַיִּתֵּן עַל שִׁבְעִים אִישׁ הַזְּקֵנִים וַיְהִי כְּנוֹחַ עֲלֵיהֶם הָרוּחַ וַיִּתְנַבְּאוּ וְלֹא יָסָפוּ` (במדבר יא:כה)
`וַיִּשָּׁאֲרוּ שְׁנֵי אֲנָשִׁים בַּמַּחֲנֶה שֵׁם הָאֶחָד אֶלְדָּד וְשֵׁם הַשֵּׁנִי מֵידָד וַתָּנַח עֲלֵיהֶם הָרוּחַ וְהֵמָּה בַּכְּתֻבִים וְלֹא יָצְאוּ הָאֹהֱלָה וַיִּתְנַבְּאוּ בַּמַּחֲנֶה` (במדבר יא:כו)
`וַיֹּאמֶר לוֹ מֹשֶׁה הַמְקַנֵּא אַתָּה לִי וּמִי יִתֵּן כָּל עַם יְהוָה נְבִיאִים כִּי יִתֵּן יְהוָה אֶת רוּחוֹ עֲלֵיהֶם` (במדבר יא:כט)
`וְרוּחַ נָסַע מֵאֵת יְהוָה וַיָּגָז שַׂלְוִים מִן הַיָּם וַיִּטֹּשׁ עַל הַמַּחֲנֶה כְּדֶרֶךְ יוֹם כֹּה וּכְדֶרֶךְ יוֹם כֹּה סְבִיבוֹת הַמַּחֲנֶה וּכְאַמָּתַיִם עַל פְּנֵי הָאָרֶץ` (במדבר יא:לא)
`וְעַבְדִּי כָלֵב עֵקֶב הָיְתָה רוּחַ אַחֶרֶת עִמּוֹ וַיְמַלֵּא אַחֲרָי וַהֲבִיאֹתִיו אֶל הָאָרֶץ אֲשֶׁר בָּא שָׁמָּה וְזַרְעוֹ יוֹרִשֶׁנָּה` (במדבר יד:כד)
`וַיִּפְּלוּ עַל פְּנֵיהֶם וַיֹּאמְרוּ אֵל אֱלֹהֵי הָרוּחֹת לְכָל בָּשָׂר הָאִישׁ אֶחָד יֶחֱטָא וְעַל כָּל הָעֵדָה תִּקְצֹף` (במדבר טז:כב)
`וַיִּשָּׂא בִלְעָם אֶת עֵינָיו וַיַּרְא אֶת יִשְׂרָאֵל שֹׁכֵן לִשְׁבָטָיו וַתְּהִי עָלָיו רוּחַ אֱלֹהִים` (במדבר כד:ב)
`יִפְקֹד יְהוָה אֱלֹהֵי הָרוּחֹת לְכָל בָּשָׂר אִישׁ עַל הָעֵדָה` (במדבר כז:טז)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה קַח לְךָ אֶת יְהוֹשֻׁעַ בִּן נוּן אִישׁ אֲשֶׁר רוּחַ בּוֹ וְסָמַכְתָּ אֶת יָדְךָ עָלָיו` (במדבר כז:יח)
`וְלֹא אָבָה סִיחֹן מֶלֶךְ חֶשְׁבּוֹן הַעֲבִרֵנוּ בּוֹ כִּי הִקְשָׁה יְהוָה אֱלֹהֶיךָ אֶת רוּחוֹ וְאִמֵּץ אֶת לְבָבוֹ לְמַעַן תִּתּוֹ בְיָדְךָ כַּיּוֹם הַזֶּה` (דברים ב:ל)
`וִיהוֹשֻׁעַ בִּן נוּן מָלֵא רוּחַ חָכְמָה כִּי סָמַךְ מֹשֶׁה אֶת יָדָיו עָלָיו וַיִּשְׁמְעוּ אֵלָיו בְּנֵי יִשְׂרָאֵל וַיַּעֲשׂוּ כַּאֲשֶׁר צִוָּה יְהוָה אֶת מֹשֶׁה` (דברים לד:ט)
`וַנִּשְׁמַע וַיִּמַּס לְבָבֵנוּ וְלֹא קָמָה עוֹד רוּחַ בְּאִישׁ מִפְּנֵיכֶם כִּי יְהוָה אֱלֹהֵיכֶם הוּא אֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת` (יהושע ב:יא)
`וַיְהִי כִשְׁמֹעַ כָּל מַלְכֵי הָאֱמֹרִי אֲשֶׁר בְּעֵבֶר הַיַּרְדֵּן יָמָּה וְכָל מַלְכֵי הַכְּנַעֲנִי אֲשֶׁר עַל הַיָּם אֵת אֲשֶׁר הוֹבִישׁ יְהוָה אֶת מֵי הַיַּרְדֵּן מִפְּנֵי בְנֵי יִשְׂרָאֵל עַד עָבְרָנוּ וַיִּמַּס לְבָבָם וְלֹא הָיָה בָם עוֹד רוּחַ מִפְּנֵי בְּנֵי יִשְׂרָאֵל` (יהושע ה:א)
`וַתְּהִי עָלָיו רוּחַ יְהוָה וַיִּשְׁפֹּט אֶת יִשְׂרָאֵל וַיֵּצֵא לַמִּלְחָמָה וַיִּתֵּן יְהוָה בְּיָדוֹ אֶת כּוּשַׁן רִשְׁעָתַיִם מֶלֶךְ אֲרָם וַתָּעָז יָדוֹ עַל כּוּשַׁן רִשְׁעָתָיִם` (שופטים ג:י)
`וְרוּחַ יְהוָה לָבְשָׁה אֶת גִּדְעוֹן וַיִּתְקַע בַּשּׁוֹפָר וַיִזָּעֵק אֲבִיעֶזֶר אַחֲרָיו` (שופטים ו:לד)
`בְּיֶדְכֶם נָתַן אֱלֹהִים אֶת שָׂרֵי מִדְיָן אֶת עֹרֵב וְאֶת זְאֵב וּמַה יָּכֹלְתִּי עֲשׂוֹת כָּכֶם אָז רָפְתָה רוּחָם מֵעָלָיו בְּדַבְּרוֹ הַדָּבָר הַזֶּה` (שופטים ח:ג)
`וַיִּשְׁלַח אֱלֹהִים רוּחַ רָעָה בֵּין אֲבִימֶלֶךְ וּבֵין בַּעֲלֵי שְׁכֶם וַיִּבְגְּדוּ בַעֲלֵי שְׁכֶם בַּאֲבִימֶלֶךְ` (שופטים ט:כג)
`וַתְּהִי עַל יִפְתָּח רוּחַ יְהוָה וַיַּעֲבֹר אֶת הַגִּלְעָד וְאֶת מְנַשֶּׁה וַיַּעֲבֹר אֶת מִצְפֵּה גִלְעָד וּמִמִּצְפֵּה גִלְעָד עָבַר בְּנֵי עַמּוֹן` (שופטים יא:כט)
`וַתָּחֶל רוּחַ יְהוָה לְפַעֲמוֹ בְּמַחֲנֵה דָן בֵּין צָרְעָה וּבֵין אֶשְׁתָּאֹל` (שופטים יג:כה)
`וַתִּצְלַח עָלָיו רוּחַ יְהוָה וַיְשַׁסְּעֵהוּ כְּשַׁסַּע הַגְּדִי וּמְאוּמָה אֵין בְּיָדוֹ וְלֹא הִגִּיד לְאָבִיו וּלְאִמּוֹ אֵת אֲשֶׁר עָשָׂה` (שופטים יד:ו)
`וַתִּצְלַח עָלָיו רוּחַ יְהוָה וַיֵּרֶד אַשְׁקְלוֹן וַיַּךְ מֵהֶם שְׁלֹשִׁים אִישׁ וַיִּקַּח אֶת חֲלִיצוֹתָם וַיִּתֵּן הַחֲלִיפוֹת לְמַגִּידֵי הַחִידָה וַיִּחַר אַפּוֹ וַיַּעַל בֵּית אָבִיהוּ` (שופטים יד:יט)
`הוּא בָא עַד לֶחִי וּפִלְשִׁתִּים הֵרִיעוּ לִקְרָאתוֹ וַתִּצְלַח עָלָיו רוּחַ יְהוָה וַתִּהְיֶינָה הָעֲבֹתִים אֲשֶׁר עַל זְרוֹעוֹתָיו כַּפִּשְׁתִּים אֲשֶׁר בָּעֲרוּ בָאֵשׁ וַיִּמַּסּוּ אֱסוּרָיו מֵעַל יָדָיו` (שופטים טו:יד)
`וַיִּבְקַע אֱלֹהִים אֶת הַמַּכְתֵּשׁ אֲשֶׁר בַּלֶּחִי וַיֵּצְאוּ מִמֶּנּוּ מַיִם וַיֵּשְׁתְּ וַתָּשָׁב רוּחוֹ וַיֶּחִי עַל כֵּן קָרָא שְׁמָהּ עֵין הַקּוֹרֵא אֲשֶׁר בַּלֶּחִי עַד הַיּוֹם הַזֶּה` (שופטים טו:יט)
`וַתַּעַן חַנָּה וַתֹּאמֶר לֹא אֲדֹנִי אִשָּׁה קְשַׁת רוּחַ אָנֹכִי וְיַיִן וְשֵׁכָר לֹא שָׁתִיתִי וָאֶשְׁפֹּךְ אֶת נַפְשִׁי לִפְנֵי יְהוָה` (שמואל-א א:טו)
`וְצָלְחָה עָלֶיךָ רוּחַ יְהוָה וְהִתְנַבִּיתָ עִמָּם וְנֶהְפַּכְתָּ לְאִישׁ אַחֵר` (שמואל-א י:ו)
`וַיָּבֹאוּ שָׁם הַגִּבְעָתָה וְהִנֵּה חֶבֶל נְבִאִים לִקְרָאתוֹ וַתִּצְלַח עָלָיו רוּחַ אֱלֹהִים וַיִּתְנַבֵּא בְּתוֹכָם` (שמואל-א י:י)
`וַתִּצְלַח רוּחַ אֱלֹהִים עַל שָׁאוּל בְּשָׁמְעוֹ אֶת הַדְּבָרִים הָאֵלֶּה וַיִּחַר אַפּוֹ מְאֹד` (שמואל-א יא:ו)
`וַיִּקַּח שְׁמוּאֵל אֶת קֶרֶן הַשֶּׁמֶן וַיִּמְשַׁח אֹתוֹ בְּקֶרֶב אֶחָיו וַתִּצְלַח רוּחַ יְהוָה אֶל דָּוִד מֵהַיּוֹם הַהוּא וָמָעְלָה וַיָּקָם שְׁמוּאֵל וַיֵּלֶךְ הָרָמָתָה` (שמואל-א טז:יג)
`וְרוּחַ יְהוָה סָרָה מֵעִם שָׁאוּל וּבִעֲתַתּוּ רוּחַ רָעָה מֵאֵת יְהוָה` (שמואל-א טז:יד)
`וַיֹּאמְרוּ עַבְדֵי שָׁאוּל אֵלָיו הִנֵּה נָא רוּחַ אֱלֹהִים רָעָה מְבַעִתֶּךָ` (שמואל-א טז:טו)
`יֹאמַר נָא אֲדֹנֵנוּ עֲבָדֶיךָ לְפָנֶיךָ יְבַקְשׁוּ אִישׁ יֹדֵעַ מְנַגֵּן בַּכִּנּוֹר וְהָיָה בִּהְיוֹת עָלֶיךָ רוּחַ אֱלֹהִים רָעָה וְנִגֵּן בְּיָדוֹ וְטוֹב לָךְ` (שמואל-א טז:טז)
`וְהָיָה בִּהְיוֹת רוּחַ אֱלֹהִים אֶל שָׁאוּל וְלָקַח דָּוִד אֶת הַכִּנּוֹר וְנִגֵּן בְּיָדוֹ וְרָוַח לְשָׁאוּל וְטוֹב לוֹ וְסָרָה מֵעָלָיו רוּחַ הָרָעָה` (שמואל-א טז:כג)
`וַיְהִי מִמָּחֳרָת וַתִּצְלַח רוּחַ אֱלֹהִים רָעָה אֶל שָׁאוּל וַיִּתְנַבֵּא בְתוֹךְ הַבַּיִת וְדָוִד מְנַגֵּן בְּיָדוֹ כְּיוֹם בְּיוֹם וְהַחֲנִית בְּיַד שָׁאוּל` (שמואל-א יח:י)
`וַתְּהִי רוּחַ יְהוָה רָעָה אֶל שָׁאוּל וְהוּא בְּבֵיתוֹ יוֹשֵׁב וַחֲנִיתוֹ בְּיָדוֹ וְדָוִד מְנַגֵּן בְּיָד` (שמואל-א יט:ט)
`וַיִּשְׁלַח שָׁאוּל מַלְאָכִים לָקַחַת אֶת דָּוִד וַיַּרְא אֶת לַהֲקַת הַנְּבִיאִים נִבְּאִים וּשְׁמוּאֵל עֹמֵד נִצָּב עֲלֵיהֶם וַתְּהִי עַל מַלְאֲכֵי שָׁאוּל רוּחַ אֱלֹהִים וַיִּתְנַבְּאוּ גַּם הֵמָּה` (שמואל-א יט:כ)
`וַיֵּלֶךְ שָׁם אֶל נֹוִית בָּרָמָה וַתְּהִי עָלָיו גַּם הוּא רוּחַ אֱלֹהִים וַיֵּלֶךְ הָלוֹךְ וַיִּתְנַבֵּא עַד בֹּאוֹ בְּנֹוִית בָּרָמָה` (שמואל-א יט:כג)
`וַיִּתְּנוּ לוֹ פֶלַח דְּבֵלָה וּשְׁנֵי צִמֻּקִים וַיֹּאכַל וַתָּשָׁב רוּחוֹ אֵלָיו כִּי לֹא אָכַל לֶחֶם וְלֹא שָׁתָה מַיִם שְׁלֹשָׁה יָמִים וּשְׁלֹשָׁה לֵילוֹת` (שמואל-א ל:יב)
`וַיִּרְכַּב עַל כְּרוּב וַיָּעֹף וַיֵּרָא עַל כַּנְפֵי רוּחַ` (שמואל-ב כב:יא)
`וַיֵּרָאוּ אֲפִקֵי יָם יִגָּלוּ מֹסְדוֹת תֵּבֵל בְּגַעֲרַת יְהוָה מִנִּשְׁמַת רוּחַ אַפּוֹ` (שמואל-ב כב:טז)
`רוּחַ יְהוָה דִּבֶּר בִּי וּמִלָּתוֹ עַל לְשׁוֹנִי` (שמואל-ב כג:ב)
`וּמַאֲכַל שֻׁלְחָנוֹ וּמוֹשַׁב עֲבָדָיו וּמַעֲמַד מְשָׁרְתוֹ וּמַלְבֻּשֵׁיהֶם וּמַשְׁקָיו וְעֹלָתוֹ אֲשֶׁר יַעֲלֶה בֵּית יְהוָה וְלֹא הָיָה בָהּ עוֹד רוּחַ` (מלכים-א י:ה)
`וְהָיָה אֲנִי אֵלֵךְ מֵאִתָּךְ וְרוּחַ יְהוָה יִשָּׂאֲךָ עַל אֲשֶׁר לֹא אֵדָע וּבָאתִי לְהַגִּיד לְאַחְאָב וְלֹא יִמְצָאֲךָ וַהֲרָגָנִי וְעַבְדְּךָ יָרֵא אֶת יְהוָה מִנְּעֻרָי` (מלכים-א יח:יב)
`וַיְהִי עַד כֹּה וְעַד כֹּה וְהַשָּׁמַיִם הִתְקַדְּרוּ עָבִים וְרוּחַ וַיְהִי גֶּשֶׁם גָּדוֹל וַיִּרְכַּב אַחְאָב וַיֵּלֶךְ יִזְרְעֶאלָה` (מלכים-א יח:מה)
`וַיֹּאמֶר צֵא וְעָמַדְתָּ בָהָר לִפְנֵי יְהוָה וְהִנֵּה יְהוָה עֹבֵר וְרוּחַ גְּדוֹלָה וְחָזָק מְפָרֵק הָרִים וּמְשַׁבֵּר סְלָעִים לִפְנֵי יְהוָה לֹא בָרוּחַ יְהוָה וְאַחַר הָרוּחַ רַעַשׁ לֹא בָרַעַשׁ יְהוָה` (מלכים-א יט:יא)
`וַתָּבֹא אֵלָיו אִיזֶבֶל אִשְׁתּוֹ וַתְּדַבֵּר אֵלָיו מַה זֶּה רוּחֲךָ סָרָה וְאֵינְךָ אֹכֵל לָחֶם` (מלכים-א כא:ה)
`וַיֵּצֵא הָרוּחַ וַיַּעֲמֹד לִפְנֵי יְהוָה וַיֹּאמֶר אֲנִי אֲפַתֶּנּוּ וַיֹּאמֶר יְהוָה אֵלָיו בַּמָּה` (מלכים-א כב:כא)
`וַיֹּאמֶר אֵצֵא וְהָיִיתִי רוּחַ שֶׁקֶר בְּפִי כָּל נְבִיאָיו וַיֹּאמֶר תְּפַתֶּה וְגַם תּוּכָל צֵא וַעֲשֵׂה כֵן` (מלכים-א כב:כב)
`וְעַתָּה הִנֵּה נָתַן יְהוָה רוּחַ שֶׁקֶר בְּפִי כָּל נְבִיאֶיךָ אֵלֶּה וַיהוָה דִּבֶּר עָלֶיךָ רָעָה` (מלכים-א כב:כג)
`וַיִּגַּשׁ צִדְקִיָּהוּ בֶן כְּנַעֲנָה וַיַּכֶּה אֶת מִיכָיְהוּ עַל הַלֶּחִי וַיֹּאמֶר אֵי זֶה עָבַר רוּחַ יְהוָה מֵאִתִּי לְדַבֵּר אוֹתָךְ` (מלכים-א כב:כד)
`וַיְהִי כְעָבְרָם וְאֵלִיָּהוּ אָמַר אֶל אֱלִישָׁע שְׁאַל מָה אֶעֱשֶׂה לָּךְ בְּטֶרֶם אֶלָּקַח מֵעִמָּךְ וַיֹּאמֶר אֱלִישָׁע וִיהִי נָא פִּי שְׁנַיִם בְּרוּחֲךָ אֵלָי` (מלכים-ב ב:ט)
`וַיִּרְאֻהוּ בְנֵי הַנְּבִיאִים אֲשֶׁר בִּירִיחוֹ מִנֶּגֶד וַיֹּאמְרוּ נָחָה רוּחַ אֵלִיָּהוּ עַל אֱלִישָׁע וַיָּבֹאוּ לִקְרָאתוֹ וַיִּשְׁתַּחֲווּ לוֹ אָרְצָה` (מלכים-ב ב:טו)
`וַיֹּאמְרוּ אֵלָיו הִנֵּה נָא יֵשׁ אֶת עֲבָדֶיךָ חֲמִשִּׁים אֲנָשִׁים בְּנֵי חַיִל יֵלְכוּ נָא וִיבַקְשׁוּ אֶת אֲדֹנֶיךָ פֶּן נְשָׂאוֹ רוּחַ יְהוָה וַיַּשְׁלִכֵהוּ בְּאַחַד הֶהָרִים אוֹ בְּאַחַת הַגְּיָאוֹת וַיֹּאמֶר לֹא תִשְׁלָחוּ` (מלכים-ב ב:טז)
`כִּי כֹה אָמַר יְהוָה לֹא תִרְאוּ רוּחַ וְלֹא תִרְאוּ גֶשֶׁם וְהַנַּחַל הַהוּא יִמָּלֵא מָיִם וּשְׁתִיתֶם אַתֶּם וּמִקְנֵיכֶם וּבְהֶמְתְּכֶם` (מלכים-ב ג:יז)
`הִנְנִי נֹתֵן בּוֹ רוּחַ וְשָׁמַע שְׁמוּעָה וְשָׁב לְאַרְצוֹ וְהִפַּלְתִּיו בַּחֶרֶב בְּאַרְצוֹ` (מלכים-ב יט:ז)
`אִם רָחַץ אֲדֹנָי אֵת צֹאַת בְּנוֹת צִיּוֹן וְאֶת דְּמֵי יְרוּשָׁלִַם יָדִיחַ מִקִּרְבָּהּ בְּרוּחַ מִשְׁפָּט וּבְרוּחַ בָּעֵר` (ישעיהו ד:ד)
`וַיֻּגַּד לְבֵית דָּוִד לֵאמֹר נָחָה אֲרָם עַל אֶפְרָיִם וַיָּנַע לְבָבוֹ וּלְבַב עַמּוֹ כְּנוֹעַ עֲצֵי יַעַר מִפְּנֵי רוּחַ` (ישעיהו ז:ב)
`וְנָחָה עָלָיו רוּחַ יְהוָה רוּחַ חָכְמָה וּבִינָה רוּחַ עֵצָה וּגְבוּרָה רוּחַ דַּעַת וְיִרְאַת יְהוָה` (ישעיהו יא:ב)
`וְשָׁפַט בְּצֶדֶק דַּלִּים וְהוֹכִיחַ בְּמִישׁוֹר לְעַנְוֵי אָרֶץ וְהִכָּה אֶרֶץ בְּשֵׁבֶט פִּיו וּבְרוּחַ שְׂפָתָיו יָמִית רָשָׁע` (ישעיהו יא:ד)
`וְהֶחֱרִים יְהוָה אֵת לְשׁוֹן יָם מִצְרַיִם וְהֵנִיף יָדוֹ עַל הַנָּהָר בַּעְיָם רוּחוֹ וְהִכָּהוּ לְשִׁבְעָה נְחָלִים וְהִדְרִיךְ בַּנְּעָלִים` (ישעיהו יא:טו)
`לְאֻמִּים כִּשְׁאוֹן מַיִם רַבִּים יִשָּׁאוּן וְגָעַר בּוֹ וְנָס מִמֶּרְחָק וְרֻדַּף כְּמֹץ הָרִים לִפְנֵי רוּחַ וּכְגַלְגַּל לִפְנֵי סוּפָה` (ישעיהו יז:יג)
`וְנָבְקָה רוּחַ מִצְרַיִם בְּקִרְבּוֹ וַעֲצָתוֹ אֲבַלֵּעַ וְדָרְשׁוּ אֶל הָאֱלִילִים וְאֶל הָאִטִּים וְאֶל הָאֹבוֹת וְאֶל הַיִּדְּעֹנִים` (ישעיהו יט:ג)
`יְהוָה מָסַךְ בְּקִרְבָּהּ רוּחַ עִוְעִים וְהִתְעוּ אֶת מִצְרַיִם בְּכָל מַעֲשֵׂהוּ כְּהִתָּעוֹת שִׁכּוֹר בְּקִיאוֹ` (ישעיהו יט:יד)
`כִּי הָיִיתָ מָעוֹז לַדָּל מָעוֹז לָאֶבְיוֹן בַּצַּר לוֹ מַחְסֶה מִזֶּרֶם צֵל מֵחֹרֶב כִּי רוּחַ עָרִיצִים כְּזֶרֶם קִיר` (ישעיהו כה:ד)
`נַפְשִׁי אִוִּיתִיךָ בַּלַּיְלָה אַף רוּחִי בְקִרְבִּי אֲשַׁחֲרֶךָּ כִּי כַּאֲשֶׁר מִשְׁפָּטֶיךָ לָאָרֶץ צֶדֶק לָמְדוּ יֹשְׁבֵי תֵבֵל` (ישעיהו כו:ט)
`הָרִינוּ חַלְנוּ כְּמוֹ יָלַדְנוּ רוּחַ יְשׁוּעֹת בַּל נַעֲשֶׂה אֶרֶץ וּבַל יִפְּלוּ יֹשְׁבֵי תֵבֵל` (ישעיהו כו:יח)
`בְּסַּאסְּאָה בְּשַׁלְחָהּ תְּרִיבֶנָּה הָגָה בְּרוּחוֹ הַקָּשָׁה בְּיוֹם קָדִים` (ישעיהו כז:ח)
`וּלְרוּחַ מִשְׁפָּט לַיּוֹשֵׁב עַל הַמִּשְׁפָּט וְלִגְבוּרָה מְשִׁיבֵי מִלְחָמָה שָׁעְרָה` (ישעיהו כח:ו)
`כִּי נָסַךְ עֲלֵיכֶם יְהוָה רוּחַ תַּרְדֵּמָה וַיְעַצֵּם אֶת עֵינֵיכֶם אֶת הַנְּבִיאִים וְאֶת רָאשֵׁיכֶם הַחֹזִים כִּסָּה` (ישעיהו כט:י)
`וְיָדְעוּ תֹעֵי רוּחַ בִּינָה וְרוֹגְנִים יִלְמְדוּ לֶקַח` (ישעיהו כט:כד)
`הוֹי בָּנִים סוֹרְרִים נְאֻם יְהוָה לַעֲשׂוֹת עֵצָה וְלֹא מִנִּי וְלִנְסֹךְ מַסֵּכָה וְלֹא רוּחִי לְמַעַן סְפוֹת חַטָּאת עַל חַטָּאת` (ישעיהו ל:א)
`וְרוּחוֹ כְּנַחַל שׁוֹטֵף עַד צַוָּאר יֶחֱצֶה לַהֲנָפָה גוֹיִם בְּנָפַת שָׁוְא וְרֶסֶן מַתְעֶה עַל לְחָיֵי עַמִּים` (ישעיהו ל:כח)
`וּמִצְרַיִם אָדָם וְלֹא אֵל וְסוּסֵיהֶם בָּשָׂר וְלֹא רוּחַ וַיהוָה יַטֶּה יָדוֹ וְכָשַׁל עוֹזֵר וְנָפַל עָזֻר וְיַחְדָּו כֻּלָּם יִכְלָיוּן` (ישעיהו לא:ג)
`וְהָיָה אִישׁ כְּמַחֲבֵא רוּחַ וְסֵתֶר זָרֶם כְּפַלְגֵי מַיִם בְּצָיוֹן כְּצֵל סֶלַע כָּבֵד בְּאֶרֶץ עֲיֵפָה` (ישעיהו לב:ב)
`עַד יֵעָרֶה עָלֵינוּ רוּחַ מִמָּרוֹם וְהָיָה מִדְבָּר לַכַּרְמֶל וְכַרְמֶל לַיַּעַר יֵחָשֵׁב` (ישעיהו לב:טו)
`תַּהֲרוּ חֲשַׁשׁ תֵּלְדוּ קַשׁ רוּחֲכֶם אֵשׁ תֹּאכַלְכֶם` (ישעיהו לג:יא)
`דִּרְשׁוּ מֵעַל סֵפֶר יְהוָה וּקְרָאוּ אַחַת מֵהֵנָּה לֹא נֶעְדָּרָה אִשָּׁה רְעוּתָהּ לֹא פָקָדוּ כִּי פִי הוּא צִוָּה וְרוּחוֹ הוּא קִבְּצָן` (ישעיהו לד:טז)
`הִנְנִי נוֹתֵן בּוֹ רוּחַ וְשָׁמַע שְׁמוּעָה וְשָׁב אֶל אַרְצוֹ וְהִפַּלְתִּיו בַּחֶרֶב בְּאַרְצוֹ` (ישעיהו לז:ז)
`אֲדֹנָי עֲלֵיהֶם יִחְיוּ וּלְכָל בָּהֶן חַיֵּי רוּחִי וְתַחֲלִימֵנִי וְהַחֲיֵנִי` (ישעיהו לח:טז)
`יָבֵשׁ חָצִיר נָבֵל צִיץ כִּי רוּחַ יְהוָה נָשְׁבָה בּוֹ אָכֵן חָצִיר הָעָם` (ישעיהו מ:ז)
`מִי תִכֵּן אֶת רוּחַ יְהוָה וְאִישׁ עֲצָתוֹ יוֹדִיעֶנּוּ` (ישעיהו מ:יג)
`תִּזְרֵם וְרוּחַ תִּשָּׂאֵם וּסְעָרָה תָּפִיץ אוֹתָם וְאַתָּה תָּגִיל בַּיהוָה בִּקְדוֹשׁ יִשְׂרָאֵל תִּתְהַלָּל` (ישעיהו מא:טז)
`הֵן כֻּלָּם אָוֶן אֶפֶס מַעֲשֵׂיהֶם רוּחַ וָתֹהוּ נִסְכֵּיהֶם` (ישעיהו מא:כט)
`הֵן עַבְדִּי אֶתְמָךְ בּוֹ בְּחִירִי רָצְתָה נַפְשִׁי נָתַתִּי רוּחִי עָלָיו מִשְׁפָּט לַגּוֹיִם יוֹצִיא` (ישעיהו מב:א)
`כֹּה אָמַר הָאֵל יְהוָה בּוֹרֵא הַשָּׁמַיִם וְנוֹטֵיהֶם רֹקַע הָאָרֶץ וְצֶאֱצָאֶיהָ נֹתֵן נְשָׁמָה לָעָם עָלֶיהָ וְרוּחַ לַהֹלְכִים בָּהּ` (ישעיהו מב:ה)
`כִּי אֶצָּק מַיִם עַל צָמֵא וְנֹזְלִים עַל יַבָּשָׁה אֶצֹּק רוּחִי עַל זַרְעֶךָ וּבִרְכָתִי עַל צֶאֱצָאֶיךָ` (ישעיהו מד:ג)
`קִרְבוּ אֵלַי שִׁמְעוּ זֹאת לֹא מֵרֹאשׁ בַּסֵּתֶר דִּבַּרְתִּי מֵעֵת הֱיוֹתָהּ שָׁם אָנִי וְעַתָּה אֲדֹנָי יְהוִה שְׁלָחַנִי וְרוּחוֹ` (ישעיהו מח:טז)
`כִּי כְאִשָּׁה עֲזוּבָה וַעֲצוּבַת רוּחַ קְרָאָךְ יְהוָה וְאֵשֶׁת נְעוּרִים כִּי תִמָּאֵס אָמַר אֱלֹהָיִךְ` (ישעיהו נד:ו)
`בְּזַעֲקֵךְ יַצִּילֻךְ קִבּוּצַיִךְ וְאֶת כֻּלָּם יִשָּׂא רוּחַ יִקַּח הָבֶל וְהַחוֹסֶה בִי יִנְחַל אֶרֶץ וְיִירַשׁ הַר קָדְשִׁי` (ישעיהו נז:יג)
`כִּי כֹה אָמַר רָם וְנִשָּׂא שֹׁכֵן עַד וְקָדוֹשׁ שְׁמוֹ מָרוֹם וְקָדוֹשׁ אֶשְׁכּוֹן וְאֶת דַּכָּא וּשְׁפַל רוּחַ לְהַחֲיוֹת רוּחַ שְׁפָלִים וּלְהַחֲיוֹת לֵב נִדְכָּאִים` (ישעיהו נז:טו)
`כִּי לֹא לְעוֹלָם אָרִיב וְלֹא לָנֶצַח אֶקְּצוֹף כִּי רוּחַ מִלְּפָנַי יַעֲטוֹף וּנְשָׁמוֹת אֲנִי עָשִׂיתִי` (ישעיהו נז:טז)
`וְיִירְאוּ מִמַּעֲרָב אֶת שֵׁם יְהוָה וּמִמִּזְרַח שֶׁמֶשׁ אֶת כְּבוֹדוֹ כִּי יָבוֹא כַנָּהָר צָר רוּחַ יְהוָה נֹסְסָה בוֹ` (ישעיהו נט:יט)
`וַאֲנִי זֹאת בְּרִיתִי אוֹתָם אָמַר יְהוָה רוּחִי אֲשֶׁר עָלֶיךָ וּדְבָרַי אֲשֶׁר שַׂמְתִּי בְּפִיךָ לֹא יָמוּשׁוּ מִפִּיךָ וּמִפִּי זַרְעֲךָ וּמִפִּי זֶרַע זַרְעֲךָ אָמַר יְהוָה מֵעַתָּה וְעַד עוֹלָם` (ישעיהו נט:כא)
`רוּחַ אֲדֹנָי יְהוִה עָלָי יַעַן מָשַׁח יְהוָה אֹתִי לְבַשֵּׂר עֲנָוִים שְׁלָחַנִי לַחֲבֹשׁ לְנִשְׁבְּרֵי לֵב לִקְרֹא לִשְׁבוּיִם דְּרוֹר וְלַאֲסוּרִים פְּקַח קוֹחַ` (ישעיהו סא:א)
`לָשׂוּם לַאֲבֵלֵי צִיּוֹן לָתֵת לָהֶם פְּאֵר תַּחַת אֵפֶר שֶׁמֶן שָׂשׂוֹן תַּחַת אֵבֶל מַעֲטֵה תְהִלָּה תַּחַת רוּחַ כֵּהָה וְקֹרָא לָהֶם אֵילֵי הַצֶּדֶק מַטַּע יְהוָה לְהִתְפָּאֵר` (ישעיהו סא:ג)
`וְהֵמָּה מָרוּ וְעִצְּבוּ אֶת רוּחַ קָדְשׁוֹ וַיֵּהָפֵךְ לָהֶם לְאוֹיֵב הוּא נִלְחַם בָּם` (ישעיהו סג:י)
`וַיִּזְכֹּר יְמֵי עוֹלָם מֹשֶׁה עַמּוֹ אַיֵּה הַמַּעֲלֵם מִיָּם אֵת רֹעֵי צֹאנוֹ אַיֵּה הַשָּׂם בְּקִרְבּוֹ אֶת רוּחַ קָדְשׁוֹ` (ישעיהו סג:יא)
`כַּבְּהֵמָה בַּבִּקְעָה תֵרֵד רוּחַ יְהוָה תְּנִיחֶנּוּ כֵּן נִהַגְתָּ עַמְּךָ לַעֲשׂוֹת לְךָ שֵׁם תִּפְאָרֶת` (ישעיהו סג:יד)
`וַנְּהִי כַטָּמֵא כֻּלָּנוּ וּכְבֶגֶד עִדִּים כָּל צִדְקֹתֵינוּ וַנָּבֶל כֶּעָלֶה כֻּלָּנוּ וַעֲוֺנֵנוּ כָּרוּחַ יִשָּׂאֻנוּ` (ישעיהו סד:ו)
`הִנֵּה עֲבָדַי יָרֹנּוּ מִטּוּב לֵב וְאַתֶּם תִּצְעֲקוּ מִכְּאֵב לֵב וּמִשֵּׁבֶר רוּחַ תְּיֵלִילוּ` (ישעיהו סה:יד)
`וְאֶת כָּל אֵלֶּה יָדִי עָשָׂתָה וַיִּהְיוּ כָל אֵלֶּה נְאֻם יְהוָה וְאֶל זֶה אַבִּיט אֶל עָנִי וּנְכֵה רוּחַ וְחָרֵד עַל דְּבָרִי` (ישעיהו סו:ב)
`פֶּרֶה לִמֻּד מִדְבָּר בְּאַוַּת נַפְשׁוֹ שָׁאֲפָה רוּחַ תַּאֲנָתָהּ מִי יְשִׁיבֶנָּה כָּל מְבַקְשֶׁיהָ לֹא יִיעָפוּ בְּחָדְשָׁהּ יִמְצָאוּנְהָ` (ירמיהו ב:כד)
`בָּעֵת הַהִיא יֵאָמֵר לָעָם הַזֶּה וְלִירוּשָׁלִַם רוּחַ צַח שְׁפָיִים בַּמִּדְבָּר דֶּרֶךְ בַּת עַמִּי לוֹא לִזְרוֹת וְלוֹא לְהָבַר` (ירמיהו ד:יא)
`רוּחַ מָלֵא מֵאֵלֶּה יָבוֹא לִי עַתָּה גַּם אֲנִי אֲדַבֵּר מִשְׁפָּטִים אוֹתָם` (ירמיהו ד:יב)
`וְהַנְּבִיאִים יִהְיוּ לְרוּחַ וְהַדִּבֵּר אֵין בָּהֶם כֹּה יֵעָשֶׂה לָהֶם` (ירמיהו ה:יג)
`לְקוֹל תִּתּוֹ הֲמוֹן מַיִם בַּשָּׁמַיִם וַיַּעֲלֶה נְשִׂאִים מִקְצֵה אֶרֶץ בְּרָקִים לַמָּטָר עָשָׂה וַיּוֹצֵא רוּחַ מֵאֹצְרֹתָיו` (ירמיהו י:יג)
`נִבְעַר כָּל אָדָם מִדַּעַת הֹבִישׁ כָּל צוֹרֵף מִפָּסֶל כִּי שֶׁקֶר נִסְכּוֹ וְלֹא רוּחַ בָּם` (ירמיהו י:יד)
`וַאֲפִיצֵם כְּקַשׁ עוֹבֵר לְרוּחַ מִדְבָּר` (ירמיהו יג:כד)
`וּפְרָאִים עָמְדוּ עַל שְׁפָיִם שָׁאֲפוּ רוּחַ כַּתַּנִּים כָּלוּ עֵינֵיהֶם כִּי אֵין עֵשֶׂב` (ירמיהו יד:ו)
`כְּרוּחַ קָדִים אֲפִיצֵם לִפְנֵי אוֹיֵב עֹרֶף וְלֹא פָנִים אֶרְאֵם בְּיוֹם אֵידָם` (ירמיהו יח:יז)
`כָּל רֹעַיִךְ תִּרְעֶה רוּחַ וּמְאַהֲבַיִךְ בַּשְּׁבִי יֵלֵכוּ כִּי אָז תֵּבֹשִׁי וְנִכְלַמְתְּ מִכֹּל רָעָתֵךְ` (ירמיהו כב:כב)
`וְהָיוּ גְמַלֵּיהֶם לָבַז וַהֲמוֹן מִקְנֵיהֶם לְשָׁלָל וְזֵרִתִים לְכָל רוּחַ קְצוּצֵי פֵאָה וּמִכָּל עֲבָרָיו אָבִיא אֶת אֵידָם נְאֻם יְהוָה` (ירמיהו מט:לב)
`וְהֵבֵאתִי אֶל עֵילָם אַרְבַּע רוּחוֹת מֵאַרְבַּע קְצוֹת הַשָּׁמַיִם וְזֵרִתִים לְכֹל הָרֻחוֹת הָאֵלֶּה וְלֹא יִהְיֶה הַגּוֹי אֲשֶׁר לֹא יָבוֹא שָׁם נִדְּחֵי עוֹלָם` (ירמיהו מט:לו)
`כֹּה אָמַר יְהוָה הִנְנִי מֵעִיר עַל בָּבֶל וְאֶל יֹשְׁבֵי לֵב קָמָי רוּחַ מַשְׁחִית` (ירמיהו נא:א)
`הָבֵרוּ הַחִצִּים מִלְאוּ הַשְּׁלָטִים הֵעִיר יְהוָה אֶת רוּחַ מַלְכֵי מָדַי כִּי עַל בָּבֶל מְזִמָּתוֹ לְהַשְׁחִיתָהּ כִּי נִקְמַת יְהוָה הִיא נִקְמַת הֵיכָלוֹ` (ירמיהו נא:יא)
`לְקוֹל תִּתּוֹ הֲמוֹן מַיִם בַּשָּׁמַיִם וַיַּעַל נְשִׂאִים מִקְצֵה אָרֶץ בְּרָקִים לַמָּטָר עָשָׂה וַיֹּצֵא רוּחַ מֵאֹצְרֹתָיו` (ירמיהו נא:טז)
`נִבְעַר כָּל אָדָם מִדַּעַת הֹבִישׁ כָּל צֹרֵף מִפָּסֶל כִּי שֶׁקֶר נִסְכּוֹ וְלֹא רוּחַ בָּם` (ירמיהו נא:יז)
`וַיִּהְיוּ הָרִמֹּנִים תִּשְׁעִים וְשִׁשָּׁה רוּחָה כָּל הָרִמּוֹנִים מֵאָה עַל הַשְּׂבָכָה סָבִיב` (ירמיהו נב:כג)
`וָאֵרֶא וְהִנֵּה רוּחַ סְעָרָה בָּאָה מִן הַצָּפוֹן עָנָן גָּדוֹל וְאֵשׁ מִתְלַקַּחַת וְנֹגַהּ לוֹ סָבִיב וּמִתּוֹכָהּ כְּעֵין הַחַשְׁמַל מִתּוֹךְ הָאֵשׁ` (יחזקאל א:ד)
`וְאִישׁ אֶל עֵבֶר פָּנָיו יֵלֵכוּ אֶל אֲשֶׁר יִהְיֶה שָׁמָּה הָרוּחַ לָלֶכֶת יֵלֵכוּ לֹא יִסַּבּוּ בְּלֶכְתָּן` (יחזקאל א:יב)
`עַל אֲשֶׁר יִהְיֶה שָּׁם הָרוּחַ לָלֶכֶת יֵלֵכוּ שָׁמָּה הָרוּחַ לָלֶכֶת וְהָאוֹפַנִּים יִנָּשְׂאוּ לְעֻמָּתָם כִּי רוּחַ הַחַיָּה בָּאוֹפַנִּים` (יחזקאל א:כ)
`בְּלֶכְתָּם יֵלֵכוּ וּבְעָמְדָם יַעֲמֹדוּ וּבְהִנָּשְׂאָם מֵעַל הָאָרֶץ יִנָּשְׂאוּ הָאוֹפַנִּים לְעֻמָּתָם כִּי רוּחַ הַחַיָּה בָּאוֹפַנִּים` (יחזקאל א:כא)
`וַתָּבֹא בִי רוּחַ כַּאֲשֶׁר דִּבֶּר אֵלַי וַתַּעֲמִדֵנִי עַל רַגְלָי וָאֶשְׁמַע אֵת מִדַּבֵּר אֵלָי` (יחזקאל ב:ב)
`וַתִּשָּׂאֵנִי רוּחַ וָאֶשְׁמַע אַחֲרַי קוֹל רַעַשׁ גָּדוֹל בָּרוּךְ כְּבוֹד יְהוָה מִמְּקוֹמוֹ` (יחזקאל ג:יב)
`וְרוּחַ נְשָׂאַתְנִי וַתִּקָּחֵנִי וָאֵלֵךְ מַר בַּחֲמַת רוּחִי וְיַד יְהוָה עָלַי חָזָקָה` (יחזקאל ג:יד)
`וַתָּבֹא בִי רוּחַ וַתַּעֲמִדֵנִי עַל רַגְלָי וַיְדַבֵּר אֹתִי וַיֹּאמֶר אֵלַי בֹּא הִסָּגֵר בְּתוֹךְ בֵּיתֶךָ` (יחזקאל ג:כד)
`שְׁלִשִׁית בָּאוּר תַּבְעִיר בְּתוֹךְ הָעִיר כִּמְלֹאת יְמֵי הַמָּצוֹר וְלָקַחְתָּ אֶת הַשְּׁלִשִׁית תַּכֶּה בַחֶרֶב סְבִיבוֹתֶיהָ וְהַשְּׁלִשִׁית תִּזְרֶה לָרוּחַ וְחֶרֶב אָרִיק אַחֲרֵיהֶם` (יחזקאל ה:ב)
`לָכֵן אָבוֹת יֹאכְלוּ בָנִים בְּתוֹכֵךְ וּבָנִים יֹאכְלוּ אֲבוֹתָם וְעָשִׂיתִי בָךְ שְׁפָטִים וְזֵרִיתִי אֶת כָּל שְׁאֵרִיתֵךְ לְכָל רוּחַ` (יחזקאל ה:י)
`שְׁלִשִׁתֵיךְ בַּדֶּבֶר יָמוּתוּ וּבָרָעָב יִכְלוּ בְתוֹכֵךְ וְהַשְּׁלִשִׁית בַּחֶרֶב יִפְּלוּ סְבִיבוֹתָיִךְ וְהַשְּׁלִישִׁית לְכָל רוּחַ אֱזָרֶה וְחֶרֶב אָרִיק אַחֲרֵיהֶם` (יחזקאל ה:יב)
`וַיִּשְׁלַח תַּבְנִית יָד וַיִּקָּחֵנִי בְּצִיצִת רֹאשִׁי וַתִּשָּׂא אֹתִי רוּחַ בֵּין הָאָרֶץ וּבֵין הַשָּׁמַיִם וַתָּבֵא אֹתִי יְרוּשָׁלְַמָה בְּמַרְאוֹת אֱלֹהִים אֶל פֶּתַח שַׁעַר הַפְּנִימִית הַפּוֹנֶה צָפוֹנָה אֲשֶׁר שָׁם מוֹשַׁב סֵמֶל הַקִּנְאָה הַמַּקְנֶה` (יחזקאל ח:ג)
`בְּעָמְדָם יַעֲמֹדוּ וּבְרוֹמָם יֵרוֹמּוּ אוֹתָם כִּי רוּחַ הַחַיָּה בָּהֶם` (יחזקאל י:יז)
`וַתִּשָּׂא אֹתִי רוּחַ וַתָּבֵא אֹתִי אֶל שַׁעַר בֵּית יְהוָה הַקַּדְמוֹנִי הַפּוֹנֶה קָדִימָה וְהִנֵּה בְּפֶתַח הַשַּׁעַר עֶשְׂרִים וַחֲמִשָּׁה אִישׁ וָאֶרְאֶה בְתוֹכָם אֶת יַאֲזַנְיָה בֶן עַזֻּר וְאֶת פְּלַטְיָהוּ בֶן בְּנָיָהוּ שָׂרֵי הָעָם` (יחזקאל יא:א)
`וַתִּפֹּל עָלַי רוּחַ יְהוָה וַיֹּאמֶר אֵלַי אֱמֹר כֹּה אָמַר יְהוָה כֵּן אֲמַרְתֶּם בֵּית יִשְׂרָאֵל וּמַעֲלוֹת רוּחֲכֶם אֲנִי יְדַעְתִּיהָ` (יחזקאל יא:ה)
`וְנָתַתִּי לָהֶם לֵב אֶחָד וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי לֵב הָאֶבֶן מִבְּשָׂרָם וְנָתַתִּי לָהֶם לֵב בָּשָׂר` (יחזקאל יא:יט)
`וְרוּחַ נְשָׂאַתְנִי וַתְּבִיאֵנִי כַשְׂדִּימָה אֶל הַגּוֹלָה בַּמַּרְאֶה בְּרוּחַ אֱלֹהִים וַיַּעַל מֵעָלַי הַמַּרְאֶה אֲשֶׁר רָאִיתִי` (יחזקאל יא:כד)
`וְכֹל אֲשֶׁר סְבִיבֹתָיו עֶזְרֹה וְכָל אֲגַפָּיו אֱזָרֶה לְכָל רוּחַ וְחֶרֶב אָרִיק אַחֲרֵיהֶם` (יחזקאל יב:יד)
`כֹּה אָמַר אֲדֹנָי יְהוִה הוֹי עַל הַנְּבִיאִים הַנְּבָלִים אֲשֶׁר הֹלְכִים אַחַר רוּחָם וּלְבִלְתִּי רָאוּ` (יחזקאל יג:ג)
`אֱמֹר אֶל טָחֵי תָפֵל וְיִפֹּל הָיָה גֶּשֶׁם שׁוֹטֵף וְאַתֵּנָה אַבְנֵי אֶלְגָּבִישׁ תִּפֹּלְנָה וְרוּחַ סְעָרוֹת תְּבַקֵּעַ` (יחזקאל יג:יא)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה וּבִקַּעְתִּי רוּחַ סְעָרוֹת בַּחֲמָתִי וְגֶשֶׁם שֹׁטֵף בְּאַפִּי יִהְיֶה וְאַבְנֵי אֶלְגָּבִישׁ בְּחֵמָה לְכָלָה` (יחזקאל יג:יג)
`וְהִנֵּה שְׁתוּלָה הֲתִצְלָח הֲלוֹא כְגַעַת בָּהּ רוּחַ הַקָּדִים תִּיבַשׁ יָבֹשׁ עַל עֲרֻגֹת צִמְחָהּ תִּיבָשׁ` (יחזקאל יז:י)
`וְאֵת כָּל מִבְרָחוֹ בְּכָל אֲגַפָּיו בַּחֶרֶב יִפֹּלוּ וְהַנִּשְׁאָרִים לְכָל רוּחַ יִפָּרֵשׂוּ וִידַעְתֶּם כִּי אֲנִי יְהוָה דִּבַּרְתִּי` (יחזקאל יז:כא)
`הַשְׁלִיכוּ מֵעֲלֵיכֶם אֶת כָּל פִּשְׁעֵיכֶם אֲשֶׁר פְּשַׁעְתֶּם בָּם וַעֲשׂוּ לָכֶם לֵב חָדָשׁ וְרוּחַ חֲדָשָׁה וְלָמָּה תָמֻתוּ בֵּית יִשְׂרָאֵל` (יחזקאל יח:לא)
`וַתֻּתַּשׁ בְּחֵמָה לָאָרֶץ הֻשְׁלָכָה וְרוּחַ הַקָּדִים הוֹבִישׁ פִּרְיָהּ הִתְפָּרְקוּ וְיָבֵשׁוּ מַטֵּה עֻזָּהּ אֵשׁ אֲכָלָתְהוּ` (יחזקאל יט:יב)
`וְהָעֹלָה עַל רוּחֲכֶם הָיוֹ לֹא תִהְיֶה אֲשֶׁר אַתֶּם אֹמְרִים נִהְיֶה כַגּוֹיִם כְּמִשְׁפְּחוֹת הָאֲרָצוֹת לְשָׁרֵת עֵץ וָאָבֶן` (יחזקאל כ:לב)
`וְהָיָה כִּי יֹאמְרוּ אֵלֶיךָ עַל מָה אַתָּה נֶאֱנָח וְאָמַרְתָּ אֶל שְׁמוּעָה כִי בָאָה וְנָמֵס כָּל לֵב וְרָפוּ כָל יָדַיִם וְכִהֲתָה כָל רוּחַ וְכָל בִּרְכַּיִם תֵּלַכְנָה מַּיִם הִנֵּה בָאָה וְנִהְיָתָה נְאֻם אֲדֹנָי יְהוִה` (יחזקאל כא:ז)
`בְּמַיִם רַבִּים הֱבִיאוּךְ הַשָּׁטִים אֹתָךְ רוּחַ הַקָּדִים שְׁבָרֵךְ בְּלֵב יַמִּים` (יחזקאל כז:כו)
`וְנָתַתִּי לָכֶם לֵב חָדָשׁ וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי אֶת לֵב הָאֶבֶן מִבְּשַׂרְכֶם וְנָתַתִּי לָכֶם לֵב בָּשָׂר` (יחזקאל לו:כו)
`וְאֶת רוּחִי אֶתֵּן בְּקִרְבְּכֶם וְעָשִׂיתִי אֵת אֲשֶׁר בְּחֻקַּי תֵּלֵכוּ וּמִשְׁפָּטַי תִּשְׁמְרוּ וַעֲשִׂיתֶם` (יחזקאל לו:כז)
`הָיְתָה עָלַי יַד יְהוָה וַיּוֹצִאֵנִי בְרוּחַ יְהוָה וַיְנִיחֵנִי בְּתוֹךְ הַבִּקְעָה וְהִיא מְלֵאָה עֲצָמוֹת` (יחזקאל לז:א)
`כֹּה אָמַר אֲדֹנָי יְהוִה לָעֲצָמוֹת הָאֵלֶּה הִנֵּה אֲנִי מֵבִיא בָכֶם רוּחַ וִחְיִיתֶם` (יחזקאל לז:ה)
`וְנָתַתִּי עֲלֵיכֶם גִּדִים וְהַעֲלֵתִי עֲלֵיכֶם בָּשָׂר וְקָרַמְתִּי עֲלֵיכֶם עוֹר וְנָתַתִּי בָכֶם רוּחַ וִחְיִיתֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה` (יחזקאל לז:ו)
`וְרָאִיתִי וְהִנֵּה עֲלֵיהֶם גִּדִים וּבָשָׂר עָלָה וַיִּקְרַם עֲלֵיהֶם עוֹר מִלְמָעְלָה וְרוּחַ אֵין בָּהֶם` (יחזקאל לז:ח)
`וַיֹּאמֶר אֵלַי הִנָּבֵא אֶל הָרוּחַ הִנָּבֵא בֶן אָדָם וְאָמַרְתָּ אֶל הָרוּחַ כֹּה אָמַר אֲדֹנָי יְהוִה מֵאַרְבַּע רוּחוֹת בֹּאִי הָרוּחַ וּפְחִי בַּהֲרוּגִים הָאֵלֶּה וְיִחְיוּ` (יחזקאל לז:ט)
`וְהִנַּבֵּאתִי כַּאֲשֶׁר צִוָּנִי וַתָּבוֹא בָהֶם הָרוּחַ וַיִּחְיוּ וַיַּעַמְדוּ עַל רַגְלֵיהֶם חַיִל גָּדוֹל מְאֹד מְאֹד` (יחזקאל לז:י)
`וְנָתַתִּי רוּחִי בָכֶם וִחְיִיתֶם וְהִנַּחְתִּי אֶתְכֶם עַל אַדְמַתְכֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה דִּבַּרְתִּי וְעָשִׂיתִי נְאֻם יְהוָה` (יחזקאל לז:יד)
`וְלֹא אַסְתִּיר עוֹד פָּנַי מֵהֶם אֲשֶׁר שָׁפַכְתִּי אֶת רוּחִי עַל בֵּית יִשְׂרָאֵל נְאֻם אֲדֹנָי יְהוִה` (יחזקאל לט:כט)
`מָדַד רוּחַ הַקָּדִים בִּקְנֵה הַמִּדָּה חֲמֵשׁ אֵמוֹת קָנִים בִּקְנֵה הַמִּדָּה סָבִיב` (יחזקאל מב:טז)
`מָדַד רוּחַ הַצָּפוֹן חֲמֵשׁ מֵאוֹת קָנִים בִּקְנֵה הַמִּדָּה סָבִיב` (יחזקאל מב:יז)
`אֵת רוּחַ הַדָּרוֹם מָדָד חֲמֵשׁ מֵאוֹת קָנִים בִּקְנֵה הַמִּדָּה` (יחזקאל מב:יח)
`סָבַב אֶל רוּחַ הַיָּם מָדַד חֲמֵשׁ מֵאוֹת קָנִים בִּקְנֵה הַמִּדָּה` (יחזקאל מב:יט)
`לְאַרְבַּע רוּחוֹת מְדָדוֹ חוֹמָה לוֹ סָבִיב סָבִיב אֹרֶךְ חֲמֵשׁ מֵאוֹת וְרֹחַב חֲמֵשׁ מֵאוֹת לְהַבְדִּיל בֵּין הַקֹּדֶשׁ לְחֹל` (יחזקאל מב:כ)
`וַתִּשָּׂאֵנִי רוּחַ וַתְּבִיאֵנִי אֶל הֶחָצֵר הַפְּנִימִי וְהִנֵּה מָלֵא כְבוֹד יְהוָה הַבָּיִת` (יחזקאל מג:ה)
`עַמִּי בְּעֵצוֹ יִשְׁאָל וּמַקְלוֹ יַגִּיד לוֹ כִּי רוּחַ זְנוּנִים הִתְעָה וַיִּזְנוּ מִתַּחַת אֱלֹהֵיהֶם` (הושע ד:יב)
`צָרַר רוּחַ אוֹתָהּ בִּכְנָפֶיהָ וְיֵבֹשׁוּ מִזִּבְחוֹתָם` (הושע ד:יט)
`לֹא יִתְּנוּ מַעַלְלֵיהֶם לָשׁוּב אֶל אֱלֹהֵיהֶם כִּי רוּחַ זְנוּנִים בְּקִרְבָּם וְאֶת יְהוָה לֹא יָדָעוּ` (הושע ה:ד)
`כִּי רוּחַ יִזְרָעוּ וְסוּפָתָה יִקְצֹרוּ קָמָה אֵין לוֹ צֶמַח בְּלִי יַעֲשֶׂה קֶּמַח אוּלַי יַעֲשֶׂה זָרִים יִבְלָעֻהוּ` (הושע ח:ז)
`בָּאוּ יְמֵי הַפְּקֻדָּה בָּאוּ יְמֵי הַשִׁלֻּם יֵדְעוּ יִשְׂרָאֵל אֱוִיל הַנָּבִיא מְשֻׁגָּע אִישׁ הָרוּחַ עַל רֹב עֲוֺנְךָ וְרַבָּה מַשְׂטֵמָה` (הושע ט:ז)
`אֶפְרַיִם רֹעֶה רוּחַ וְרֹדֵף קָדִים כָּל הַיּוֹם כָּזָב וָשֹׁד יַרְבֶּה וּבְרִית עִם אַשּׁוּר יִכְרֹתוּ וְשֶׁמֶן לְמִצְרַיִם יוּבָל` (הושע יב:א)
`כִּי הוּא בֵּן אַחִים יַפְרִיא יָבוֹא קָדִים רוּחַ יְהוָה מִמִּדְבָּר עֹלֶה וְיֵבוֹשׁ מְקוֹרוֹ וְיֶחֱרַב מַעְיָנוֹ הוּא יִשְׁסֶה אוֹצַר כָּל כְּלִי חֶמְדָּה` (הושע יג:טו)
`וְהָיָה אַחֲרֵי כֵן אֶשְׁפּוֹךְ אֶת רוּחִי עַל כָּל בָּשָׂר וְנִבְּאוּ בְּנֵיכֶם וּבְנוֹתֵיכֶם זִקְנֵיכֶם חֲלֹמוֹת יַחֲלֹמוּן בַּחוּרֵיכֶם חֶזְיֹנוֹת יִרְאוּ` (יואל ב:כח)
`וְגַם עַל הָעֲבָדִים וְעַל הַשְּׁפָחוֹת בַּיָּמִים הָהֵמָּה אֶשְׁפּוֹךְ אֶת רוּחִי` (יואל ב:כט)
`כִּי הִנֵּה יוֹצֵר הָרִים וּבֹרֵא רוּחַ וּמַגִּיד לְאָדָם מַה שֵּׂחוֹ עֹשֵׂה שַׁחַר עֵיפָה וְדֹרֵךְ עַל בָּמֳתֵי אָרֶץ יְהוָה אֱלֹהֵי צְבָאוֹת שְׁמוֹ` (עמוס ד:יג)
`וַיהוָה הֵטִיל רוּחַ גְּדוֹלָה אֶל הַיָּם וַיְהִי סַעַר גָּדוֹל בַּיָּם וְהָאֳנִיָּה חִשְּׁבָה לְהִשָּׁבֵר` (יונה א:ד)
`וַיְהִי כִּזְרֹחַ הַשֶּׁמֶשׁ וַיְמַן אֱלֹהִים רוּחַ קָדִים חֲרִישִׁית וַתַּךְ הַשֶּׁמֶשׁ עַל רֹאשׁ יוֹנָה וַיִּתְעַלָּף וַיִּשְׁאַל אֶת נַפְשׁוֹ לָמוּת וַיֹּאמֶר טוֹב מוֹתִי מֵחַיָּי` (יונה ד:ח)
`הֶאָמוּר בֵּית יַעֲקֹב הֲקָצַר רוּחַ יְהוָה אִם אֵלֶּה מַעֲלָלָיו הֲלוֹא דְבָרַי יֵיטִיבוּ עִם הַיָּשָׁר הוֹלֵךְ` (מיכה ב:ז)
`לוּ אִישׁ הֹלֵךְ רוּחַ וָשֶׁקֶר כִּזֵּב אַטִּף לְךָ לַיַּיִן וְלַשֵּׁכָר וְהָיָה מַטִּיף הָעָם הַזֶּה` (מיכה ב:יא)
`וְאוּלָם אָנֹכִי מָלֵאתִי כֹחַ אֶת רוּחַ יְהוָה וּמִשְׁפָּט וּגְבוּרָה לְהַגִּיד לְיַעֲקֹב פִּשְׁעוֹ וּלְיִשְׂרָאֵל חַטָּאתוֹ` (מיכה ג:ח)
`אָז חָלַף רוּחַ וַיַּעֲבֹר וְאָשֵׁם זוּ כֹחוֹ לֵאלֹהוֹ` (חבקוק א:יא)
`הוֹי אֹמֵר לָעֵץ הָקִיצָה עוּרִי לְאֶבֶן דּוּמָם הוּא יוֹרֶה הִנֵּה הוּא תָּפוּשׂ זָהָב וָכֶסֶף וְכָל רוּחַ אֵין בְּקִרְבּוֹ` (חבקוק ב:יט)
`וַיָּעַר יְהוָה אֶת רוּחַ זְרֻבָּבֶל בֶּן שַׁלְתִּיאֵל פַּחַת יְהוּדָה וְאֶת רוּחַ יְהוֹשֻׁעַ בֶּן יְהוֹצָדָק הַכֹּהֵן הַגָּדוֹל וְאֶת רוּחַ כֹּל שְׁאֵרִית הָעָם וַיָּבֹאוּ וַיַּעֲשׂוּ מְלָאכָה בְּבֵית יְהוָה צְבָאוֹת אֱלֹהֵיהֶם` (חגי א:יד)
`אֶת הַדָּבָר אֲשֶׁר כָּרַתִּי אִתְּכֶם בְּצֵאתְכֶם מִמִּצְרַיִם וְרוּחִי עֹמֶדֶת בְּתוֹכְכֶם אַל תִּירָאוּ` (חגי ב:ה)
`הוֹי הוֹי וְנֻסוּ מֵאֶרֶץ צָפוֹן נְאֻם יְהוָה כִּי כְּאַרְבַּע רוּחוֹת הַשָּׁמַיִם פֵּרַשְׂתִּי אֶתְכֶם נְאֻם יְהוָה` (זכריה ב:ו)
`וַיַּעַן וַיֹּאמֶר אֵלַי לֵאמֹר זֶה דְּבַר יְהוָה אֶל זְרֻבָּבֶל לֵאמֹר לֹא בְחַיִל וְלֹא בְכֹחַ כִּי אִם בְּרוּחִי אָמַר יְהוָה צְבָאוֹת` (זכריה ד:ו)
`וָאֶשָּׂא עֵינַי וָאֵרֶא וְהִנֵּה שְׁתַּיִם נָשִׁים יוֹצְאוֹת וְרוּחַ בְּכַנְפֵיהֶם וְלָהֵנָּה כְנָפַיִם כְּכַנְפֵי הַחֲסִידָה וַתִּשֶּׂאנָה אֶת הָאֵיפָה בֵּין הָאָרֶץ וּבֵין הַשָּׁמָיִם` (זכריה ה:ט)
`וַיַּעַן הַמַּלְאָךְ וַיֹּאמֶר אֵלָי אֵלֶּה אַרְבַּע רֻחוֹת הַשָּׁמַיִם יוֹצְאוֹת מֵהִתְיַצֵּב עַל אֲדוֹן כָּל הָאָרֶץ` (זכריה ו:ה)
`וַיַּזְעֵק אֹתִי וַיְדַבֵּר אֵלַי לֵאמֹר רְאֵה הַיּוֹצְאִים אֶל אֶרֶץ צָפוֹן הֵנִיחוּ אֶת רוּחִי בְּאֶרֶץ צָפוֹן` (זכריה ו:ח)
`וְלִבָּם שָׂמוּ שָׁמִיר מִשְּׁמוֹעַ אֶת הַתּוֹרָה וְאֶת הַדְּבָרִים אֲשֶׁר שָׁלַח יְהוָה צְבָאוֹת בְּרוּחוֹ בְּיַד הַנְּבִיאִים הָרִאשֹׁנִים וַיְהִי קֶצֶף גָּדוֹל מֵאֵת יְהוָה צְבָאוֹת` (זכריה ז:יב)
`מַשָּׂא דְבַר יְהוָה עַל יִשְׂרָאֵל נְאֻם יְהוָה נֹטֶה שָׁמַיִם וְיֹסֵד אָרֶץ וְיֹצֵר רוּחַ אָדָם בְּקִרְבּוֹ` (זכריה יב:א)
`וְשָׁפַכְתִּי עַל בֵּית דָּוִיד וְעַל יוֹשֵׁב יְרוּשָׁלִַם רוּחַ חֵן וְתַחֲנוּנִים וְהִבִּיטוּ אֵלַי אֵת אֲשֶׁר דָּקָרוּ וְסָפְדוּ עָלָיו כְּמִסְפֵּד עַל הַיָּחִיד וְהָמֵר עָלָיו כְּהָמֵר עַל הַבְּכוֹר` (זכריה יב:י)
`וְהָיָה בַיּוֹם הַהוּא נְאֻם יְהוָה צְבָאוֹת אַכְרִית אֶת שְׁמוֹת הָעֲצַבִּים מִן הָאָרֶץ וְלֹא יִזָּכְרוּ עוֹד וְגַם אֶת הַנְּבִיאִים וְאֶת רוּחַ הַטֻּמְאָה אַעֲבִיר מִן הָאָרֶץ` (זכריה יג:ב)
`וְלֹא אֶחָד עָשָׂה וּשְׁאָר רוּחַ לוֹ וּמָה הָאֶחָד מְבַקֵּשׁ זֶרַע אֱלֹהִים וְנִשְׁמַרְתֶּם בְּרוּחֲכֶם וּבְאֵשֶׁת נְעוּרֶיךָ אַל יִבְגֹּד` (מלאכי ב:טו)
`כִּי שָׂנֵא שַׁלַּח אָמַר יְהוָה אֱלֹהֵי יִשְׂרָאֵל וְכִסָּה חָמָס עַל לְבוּשׁוֹ אָמַר יְהוָה צְבָאוֹת וְנִשְׁמַרְתֶּם בְּרוּחֲכֶם וְלֹא תִבְגֹּדוּ` (מלאכי ב:טז)
`וַיָּעַר אֱלֹהֵי יִשְׂרָאֵל אֶת רוּחַ פּוּל מֶלֶךְ אַשּׁוּר וְאֶת רוּחַ תִּלְּגַת פִּלְנֶסֶר מֶלֶךְ אַשּׁוּר וַיַּגְלֵם לָראוּבֵנִי וְלַגָּדִי וְלַחֲצִי שֵׁבֶט מְנַשֶּׁה וַיְבִיאֵם לַחְלַח וְחָבוֹר וְהָרָא וּנְהַר גּוֹזָן עַד הַיּוֹם הַזֶּה` (דברי-הימים-א ה:כו)
`לְאַרְבַּע רוּחוֹת יִהְיוּ הַשֹּׁעֲרִים מִזְרָח יָמָּה צָפוֹנָה וָנֶגְבָּה` (דברי-הימים-א ט:כד)
`וְרוּחַ לָבְשָׁה אֶת עֲמָשַׂי רֹאשׁ הַשְּׁלוֹשִׁים לְךָ דָוִיד וְעִמְּךָ בֶן יִשַׁי שָׁלוֹם שָׁלוֹם לְךָ וְשָׁלוֹם לְעֹזְרֶךָ כִּי עֲזָרְךָ אֱלֹהֶיךָ וַיְקַבְּלֵם דָּוִיד וַיִּתְּנֵם בְּרָאשֵׁי הַגְּדוּד` (דברי-הימים-א יב:יח)
`וְתַבְנִית כֹּל אֲשֶׁר הָיָה בָרוּחַ עִמּוֹ לְחַצְרוֹת בֵּית יְהוָה וּלְכָל הַלְּשָׁכוֹת סָבִיב לְאֹצְרוֹת בֵּית הָאֱלֹהִים וּלְאֹצְרוֹת הַקֳּדָשִׁים` (דברי-הימים-א כח:יב)
`וּמַאֲכַל שֻׁלְחָנוֹ וּמוֹשַׁב עֲבָדָיו וּמַעֲמַד מְשָׁרְתָיו וּמַלְבּוּשֵׁיהֶם וּמַשְׁקָיו וּמַלְבּוּשֵׁיהֶם וַעֲלִיָּתוֹ אֲשֶׁר יַעֲלֶה בֵּית יְהוָה וְלֹא הָיָה עוֹד בָּהּ רוּחַ` (דברי-הימים-ב ט:ד)
`וַעֲזַרְיָהוּ בֶּן עוֹדֵד הָיְתָה עָלָיו רוּחַ אֱלֹהִים` (דברי-הימים-ב טו:א)
`וַיֵּצֵא הָרוּחַ וַיַּעֲמֹד לִפְנֵי יְהוָה וַיֹּאמֶר אֲנִי אֲפַתֶּנּוּ וַיֹּאמֶר יְהוָה אֵלָיו בַּמָּה` (דברי-הימים-ב יח:כ)
`וַיֹּאמֶר אֵצֵא וְהָיִיתִי לְרוּחַ שֶׁקֶר בְּפִי כָּל נְבִיאָיו וַיֹּאמֶר תְּפַתֶּה וְגַם תּוּכָל צֵא וַעֲשֵׂה כֵן` (דברי-הימים-ב יח:כא)
`וְעַתָּה הִנֵּה נָתַן יְהוָה רוּחַ שֶׁקֶר בְּפִי נְבִיאֶיךָ אֵלֶּה וַיהוָה דִּבֶּר עָלֶיךָ רָעָה` (דברי-הימים-ב יח:כב)
`וַיִּגַּשׁ צִדְקִיָּהוּ בֶן כְּנַעֲנָה וַיַּךְ אֶת מִיכָיְהוּ עַל הַלֶּחִי וַיֹּאמֶר אֵי זֶה הַדֶּרֶךְ עָבַר רוּחַ יְהוָה מֵאִתִּי לְדַבֵּר אֹתָךְ` (דברי-הימים-ב יח:כג)
`וְיַחֲזִיאֵל בֶּן זְכַרְיָהוּ בֶּן בְּנָיָה בֶּן יְעִיאֵל בֶּן מַתַּנְיָה הַלֵּוִי מִן בְּנֵי אָסָף הָיְתָה עָלָיו רוּחַ יְהוָה בְּתוֹךְ הַקָּהָל` (דברי-הימים-ב כ:יד)
`וַיָּעַר יְהוָה עַל יְהוֹרָם אֵת רוּחַ הַפְּלִשְׁתִּים וְהָעַרְבִים אֲשֶׁר עַל יַד כּוּשִׁים` (דברי-הימים-ב כא:טז)
`וְרוּחַ אֱלֹהִים לָבְשָׁה אֶת זְכַרְיָה בֶּן יְהוֹיָדָע הַכֹּהֵן וַיַּעֲמֹד מֵעַל לָעָם וַיֹּאמֶר לָהֶם כֹּה אָמַר הָאֱלֹהִים לָמָה אַתֶּם עֹבְרִים אֶת מִצְוֺת יְהוָה וְלֹא תַצְלִיחוּ כִּי עֲזַבְתֶּם אֶת יְהוָה וַיַּעֲזֹב אֶתְכֶם` (דברי-הימים-ב כד:כ)
`וּבִשְׁנַת אַחַת לְכוֹרֶשׁ מֶלֶךְ פָּרַס לִכְלוֹת דְּבַר יְהוָה בְּפִי יִרְמְיָהוּ הֵעִיר יְהוָה אֶת רוּחַ כּוֹרֶשׁ מֶלֶךְ פָּרַס וַיַּעֲבֶר קוֹל בְּכָל מַלְכוּתוֹ וְגַם בְּמִכְתָּב לֵאמֹר` (דברי-הימים-ב לו:כב)
`לֹא כֵן הָרְשָׁעִים כִּי אִם כַּמֹּץ אֲ‍שֶׁר תִּדְּפֶנּוּ רוּחַ` (תהילים א:ד)
`יַמְטֵר עַל רְשָׁעִים פַּחִים אֵשׁ וְגָפְרִית וְרוּחַ זִלְעָפוֹת מְנָת כּוֹסָם` (תהילים יא:ו)
`וַיִּרְכַּב עַל כְּרוּב וַיָּעֹף וַיֵּדֶא עַל כַּנְפֵי רוּחַ` (תהילים יח:י)
`וַיֵּרָאוּ אֲפִיקֵי מַיִם וַיִּגָּלוּ מוֹסְדוֹת תֵּבֵל מִגַּעֲרָתְךָ יְהוָה מִנִּשְׁמַת רוּחַ אַפֶּךָ` (תהילים יח:טו)
`וְאֶשְׁחָקֵם כְּעָפָר עַל פְּנֵי רוּחַ כְּטִיט חוּצוֹת אֲרִיקֵם` (תהילים יח:מב)
`בְּיָדְךָ אַפְקִיד רוּחִי פָּדִיתָה אוֹתִי יְהוָה אֵל אֱמֶת` (תהילים לא:ה)
`אַשְׁרֵי אָדָם לֹא יַחְשֹׁב יְהוָה לוֹ עָוֺן וְאֵין בְּרוּחוֹ רְמִיָּה` (תהילים לב:ב)
`בִּדְבַר יְהוָה שָׁמַיִם נַעֲשׂוּ וּבְרוּחַ פִּיו כָּל צְבָאָם` (תהילים לג:ו)
`קָרוֹב יְהוָה לְנִשְׁבְּרֵי לֵב וְאֶת דַּכְּאֵי רוּחַ יוֹשִׁיעַ` (תהילים לד:יח)
`יִהְיוּ כְּמֹץ לִפְנֵי רוּחַ וּמַלְאַךְ יְהוָה דּוֹחֶה` (תהילים לה:ה)
`בְּרוּחַ קָדִים תְּשַׁבֵּר אֳנִיּוֹת תַּרְשִׁישׁ` (תהילים מח:ז)
`לֵב טָהוֹר בְּרָא לִי אֱלֹהִים וְרוּחַ נָכוֹן חַדֵּשׁ בְּקִרְבִּי` (תהילים נא:י)
`אַל תַּשְׁלִיכֵנִי מִלְּפָנֶיךָ וְרוּחַ קָדְשְׁךָ אַל תִּקַּח מִמֶּנִּי` (תהילים נא:יא)
`הָשִׁיבָה לִּי שְׂשׂוֹן יִשְׁעֶךָ וְרוּחַ נְדִיבָה תִסְמְכֵנִי` (תהילים נא:יב)
`זִבְחֵי אֱלֹהִים רוּחַ נִשְׁבָּרָה לֵב נִשְׁבָּר וְנִדְכֶּה אֱלֹהִים לֹא תִבְזֶה` (תהילים נא:יז)
`אָחִישָׁה מִפְלָט לִי מֵרוּחַ סֹעָה מִסָּעַר` (תהילים נה:ח)
`יִבְצֹר רוּחַ נְגִידִים נוֹרָא לְמַלְכֵי אָרֶץ` (תהילים עו:יב)
`אֶזְכְּרָה אֱלֹהִים וְאֶהֱמָיָה אָשִׂיחָה וְתִתְעַטֵּף רוּחִי סֶלָה` (תהילים עז:ג)
`אֶזְכְּרָה נְגִינָתִי בַּלָּיְלָה עִם לְבָבִי אָשִׂיחָה וַיְחַפֵּשׂ רוּחִי` (תהילים עז:ו)
`וְלֹא יִהְיוּ כַּאֲבוֹתָם דּוֹר סוֹרֵר וּמֹרֶה דּוֹר לֹא הֵכִין לִבּוֹ וְלֹא נֶאֶמְנָה אֶת אֵל רוּחוֹ` (תהילים עח:ח)
`וַיִּזְכֹּר כִּי בָשָׂר הֵמָּה רוּחַ הוֹלֵךְ וְלֹא יָשׁוּב` (תהילים עח:לט)
`אֱ‍לֹהַי שִׁיתֵמוֹ כַגַּלְגַּל כְּקַשׁ לִפְנֵי רוּחַ` (תהילים פג:יג)
`כִּי רוּחַ עָבְרָה בּוֹ וְאֵינֶנּוּ וְלֹא יַכִּירֶנּוּ עוֹד מְקוֹמוֹ` (תהילים קג:טז)
`הַמְקָרֶה בַמַּיִם עֲ‍לִיּוֹתָיו הַשָּׂם עָבִים רְכוּבוֹ הַמְהַלֵּךְ עַל כַּנְפֵי רוּחַ` (תהילים קד:ג)
`עֹשֶׂה מַלְאָכָיו רוּחוֹת מְשָׁרְתָיו אֵשׁ לֹהֵט` (תהילים קד:ד)
`תַּסְתִּיר פָּנֶיךָ יִבָּהֵלוּן תֹּסֵף רוּחָם יִגְוָעוּן וְאֶל עֲפָרָם יְשׁוּבוּן` (תהילים קד:כט)
`תְּשַׁלַּח רוּחֲךָ יִבָּרֵאוּן וּתְחַדֵּשׁ פְּנֵי אֲדָמָה` (תהילים קד:ל)
`כִּי הִמְרוּ אֶת רוּחוֹ וַיְבַטֵּא בִּשְׂפָתָיו` (תהילים קו:לג)
`וַיֹּאמֶר וַיַּעֲמֵד רוּחַ סְעָרָה וַתְּרוֹמֵם גַּלָּיו` (תהילים קז:כה)
`מַעֲלֶה נְשִׂאִים מִקְצֵה הָאָרֶץ בְּרָקִים לַמָּטָר עָשָׂה מוֹצֵא רוּחַ מֵאוֹצְרוֹתָיו` (תהילים קלה:ז)
`אָזְנַיִם לָהֶם וְלֹא יַאֲזִינוּ אַף אֵין יֶשׁ רוּחַ בְּפִיהֶם` (תהילים קלה:יז)
`אָנָה אֵלֵךְ מֵרוּחֶךָ וְאָנָה מִפָּנֶיךָ אֶבְרָח` (תהילים קלט:ז)
`בְּהִתְעַטֵּף עָלַי רוּחִי וְאַתָּה יָדַעְתָּ נְתִיבָתִי בְּאֹרַח זוּ אֲהַלֵּךְ טָמְנוּ פַח לִי` (תהילים קמב:ג)
`וַתִּתְעַטֵּף עָלַי רוּחִי בְּתוֹכִי יִשְׁתּוֹמֵם לִבִּי` (תהילים קמג:ד)
`מַהֵר עֲנֵנִי יְהוָה כָּלְתָה רוּחִי אַל תַּסְתֵּר פָּנֶיךָ מִמֶּנִּי וְנִמְשַׁלְתִּי עִם יֹרְדֵי בוֹר` (תהילים קמג:ז)
`לַמְּדֵנִי לַעֲשׂוֹת רְצוֹנֶךָ כִּי אַתָּה אֱלוֹהָי רוּחֲךָ טוֹבָה תַּנְחֵנִי בְּאֶרֶץ מִישׁוֹר` (תהילים קמג:י)
`תֵּצֵא רוּחוֹ יָשֻׁב לְאַדְמָתוֹ בַּיּוֹם הַהוּא אָבְדוּ עֶשְׁתֹּנֹתָיו` (תהילים קמו:ד)
`יִשְׁלַח דְּבָרוֹ וְיַמְסֵם יַשֵּׁב רוּחוֹ יִזְּלוּ מָיִם` (תהילים קמז:יח)
`אֵשׁ וּבָרָד שֶׁלֶג וְקִיטוֹר רוּחַ סְעָרָה עֹשָׂה דְבָרוֹ` (תהילים קמח:ח)
`וְהִנֵּה רוּחַ גְּדוֹלָה בָּאָה מֵעֵבֶר הַמִּדְבָּר וַיִּגַּע בְּאַרְבַּע פִּנּוֹת הַבַּיִת וַיִּפֹּל עַל הַנְּעָרִים וַיָּמוּתוּ וָאִמָּלְטָה רַק אֲנִי לְבַדִּי לְהַגִּיד לָךְ` (איוב א:יט)
`מִנִּשְׁמַת אֱלוֹהַ יֹאבֵדוּ וּמֵרוּחַ אַפּוֹ יִכְלוּ` (איוב ד:ט)
`וְרוּחַ עַל פָּנַי יַחֲלֹף תְּסַמֵּר שַׂעֲרַת בְּשָׂרִי` (איוב ד:טו)
`כִּי חִצֵּי שַׁדַּי עִמָּדִי אֲשֶׁר חֲמָתָם שֹׁתָה רוּחִי בִּעוּתֵי אֱלוֹהַּ יַעַרְכוּנִי` (איוב ו:ד)
`הַלְהוֹכַח מִלִּים תַּחְשֹׁבוּ וּלְרוּחַ אִמְרֵי נֹאָשׁ` (איוב ו:כו)
`זְכֹר כִּי רוּחַ חַיָּי לֹא תָשׁוּב עֵינִי לִרְאוֹת טוֹב` (איוב ז:ז)
`גַּם אֲנִי לֹא אֶחֱשָׂךְ פִּי אֲ‍דַבְּרָה בְּצַר רוּחִי אָשִׂיחָה בְּמַר נַפְשִׁי` (איוב ז:יא)
`עַד אָן תְּמַלֶּל אֵלֶּה וְרוּחַ כַּבִּיר אִמְרֵי פִיךָ` (איוב ח:ב)
`לֹא יִתְּנֵנִי הָשֵׁב רוּחִי כִּי יַשְׂבִּעַנִי מַמְּרֹרִים` (איוב ט:יח)
`חַיִּים וָחֶסֶד עָשִׂיתָ עִמָּדִי וּפְקֻדָּתְךָ שָׁמְרָה רוּחִי` (איוב י:יב)
`אֲשֶׁר בְּיָדוֹ נֶפֶשׁ כָּל חָי וְרוּחַ כָּל בְּשַׂר אִישׁ` (איוב יב:י)
`הֶחָכָם יַעֲנֶה דַעַת רוּחַ וִימַלֵּא קָדִים בִּטְנוֹ` (איוב טו:ב)
`כִּי תָשִׁיב אֶל אֵל רוּחֶךָ וְהֹצֵאתָ מִפִּיךָ מִלִּין` (איוב טו:יג)
`לֹא יָסוּר מִנִּי חֹשֶׁךְ יֹנַקְתּוֹ תְּיַבֵּשׁ שַׁלְהָבֶת וְיָסוּר בְּרוּחַ פִּיו` (איוב טו:ל)
`הֲקֵץ לְדִבְרֵי רוּחַ אוֹ מַה יַּמְרִיצְךָ כִּי תַעֲנֶה` (איוב טז:ג)
`רוּחִי חֻבָּלָה יָמַי נִזְעָכוּ קְבָרִים לִי` (איוב יז:א)
`רוּחִי זָרָה לְאִשְׁתִּי וְחַנֹּתִי לִבְנֵי בִטְנִי` (איוב יט:יז)
`מוּסַר כְּלִמָּתִי אֶשְׁמָע וְרוּחַ מִבִּינָתִי יַעֲנֵנִי` (איוב כ:ג)
`הֶאָנֹכִי לְאָדָם שִׂיחִי וְאִם מַדּוּעַ לֹא תִקְצַר רוּחִי` (איוב כא:ד)
`יִהְיוּ כְּתֶבֶן לִפְנֵי רוּחַ וּכְמֹץ גְּנָבַתּוּ סוּפָה` (איוב כא:יח)
`בְּרוּחוֹ שָׁמַיִם שִׁפְרָה חֹלֲלָה יָדוֹ נָחָשׁ בָּרִיחַ` (איוב כו:יג)
`כִּי כָל עוֹד נִשְׁמָתִי בִי וְרוּחַ אֱלוֹהַּ בְּאַפִּי` (איוב כז:ג)
`לַעֲשׂוֹת לָרוּחַ מִשְׁקָל וּמַיִם תִּכֵּן בְּמִדָּה` (איוב כח:כה)
`הָהְפַּךְ עָלַי בַּלָּהוֹת תִּרְדֹּף כָּרוּחַ נְדִבָתִי וּכְעָב עָבְרָה יְשֻׁעָתִי` (איוב ל:טו)
`תִּשָּׂאֵנִי אֶל רוּחַ תַּרְכִּיבֵנִי וּתְמֹגְגֵנִי תְּשֻׁוֶּה` (איוב ל:כב)
`אָכֵן רוּחַ הִיא בֶאֱנוֹשׁ וְנִשְׁמַת שַׁדַּי תְּבִינֵם` (איוב לב:ח)
`כִּי מָלֵתִי מִלִּים הֱצִיקַתְנִי רוּחַ בִּטְנִי` (איוב לב:יח)
`רוּחַ אֵל עָשָׂתְנִי וְנִשְׁמַת שַׁדַּי תְּחַיֵּנִי` (איוב לג:ד)
`אִם יָשִׂים אֵלָיו לִבּוֹ רוּחוֹ וְנִשְׁמָתוֹ אֵלָיו יֶאֱסֹף` (איוב לד:יד)
`וְעַתָּה לֹא רָאוּ אוֹר בָּהִיר הוּא בַּשְּׁחָקִים וְרוּחַ עָבְרָה וַתְּטַהֲרֵם` (איוב לז:כא)
`אֶחָד בְּאֶחָד יִגַּשׁוּ וְרוּחַ לֹא יָבוֹא בֵינֵיהֶם` (איוב מא:טז)
`תָּשׁוּבוּ לְתוֹכַחְתִּי הִנֵּה אַבִּיעָה לָכֶם רוּחִי אוֹדִיעָה דְבָרַי אֶתְכֶם` (משלי א:כג)
`הוֹלֵךְ רָכִיל מְגַלֶּה סּוֹד וְנֶאֱמַן רוּחַ מְכַסֶּה דָבָר` (משלי יא:יג)
`עוֹכֵר בֵּיתוֹ יִנְחַל רוּחַ וְעֶבֶד אֱוִיל לַחֲכַם לֵב` (משלי יא:כט)
`אֶרֶךְ אַפַּיִם רַב תְּבוּנָה וּקְצַר רוּחַ מֵרִים אִוֶּלֶת` (משלי יד:כט)
`מַרְפֵּא לָשׁוֹן עֵץ חַיִּים וְסֶלֶף בָּהּ שֶׁבֶר בְּרוּחַ` (משלי טו:ד)
`לֵב שָׂמֵחַ יֵיטִב פָּנִים וּבְעַצְּבַת לֵב רוּחַ נְכֵאָה` (משלי טו:יג)
`כָּל דַּרְכֵי אִישׁ זַךְ בְּעֵינָיו וְתֹכֵן רוּחוֹת יְהוָה` (משלי טז:ב)
`לִפְנֵי שֶׁבֶר גָּאוֹן וְלִפְנֵי כִשָּׁלוֹן גֹּבַהּ רוּחַ` (משלי טז:יח)
`טוֹב שְׁפַל רוּחַ אֶת עֲנִיִּים מֵחַלֵּק שָׁלָל אֶת גֵּאִים` (משלי טז:יט)
`טוֹב אֶרֶךְ אַפַּיִם מִגִּבּוֹר וּמֹשֵׁל בְּרוּחוֹ מִלֹּכֵד עִיר` (משלי טז:לב)
`לֵב שָׂמֵחַ יֵיטִב גֵּהָה וְרוּחַ נְכֵאָה תְּיַבֶּשׁ גָּרֶם` (משלי יז:כב)
`חוֹשֵׂךְ אֲמָרָיו יוֹדֵעַ דָּעַת וְקַר רוּחַ אִישׁ תְּבוּנָה` (משלי יז:כז)
`רוּחַ אִישׁ יְכַלְכֵּל מַחֲלֵהוּ וְרוּחַ נְכֵאָה מִי יִשָּׂאֶנָּה` (משלי יח:יד)
`נְשִׂיאִים וְרוּחַ וְגֶשֶׁם אָיִן אִישׁ מִתְהַלֵּל בְּמַתַּת שָׁקֶר` (משלי כה:יד)
`רוּחַ צָפוֹן תְּחוֹלֵל גָּשֶׁם וּפָנִים נִזְעָמִים לְשׁוֹן סָתֶר` (משלי כה:כג)
`עִיר פְּרוּצָה אֵין חוֹמָה אִישׁ אֲשֶׁר אֵין מַעְצָר לְרוּחוֹ` (משלי כה:כח)
`צֹפְנֶיהָ צָפַן רוּחַ וְשֶׁמֶן יְמִינוֹ יִקְרָא` (משלי כז:טז)
`כָּל רוּחוֹ יוֹצִיא כְסִיל וְחָכָם בְּאָחוֹר יְשַׁבְּחֶנָּה` (משלי כט:יא)
`גַּאֲוַת אָדָם תַּשְׁפִּילֶנּוּ וּשְׁפַל רוּחַ יִתְמֹךְ כָּבוֹד` (משלי כט:כג)
`מִי עָלָה שָׁמַיִם וַיֵּרַד מִי אָסַף רוּחַ בְּחָפְנָיו מִי צָרַר מַיִם בַּשִּׂמְלָה מִי הֵקִים כָּל אַפְסֵי אָרֶץ מַה שְּׁמוֹ וּמַה שֶּׁם בְּנוֹ כִּי תֵדָע` (משלי ל:ד)
`הוֹלֵךְ אֶל דָּרוֹם וְסוֹבֵב אֶל צָפוֹן סוֹבֵב סֹבֵב הוֹלֵךְ הָרוּחַ וְעַל סְבִיבֹתָיו שָׁב הָרוּחַ` (קהלת א:ו)
`רָאִיתִי אֶת כָּל הַמַּעֲשִׂים שֶׁנַּעֲשׂוּ תַּחַת הַשָּׁמֶשׁ וְהִנֵּה הַכֹּל הֶבֶל וּרְעוּת רוּחַ` (קהלת א:יד)
`וָאֶתְּנָה לִבִּי לָדַעַת חָכְמָה וְדַעַת הוֹלֵלוֹת וְשִׂכְלוּת יָדַעְתִּי שֶׁגַּם זֶה הוּא רַעְיוֹן רוּחַ` (קהלת א:יז)
`וּפָנִיתִי אֲנִי בְּכָל מַעֲשַׂי שֶׁעָשׂוּ יָדַי וּבֶעָמָל שֶׁעָמַלְתִּי לַעֲשׂוֹת וְהִנֵּה הַכֹּל הֶבֶל וּרְעוּת רוּחַ וְאֵין יִתְרוֹן תַּחַת הַשָּׁמֶשׁ` (קהלת ב:יא)
`וְשָׂנֵאתִי אֶת הַחַיִּים כִּי רַע עָלַי הַמַּעֲשֶׂה שֶׁנַּעֲשָׂה תַּחַת הַשָּׁמֶשׁ כִּי הַכֹּל הֶבֶל וּרְעוּת רוּחַ` (קהלת ב:יז)
`כִּי לְאָדָם שֶׁטּוֹב לְפָנָיו נָתַן חָכְמָה וְדַעַת וְשִׂמְחָה וְלַחוֹטֶא נָתַן עִנְיָן לֶאֱסוֹף וְלִכְנוֹס לָתֵת לְטוֹב לִפְנֵי הָאֱלֹהִים גַּם זֶה הֶבֶל וּרְעוּת רוּחַ` (קהלת ב:כו)
`כִּי מִקְרֶה בְנֵי הָאָדָם וּמִקְרֶה הַבְּהֵמָה וּמִקְרֶה אֶחָד לָהֶם כְּמוֹת זֶה כֵּן מוֹת זֶה וְרוּחַ אֶחָד לַכֹּל וּמוֹתַר הָאָדָם מִן הַבְּהֵמָה אָיִן כִּי הַכֹּל הָבֶל` (קהלת ג:יט)
`מִי יוֹדֵעַ רוּחַ בְּנֵי הָאָדָם הָעֹלָה הִיא לְמָעְלָה וְרוּחַ הַבְּהֵמָה הַיֹּרֶדֶת הִיא לְמַטָּה לָאָרֶץ` (קהלת ג:כא)
`וְרָאִיתִי אֲנִי אֶת כָּל עָמָל וְאֵת כָּל כִּשְׁרוֹן הַמַּעֲשֶׂה כִּי הִיא קִנְאַת אִישׁ מֵרֵעֵהוּ גַּם זֶה הֶבֶל וּרְעוּת רוּחַ` (קהלת ד:ד)
`טוֹב מְלֹא כַף נָחַת מִמְּלֹא חָפְנַיִם עָמָל וּרְעוּת רוּחַ` (קהלת ד:ו)
`אֵין קֵץ לְכָל הָעָם לְכֹל אֲשֶׁר הָיָה לִפְנֵיהֶם גַּם הָאַחֲרוֹנִים לֹא יִשְׂמְחוּ בוֹ כִּי גַם זֶה הֶבֶל וְרַעְיוֹן רוּחַ` (קהלת ד:טז)
`וְגַם זֹה רָעָה חוֹלָה כָּל עֻמַּת שֶׁבָּא כֵּן יֵלֵךְ וּמַה יִּתְרוֹן לוֹ שֶׁיַּעֲמֹל לָרוּחַ` (קהלת ה:טז)
`טוֹב מַרְאֵה עֵינַיִם מֵהֲלָךְ נָפֶשׁ גַּם זֶה הֶבֶל וּרְעוּת רוּחַ` (קהלת ו:ט)
`טוֹב אַחֲרִית דָּבָר מֵרֵאשִׁיתוֹ טוֹב אֶרֶךְ רוּחַ מִגְּבַהּ רוּחַ` (קהלת ז:ח)
`אַל תְּבַהֵל בְּרוּחֲךָ לִכְעוֹס כִּי כַעַס בְּחֵיק כְּסִילִים יָנוּחַ` (קהלת ז:ט)
`אֵין אָדָם שַׁלִּיט בָּרוּחַ לִכְלוֹא אֶת הָרוּחַ וְאֵין שִׁלְטוֹן בְּיוֹם הַמָּוֶת וְאֵין מִשְׁלַחַת בַּמִּלְחָמָה וְלֹא יְמַלֵּט רֶשַׁע אֶת בְּעָלָיו` (קהלת ח:ח)
`אִם רוּחַ הַמּוֹשֵׁל תַּעֲלֶה עָלֶיךָ מְקוֹמְךָ אַל תַּנַּח כִּי מַרְפֵּא יַנִּיחַ חֲטָאִים גְּדוֹלִים` (קהלת י:ד)
`שֹׁמֵר רוּחַ לֹא יִזְרָע וְרֹאֶה בֶעָבִים לֹא יִקְצוֹר` (קהלת יא:ד)
`כַּאֲשֶׁר אֵינְךָ יוֹדֵעַ מַה דֶּרֶךְ הָרוּחַ כַּעֲצָמִים בְּבֶטֶן הַמְּלֵאָה כָּכָה לֹא תֵדַע אֶת מַעֲשֵׂה הָאֱלֹהִים אֲשֶׁר יַעֲשֶׂה אֶת הַכֹּל` (קהלת יא:ה)
`וְיָשֹׁב הֶעָפָר עַל הָאָרֶץ כְּשֶׁהָיָה וְהָרוּחַ תָּשׁוּב אֶל הָאֱלֹהִים אֲשֶׁר נְתָנָהּ` (קהלת יב:ז)
`רוּחַ אַפֵּינוּ מְשִׁיחַ יְהוָה נִלְכַּד בִּשְׁחִיתוֹתָם אֲשֶׁר אָמַרְנוּ בְּצִלּוֹ נִחְיֶה בַגּוֹיִם` (איכה ד:כ)
`וּבִשְׁנַת שְׁתַּיִם לְמַלְכוּת נְבֻכַדְנֶצַּר חָלַם נְבֻכַדְנֶצַּר חֲלֹמוֹת וַתִּתְפָּעֶם רוּחוֹ וּשְׁנָתוֹ נִהְיְתָה עָלָיו` (דניאל ב:א)
`וַיֹּאמֶר לָהֶם הַמֶּלֶךְ חֲלוֹם חָלָמְתִּי וַתִּפָּעֶם רוּחִי לָדַעַת אֶת הַחֲלוֹם` (דניאל ב:ג)
`וּצְפִיר הָעִזִּים הִגְדִּיל עַד מְאֹד וּכְעָצְמוֹ נִשְׁבְּרָה הַקֶּרֶן הַגְּדוֹלָה וַתַּעֲלֶנָה חָזוּת אַרְבַּע תַּחְתֶּיהָ לְאַרְבַּע רוּחוֹת הַשָּׁמָיִם` (דניאל ח:ח)
`וּכְעָמְדוֹ תִּשָּׁבֵר מַלְכוּתוֹ וְתֵחָץ לְאַרְבַּע רוּחוֹת הַשָּׁמָיִם וְלֹא לְאַחֲרִיתוֹ וְלֹא כְמָשְׁלוֹ אֲשֶׁר מָשָׁל כִּי תִנָּתֵשׁ מַלְכוּתוֹ וְלַאֲחֵרִים מִלְּבַד אֵלֶּה` (דניאל יא:ד)
`וּבִשְׁנַת אַחַת לְכוֹרֶשׁ מֶלֶךְ פָּרַס לִכְלוֹת דְּבַר יְהוָה מִפִּי יִרְמְיָה הֵעִיר יְהוָה אֶת רוּחַ כֹּרֶשׁ מֶלֶךְ פָּרַס וַיַּעֲבֶר קוֹל בְּכָל מַלְכוּתוֹ וְגַם בְּמִכְתָּב לֵאמֹר` (עזרא א:א)
`וַיָּקוּמוּ רָאשֵׁי הָאָבוֹת לִיהוּדָה וּבִנְיָמִן וְהַכֹּהֲנִים וְהַלְוִיִּם לְכֹל הֵעִיר הָאֱלֹהִים אֶת רוּחוֹ לַעֲלוֹת לִבְנוֹת אֶת בֵּית יְהוָה אֲשֶׁר בִּירוּשָׁלִָם` (עזרא א:ה)
`וְרוּחֲךָ הַטּוֹבָה נָתַתָּ לְהַשְׂכִּילָם וּמַנְךָ לֹא מָנַעְתָּ מִפִּיהֶם וּמַיִם נָתַתָּה לָהֶם לִצְמָאָם` (נחמיה ט:כ)
`וַתִּמְשֹׁךְ עֲלֵיהֶם שָׁנִים רַבּוֹת וַתָּעַד בָּם בְּרוּחֲךָ בְּיַד נְבִיאֶיךָ וְלֹא הֶאֱזִינוּ וַתִּתְּנֵם בְּיַד עַמֵּי הָאֲרָצֹת` (נחמיה ט:ל)
