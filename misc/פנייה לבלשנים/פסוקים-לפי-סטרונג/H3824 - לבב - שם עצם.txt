`הֲלֹא הוּא אָמַר לִי אֲחֹתִי הִוא וְהִיא גַם הִוא אָמְרָה אָחִי הוּא בְּתָם לְבָבִי וּבְנִקְיֹן כַּפַּי עָשִׂיתִי זֹאת` (בראשית כ:ה)
`וַיֹּאמֶר אֵלָיו הָאֱלֹהִים בַּחֲלֹם גַּם אָנֹכִי יָדַעְתִּי כִּי בְתָם לְבָבְךָ עָשִׂיתָ זֹּאת וָאֶחְשֹׂךְ גַּם אָנֹכִי אוֹתְךָ מֵחֲטוֹ לִי עַל כֵּן לֹא נְתַתִּיךָ לִנְגֹּעַ אֵלֶיהָ` (בראשית כ:ו)
`וַיֹּאמֶר לָבָן לְיַעֲקֹב מֶה עָשִׂיתָ וַתִּגְנֹב אֶת לְבָבִי וַתְּנַהֵג אֶת בְּנֹתַי כִּשְׁבֻיוֹת חָרֶב` (בראשית לא:כו)
`וַיֻּגַּד לְמֶלֶךְ מִצְרַיִם כִּי בָרַח הָעָם וַיֵּהָפֵךְ לְבַב פַּרְעֹה וַעֲבָדָיו אֶל הָעָם וַיֹּאמרוּ מַה זֹּאת עָשִׂינוּ כִּי שִׁלַּחְנוּ אֶת יִשְׂרָאֵל מֵעָבְדֵנוּ` (שמות יד:ה)
`לֹא תִשְׂנָא אֶת אָחִיךָ בִּלְבָבֶךָ הוֹכֵחַ תּוֹכִיחַ אֶת עֲמִיתֶךָ וְלֹא תִשָּׂא עָלָיו חֵטְא` (ויקרא יט:יז)
`וְהַנִּשְׁאָרִים בָּכֶם וְהֵבֵאתִי מֹרֶךְ בִּלְבָבָם בְּאַרְצֹת אֹיְבֵיהֶם וְרָדַף אֹתָם קוֹל עָלֶה נִדָּף וְנָסוּ מְנֻסַת חֶרֶב וְנָפְלוּ וְאֵין רֹדֵף` (ויקרא כו:לו)
`אַף אֲנִי אֵלֵךְ עִמָּם בְּקֶרִי וְהֵבֵאתִי אֹתָם בְּאֶרֶץ אֹיְבֵיהֶם אוֹ אָז יִכָּנַע לְבָבָם הֶעָרֵל וְאָז יִרְצוּ אֶת עֲוֺנָם` (ויקרא כו:מא)
`וְהָיָה לָכֶם לְצִיצִת וּרְאִיתֶם אֹתוֹ וּזְכַרְתֶּם אֶת כָּל מִצְוֺת יְהוָה וַעֲשִׂיתֶם אֹתָם וְלֹא תָתֻרוּ אַחֲרֵי לְבַבְכֶם וְאַחֲרֵי עֵינֵיכֶם אֲשֶׁר אַתֶּם זֹנִים אַחֲרֵיהֶם` (במדבר טו:לט)
`אָנָה אֲנַחְנוּ עֹלִים אַחֵינוּ הֵמַסּוּ אֶת לְבָבֵנוּ לֵאמֹר עַם גָּדוֹל וָרָם מִמֶּנּוּ עָרִים גְּדֹלֹת וּבְצוּרֹת בַּשָּׁמָיִם וְגַם בְּנֵי עֲנָקִים רָאִינוּ שָׁם` (דברים א:כח)
`וְלֹא אָבָה סִיחֹן מֶלֶךְ חֶשְׁבּוֹן הַעֲבִרֵנוּ בּוֹ כִּי הִקְשָׁה יְהוָה אֱלֹהֶיךָ אֶת רוּחוֹ וְאִמֵּץ אֶת לְבָבוֹ לְמַעַן תִּתּוֹ בְיָדְךָ כַּיּוֹם הַזֶּה` (דברים ב:ל)
`רַק הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד פֶּן תִּשְׁכַּח אֶת הַדְּבָרִים אֲשֶׁר רָאוּ עֵינֶיךָ וּפֶן יָסוּרוּ מִלְּבָבְךָ כֹּל יְמֵי חַיֶּיךָ וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ` (דברים ד:ט)
`וּבִקַּשְׁתֶּם מִשָּׁם אֶת יְהוָה אֱלֹהֶיךָ וּמָצָאתָ כִּי תִדְרְשֶׁנּוּ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים ד:כט)
`וְיָדַעְתָּ הַיּוֹם וַהֲשֵׁבֹתָ אֶל לְבָבֶךָ כִּי יְהוָה הוּא הָאֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת אֵין עוֹד` (דברים ד:לט)
`מִי יִתֵּן וְהָיָה לְבָבָם זֶה לָהֶם לְיִרְאָה אֹתִי וְלִשְׁמֹר אֶת כָּל מִצְוֺתַי כָּל הַיָּמִים לְמַעַן יִיטַב לָהֶם וְלִבְנֵיהֶם לְעֹלָם` (דברים ה:כט)
`וְאָהַבְתָּ אֵת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ וּבְכָל מְאֹדֶךָ` (דברים ו:ה)
`וְהָיוּ הַדְּבָרִים הָאֵלֶּה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם עַל לְבָבֶךָ` (דברים ו:ו)
`כִּי תֹאמַר בִּלְבָבְךָ רַבִּים הַגּוֹיִם הָאֵלֶּה מִמֶּנִּי אֵיכָה אוּכַל לְהוֹרִישָׁם` (דברים ז:יז)
`וְזָכַרְתָּ אֶת כָּל הַדֶּרֶךְ אֲשֶׁר הֹלִיכֲךָ יְהוָה אֱלֹהֶיךָ זֶה אַרְבָּעִים שָׁנָה בַּמִּדְבָּר לְמַעַן עַנֹּתְךָ לְנַסֹּתְךָ לָדַעַת אֶת אֲשֶׁר בִּלְבָבְךָ הֲתִשְׁמֹר מִצְוֹתוֹ אִם לֹא` (דברים ח:ב)
`וְיָדַעְתָּ עִם לְבָבֶךָ כִּי כַּאֲשֶׁר יְיַסֵּר אִישׁ אֶת בְּנוֹ יְהוָה אֱלֹהֶיךָ מְיַסְּרֶךָּ` (דברים ח:ה)
`וְרָם לְבָבֶךָ וְשָׁכַחְתָּ אֶת יְהוָה אֱלֹהֶיךָ הַמּוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים` (דברים ח:יד)
`וְאָמַרְתָּ בִּלְבָבֶךָ כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה` (דברים ח:יז)
`אַל תֹּאמַר בִּלְבָבְךָ בַּהֲדֹף יְהוָה אֱלֹהֶיךָ אֹתָם מִלְּפָנֶיךָ לֵאמֹר בְּצִדְקָתִי הֱבִיאַנִי יְהוָה לָרֶשֶׁת אֶת הָאָרֶץ הַזֹּאת וּבְרִשְׁעַת הַגּוֹיִם הָאֵלֶּה יְהוָה מוֹרִישָׁם מִפָּנֶיךָ` (דברים ט:ד)
`לֹא בְצִדְקָתְךָ וּבְיֹשֶׁר לְבָבְךָ אַתָּה בָא לָרֶשֶׁת אֶת אַרְצָם כִּי בְּרִשְׁעַת הַגּוֹיִם הָאֵלֶּה יְהוָה אֱלֹהֶיךָ מוֹרִישָׁם מִפָּנֶיךָ וּלְמַעַן הָקִים אֶת הַדָּבָר אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֶיךָ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב` (דברים ט:ה)
`וְעַתָּה יִשְׂרָאֵל מָה יְהוָה אֱלֹהֶיךָ שֹׁאֵל מֵעִמָּךְ כִּי אִם לְיִרְאָה אֶת יְהוָה אֱלֹהֶיךָ לָלֶכֶת בְּכָל דְּרָכָיו וּלְאַהֲבָה אֹתוֹ וְלַעֲבֹד אֶת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים י:יב)
`וּמַלְתֶּם אֵת עָרְלַת לְבַבְכֶם וְעָרְפְּכֶם לֹא תַקְשׁוּ עוֹד` (דברים י:טז)
`וְהָיָה אִם שָׁמֹעַ תִּשְׁמְעוּ אֶל מִצְוֺתַי אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם לְאַהֲבָה אֶת יְהוָה אֱלֹהֵיכֶם וּלְעָבְדוֹ בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם` (דברים יא:יג)
`הִשָּׁמְרוּ לָכֶם פֶּן יִפְתֶּה לְבַבְכֶם וְסַרְתֶּם וַעֲבַדְתֶּם אֱלֹהִים אֲחֵרִים וְהִשְׁתַּחֲוִיתֶם לָהֶם` (דברים יא:טז)
`וְשַׂמְתֶּם אֶת דְּבָרַי אֵלֶּה עַל לְבַבְכֶם וְעַל נַפְשְׁכֶם וּקְשַׁרְתֶּם אֹתָם לְאוֹת עַל יֶדְכֶם וְהָיוּ לְטוֹטָפֹת בֵּין עֵינֵיכֶם` (דברים יא:יח)
`לֹא תִשְׁמַע אֶל דִּבְרֵי הַנָּבִיא הַהוּא אוֹ אֶל חוֹלֵם הַחֲלוֹם הַהוּא כִּי מְנַסֶּה יְהוָה אֱלֹהֵיכֶם אֶתְכֶם לָדַעַת הֲיִשְׁכֶם אֹהֲבִים אֶת יְהוָה אֱלֹהֵיכֶם בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם` (דברים יג:ג)
`כִּי יִהְיֶה בְךָ אֶבְיוֹן מֵאַחַד אַחֶיךָ בְּאַחַד שְׁעָרֶיךָ בְּאַרְצְךָ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ לֹא תְאַמֵּץ אֶת לְבָבְךָ וְלֹא תִקְפֹּץ אֶת יָדְךָ מֵאָחִיךָ הָאֶבְיוֹן` (דברים טו:ז)
`הִשָּׁמֶר לְךָ פֶּן יִהְיֶה דָבָר עִם לְבָבְךָ בְלִיַּעַל לֵאמֹר קָרְבָה שְׁנַת הַשֶּׁבַע שְׁנַת הַשְּׁמִטָּה וְרָעָה עֵינְךָ בְּאָחִיךָ הָאֶבְיוֹן וְלֹא תִתֵּן לוֹ וְקָרָא עָלֶיךָ אֶל יְהוָה וְהָיָה בְךָ חֵטְא` (דברים טו:ט)
`נָתוֹן תִּתֵּן לוֹ וְלֹא יֵרַע לְבָבְךָ בְּתִתְּךָ לוֹ כִּי בִּגְלַל הַדָּבָר הַזֶּה יְבָרֶכְךָ יְהוָה אֱלֹהֶיךָ בְּכָל מַעֲשֶׂךָ וּבְכֹל מִשְׁלַח יָדֶךָ` (דברים טו:י)
`וְלֹא יַרְבֶּה לּוֹ נָשִׁים וְלֹא יָסוּר לְבָבוֹ וְכֶסֶף וְזָהָב לֹא יַרְבֶּה לּוֹ מְאֹד` (דברים יז:יז)
`לְבִלְתִּי רוּם לְבָבוֹ מֵאֶחָיו וּלְבִלְתִּי סוּר מִן הַמִּצְוָה יָמִין וּשְׂמֹאול לְמַעַן יַאֲרִיךְ יָמִים עַל מַמְלַכְתּוֹ הוּא וּבָנָיו בְּקֶרֶב יִשְׂרָאֵל` (דברים יז:כ)
`וְכִי תֹאמַר בִּלְבָבֶךָ אֵיכָה נֵדַע אֶת הַדָּבָר אֲשֶׁר לֹא דִבְּרוֹ יְהוָה` (דברים יח:כא)
`פֶּן יִרְדֹּף גֹּאֵל הַדָּם אַחֲרֵי הָרֹצֵחַ כִּי יֵחַם לְבָבוֹ וְהִשִּׂיגוֹ כִּי יִרְבֶּה הַדֶּרֶךְ וְהִכָּהוּ נָפֶשׁ וְלוֹ אֵין מִשְׁפַּט מָוֶת כִּי לֹא שֹׂנֵא הוּא לוֹ מִתְּמוֹל שִׁלְשׁוֹם` (דברים יט:ו)
`וְאָמַר אֲלֵהֶם שְׁמַע יִשְׂרָאֵל אַתֶּם קְרֵבִים הַיּוֹם לַמִּלְחָמָה עַל אֹיְבֵיכֶם אַל יֵרַךְ לְבַבְכֶם אַל תִּירְאוּ וְאַל תַּחְפְּזוּ וְאַל תַּעַרְצוּ מִפְּנֵיהֶם` (דברים כ:ג)
`וְיָסְפוּ הַשֹּׁטְרִים לְדַבֵּר אֶל הָעָם וְאָמְרוּ מִי הָאִישׁ הַיָּרֵא וְרַךְ הַלֵּבָב יֵלֵךְ וְיָשֹׁב לְבֵיתוֹ וְלֹא יִמַּס אֶת לְבַב אֶחָיו כִּלְבָבוֹ` (דברים כ:ח)
`הַיּוֹם הַזֶּה יְהוָה אֱלֹהֶיךָ מְצַוְּךָ לַעֲשׂוֹת אֶת הַחֻקִּים הָאֵלֶּה וְאֶת הַמִּשְׁפָּטִים וְשָׁמַרְתָּ וְעָשִׂיתָ אוֹתָם בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים כו:טז)
`יַכְּכָה יְהוָה בְּשִׁגָּעוֹן וּבְעִוָּרוֹן וּבְתִמְהוֹן לֵבָב` (דברים כח:כח)
`תַּחַת אֲשֶׁר לֹא עָבַדְתָּ אֶת יְהוָה אֱלֹהֶיךָ בְּשִׂמְחָה וּבְטוּב לֵבָב מֵרֹב כֹּל` (דברים כח:מז)
`בַּבֹּקֶר תֹּאמַר מִי יִתֵּן עֶרֶב וּבָעֶרֶב תֹּאמַר מִי יִתֵּן בֹּקֶר מִפַּחַד לְבָבְךָ אֲשֶׁר תִּפְחָד וּמִמַּרְאֵה עֵינֶיךָ אֲשֶׁר תִּרְאֶה` (דברים כח:סז)
`פֶּן יֵשׁ בָּכֶם אִישׁ אוֹ אִשָּׁה אוֹ מִשְׁפָּחָה אוֹ שֵׁבֶט אֲשֶׁר לְבָבוֹ פֹנֶה הַיּוֹם מֵעִם יְהוָה אֱלֹהֵינוּ לָלֶכֶת לַעֲבֹד אֶת אֱלֹהֵי הַגּוֹיִם הָהֵם פֶּן יֵשׁ בָּכֶם שֹׁרֶשׁ פֹּרֶה רֹאשׁ וְלַעֲנָה` (דברים כט:יח)
`וְהָיָה בְּשָׁמְעוֹ אֶת דִּבְרֵי הָאָלָה הַזֹּאת וְהִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר שָׁלוֹם יִהְיֶה לִּי כִּי בִּשְׁרִרוּת לִבִּי אֵלֵךְ לְמַעַן סְפוֹת הָרָוָה אֶת הַצְּמֵאָה` (דברים כט:יט)
`וְהָיָה כִי יָבֹאוּ עָלֶיךָ כָּל הַדְּבָרִים הָאֵלֶּה הַבְּרָכָה וְהַקְּלָלָה אֲשֶׁר נָתַתִּי לְפָנֶיךָ וַהֲשֵׁבֹתָ אֶל לְבָבֶךָ בְּכָל הַגּוֹיִם אֲשֶׁר הִדִּיחֲךָ יְהוָה אֱלֹהֶיךָ שָׁמָּה` (דברים ל:א)
`וְשַׁבְתָּ עַד יְהוָה אֱלֹהֶיךָ וְשָׁמַעְתָּ בְקֹלוֹ כְּכֹל אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם אַתָּה וּבָנֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים ל:ב)
`וּמָל יְהוָה אֱלֹהֶיךָ אֶת לְבָבְךָ וְאֶת לְבַב זַרְעֶךָ לְאַהֲבָה אֶת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ לְמַעַן חַיֶּיךָ` (דברים ל:ו)
`כִּי תִשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ לִשְׁמֹר מִצְוֺתָיו וְחֻקֹּתָיו הַכְּתוּבָה בְּסֵפֶר הַתּוֹרָה הַזֶּה כִּי תָשׁוּב אֶל יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים ל:י)
`כִּי קָרוֹב אֵלֶיךָ הַדָּבָר מְאֹד בְּפִיךָ וּבִלְבָבְךָ לַעֲשֹׂתוֹ` (דברים ל:יד)
`וְאִם יִפְנֶה לְבָבְךָ וְלֹא תִשְׁמָע וְנִדַּחְתָּ וְהִשְׁתַּחֲוִיתָ לֵאלֹהִים אֲחֵרִים וַעֲבַדְתָּם` (דברים ל:יז)
`וַיֹּאמֶר אֲלֵהֶם שִׂימוּ לְבַבְכֶם לְכָל הַדְּבָרִים אֲשֶׁר אָנֹכִי מֵעִיד בָּכֶם הַיּוֹם אֲשֶׁר תְּצַוֻּם אֶת בְּנֵיכֶם לִשְׁמֹר לַעֲשׂוֹת אֶת כָּל דִּבְרֵי הַתּוֹרָה הַזֹּאת` (דברים לב:מו)
`וַנִּשְׁמַע וַיִּמַּס לְבָבֵנוּ וְלֹא קָמָה עוֹד רוּחַ בְּאִישׁ מִפְּנֵיכֶם כִּי יְהוָה אֱלֹהֵיכֶם הוּא אֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת` (יהושע ב:יא)
`וַיְהִי כִשְׁמֹעַ כָּל מַלְכֵי הָאֱמֹרִי אֲשֶׁר בְּעֵבֶר הַיַּרְדֵּן יָמָּה וְכָל מַלְכֵי הַכְּנַעֲנִי אֲשֶׁר עַל הַיָּם אֵת אֲשֶׁר הוֹבִישׁ יְהוָה אֶת מֵי הַיַּרְדֵּן מִפְּנֵי בְנֵי יִשְׂרָאֵל עַד עָבְרָנוּ וַיִּמַּס לְבָבָם וְלֹא הָיָה בָם עוֹד רוּחַ מִפְּנֵי בְּנֵי יִשְׂרָאֵל` (יהושע ה:א)
`וַיַּכּוּ מֵהֶם אַנְשֵׁי הָעַי כִּשְׁלֹשִׁים וְשִׁשָּׁה אִישׁ וַיִּרְדְּפוּם לִפְנֵי הַשַּׁעַר עַד הַשְּׁבָרִים וַיַּכּוּם בַּמּוֹרָד וַיִּמַּס לְבַב הָעָם וַיְהִי לְמָיִם` (יהושע ז:ה)
`בֶּן אַרְבָּעִים שָׁנָה אָנֹכִי בִּשְׁלֹחַ מֹשֶׁה עֶבֶד יְהוָה אֹתִי מִקָּדֵשׁ בַּרְנֵעַ לְרַגֵּל אֶת הָאָרֶץ וָאָשֵׁב אֹתוֹ דָּבָר כַּאֲשֶׁר עִם לְבָבִי` (יהושע יד:ז)
`רַק שִׁמְרוּ מְאֹד לַעֲשׂוֹת אֶת הַמִּצְוָה וְאֶת הַתּוֹרָה אֲשֶׁר צִוָּה אֶתְכֶם מֹשֶׁה עֶבֶד יְהוָה לְאַהֲבָה אֶת יְהוָה אֱלֹהֵיכֶם וְלָלֶכֶת בְּכָל דְּרָכָיו וְלִשְׁמֹר מִצְוֺתָיו וּלְדָבְקָה בוֹ וּלְעָבְדוֹ בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם` (יהושע כב:ה)
`וְהִנֵּה אָנֹכִי הוֹלֵךְ הַיּוֹם בְּדֶרֶךְ כָּל הָאָרֶץ וִידַעְתֶּם בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם כִּי לֹא נָפַל דָּבָר אֶחָד מִכֹּל הַדְּבָרִים הַטּוֹבִים אֲשֶׁר דִּבֶּר יְהוָה אֱלֹהֵיכֶם עֲלֵיכֶם הַכֹּל בָּאוּ לָכֶם לֹא נָפַל מִמֶּנּוּ דָּבָר אֶחָד` (יהושע כג:יד)
`וְעַתָּה הָסִירוּ אֶת אֱלֹהֵי הַנֵּכָר אֲשֶׁר בְּקִרְבְּכֶם וְהַטּוּ אֶת לְבַבְכֶם אֶל יְהוָה אֱלֹהֵי יִשְׂרָאֵל` (יהושע כד:כג)
`וַיַּשְׁכֵּם בַּבֹּקֶר בַּיּוֹם הַחֲמִישִׁי לָלֶכֶת וַיֹּאמֶר אֲבִי הַנַּעֲרָה סְעָד נָא לְבָבְךָ וְהִתְמַהְמְהוּ עַד נְטוֹת הַיּוֹם וַיֹּאכְלוּ שְׁנֵיהֶם` (שופטים יט:ח)
`וַיָּקָם הָאִישׁ לָלֶכֶת הוּא וּפִילַגְשׁוֹ וְנַעֲרוֹ וַיֹּאמֶר לוֹ חֹתְנוֹ אֲבִי הַנַּעֲרָה הִנֵּה נָא רָפָה הַיּוֹם לַעֲרֹב לִינוּ נָא הִנֵּה חֲנוֹת הַיּוֹם לִין פֹּה וְיִיטַב לְבָבֶךָ וְהִשְׁכַּמְתֶּם מָחָר לְדַרְכְּכֶם וְהָלַכְתָּ לְאֹהָלֶךָ` (שופטים יט:ט)
`וַיֹּאמֶר לָהּ אֶלְקָנָה אִישָׁהּ חַנָּה לָמֶה תִבְכִּי וְלָמֶה לֹא תֹאכְלִי וְלָמֶה יֵרַע לְבָבֵךְ הֲלוֹא אָנֹכִי טוֹב לָךְ מֵעֲשָׂרָה בָּנִים` (שמואל-א א:ח)
`וַהֲקִימֹתִי לִי כֹּהֵן נֶאֱמָן כַּאֲשֶׁר בִּלְבָבִי וּבְנַפְשִׁי יַעֲשֶׂה וּבָנִיתִי לוֹ בַּיִת נֶאֱמָן וְהִתְהַלֵּךְ לִפְנֵי מְשִׁיחִי כָּל הַיָּמִים` (שמואל-א ב:לה)
`וְלָמָּה תְכַבְּדוּ אֶת לְבַבְכֶם כַּאֲשֶׁר כִּבְּדוּ מִצְרַיִם וּפַרְעֹה אֶת לִבָּם הֲלוֹא כַּאֲשֶׁר הִתְעַלֵּל בָּהֶם וַיְשַׁלְּחוּם וַיֵּלֵכוּ` (שמואל-א ו:ו)
`וַיֹּאמֶר שְׁמוּאֵל אֶל כָּל בֵּית יִשְׂרָאֵל לֵאמֹר אִם בְּכָל לְבַבְכֶם אַתֶּם שָׁבִים אֶל יְהוָה הָסִירוּ אֶת אֱלֹהֵי הַנֵּכָר מִתּוֹכְכֶם וְהָעַשְׁתָּרוֹת וְהָכִינוּ לְבַבְכֶם אֶל יְהוָה וְעִבְדֻהוּ לְבַדּוֹ וְיַצֵּל אֶתְכֶם מִיַּד פְּלִשְׁתִּים` (שמואל-א ז:ג)
`וַיַּעַן שְׁמוּאֵל אֶת שָׁאוּל וַיֹּאמֶר אָנֹכִי הָרֹאֶה עֲלֵה לְפָנַי הַבָּמָה וַאֲכַלְתֶּם עִמִּי הַיּוֹם וְשִׁלַּחְתִּיךָ בַבֹּקֶר וְכֹל אֲשֶׁר בִּלְבָבְךָ אַגִּיד לָךְ` (שמואל-א ט:יט)
`וַיֹּאמֶר שְׁמוּאֵל אֶל הָעָם אַל תִּירָאוּ אַתֶּם עֲשִׂיתֶם אֵת כָּל הָרָעָה הַזֹּאת אַךְ אַל תָּסוּרוּ מֵאַחֲרֵי יְהוָה וַעֲבַדְתֶּם אֶת יְהוָה בְּכָל לְבַבְכֶם` (שמואל-א יב:כ)
`אַךְ יְראוּ אֶת יְהוָה וַעֲבַדְתֶּם אֹתוֹ בֶּאֱמֶת בְּכָל לְבַבְכֶם כִּי רְאוּ אֵת אֲשֶׁר הִגְדִּל עִמָּכֶם` (שמואל-א יב:כד)
`וְעַתָּה מַמְלַכְתְּךָ לֹא תָקוּם בִּקֵּשׁ יְהוָה לוֹ אִישׁ כִּלְבָבוֹ וַיְצַוֵּהוּ יְהוָה לְנָגִיד עַל עַמּוֹ כִּי לֹא שָׁמַרְתָּ אֵת אֲשֶׁר צִוְּךָ יְהוָה` (שמואל-א יג:יד)
`וַיֹּאמֶר לוֹ נֹשֵׂא כֵלָיו עֲשֵׂה כָּל אֲשֶׁר בִּלְבָבֶךָ נְטֵה לָךְ הִנְנִי עִמְּךָ כִּלְבָבֶךָ` (שמואל-א יד:ז)
`וַיֹּאמֶר יְהוָה אֶל שְׁמוּאֵל אַל תַּבֵּט אֶל מַרְאֵהוּ וְאֶל גְּבֹהַּ קוֹמָתוֹ כִּי מְאַסְתִּיהוּ כִּי לֹא אֲשֶׁר יִרְאֶה הָאָדָם כִּי הָאָדָם יִרְאֶה לַעֵינַיִם וַיהוָה יִרְאֶה לַלֵּבָב` (שמואל-א טז:ז)
`וַיִּשְׁמַע אֱלִיאָב אָחִיו הַגָּדוֹל בְּדַבְּרוֹ אֶל הָאֲנָשִׁים וַיִּחַר אַף אֱלִיאָב בְּדָוִד וַיֹּאמֶר לָמָּה זֶּה יָרַדְתָּ וְעַל מִי נָטַשְׁתָּ מְעַט הַצֹּאן הָהֵנָּה בַּמִּדְבָּר אֲנִי יָדַעְתִּי אֶת זְדֹנְךָ וְאֵת רֹעַ לְבָבֶךָ כִּי לְמַעַן רְאוֹת הַמִּלְחָמָה יָרָדְתָּ` (שמואל-א יז:כח)
`וַיָּשֶׂם דָּוִד אֶת הַדְּבָרִים הָאֵלֶּה בִּלְבָבוֹ וַיִּרָא מְאֹד מִפְּנֵי אָכִישׁ מֶלֶךְ גַּת` (שמואל-א כא:יב)
`וַיֹּאמֶר נָתָן אֶל הַמֶּלֶךְ כֹּל אֲשֶׁר בִּלְבָבְךָ לֵךְ עֲשֵׂה כִּי יְהוָה עִמָּךְ` (שמואל-ב ז:ג)
`וַיַּט אֶת לְבַב כָּל אִישׁ יְהוּדָה כְּאִישׁ אֶחָד וַיִּשְׁלְחוּ אֶל הַמֶּלֶךְ שׁוּב אַתָּה וְכָל עֲבָדֶיךָ` (שמואל-ב יט:יד)
`לְמַעַן יָקִים יְהוָה אֶת דְּבָרוֹ אֲשֶׁר דִּבֶּר עָלַי לֵאמֹר אִם יִשְׁמְרוּ בָנֶיךָ אֶת דַּרְכָּם לָלֶכֶת לְפָנַי בֶּאֱמֶת בְּכָל לְבָבָם וּבְכָל נַפְשָׁם לֵאמֹר לֹא יִכָּרֵת לְךָ אִישׁ מֵעַל כִּסֵּא יִשְׂרָאֵל` (מלכים-א ב:ד)
`וַיֹּאמֶר הַמֶּלֶךְ אֶל שִׁמְעִי אַתָּה יָדַעְתָּ אֵת כָּל הָרָעָה אֲשֶׁר יָדַע לְבָבְךָ אֲשֶׁר עָשִׂיתָ לְדָוִד אָבִי וְהֵשִׁיב יְהוָה אֶת רָעָתְךָ בְּרֹאשֶׁךָ` (מלכים-א ב:מד)
`וַיֹּאמֶר שְׁלֹמֹה אַתָּה עָשִׂיתָ עִם עַבְדְּךָ דָוִד אָבִי חֶסֶד גָּדוֹל כַּאֲשֶׁר הָלַךְ לְפָנֶיךָ בֶּאֱמֶת וּבִצְדָקָה וּבְיִשְׁרַת לֵבָב עִמָּךְ וַתִּשְׁמָר לוֹ אֶת הַחֶסֶד הַגָּדוֹל הַזֶּה וַתִּתֶּן לוֹ בֵן יֹשֵׁב עַל כִּסְאוֹ כַּיּוֹם הַזֶּה` (מלכים-א ג:ו)
`וַיְהִי עִם לְבַב דָּוִד אָבִי לִבְנוֹת בַּיִת לְשֵׁם יְהוָה אֱלֹהֵי יִשְׂרָאֵל` (מלכים-א ח:יז)
`וַיֹּאמֶר יְהוָה אֶל דָּוִד אָבִי יַעַן אֲשֶׁר הָיָה עִם לְבָבְךָ לִבְנוֹת בַּיִת לִשְׁמִי הֱטִיבֹתָ כִּי הָיָה עִם לְבָבֶךָ` (מלכים-א ח:יח)
`כָּל תְּפִלָּה כָל תְּחִנָּה אֲשֶׁר תִהְיֶה לְכָל הָאָדָם לְכֹל עַמְּךָ יִשְׂרָאֵל אֲשֶׁר יֵדְעוּן אִישׁ נֶגַע לְבָבוֹ וּפָרַשׂ כַּפָּיו אֶל הַבַּיִת הַזֶּה` (מלכים-א ח:לח)
`וְאַתָּה תִּשְׁמַע הַשָּׁמַיִם מְכוֹן שִׁבְתֶּךָ וְסָלַחְתָּ וְעָשִׂיתָ וְנָתַתָּ לָאִישׁ כְּכָל דְּרָכָיו אֲשֶׁר תֵּדַע אֶת לְבָבוֹ כִּי אַתָּה יָדַעְתָּ לְבַדְּךָ אֶת לְבַב כָּל בְּנֵי הָאָדָם` (מלכים-א ח:לט)
`וְשָׁבוּ אֵלֶיךָ בְּכָל לְבָבָם וּבְכָל נַפְשָׁם בְּאֶרֶץ אֹיְבֵיהֶם אֲשֶׁר שָׁבוּ אֹתָם וְהִתְפַּלְלוּ אֵלֶיךָ דֶּרֶךְ אַרְצָם אֲשֶׁר נָתַתָּה לַאֲבוֹתָם הָעִיר אֲשֶׁר בָּחַרְתָּ וְהַבַּיִת אֲשֶׁר בָּנִיתָ לִשְׁמֶךָ` (מלכים-א ח:מח)
`לְהַטּוֹת לְבָבֵנוּ אֵלָיו לָלֶכֶת בְּכָל דְּרָכָיו וְלִשְׁמֹר מִצְוֺתָיו וְחֻקָּיו וּמִשְׁפָּטָיו אֲשֶׁר צִוָּה אֶת אֲבֹתֵינוּ` (מלכים-א ח:נח)
`וְהָיָה לְבַבְכֶם שָׁלֵם עִם יְהוָה אֱלֹהֵינוּ לָלֶכֶת בְּחֻקָּיו וְלִשְׁמֹר מִצְוֺתָיו כַּיּוֹם הַזֶּה` (מלכים-א ח:סא)
`וְאַתָּה אִם תֵּלֵךְ לְפָנַי כַּאֲשֶׁר הָלַךְ דָּוִד אָבִיךָ בְּתָם לֵבָב וּבְיֹשֶׁר לַעֲשׂוֹת כְּכֹל אֲשֶׁר צִוִּיתִיךָ חֻקַּי וּמִשְׁפָּטַי תִּשְׁמֹר` (מלכים-א ט:ד)
`וַתָּבֹא יְרוּשָׁלְַמָה בְּחַיִל כָּבֵד מְאֹד גְּמַלִּים נֹשְׂאִים בְּשָׂמִים וְזָהָב רַב מְאֹד וְאֶבֶן יְקָרָה וַתָּבֹא אֶל שְׁלֹמֹה וַתְּדַבֵּר אֵלָיו אֵת כָּל אֲשֶׁר הָיָה עִם לְבָבָהּ` (מלכים-א י:ב)
`מִן הַגּוֹיִם אֲשֶׁר אָמַר יְהוָה אֶל בְּנֵי יִשְׂרָאֵל לֹא תָבֹאוּ בָהֶם וְהֵם לֹא יָבֹאוּ בָכֶם אָכֵן יַטּוּ אֶת לְבַבְכֶם אַחֲרֵי אֱלֹהֵיהֶם בָּהֶם דָּבַק שְׁלֹמֹה לְאַהֲבָה` (מלכים-א יא:ב)
`וַיְהִי לְעֵת זִקְנַת שְׁלֹמֹה נָשָׁיו הִטּוּ אֶת לְבָבוֹ אַחֲרֵי אֱלֹהִים אֲחֵרִים וְלֹא הָיָה לְבָבוֹ שָׁלֵם עִם יְהוָה אֱלֹהָיו כִּלְבַב דָּוִיד אָבִיו` (מלכים-א יא:ד)
`וַיִּתְאַנַּף יְהוָה בִּשְׁלֹמֹה כִּי נָטָה לְבָבוֹ מֵעִם יְהוָה אֱלֹהֵי יִשְׂרָאֵל הַנִּרְאָה אֵלָיו פַּעֲמָיִם` (מלכים-א יא:ט)
`וָאֶקְרַע אֶת הַמַּמְלָכָה מִבֵּית דָּוִד וָאֶתְּנֶהָ לָךְ וְלֹא הָיִיתָ כְּעַבְדִּי דָוִד אֲשֶׁר שָׁמַר מִצְוֺתַי וַאֲשֶׁר הָלַךְ אַחֲרַי בְּכָל לְבָבוֹ לַעֲשׂוֹת רַק הַיָּשָׁר בְּעֵינָי` (מלכים-א יד:ח)
`וַיֵּלֶךְ בְּכָל חַטֹּאות אָבִיו אֲשֶׁר עָשָׂה לְפָנָיו וְלֹא הָיָה לְבָבוֹ שָׁלֵם עִם יְהוָה אֱלֹהָיו כִּלְבַב דָּוִד אָבִיו` (מלכים-א טו:ג)
`וְהַבָּמוֹת לֹא סָרוּ רַק לְבַב אָסָא הָיָה שָׁלֵם עִם יְהוָה כָּל יָמָיו` (מלכים-א טו:יד)
`וַיֵּלֶךְ מִשָּׁם וַיִּמְצָא אֶת יְהוֹנָדָב בֶּן רֵכָב לִקְרָאתוֹ וַיְבָרְכֵהוּ וַיֹּאמֶר אֵלָיו הֲיֵשׁ אֶת לְבָבְךָ יָשָׁר כַּאֲשֶׁר לְבָבִי עִם לְבָבֶךָ וַיֹּאמֶר יְהוֹנָדָב יֵשׁ וָיֵשׁ תְּנָה אֶת יָדֶךָ וַיִּתֵּן יָדוֹ וַיַּעֲלֵהוּ אֵלָיו אֶל הַמֶּרְכָּבָה` (מלכים-ב י:טו)
`וַיֹּאמֶר יְהוָה אֶל יֵהוּא יַעַן אֲשֶׁר הֱטִיבֹתָ לַעֲשׂוֹת הַיָּשָׁר בְּעֵינַי כְּכֹל אֲשֶׁר בִּלְבָבִי עָשִׂיתָ לְבֵית אַחְאָב בְּנֵי רְבִעִים יֵשְׁבוּ לְךָ עַל כִּסֵּא יִשְׂרָאֵל` (מלכים-ב י:ל)
`וְיֵהוּא לֹא שָׁמַר לָלֶכֶת בְּתוֹרַת יְהוָה אֱלֹהֵי יִשְׂרָאֵל בְּכָל לְבָבוֹ לֹא סָר מֵעַל חַטֹּאות יָרָבְעָם אֲשֶׁר הֶחֱטִיא אֶת יִשְׂרָאֵל` (מלכים-ב י:לא)
`אָנָּה יְהוָה זְכָר נָא אֵת אֲשֶׁר הִתְהַלַּכְתִּי לְפָנֶיךָ בֶּאֱמֶת וּבְלֵבָב שָׁלֵם וְהַטּוֹב בְּעֵינֶיךָ עָשִׂיתִי וַיֵּבְךְּ חִזְקִיָּהוּ בְּכִי גָדוֹל` (מלכים-ב כ:ג)
`יַעַן רַךְ לְבָבְךָ וַתִּכָּנַע מִפְּנֵי יְהוָה בְּשָׁמְעֲךָ אֲשֶׁר דִּבַּרְתִּי עַל הַמָּקוֹם הַזֶּה וְעַל יֹשְׁבָיו לִהְיוֹת לְשַׁמָּה וְלִקְלָלָה וַתִּקְרַע אֶת בְּגָדֶיךָ וַתִּבְכֶּה לְפָנָי וְגַם אָנֹכִי שָׁמַעְתִּי נְאֻם יְהוָה` (מלכים-ב כב:יט)
`וְכָמֹהוּ לֹא הָיָה לְפָנָיו מֶלֶךְ אֲשֶׁר שָׁב אֶל יְהוָה בְּכָל לְבָבוֹ וּבְכָל נַפְשׁוֹ וּבְכָל מְאֹדוֹ כְּכֹל תּוֹרַת מֹשֶׁה וְאַחֲרָיו לֹא קָם כָּמֹהוּ` (מלכים-ב כג:כה)
`עַל מֶה תֻכּוּ עוֹד תּוֹסִיפוּ סָרָה כָּל רֹאשׁ לָחֳלִי וְכָל לֵבָב דַּוָּי` (ישעיהו א:ה)
`הַשְׁמֵן לֵב הָעָם הַזֶּה וְאָזְנָיו הַכְבֵּד וְעֵינָיו הָשַׁע פֶּן יִרְאֶה בְעֵינָיו וּבְאָזְנָיו יִשְׁמָע וּלְבָבוֹ יָבִין וָשָׁב וְרָפָא לוֹ` (ישעיהו ו:י)
`וַיֻּגַּד לְבֵית דָּוִד לֵאמֹר נָחָה אֲרָם עַל אֶפְרָיִם וַיָּנַע לְבָבוֹ וּלְבַב עַמּוֹ כְּנוֹעַ עֲצֵי יַעַר מִפְּנֵי רוּחַ` (ישעיהו ז:ב)
`וְאָמַרְתָּ אֵלָיו הִשָּׁמֵר וְהַשְׁקֵט אַל תִּירָא וּלְבָבְךָ אַל יֵרַךְ מִשְּׁנֵי זַנְבוֹת הָאוּדִים הָעֲשֵׁנִים הָאֵלֶּה בָּחֳרִי אַף רְצִין וַאֲרָם וּבֶן רְמַלְיָהוּ` (ישעיהו ז:ד)
`וְיָדְעוּ הָעָם כֻּלּוֹ אֶפְרַיִם וְיוֹשֵׁב שֹׁמְרוֹן בְּגַאֲוָה וּבְגֹדֶל לֵבָב לֵאמֹר` (ישעיהו ט:ט)
`וְהוּא לֹא כֵן יְדַמֶּה וּלְבָבוֹ לֹא כֵן יַחְשֹׁב כִּי לְהַשְׁמִיד בִּלְבָבוֹ וּלְהַכְרִית גּוֹיִם לֹא מְעָט` (ישעיהו י:ז)
`וְהָיָה כִּי יְבַצַּע אֲדֹנָי אֶת כָּל מַעֲשֵׂהוּ בְּהַר צִיּוֹן וּבִירוּשָׁלִָם אֶפְקֹד עַל פְּרִי גֹדֶל לְבַב מֶלֶךְ אַשּׁוּר וְעַל תִּפְאֶרֶת רוּם עֵינָיו` (ישעיהו י:יב)
`עַל כֵּן כָּל יָדַיִם תִּרְפֶּינָה וְכָל לְבַב אֱנוֹשׁ יִמָּס` (ישעיהו יג:ז)
`וְאַתָּה אָמַרְתָּ בִלְבָבְךָ הַשָּׁמַיִם אֶעֱלֶה מִמַּעַל לְכוֹכְבֵי אֵל אָרִים כִּסְאִי וְאֵשֵׁב בְּהַר מוֹעֵד בְּיַרְכְּתֵי צָפוֹן` (ישעיהו יד:יג)
`מַשָּׂא מִצְרָיִם הִנֵּה יְהוָה רֹכֵב עַל עָב קַל וּבָא מִצְרַיִם וְנָעוּ אֱלִילֵי מִצְרַיִם מִפָּנָיו וּלְבַב מִצְרַיִם יִמַּס בְּקִרְבּוֹ` (ישעיהו יט:א)
`תָּעָה לְבָבִי פַּלָּצוּת בִּעֲתָתְנִי אֵת נֶשֶׁף חִשְׁקִי שָׂם לִי לַחֲרָדָה` (ישעיהו כא:ד)
`הַשִּׁיר יִהְיֶה לָכֶם כְּלֵיל הִתְקַדֶּשׁ חָג וְשִׂמְחַת לֵבָב כַּהוֹלֵךְ בֶּחָלִיל לָבוֹא בְהַר יְהוָה אֶל צוּר יִשְׂרָאֵל` (ישעיהו ל:כט)
`וּלְבַב נִמְהָרִים יָבִין לָדָעַת וּלְשׁוֹן עִלְּגִים תְּמַהֵר לְדַבֵּר צָחוֹת` (ישעיהו לב:ד)
`וְעַתָּה שִׁמְעִי זֹאת עֲדִינָה הַיּוֹשֶׁבֶת לָבֶטַח הָאֹמְרָה בִּלְבָבָהּ אֲנִי וְאַפְסִי עוֹד לֹא אֵשֵׁב אַלְמָנָה וְלֹא אֵדַע שְׁכוֹל` (ישעיהו מז:ח)
`וְאָמַרְתְּ בִּלְבָבֵךְ מִי יָלַד לִי אֶת אֵלֶּה וַאֲנִי שְׁכוּלָה וְגַלְמוּדָה גֹּלָה וְסוּרָה וְאֵלֶּה מִי גִדֵּל הֵן אֲנִי נִשְׁאַרְתִּי לְבַדִּי אֵלֶּה אֵיפֹה הֵם` (ישעיהו מט:כא)
`אָז תִּרְאִי וְנָהַרְתְּ וּפָחַד וְרָחַב לְבָבֵךְ כִּי יֵהָפֵךְ עָלַיִךְ הֲמוֹן יָם חֵיל גּוֹיִם יָבֹאוּ לָךְ` (ישעיהו ס:ה)
`הִמֹּלוּ לַיהֹוָה וְהָסִרוּ עָרְלוֹת לְבַבְכֶם אִישׁ יְהוּדָה וְיֹשְׁבֵי יְרוּשָׁלִָם פֶּן תֵּצֵא כָאֵשׁ חֲמָתִי וּבָעֲרָה וְאֵין מְכַבֶּה מִפְּנֵי רֹעַ מַעַלְלֵיכֶם` (ירמיהו ד:ד)
`וְלֹא אָמְרוּ בִלְבָבָם נִירָא נָא אֶת יְהוָה אֱלֹהֵינוּ הַנֹּתֵן גֶּשֶׁם וְיֹרֶה וּמַלְקוֹשׁ בְּעִתּוֹ שְׁבֻעוֹת חֻקּוֹת קָצִיר יִשְׁמָר לָנוּ` (ירמיהו ה:כד)
`וְכִי תֹאמְרִי בִּלְבָבֵךְ מַדּוּעַ קְרָאֻנִי אֵלֶּה בְּרֹב עֲוֺנֵךְ נִגְלוּ שׁוּלַיִךְ נֶחְמְסוּ עֲקֵבָיִךְ` (ירמיהו יג:כב)
`נִמְצְאוּ דְבָרֶיךָ וָאֹכְלֵם וַיְהִי דְבָרֶיךָ לִי לְשָׂשׂוֹן וּלְשִׂמְחַת לְבָבִי כִּי נִקְרָא שִׁמְךָ עָלַי יְהוָה אֱלֹהֵי צְבָאוֹת` (ירמיהו טו:טז)
`וּבִקַּשְׁתֶּם אֹתִי וּמְצָאתֶם כִּי תִדְרְשֻׁנִי בְּכָל לְבַבְכֶם` (ירמיהו כט:יג)
`וְכָרַתִּי לָהֶם בְּרִית עוֹלָם אֲשֶׁר לֹא אָשׁוּב מֵאַחֲרֵיהֶם לְהֵיטִיבִי אוֹתָם וְאֶת יִרְאָתִי אֶתֵּן בִּלְבָבָם לְבִלְתִּי סוּר מֵעָלָי` (ירמיהו לב:מ)
`וּפֶן יֵרַךְ לְבַבְכֶם וְתִירְאוּ בַּשְּׁמוּעָה הַנִּשְׁמַעַת בָּאָרֶץ וּבָא בַשָּׁנָה הַשְּׁמוּעָה וְאַחֲרָיו בַּשָּׁנָה הַשְּׁמוּעָה וְחָמָס בָּאָרֶץ וּמֹשֵׁל עַל מֹשֵׁל` (ירמיהו נא:מו)
`פְּלֵטִים מֵחֶרֶב הִלְכוּ אַל תַּעֲמֹדוּ זִכְרוּ מֵרָחוֹק אֶת יְהוָה וִירוּשָׁלִַם תַּעֲלֶה עַל לְבַבְכֶם` (ירמיהו נא:נ)
`וַיֹּאמֶר אֵלָי בֶּן אָדָם אֶת כָּל דְּבָרַי אֲשֶׁר אֲדַבֵּר אֵלֶיךָ קַח בִּלְבָבְךָ וּבְאָזְנֶיךָ שְׁמָע` (יחזקאל ג:י)
`בְּרֹב חָכְמָתְךָ בִּרְכֻלָּתְךָ הִרְבִּיתָ חֵילֶךָ וַיִּגְבַּהּ לְבָבְךָ בְּחֵילֶךָ` (יחזקאל כח:ה)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה יַעַן תִּתְּךָ אֶת לְבָבְךָ כְּלֵב אֱלֹהִים` (יחזקאל כח:ו)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה יַעַן אֲשֶׁר גָּבַהְתָּ בְּקוֹמָה וַיִּתֵּן צַמַּרְתּוֹ אֶל בֵּין עֲבוֹתִים וְרָם לְבָבוֹ בְּגָבְהוֹ` (יחזקאל לא:י)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה אִם לֹא בְּאֵשׁ קִנְאָתִי דִבַּרְתִּי עַל שְׁאֵרִית הַגּוֹיִם וְעַל אֱדוֹם כֻּלָּא אֲשֶׁר נָתְנוּ אֶת אַרְצִי לָהֶם לְמוֹרָשָׁה בְּשִׂמְחַת כָּל לֵבָב בִּשְׁאָט נֶפֶשׁ לְמַעַן מִגְרָשָׁהּ לָבַז` (יחזקאל לו:ה)
`כֹּה אָמַר אֲדֹנָי יְהוִה וְהָיָה בַּיּוֹם הַהוּא יַעֲלוּ דְבָרִים עַל לְבָבֶךָ וְחָשַׁבְתָּ מַחֲשֶׁבֶת רָעָה` (יחזקאל לח:י)
`וּבַל יֹאמְרוּ לִלְבָבָם כָּל רָעָתָם זָכָרְתִּי עַתָּה סְבָבוּם מַעַלְלֵיהֶם נֶגֶד פָּנַי הָיוּ` (הושע ז:ב)
`וְגַם עַתָּה נְאֻם יְהוָה שֻׁבוּ עָדַי בְּכָל לְבַבְכֶם וּבְצוֹם וּבְבְכִי וּבְמִסְפֵּד` (יואל ב:יב)
`וְקִרְעוּ לְבַבְכֶם וְאַל בִּגְדֵיכֶם וְשׁוּבוּ אֶל יְהוָה אֱלֹהֵיכֶם כִּי חַנּוּן וְרַחוּם הוּא אֶרֶךְ אַפַּיִם וְרַב חֶסֶד וְנִחָם עַל הָרָעָה` (יואל ב:יג)
`וַתַּשְׁלִיכֵנִי מְצוּלָה בִּלְבַב יַמִּים וְנָהָר יְסֹבְבֵנִי כָּל מִשְׁבָּרֶיךָ וְגַלֶּיךָ עָלַי עָבָרוּ` (יונה ב:ג)
`וְהֻצַּב גֻּלְּתָה הֹעֲלָתָה וְאַמְהֹתֶיהָ מְנַהֲגוֹת כְּקוֹל יוֹנִים מְתֹפְפֹת עַל לִבְבֵהֶן` (נחום ב:ז)
`וְהָיָה בָּעֵת הַהִיא אֲחַפֵּשׂ אֶת יְרוּשָׁלִַם בַּנֵּרוֹת וּפָקַדְתִּי עַל הָאֲנָשִׁים הַקֹּפְאִים עַל שִׁמְרֵיהֶם הָאֹמְרִים בִּלְבָבָם לֹא יֵיטִיב יְהוָה וְלֹא יָרֵעַ` (צפניה א:יב)
`זֹאת הָעִיר הָעַלִּיזָה הַיּוֹשֶׁבֶת לָבֶטַח הָאֹמְרָה בִּלְבָבָהּ אֲנִי וְאַפְסִי עוֹד אֵיךְ הָיְתָה לְשַׁמָּה מַרְבֵּץ לַחַיָּה כֹּל עוֹבֵר עָלֶיהָ יִשְׁרֹק יָנִיעַ יָדוֹ` (צפניה ב:טו)
`וְעַתָּה כֹּה אָמַר יְהוָה צְבָאוֹת שִׂימוּ לְבַבְכֶם עַל דַּרְכֵיכֶם` (חגי א:ה)
`כֹּה אָמַר יְהוָה צְבָאוֹת שִׂימוּ לְבַבְכֶם עַל דַּרְכֵיכֶם` (חגי א:ז)
`וְעַתָּה שִׂימוּ נָא לְבַבְכֶם מִן הַיּוֹם הַזֶּה וָמָעְלָה מִטֶּרֶם שׂוּם אֶבֶן אֶל אֶבֶן בְּהֵיכַל יְהוָה` (חגי ב:טו)
`שִׂימוּ נָא לְבַבְכֶם מִן הַיּוֹם הַזֶּה וָמָעְלָה מִיּוֹם עֶשְׂרִים וְאַרְבָּעָה לַתְּשִׁיעִי לְמִן הַיּוֹם אֲשֶׁר יֻסַּד הֵיכַל יְהוָה שִׂימוּ לְבַבְכֶם` (חגי ב:יח)
`וְאַלְמָנָה וְיָתוֹם גֵּר וְעָנִי אַל תַּעֲשֹׁקוּ וְרָעַת אִישׁ אָחִיו אַל תַּחְשְׁבוּ בִּלְבַבְכֶם` (זכריה ז:י)
`וְאִישׁ אֶת רָעַת רֵעֵהוּ אַל תַּחְשְׁבוּ בִּלְבַבְכֶם וּשְׁבֻעַת שֶׁקֶר אַל תֶּאֱהָבוּ כִּי אֶת כָּל אֵלֶּה אֲשֶׁר שָׂנֵאתִי נְאֻם יְהוָה` (זכריה ח:יז)
`וַיֵּצֵא דָוִיד לִפְנֵיהֶם וַיַּעַן וַיֹּאמֶר לָהֶם אִם לְשָׁלוֹם בָּאתֶם אֵלַי לְעָזְרֵנִי יִהְיֶה לִּי עֲלֵיכֶם לֵבָב לְיָחַד וְאִם לְרַמּוֹתַנִי לְצָרַי בְּלֹא חָמָס בְּכַפַּי יֵרֶא אֱלֹהֵי אֲבוֹתֵינוּ וְיוֹכַח` (דברי-הימים-א יב:יז)
`כָּל אֵלֶּה אַנְשֵׁי מִלְחָמָה עֹדְרֵי מַעֲרָכָה בְּלֵבָב שָׁלֵם בָּאוּ חֶבְרוֹנָה לְהַמְלִיךְ אֶת דָּוִיד עַל כָּל יִשְׂרָאֵל וְגַם כָּל שֵׁרִית יִשְׂרָאֵל לֵב אֶחָד לְהַמְלִיךְ אֶת דָּוִיד` (דברי-הימים-א יב:לח)
`וַיֹּאמֶר נָתָן אֶל דָּוִיד כֹּל אֲשֶׁר בִּלְבָבְךָ עֲשֵׂה כִּי הָאֱלֹהִים עִמָּךְ` (דברי-הימים-א יז:ב)
`וַיֹּאמֶר דָּוִיד לִשְׁלֹמֹה בְּנוֹ אֲנִי הָיָה עִם לְבָבִי לִבְנוֹת בַּיִת לְשֵׁם יְהוָה אֱלֹהָי` (דברי-הימים-א כב:ז)
`עַתָּה תְּנוּ לְבַבְכֶם וְנַפְשְׁכֶם לִדְרוֹשׁ לַיהוָה אֱלֹהֵיכֶם וְקוּמוּ וּבְנוּ אֶת מִקְדַּשׁ יְהוָה הָאֱלֹהִים לְהָבִיא אֶת אֲרוֹן בְּרִית יְהוָה וּכְלֵי קֹדֶשׁ הָאֱלֹהִים לַבַּיִת הַנִּבְנֶה לְשֵׁם יְהוָה` (דברי-הימים-א כב:יט)
`וַיָּקָם דָּוִיד הַמֶּלֶךְ עַל רַגְלָיו וַיֹּאמֶר שְׁמָעוּנִי אַחַי וְעַמִּי אֲנִי עִם לְבָבִי לִבְנוֹת בֵּית מְנוּחָה לַאֲרוֹן בְּרִית יְהוָה וְלַהֲדֹם רַגְלֵי אֱלֹהֵינוּ וַהֲכִינוֹתִי לִבְנוֹת` (דברי-הימים-א כח:ב)
`וְאַתָּה שְׁלֹמֹה בְנִי דַּע אֶת אֱלֹהֵי אָבִיךָ וְעָבְדֵהוּ בְּלֵב שָׁלֵם וּבְנֶפֶשׁ חֲפֵצָה כִּי כָל לְבָבוֹת דּוֹרֵשׁ יְהוָה וְכָל יֵצֶר מַחֲשָׁבוֹת מֵבִין אִם תִּדְרְשֶׁנּוּ יִמָּצֵא לָךְ וְאִם תַּעַזְבֶנּוּ יַזְנִיחֲךָ לָעַד` (דברי-הימים-א כח:ט)
`וְיָדַעְתִּי אֱלֹהַי כִּי אַתָּה בֹּחֵן לֵבָב וּמֵישָׁרִים תִּרְצֶה אֲנִי בְּיֹשֶׁר לְבָבִי הִתְנַדַּבְתִּי כָל אֵלֶּה וְעַתָּה עַמְּךָ הַנִּמְצְאוּ פֹה רָאִיתִי בְשִׂמְחָה לְהִתְנַדֶּב לָךְ` (דברי-הימים-א כט:יז)
`יְהוָה אֱלֹהֵי אַבְרָהָם יִצְחָק וְיִשְׂרָאֵל אֲבֹתֵינוּ שֳׁמְרָה זֹּאת לְעוֹלָם לְיֵצֶר מַחְשְׁבוֹת לְבַב עַמֶּךָ וְהָכֵן לְבָבָם אֵלֶיךָ` (דברי-הימים-א כט:יח)
`וְלִשְׁלֹמֹה בְנִי תֵּן לֵבָב שָׁלֵם לִשְׁמוֹר מִצְוֺתֶיךָ עֵדְוֺתֶיךָ וְחֻקֶּיךָ וְלַעֲשׂוֹת הַכֹּל וְלִבְנוֹת הַבִּירָה אֲשֶׁר הֲכִינוֹתִי` (דברי-הימים-א כט:יט)
`וַיֹּאמֶר אֱלֹהִים לִשְׁלֹמֹה יַעַן אֲשֶׁר הָיְתָה זֹאת עִם לְבָבֶךָ וְלֹא שָׁאַלְתָּ עֹשֶׁר נְכָסִים וְכָבוֹד וְאֵת נֶפֶשׁ שֹׂנְאֶיךָ וְגַם יָמִים רַבִּים לֹא שָׁאָלְתָּ וַתִּשְׁאַל לְךָ חָכְמָה וּמַדָּע אֲשֶׁר תִּשְׁפּוֹט אֶת עַמִּי אֲשֶׁר הִמְלַכְתִּיךָ עָלָיו` (דברי-הימים-ב א:יא)
`וַיְהִי עִם לְבַב דָּוִיד אָבִי לִבְנוֹת בַּיִת לְשֵׁם יְהוָה אֱלֹהֵי יִשְׂרָאֵל` (דברי-הימים-ב ו:ז)
`וַיֹּאמֶר יְהוָה אֶל דָּוִיד אָבִי יַעַן אֲשֶׁר הָיָה עִם לְבָבְךָ לִבְנוֹת בַּיִת לִשְׁמִי הֱ‍טִיבוֹתָ כִּי הָיָה עִם לְבָבֶךָ` (דברי-הימים-ב ו:ח)
`וְאַתָּה תִּשְׁמַע מִן הַשָּׁמַיִם מְכוֹן שִׁבְתֶּךָ וְסָלַחְתָּ וְנָתַתָּה לָאִישׁ כְּכָל דְּרָכָיו אֲשֶׁר תֵּדַע אֶת לְבָבוֹ כִּי אַתָּה לְבַדְּךָ יָדַעְתָּ אֶת לְבַב בְּנֵי הָאָדָם` (דברי-הימים-ב ו:ל)
`וְהֵשִׁיבוּ אֶל לְבָבָם בָּאָרֶץ אֲשֶׁר נִשְׁבּוּ שָׁם וְשָׁבוּ וְהִתְחַנְּנוּ אֵלֶיךָ בְּאֶרֶץ שִׁבְיָם לֵאמֹר חָטָאנוּ הֶעֱוִינוּ וְרָשָׁעְנוּ` (דברי-הימים-ב ו:לז)
`וּמַלְכַּת שְׁבָא שָׁמְעָה אֶת שֵׁמַע שְׁלֹמֹה וַתָּבוֹא לְנַסּוֹת אֶת שְׁלֹמֹה בְחִידוֹת בִּירוּשָׁלִַם בְּחַיִל כָּבֵד מְאֹד וּגְמַלִּים נֹשְׂאִים בְּשָׂמִים וְזָהָב לָרֹב וְאֶבֶן יְקָרָה וַתָּבוֹא אֶל שְׁלֹמֹה וַתְּדַבֵּר עִמּוֹ אֵת כָּל אֲשֶׁר הָיָה עִם לְבָבָהּ` (דברי-הימים-ב ט:א)
`וְאַחֲרֵיהֶם מִכֹּל שִׁבְטֵי יִשְׂרָאֵל הַנֹּתְנִים אֶת לְבָבָם לְבַקֵּשׁ אֶת יְהוָה אֱלֹהֵי יִשְׂרָאֵל בָּאוּ יְרוּשָׁלִַם לִזְבּוֹחַ לַיהוָה אֱלֹהֵי אֲבוֹתֵיהֶם` (דברי-הימים-ב יא:טז)
`וַיִּקָּבְצוּ עָלָיו אֲנָשִׁים רֵקִים בְּנֵי בְלִיַּעַל וַיִּתְאַמְּצוּ עַל רְחַבְעָם בֶּן שְׁלֹמֹה וּרְחַבְעָם הָיָה נַעַר וְרַךְ לֵבָב וְלֹא הִתְחַזַּק לִפְנֵיהֶם` (דברי-הימים-ב יג:ז)
`וַיָּבֹאוּ בַבְּרִית לִדְרוֹשׁ אֶת יְהוָה אֱלֹהֵי אֲבוֹתֵיהֶם בְּכָל לְבָבָם וּבְכָל נַפְשָׁם` (דברי-הימים-ב טו:יב)
`וַיִּשְׂמְחוּ כָל יְהוּדָה עַל הַשְּׁבוּעָה כִּי בְכָל לְבָבָם נִשְׁבָּעוּ וּבְכָל רְצוֹנָם בִּקְשֻׁהוּ וַיִּמָּצֵא לָהֶם וַיָּנַח יְהוָה לָהֶם מִסָּבִיב` (דברי-הימים-ב טו:טו)
`וְהַבָּמוֹת לֹא סָרוּ מִיִּשְׂרָאֵל רַק לְבַב אָסָא הָיָה שָׁלֵם כָּל יָמָיו` (דברי-הימים-ב טו:יז)
`כִּי יְהוָה עֵינָיו מְשֹׁטְטוֹת בְּכָל הָאָרֶץ לְהִתְחַזֵּק עִם לְבָבָם שָׁלֵם אֵלָיו נִסְכַּלְתָּ עַל זֹאת כִּי מֵעַתָּה יֵשׁ עִמְּךָ מִלְחָמוֹת` (דברי-הימים-ב טז:ט)
`אֲבָל דְּבָרִים טוֹבִים נִמְצְאוּ עִמָּךְ כִּי בִעַרְתָּ הָאֲשֵׁרוֹת מִן הָאָרֶץ וַהֲכִינוֹתָ לְבָבְךָ לִדְרֹשׁ הָאֱלֹהִים` (דברי-הימים-ב יט:ג)
`וַיְצַו עֲלֵיהֶם לֵאמֹר כֹּה תַעֲשׂוּן בְּיִרְאַת יְהוָה בֶּאֱמוּנָה וּבְלֵבָב שָׁלֵם` (דברי-הימים-ב יט:ט)
`אַךְ הַבָּמוֹת לֹא סָרוּ וְעוֹד הָעָם לֹא הֵכִינוּ לְבָבָם לֵאלֹהֵי אֲבֹתֵיהֶם` (דברי-הימים-ב כ:לג)
`וַיְבַקֵּשׁ אֶת אֲחַזְיָהוּ וַיִּלְכְּדֻהוּ וְהוּא מִתְחַבֵּא בְשֹׁמְרוֹן וַיְבִאֻהוּ אֶל יֵהוּא וַיְמִתֻהוּ וַיִּקְבְּרֻהוּ כִּי אָמְרוּ בֶּן יְהוֹשָׁפָט הוּא אֲשֶׁר דָּרַשׁ אֶת יְהוָה בְּכָל לְבָבוֹ וְאֵין לְבֵית אֲחַזְיָהוּ לַעְצֹר כֹּחַ לְמַמְלָכָה` (דברי-הימים-ב כב:ט)
`וַיַּעַשׂ הַיָּשָׁר בְּעֵינֵי יְהוָה רַק לֹא בְּלֵבָב שָׁלֵם` (דברי-הימים-ב כה:ב)
`עַתָּה עִם לְבָבִי לִכְרוֹת בְּרִית לַיהוָה אֱלֹהֵי יִשְׂרָאֵל וְיָשֹׁב מִמֶּנּוּ חֲרוֹן אַפּוֹ` (דברי-הימים-ב כט:י)
`רַק הַכֹּהֲנִים הָיוּ לִמְעָט וְלֹא יָכְלוּ לְהַפְשִׁיט אֶת כָּל הָעֹלוֹת וַיְּחַזְּקוּם אֲחֵיהֶם הַלְוִיִּם עַד כְּלוֹת הַמְּלָאכָה וְעַד יִתְקַדְּשׁוּ הַכֹּהֲנִים כִּי הַלְוִיִּם יִשְׁרֵי לֵבָב לְהִתְקַדֵּשׁ מֵהַכֹּהֲנִים` (דברי-הימים-ב כט:לד)
`כָּל לְבָבוֹ הֵכִין לִדְרוֹשׁ הָאֱלֹהִים יְהוָה אֱלֹהֵי אֲבוֹתָיו וְלֹא כְּטָהֳרַת הַקֹּדֶשׁ` (דברי-הימים-ב ל:יט)
`וּבְכָל מַעֲשֶׂה אֲשֶׁר הֵחֵל בַּעֲבוֹדַת בֵּית הָאֱלֹהִים וּבַתּוֹרָה וּבַמִּצְוָה לִדְרֹשׁ לֵאלֹהָיו בְּכָל לְבָבוֹ עָשָׂה וְהִצְלִיחַ` (דברי-הימים-ב לא:כא)
`וַיִּתֵּן שָׂרֵי מִלְחָמוֹת עַל הָעָם וַיִּקְבְּצֵם אֵלָיו אֶל רְחוֹב שַׁעַר הָעִיר וַיְדַבֵּר עַל לְבָבָם לֵאמֹר` (דברי-הימים-ב לב:ו)
`וְכֵן בִּמְלִיצֵי שָׂרֵי בָּבֶל הַמְשַׁלְּחִים עָלָיו לִדְרֹשׁ הַמּוֹפֵת אֲשֶׁר הָיָה בָאָרֶץ עֲזָבוֹ הָאֱלֹהִים לְנַסּוֹתוֹ לָדַעַת כָּל בִּלְבָבוֹ` (דברי-הימים-ב לב:לא)
`יַעַן רַךְ לְבָבְךָ וַתִּכָּנַע מִלִּפְנֵי אֱלֹהִים בְּשָׁמְעֲךָ אֶת דְּבָרָיו עַל הַמָּקוֹם הַזֶּה וְעַל יֹשְׁבָיו וַתִּכָּנַע לְפָנַי וַתִּקְרַע אֶת בְּגָדֶיךָ וַתֵּבְךְּ לְפָנָי וְגַם אֲנִי שָׁמַעְתִּי נְאֻם יְהוָה` (דברי-הימים-ב לד:כז)
`וַיַּעֲמֹד הַמֶּלֶךְ עַל עָמְדוֹ וַיִּכְרֹת אֶת הַבְּרִית לִפְנֵי יְהוָה לָלֶכֶת אַחֲרֵי יְהוָה וְלִשְׁמוֹר אֶת מִצְוֺתָיו וְעֵדְוֺתָיו וְחֻקָּיו בְּכָל לְבָבוֹ וּבְכָל נַפְשׁוֹ לַעֲשׂוֹת אֶת דִּבְרֵי הַבְּרִית הַכְּתוּבִים עַל הַסֵּפֶר הַזֶּה` (דברי-הימים-ב לד:לא)
`וְגַם בַּמֶּלֶךְ נְבוּכַדְנֶאצַּר מָרָד אֲשֶׁר הִשְׁבִּיעוֹ בֵּאלֹהִים וַיֶּקֶשׁ אֶת עָרְפּוֹ וַיְאַמֵּץ אֶת לְבָבוֹ מִשּׁוּב אֶל יְהוָה אֱלֹהֵי יִשְׂרָאֵל` (דברי-הימים-ב לו:יג)
`רִגְזוּ וְאַל תֶּחֱטָאוּ אִמְרוּ בִלְבַבְכֶם עַל מִשְׁכַּבְכֶם וְדֹמּוּ סֶלָה` (תהילים ד:ד)
`עַד אָנָה אָשִׁית עֵצוֹת בְּנַפְשִׁי יָגוֹן בִּלְבָבִי יוֹמָם עַד אָנָה יָרוּם אֹיְבִי עָלָי` (תהילים יג:ב)
`הוֹלֵךְ תָּמִים וּפֹעֵל צֶדֶק וְדֹבֵר אֱמֶת בִּלְבָבוֹ` (תהילים טו:ב)
`יִתֶּן לְךָ כִלְבָבֶךָ וְכָל עֲצָתְךָ יְמַלֵּא` (תהילים כ:ד)
`יֹאכְלוּ עֲנָוִים וְיִשְׂבָּעוּ יְהַלְלוּ יְהוָה דֹּרְשָׁיו יְחִי לְבַבְכֶם לָעַד` (תהילים כב:כו)
`נְקִי כַפַּיִם וּבַר לֵבָב אֲשֶׁר לֹא נָשָׂא לַשָּׁוְא נַפְשִׁי וְלֹא נִשְׁבַּע לְמִרְמָה` (תהילים כד:ד)
`צָרוֹת לְבָבִי הִרְחִיבוּ מִמְּצוּקוֹתַי הוֹצִיאֵנִי` (תהילים כה:יז)
`אַל תִּמְשְׁכֵנִי עִם רְשָׁעִים וְעִם פֹּעֲלֵי אָוֶן דֹּבְרֵי שָׁלוֹם עִם רֵעֵיהֶם וְרָעָה בִּלְבָבָם` (תהילים כח:ג)
`חִזְקוּ וְיַאֲמֵץ לְבַבְכֶם כָּל הַמְיַחֲלִים לַיהוָה` (תהילים לא:כד)
`בִּטְחוּ בוֹ בְכָל עֵת עָם שִׁפְכוּ לְפָנָיו לְבַבְכֶם אֱלֹהִים מַחֲסֶה לָּנוּ סֶלָה` (תהילים סב:ח)
`רָאוּ עֲנָוִים יִשְׂמָחוּ דֹּרְשֵׁי אֱלֹהִים וִיחִי לְבַבְכֶם` (תהילים סט:לב)
`מִזְמוֹר לְאָסָף אַךְ טוֹב לְיִשְׂרָאֵל אֱלֹהִים לְבָרֵי לֵבָב` (תהילים עג:א)
`יָצָא מֵחֵלֶב עֵינֵמוֹ עָבְרוּ מַשְׂכִּיּוֹת לֵבָב` (תהילים עג:ז)
`אַךְ רִיק זִכִּיתִי לְבָבִי וָאֶרְחַץ בְּנִקָּיוֹן כַּפָּי` (תהילים עג:יג)
`כִּי יִתְחַמֵּץ לְבָבִי וְכִלְיוֹתַי אֶשְׁתּוֹנָן` (תהילים עג:כא)
`כָּלָה שְׁאֵרִי וּלְבָבִי צוּר לְבָבִי וְחֶלְקִי אֱלֹהִים לְעוֹלָם` (תהילים עג:כו)
`אֶזְכְּרָה נְגִינָתִי בַּלָּיְלָה עִם לְבָבִי אָשִׂיחָה וַיְחַפֵּשׂ רוּחִי` (תהילים עז:ו)
`וַיְנַסּוּ אֵל בִּלְבָבָם לִשְׁאָל אֹכֶל לְנַפְשָׁם` (תהילים עח:יח)
`וַיִּרְעֵם כְּתֹם לְבָבוֹ וּבִתְבוּנוֹת כַּפָּיו יַנְחֵם` (תהילים עח:עב)
`אַשְׁרֵי אָדָם עוֹז לוֹ בָךְ מְסִלּוֹת בִּלְבָבָם` (תהילים פד:ה)
`הוֹרֵנִי יְהוָה דַּרְכֶּךָ אֲהַלֵּךְ בַּאֲמִתֶּךָ יַחֵד לְבָבִי לְיִרְאָה שְׁמֶךָ` (תהילים פו:יא)
`אוֹדְךָ אֲדֹנָי אֱלֹהַי בְּכָל לְבָבִי וַאֲכַבְּדָה שִׁמְךָ לְעוֹלָם` (תהילים פו:יב)
`לִמְנוֹת יָמֵינוּ כֵּן הוֹדַע וְנָבִא לְבַב חָכְמָה` (תהילים צ:יב)
`אַל תַּקְשׁוּ לְבַבְכֶם כִּמְרִיבָה כְּיוֹם מַסָּה בַּמִּדְבָּר` (תהילים צה:ח)
`אַרְבָּעִים שָׁנָה אָקוּט בְּדוֹר וָאֹמַר עַם תֹּעֵי לֵבָב הֵם וְהֵם לֹא יָדְעוּ דְרָכָי` (תהילים צה:י)
`אַשְׂכִּילָה בְּדֶרֶךְ תָּמִים מָתַי תָּבוֹא אֵלָי אֶתְהַלֵּךְ בְּתָם לְבָבִי בְּקֶרֶב בֵּיתִי` (תהילים קא:ב)
`לֵבָב עִקֵּשׁ יָסוּר מִמֶּנִּי רָע לֹא אֵדָע` (תהילים קא:ד)
`מְלוֹשְׁנִי בַסֵּתֶר רֵעֵהוּ אוֹתוֹ אַצְמִית גְּבַהּ עֵינַיִם וּרְחַב לֵבָב אֹתוֹ לֹא אוּכָל` (תהילים קא:ה)
`וְיַיִן יְשַׂמַּח לְבַב אֱנוֹשׁ לְהַצְהִיל פָּנִים מִשָּׁמֶן וְלֶחֶם לְבַב אֱנוֹשׁ יִסְעָד` (תהילים קד:טו)
`יַעַן אֲשֶׁר לֹא זָכַר עֲשׂוֹת חָסֶד וַיִּרְדֹּף אִישׁ עָנִי וְאֶבְיוֹן וְנִכְאֵה לֵבָב לְמוֹתֵת` (תהילים קט:טז)
`הַלְלוּ יָהּ אוֹדֶה יְהוָה בְּכָל לֵבָב בְּסוֹד יְשָׁרִים וְעֵדָה` (תהילים קיא:א)
`אוֹדְךָ בְּיֹשֶׁר לֵבָב בְּלָמְדִי מִשְׁפְּטֵי צִדְקֶךָ` (תהילים קיט:ז)
`חָקְרֵנִי אֵל וְדַע לְבָבִי בְּחָנֵנִי וְדַע שַׂרְעַפָּי` (תהילים קלט:כג)
`וַיְהִי כִּי הִקִּיפוּ יְמֵי הַמִּשְׁתֶּה וַיִּשְׁלַח אִיּוֹב וַיְקַדְּשֵׁם וְהִשְׁכִּים בַּבֹּקֶר וְהֶעֱלָה עֹלוֹת מִסְפַּר כֻּלָּם כִּי אָמַר אִיּוֹב אוּלַי חָטְאוּ בָנַי וּבֵרֲכוּ אֱלֹהִים בִּלְבָבָם כָּכָה יַעֲשֶׂה אִיּוֹב כָּל הַיָּמִים` (איוב א:ה)
`חֲכַם לֵבָב וְאַמִּיץ כֹּחַ מִי הִקְשָׁה אֵלָיו וַיִּשְׁלָם` (איוב ט:ד)
`וְאֵלֶּה צָפַנְתָּ בִלְבָבֶךָ יָדַעְתִּי כִּי זֹאת עִמָּךְ` (איוב י:יג)
`גַּם לִי לֵבָב כְּמוֹכֶם לֹא נֹפֵל אָנֹכִי מִכֶּם וְאֶת מִי אֵין כְּמוֹ אֵלֶּה` (איוב יב:ג)
`יָמַי עָבְרוּ זִמֹּתַי נִתְּקוּ מוֹרָשֵׁי לְבָבִי` (איוב יז:יא)
`קַח נָא מִפִּיו תּוֹרָה וְשִׂים אֲמָרָיו בִּלְבָבֶךָ` (איוב כב:כב)
`בְּצִדְקָתִי הֶחֱזַקְתִּי וְלֹא אַרְפֶּהָ לֹא יֶחֱרַף לְבָבִי מִיָּמָי` (איוב כז:ו)
`לָכֵן אַנֲשֵׁי לֵבָב שִׁמְעוּ לִי חָלִלָה לָאֵל מֵרֶשַׁע וְשַׁדַּי מֵעָוֶל` (איוב לד:י)
`אַנְשֵׁי לֵבָב יֹאמְרוּ לִי וְגֶבֶר חָכָם שֹׁמֵעַ לִי` (איוב לד:לד)
`אַל יַלִּיזוּ מֵעֵינֶיךָ שָׁמְרֵם בְּתוֹךְ לְבָבֶךָ` (משלי ד:כא)
`אַל תַּחְמֹד יָפְיָהּ בִּלְבָבֶךָ וְאַל תִּקָּחֲךָ בְּעַפְעַפֶּיהָ` (משלי ו:כה)
`זֶה רָע בְּכֹל אֲשֶׁר נַעֲשָׂה תַּחַת הַשֶּׁמֶשׁ כִּי מִקְרֶה אֶחָד לַכֹּל וְגַם לֵב בְּנֵי הָאָדָם מָלֵא רָע וְהוֹלֵלוֹת בִּלְבָבָם בְּחַיֵּיהֶם וְאַחֲרָיו אֶל הַמֵּתִים` (קהלת ט:ג)
`נִשָּׂא לְבָבֵנוּ אֶל כַּפָּיִם אֶל אֵל בַּשָּׁמָיִם` (איכה ג:מא)
`וְעַל שִׂכְלוֹ וְהִצְלִיחַ מִרְמָה בְּיָדוֹ וּבִלְבָבוֹ יַגְדִּיל וּבְשַׁלְוָה יַשְׁחִית רַבִּים וְעַל שַׂר שָׂרִים יַעֲמֹד וּבְאֶפֶס יָד יִשָּׁבֵר` (דניאל ח:כה)
`וְנִשָּׂא הֶהָמוֹן יָרוּם לְבָבוֹ וְהִפִּיל רִבֹּאוֹת וְלֹא יָעוֹז` (דניאל יא:יב)
`וְיָעֵר כֹּחוֹ וּלְבָבוֹ עַל מֶלֶךְ הַנֶּגֶב בְּחַיִל גָּדוֹל וּמֶלֶךְ הַנֶּגֶב יִתְגָּרֶה לַמִּלְחָמָה בְּחַיִל גָּדוֹל וְעָצוּם עַד מְאֹד וְלֹא יַעֲמֹד כִּי יַחְשְׁבוּ עָלָיו מַחֲשָׁבוֹת` (דניאל יא:כה)
`וּשְׁנֵיהֶם הַמְּלָכִים לְבָבָם לְמֵרָע וְעַל שֻׁלְחָן אֶחָד כָּזָב יְדַבֵּרוּ וְלֹא תִצְלָח כִּי עוֹד קֵץ לַמּוֹעֵד` (דניאל יא:כז)
`וְיָשֹׁב אַרְצוֹ בִּרְכוּשׁ גָּדוֹל וּלְבָבוֹ עַל בְּרִית קֹדֶשׁ וְעָשָׂה וְשָׁב לְאַרְצוֹ` (דניאל יא:כח)
`כִּי עֶזְרָא הֵכִין לְבָבוֹ לִדְרוֹשׁ אֶת תּוֹרַת יְהוָה וְלַעֲשֹׂת וּלְלַמֵּד בְּיִשְׂרָאֵל חֹק וּמִשְׁפָּט` (עזרא ז:י)
`וּמָצָאתָ אֶת לְבָבוֹ נֶאֱמָן לְפָנֶיךָ וְכָרוֹת עִמּוֹ הַבְּרִית לָתֵת אֶת אֶרֶץ הַכְּנַעֲנִי הַחִתִּי הָאֱמֹרִי וְהַפְּרִזִּי וְהַיְבוּסִי וְהַגִּרְגָּשִׁי לָתֵת לְזַרְעוֹ וַתָּקֶם אֶת דְּבָרֶיךָ כִּי צַדִּיק אָתָּה` (נחמיה ט:ח)
