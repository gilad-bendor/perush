`וַיֹּאמֶר אֱלֹהִים יִשְׁרְצוּ הַמַּיִם שֶׁרֶץ נֶפֶשׁ חַיָּה וְעוֹף יְעוֹפֵף עַל הָאָרֶץ עַל פְּנֵי רְקִיעַ הַשָּׁמָיִם` (בראשית א:כ)
`וַיִּבְרָא אֱלֹהִים אֶת הַתַּנִּינִם הַגְּדֹלִים וְאֵת כָּל נֶפֶשׁ הַחַיָּה הָרֹמֶשֶׂת אֲשֶׁר שָׁרְצוּ הַמַּיִם לְמִינֵהֶם וְאֵת כָּל עוֹף כָּנָף לְמִינֵהוּ וַיַּרְא אֱלֹהִים כִּי טוֹב` (בראשית א:כא)
`וַיֹּאמֶר אֱלֹהִים תּוֹצֵא הָאָרֶץ נֶפֶשׁ חַיָּה לְמִינָהּ בְּהֵמָה וָרֶמֶשׂ וְחַיְתוֹ אֶרֶץ לְמִינָהּ וַיְהִי כֵן` (בראשית א:כד)
`וּלְכָל חַיַּת הָאָרֶץ וּלְכָל עוֹף הַשָּׁמַיִם וּלְכֹל רוֹמֵשׂ עַל הָאָרֶץ אֲשֶׁר בּוֹ נֶפֶשׁ חַיָּה אֶת כָּל יֶרֶק עֵשֶׂב לְאָכְלָה וַיְהִי כֵן` (בראשית א:ל)
`וַיִּיצֶר יְהוָה אֱלֹהִים אֶת הָאָדָם עָפָר מִן הָאֲדָמָה וַיִּפַּח בְּאַפָּיו נִשְׁמַת חַיִּים וַיְהִי הָאָדָם לְנֶפֶשׁ חַיָּה` (בראשית ב:ז)
`וַיִּצֶר יְהוָה אֱלֹהִים מִן הָאֲדָמָה כָּל חַיַּת הַשָּׂדֶה וְאֵת כָּל עוֹף הַשָּׁמַיִם וַיָּבֵא אֶל הָאָדָם לִרְאוֹת מַה יִּקְרָא לוֹ וְכֹל אֲשֶׁר יִקְרָא לוֹ הָאָדָם נֶפֶשׁ חַיָּה הוּא שְׁמוֹ` (בראשית ב:יט)
`אַךְ בָּשָׂר בְּנַפְשׁוֹ דָמוֹ לֹא תֹאכֵלוּ` (בראשית ט:ד)
`וְאַךְ אֶת דִּמְכֶם לְנַפְשֹׁתֵיכֶם אֶדְרֹשׁ מִיַּד כָּל חַיָּה אֶדְרְשֶׁנּוּ וּמִיַּד הָאָדָם מִיַּד אִישׁ אָחִיו אֶדְרֹשׁ אֶת נֶפֶשׁ הָאָדָם` (בראשית ט:ה)
`וְאֵת כָּל נֶפֶשׁ הַחַיָּה אֲשֶׁר אִתְּכֶם בָּעוֹף בַּבְּהֵמָה וּבְכָל חַיַּת הָאָרֶץ אִתְּכֶם מִכֹּל יֹצְאֵי הַתֵּבָה לְכֹל חַיַּת הָאָרֶץ` (בראשית ט:י)
`וַיֹּאמֶר אֱלֹהִים זֹאת אוֹת הַבְּרִית אֲשֶׁר אֲנִי נֹתֵן בֵּינִי וּבֵינֵיכֶם וּבֵין כָּל נֶפֶשׁ חַיָּה אֲשֶׁר אִתְּכֶם לְדֹרֹת עוֹלָם` (בראשית ט:יב)
`וְזָכַרְתִּי אֶת בְּרִיתִי אֲשֶׁר בֵּינִי וּבֵינֵיכֶם וּבֵין כָּל נֶפֶשׁ חַיָּה בְּכָל בָּשָׂר וְלֹא יִהְיֶה עוֹד הַמַּיִם לְמַבּוּל לְשַׁחֵת כָּל בָּשָׂר` (בראשית ט:טו)
`וְהָיְתָה הַקֶּשֶׁת בֶּעָנָן וּרְאִיתִיהָ לִזְכֹּר בְּרִית עוֹלָם בֵּין אֱלֹהִים וּבֵין כָּל נֶפֶשׁ חַיָּה בְּכָל בָּשָׂר אֲשֶׁר עַל הָאָרֶץ` (בראשית ט:טז)
`וַיִּקַּח אַבְרָם אֶת שָׂרַי אִשְׁתּוֹ וְאֶת לוֹט בֶּן אָחִיו וְאֶת כָּל רְכוּשָׁם אֲשֶׁר רָכָשׁוּ וְאֶת הַנֶּפֶשׁ אֲשֶׁר עָשׂוּ בְחָרָן וַיֵּצְאוּ לָלֶכֶת אַרְצָה כְּנַעַן וַיָּבֹאוּ אַרְצָה כְּנָעַן` (בראשית יב:ה)
`אִמְרִי נָא אֲחֹתִי אָתְּ לְמַעַן יִיטַב לִי בַעֲבוּרֵךְ וְחָיְתָה נַפְשִׁי בִּגְלָלֵךְ` (בראשית יב:יג)
`וַיֹּאמֶר מֶלֶךְ סְדֹם אֶל אַבְרָם תֶּן לִי הַנֶּפֶשׁ וְהָרְכֻשׁ קַח לָךְ` (בראשית יד:כא)
`וְעָרֵל זָכָר אֲשֶׁר לֹא יִמּוֹל אֶת בְּשַׂר עָרְלָתוֹ וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ אֶת בְּרִיתִי הֵפַר` (בראשית יז:יד)
`וַיְהִי כְהוֹצִיאָם אֹתָם הַחוּצָה וַיֹּאמֶר הִמָּלֵט עַל נַפְשֶׁךָ אַל תַּבִּיט אַחֲרֶיךָ וְאַל תַּעֲמֹד בְּכָל הַכִּכָּר הָהָרָה הִמָּלֵט פֶּן תִּסָּפֶה` (בראשית יט:יז)
`הִנֵּה נָא מָצָא עַבְדְּךָ חֵן בְּעֵינֶיךָ וַתַּגְדֵּל חַסְדְּךָ אֲשֶׁר עָשִׂיתָ עִמָּדִי לְהַחֲיוֹת אֶת נַפְשִׁי וְאָנֹכִי לֹא אוּכַל לְהִמָּלֵט הָהָרָה פֶּן תִּדְבָּקַנִי הָרָעָה וָמַתִּי` (בראשית יט:יט)
`הִנֵּה נָא הָעִיר הַזֹּאת קְרֹבָה לָנוּס שָׁמָּה וְהִיא מִצְעָר אִמָּלְטָה נָּא שָׁמָּה הֲלֹא מִצְעָר הִוא וּתְחִי נַפְשִׁי` (בראשית יט:כ)
`וַיְדַבֵּר אִתָּם לֵאמֹר אִם יֵשׁ אֶת נַפְשְׁכֶם לִקְבֹּר אֶת מֵתִי מִלְּפָנַי שְׁמָעוּנִי וּפִגְעוּ לִי בְּעֶפְרוֹן בֶּן צֹחַר` (בראשית כג:ח)
`וַעֲשֵׂה לִי מַטְעַמִּים כַּאֲשֶׁר אָהַבְתִּי וְהָבִיאָה לִּי וְאֹכֵלָה בַּעֲבוּר תְּבָרֶכְךָ נַפְשִׁי בְּטֶרֶם אָמוּת` (בראשית כז:ד)
`וַיֹּאמֶר יַעֲקֹב אֶל אָבִיו אָנֹכִי עֵשָׂו בְּכֹרֶךָ עָשִׂיתִי כַּאֲשֶׁר דִּבַּרְתָּ אֵלָי קוּם נָא שְׁבָה וְאָכְלָה מִצֵּידִי בַּעֲבוּר תְּבָרֲכַנִּי נַפְשֶׁךָ` (בראשית כז:יט)
`וַיֹּאמֶר הַגִּשָׁה לִּי וְאֹכְלָה מִצֵּיד בְּנִי לְמַעַן תְּבָרֶכְךָ נַפְשִׁי וַיַּגֶּשׁ לוֹ וַיֹּאכַל וַיָּבֵא לוֹ יַיִן וַיֵּשְׁתְּ` (בראשית כז:כה)
`וַיַּעַשׂ גַּם הוּא מַטְעַמִּים וַיָּבֵא לְאָבִיו וַיֹּאמֶר לְאָבִיו יָקֻם אָבִי וְיֹאכַל מִצֵּיד בְּנוֹ בַּעֲבוּר תְּבָרֲכַנִּי נַפְשֶׁךָ` (בראשית כז:לא)
`וַיִּקְרָא יַעֲקֹב שֵׁם הַמָּקוֹם פְּנִיאֵל כִּי רָאִיתִי אֱלֹהִים פָּנִים אֶל פָּנִים וַתִּנָּצֵל נַפְשִׁי` (בראשית לב:ל)
`וַתִּדְבַּק נַפְשׁוֹ בְּדִינָה בַּת יַעֲקֹב וַיֶּאֱהַב אֶת הַנַּעֲרָ וַיְדַבֵּר עַל לֵב הַנַּעֲרָ` (בראשית לד:ג)
`וַיְדַבֵּר חֲמוֹר אִתָּם לֵאמֹר שְׁכֶם בְּנִי חָשְׁקָה נַפְשׁוֹ בְּבִתְּכֶם תְּנוּ נָא אֹתָהּ לוֹ לְאִשָּׁה` (בראשית לד:ח)
`וַיְהִי בְּצֵאת נַפְשָׁהּ כִּי מֵתָה וַתִּקְרָא שְׁמוֹ בֶּן אוֹנִי וְאָבִיו קָרָא לוֹ בִנְיָמִין` (בראשית לה:יח)
`וַיִּקַּח עֵשָׂו אֶת נָשָׁיו וְאֶת בָּנָיו וְאֶת בְּנֹתָיו וְאֶת כָּל נַפְשׁוֹת בֵּיתוֹ וְאֶת מִקְנֵהוּ וְאֶת כָּל בְּהֶמְתּוֹ וְאֵת כָּל קִנְיָנוֹ אֲשֶׁר רָכַשׁ בְּאֶרֶץ כְּנָעַן וַיֵּלֶךְ אֶל אֶרֶץ מִפְּנֵי יַעֲקֹב אָחִיו` (בראשית לו:ו)
`וַיִּשְׁמַע רְאוּבֵן וַיַּצִּלֵהוּ מִיָּדָם וַיֹּאמֶר לֹא נַכֶּנּוּ נָפֶשׁ` (בראשית לז:כא)
`וַיֹּאמְרוּ אִישׁ אֶל אָחִיו אֲבָל אֲשֵׁמִים אֲנַחְנוּ עַל אָחִינוּ אֲשֶׁר רָאִינוּ צָרַת נַפְשׁוֹ בְּהִתְחַנְנוֹ אֵלֵינוּ וְלֹא שָׁמָעְנוּ עַל כֵּן בָּאָה אֵלֵינוּ הַצָּרָה הַזֹּאת` (בראשית מב:כא)
`וְעַתָּה כְּבֹאִי אֶל עַבְדְּךָ אָבִי וְהַנַּעַר אֵינֶנּוּ אִתָּנוּ וְנַפְשׁוֹ קְשׁוּרָה בְנַפְשׁוֹ` (בראשית מד:ל)
`אֵלֶּה בְּנֵי לֵאָה אֲשֶׁר יָלְדָה לְיַעֲקֹב בְּפַדַּן אֲרָם וְאֵת דִּינָה בִתּוֹ כָּל נֶפֶשׁ בָּנָיו וּבְנוֹתָיו שְׁלֹשִׁים וְשָׁלֹשׁ` (בראשית מו:טו)
`אֵלֶּה בְּנֵי זִלְפָּה אֲשֶׁר נָתַן לָבָן לְלֵאָה בִתּוֹ וַתֵּלֶד אֶת אֵלֶּה לְיַעֲקֹב שֵׁשׁ עֶשְׂרֵה נָפֶשׁ` (בראשית מו:יח)
`אֵלֶּה בְּנֵי רָחֵל אֲשֶׁר יֻלַּד לְיַעֲקֹב כָּל נֶפֶשׁ אַרְבָּעָה עָשָׂר` (בראשית מו:כב)
`אֵלֶּה בְּנֵי בִלְהָה אֲשֶׁר נָתַן לָבָן לְרָחֵל בִּתּוֹ וַתֵּלֶד אֶת אֵלֶּה לְיַעֲקֹב כָּל נֶפֶשׁ שִׁבְעָה` (בראשית מו:כה)
`כָּל הַנֶּפֶשׁ הַבָּאָה לְיַעֲקֹב מִצְרַיְמָה יֹצְאֵי יְרֵכוֹ מִלְּבַד נְשֵׁי בְנֵי יַעֲקֹב כָּל נֶפֶשׁ שִׁשִּׁים וָשֵׁשׁ` (בראשית מו:כו)
`וּבְנֵי יוֹסֵף אֲשֶׁר יֻלַּד לוֹ בְמִצְרַיִם נֶפֶשׁ שְׁנָיִם כָּל הַנֶּפֶשׁ לְבֵית יַעֲקֹב הַבָּאָה מִצְרַיְמָה שִׁבְעִים` (בראשית מו:כז)
`בְּסֹדָם אַל תָּבֹא נַפְשִׁי בִּקְהָלָם אַל תֵּחַד כְּבֹדִי כִּי בְאַפָּם הָרְגוּ אִישׁ וּבִרְצֹנָם עִקְּרוּ שׁוֹר` (בראשית מט:ו)
`וַיְהִי כָּל נֶפֶשׁ יֹצְאֵי יֶרֶךְ יַעֲקֹב שִׁבְעִים נָפֶשׁ וְיוֹסֵף הָיָה בְמִצְרָיִם` (שמות א:ה)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה בְּמִדְיָן לֵךְ שֻׁב מִצְרָיִם כִּי מֵתוּ כָּל הָאֲנָשִׁים הַמְבַקְשִׁים אֶת נַפְשֶׁךָ` (שמות ד:יט)
`וְאִם יִמְעַט הַבַּיִת מִהְיֹת מִשֶּׂה וְלָקַח הוּא וּשְׁכֵנוֹ הַקָּרֹב אֶל בֵּיתוֹ בְּמִכְסַת נְפָשֹׁת אִישׁ לְפִי אָכְלוֹ תָּכֹסּוּ עַל הַשֶּׂה` (שמות יב:ד)
`שִׁבְעַת יָמִים מַצּוֹת תֹּאכֵלוּ אַךְ בַּיּוֹם הָרִאשׁוֹן תַּשְׁבִּיתוּ שְּׂאֹר מִבָּתֵּיכֶם כִּי כָּל אֹכֵל חָמֵץ וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִיִּשְׂרָאֵל מִיּוֹם הָרִאשֹׁן עַד יוֹם הַשְּׁבִעִי` (שמות יב:טו)
`וּבַיּוֹם הָרִאשׁוֹן מִקְרָא קֹדֶשׁ וּבַיּוֹם הַשְּׁבִיעִי מִקְרָא קֹדֶשׁ יִהְיֶה לָכֶם כָּל מְלָאכָה לֹא יֵעָשֶׂה בָהֶם אַךְ אֲשֶׁר יֵאָכֵל לְכָל נֶפֶשׁ הוּא לְבַדּוֹ יֵעָשֶׂה לָכֶם` (שמות יב:טז)
`שִׁבְעַת יָמִים שְׂאֹר לֹא יִמָּצֵא בְּבָתֵּיכֶם כִּי כָּל אֹכֵל מַחְמֶצֶת וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעֲדַת יִשְׂרָאֵל בַּגֵּר וּבְאֶזְרַח הָאָרֶץ` (שמות יב:יט)
`אָמַר אוֹיֵב אֶרְדֹּף אַשִּׂיג אֲחַלֵּק שָׁלָל תִּמְלָאֵמוֹ נַפְשִׁי אָרִיק חַרְבִּי תּוֹרִישֵׁמוֹ יָדִי` (שמות טו:ט)
`זֶה הַדָּבָר אֲשֶׁר צִוָּה יְהוָה לִקְטוּ מִמֶּנּוּ אִישׁ לְפִי אָכְלוֹ עֹמֶר לַגֻּלְגֹּלֶת מִסְפַּר נַפְשֹׁתֵיכֶם אִישׁ לַאֲשֶׁר בְּאָהֳלוֹ תִּקָּחוּ` (שמות טז:טז)
`וְאִם אָסוֹן יִהְיֶה וְנָתַתָּה נֶפֶשׁ תַּחַת נָפֶשׁ` (שמות כא:כג)
`אִם כֹּפֶר יוּשַׁת עָלָיו וְנָתַן פִּדְיֹן נַפְשׁוֹ כְּכֹל אֲשֶׁר יוּשַׁת עָלָיו` (שמות כא:ל)
`וְגֵר לֹא תִלְחָץ וְאַתֶּם יְדַעְתֶּם אֶת נֶפֶשׁ הַגֵּר כִּי גֵרִים הֱיִיתֶם בְּאֶרֶץ מִצְרָיִם` (שמות כג:ט)
`כִּי תִשָּׂא אֶת רֹאשׁ בְּנֵי יִשְׂרָאֵל לִפְקֻדֵיהֶם וְנָתְנוּ אִישׁ כֹּפֶר נַפְשׁוֹ לַיהוָה בִּפְקֹד אֹתָם וְלֹא יִהְיֶה בָהֶם נֶגֶף בִּפְקֹד אֹתָם` (שמות ל:יב)
`הֶעָשִׁיר לֹא יַרְבֶּה וְהַדַּל לֹא יַמְעִיט מִמַּחֲצִית הַשָּׁקֶל לָתֵת אֶת תְּרוּמַת יְהוָה לְכַפֵּר עַל נַפְשֹׁתֵיכֶם` (שמות ל:טו)
`וְלָקַחְתָּ אֶת כֶּסֶף הַכִּפֻּרִים מֵאֵת בְּנֵי יִשְׂרָאֵל וְנָתַתָּ אֹתוֹ עַל עֲבֹדַת אֹהֶל מוֹעֵד וְהָיָה לִבְנֵי יִשְׂרָאֵל לְזִכָּרוֹן לִפְנֵי יְהוָה לְכַפֵּר עַל נַפְשֹׁתֵיכֶם` (שמות ל:טז)
`וּשְׁמַרְתֶּם אֶת הַשַּׁבָּת כִּי קֹדֶשׁ הִוא לָכֶם מְחַלְלֶיהָ מוֹת יוּמָת כִּי כָּל הָעֹשֶׂה בָהּ מְלָאכָה וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִקֶּרֶב עַמֶּיהָ` (שמות לא:יד)
`וְנֶפֶשׁ כִּי תַקְרִיב קָרְבַּן מִנְחָה לַיהוָה סֹלֶת יִהְיֶה קָרְבָּנוֹ וְיָצַק עָלֶיהָ שֶׁמֶן וְנָתַן עָלֶיהָ לְבֹנָה` (ויקרא ב:א)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל לֵאמֹר נֶפֶשׁ כִּי תֶחֱטָא בִשְׁגָגָה מִכֹּל מִצְוֺת יְהוָה אֲשֶׁר לֹא תֵעָשֶׂינָה וְעָשָׂה מֵאַחַת מֵהֵנָּה` (ויקרא ד:ב)
`וְאִם נֶפֶשׁ אַחַת תֶּחֱטָא בִשְׁגָגָה מֵעַם הָאָרֶץ בַּעֲשֹׂתָהּ אַחַת מִמִּצְוֺת יְהוָה אֲשֶׁר לֹא תֵעָשֶׂינָה וְאָשֵׁם` (ויקרא ד:כז)
`וְנֶפֶשׁ כִּי תֶחֱטָא וְשָׁמְעָה קוֹל אָלָה וְהוּא עֵד אוֹ רָאָה אוֹ יָדָע אִם לוֹא יַגִּיד וְנָשָׂא עֲוֺנוֹ` (ויקרא ה:א)
`אוֹ נֶפֶשׁ אֲשֶׁר תִּגַּע בְּכָל דָּבָר טָמֵא אוֹ בְנִבְלַת חַיָּה טְמֵאָה אוֹ בְּנִבְלַת בְּהֵמָה טְמֵאָה אוֹ בְּנִבְלַת שֶׁרֶץ טָמֵא וְנֶעְלַם מִמֶּנּוּ וְהוּא טָמֵא וְאָשֵׁם` (ויקרא ה:ב)
`אוֹ נֶפֶשׁ כִּי תִשָּׁבַע לְבַטֵּא בִשְׂפָתַיִם לְהָרַע אוֹ לְהֵיטִיב לְכֹל אֲשֶׁר יְבַטֵּא הָאָדָם בִּשְׁבֻעָה וְנֶעְלַם מִמֶּנּוּ וְהוּא יָדַע וְאָשֵׁם לְאַחַת מֵאֵלֶּה` (ויקרא ה:ד)
`נֶפֶשׁ כִּי תִמְעֹל מַעַל וְחָטְאָה בִּשְׁגָגָה מִקָּדְשֵׁי יְהוָה וְהֵבִיא אֶת אֲשָׁמוֹ לַיהוָה אַיִל תָּמִים מִן הַצֹּאן בְּעֶרְכְּךָ כֶּסֶף שְׁקָלִים בְּשֶׁקֶל הַקֹּדֶשׁ לְאָשָׁם` (ויקרא ה:טו)
`וְאִם נֶפֶשׁ כִּי תֶחֱטָא וְעָשְׂתָה אַחַת מִכָּל מִצְוֺת יְהוָה אֲשֶׁר לֹא תֵעָשֶׂינָה וְלֹא יָדַע וְאָשֵׁם וְנָשָׂא עֲוֺנוֹ` (ויקרא ה:יז)
`נֶפֶשׁ כִּי תֶחֱטָא וּמָעֲלָה מַעַל בַּיהוָה וְכִחֵשׁ בַּעֲמִיתוֹ בְּפִקָּדוֹן אוֹ בִתְשׂוּמֶת יָד אוֹ בְגָזֵל אוֹ עָשַׁק אֶת עֲמִיתוֹ` (ויקרא ו:ב)
`וְאִם הֵאָכֹל יֵאָכֵל מִבְּשַׂר זֶבַח שְׁלָמָיו בַּיּוֹם הַשְּׁלִישִׁי לֹא יֵרָצֶה הַמַּקְרִיב אֹתוֹ לֹא יֵחָשֵׁב לוֹ פִּגּוּל יִהְיֶה וְהַנֶּפֶשׁ הָאֹכֶלֶת מִמֶּנּוּ עֲוֺנָהּ תִּשָּׂא` (ויקרא ז:יח)
`וְהַנֶּפֶשׁ אֲשֶׁר תֹּאכַל בָּשָׂר מִזֶּבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה וְטֻמְאָתוֹ עָלָיו וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא ז:כ)
`וְנֶפֶשׁ כִּי תִגַּע בְּכָל טָמֵא בְּטֻמְאַת אָדָם אוֹ בִּבְהֵמָה טְמֵאָה אוֹ בְּכָל שֶׁקֶץ טָמֵא וְאָכַל מִבְּשַׂר זֶבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא ז:כא)
`כִּי כָּל אֹכֵל חֵלֶב מִן הַבְּהֵמָה אֲשֶׁר יַקְרִיב מִמֶּנָּה אִשֶּׁה לַיהוָה וְנִכְרְתָה הַנֶּפֶשׁ הָאֹכֶלֶת מֵעַמֶּיהָ` (ויקרא ז:כה)
`כָּל נֶפֶשׁ אֲשֶׁר תֹּאכַל כָּל דָּם וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא ז:כז)
`וְכֹל אֲשֶׁר אֵין לוֹ סְנַפִּיר וְקַשְׂקֶשֶׂת בַּיַּמִּים וּבַנְּחָלִים מִכֹּל שֶׁרֶץ הַמַּיִם וּמִכֹּל נֶפֶשׁ הַחַיָּה אֲשֶׁר בַּמָּיִם שֶׁקֶץ הֵם לָכֶם` (ויקרא יא:י)
`אַל תְּשַׁקְּצוּ אֶת נַפְשֹׁתֵיכֶם בְּכָל הַשֶּׁרֶץ הַשֹּׁרֵץ וְלֹא תִטַּמְּאוּ בָּהֶם וְנִטְמֵתֶם בָּם` (ויקרא יא:מג)
`כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם וְהִתְקַדִּשְׁתֶּם וִהְיִיתֶם קְדֹשִׁים כִּי קָדוֹשׁ אָנִי וְלֹא תְטַמְּאוּ אֶת נַפְשֹׁתֵיכֶם בְּכָל הַשֶּׁרֶץ הָרֹמֵשׂ עַל הָאָרֶץ` (ויקרא יא:מד)
`זֹאת תּוֹרַת הַבְּהֵמָה וְהָעוֹף וְכֹל נֶפֶשׁ הַחַיָּה הָרֹמֶשֶׂת בַּמָּיִם וּלְכָל נֶפֶשׁ הַשֹּׁרֶצֶת עַל הָאָרֶץ` (ויקרא יא:מו)
`וְהָיְתָה לָכֶם לְחֻקַּת עוֹלָם בַּחֹדֶשׁ הַשְּׁבִיעִי בֶּעָשׂוֹר לַחֹדֶשׁ תְּעַנּוּ אֶת נַפְשֹׁתֵיכֶם וְכָל מְלָאכָה לֹא תַעֲשׂוּ הָאֶזְרָח וְהַגֵּר הַגָּר בְּתוֹכְכֶם` (ויקרא טז:כט)
`שַׁבַּת שַׁבָּתוֹן הִיא לָכֶם וְעִנִּיתֶם אֶת נַפְשֹׁתֵיכֶם חֻקַּת עוֹלָם` (ויקרא טז:לא)
`וְאִישׁ אִישׁ מִבֵּית יִשְׂרָאֵל וּמִן הַגֵּר הַגָּר בְּתוֹכָם אֲשֶׁר יֹאכַל כָּל דָּם וְנָתַתִּי פָנַי בַּנֶּפֶשׁ הָאֹכֶלֶת אֶת הַדָּם וְהִכְרַתִּי אֹתָהּ מִקֶּרֶב עַמָּהּ` (ויקרא יז:י)
`כִּי נֶפֶשׁ הַבָּשָׂר בַּדָּם הִוא וַאֲנִי נְתַתִּיו לָכֶם עַל הַמִּזְבֵּחַ לְכַפֵּר עַל נַפְשֹׁתֵיכֶם כִּי הַדָּם הוּא בַּנֶּפֶשׁ יְכַפֵּר` (ויקרא יז:יא)
`עַל כֵּן אָמַרְתִּי לִבְנֵי יִשְׂרָאֵל כָּל נֶפֶשׁ מִכֶּם לֹא תֹאכַל דָּם וְהַגֵּר הַגָּר בְּתוֹכְכֶם לֹא יֹאכַל דָּם` (ויקרא יז:יב)
`כִּי נֶפֶשׁ כָּל בָּשָׂר דָּמוֹ בְנַפְשׁוֹ הוּא וָאֹמַר לִבְנֵי יִשְׂרָאֵל דַּם כָּל בָּשָׂר לֹא תֹאכֵלוּ כִּי נֶפֶשׁ כָּל בָּשָׂר דָּמוֹ הִוא כָּל אֹכְלָיו יִכָּרֵת` (ויקרא יז:יד)
`וְכָל נֶפֶשׁ אֲשֶׁר תֹּאכַל נְבֵלָה וּטְרֵפָה בָּאֶזְרָח וּבַגֵּר וְכִבֶּס בְּגָדָיו וְרָחַץ בַּמַּיִם וְטָמֵא עַד הָעֶרֶב וְטָהֵר` (ויקרא יז:טו)
`כִּי כָּל אֲשֶׁר יַעֲשֶׂה מִכֹּל הַתּוֹעֵבוֹת הָאֵלֶּה וְנִכְרְתוּ הַנְּפָשׁוֹת הָעֹשֹׂת מִקֶּרֶב עַמָּם` (ויקרא יח:כט)
`וְאֹכְלָיו עֲוֺנוֹ יִשָּׂא כִּי אֶת קֹדֶשׁ יְהוָה חִלֵּל וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא יט:ח)
`וְשֶׂרֶט לָנֶפֶשׁ לֹא תִתְּנוּ בִּבְשַׂרְכֶם וּכְתֹבֶת קַעֲקַע לֹא תִתְּנוּ בָּכֶם אֲנִי יְהוָה` (ויקרא יט:כח)
`וְהַנֶּפֶשׁ אֲשֶׁר תִּפְנֶה אֶל הָאֹבֹת וְאֶל הַיִּדְּעֹנִים לִזְנוֹת אַחֲרֵיהֶם וְנָתַתִּי אֶת פָּנַי בַּנֶּפֶשׁ הַהִוא וְהִכְרַתִּי אֹתוֹ מִקֶּרֶב עַמּוֹ` (ויקרא כ:ו)
`וְהִבְדַּלְתֶּם בֵּין הַבְּהֵמָה הַטְּהֹרָה לַטְּמֵאָה וּבֵין הָעוֹף הַטָּמֵא לַטָּהֹר וְלֹא תְשַׁקְּצוּ אֶת נַפְשֹׁתֵיכֶם בַּבְּהֵמָה וּבָעוֹף וּבְכֹל אֲשֶׁר תִּרְמֹשׂ הָאֲדָמָה אֲשֶׁר הִבְדַּלְתִּי לָכֶם לְטַמֵּא` (ויקרא כ:כה)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה אֱמֹר אֶל הַכֹּהֲנִים בְּנֵי אַהֲרֹן וְאָמַרְתָּ אֲלֵהֶם לְנֶפֶשׁ לֹא יִטַּמָּא בְּעַמָּיו` (ויקרא כא:א)
`וְעַל כָּל נַפְשֹׁת מֵת לֹא יָבֹא לְאָבִיו וּלְאִמּוֹ לֹא יִטַּמָּא` (ויקרא כא:יא)
`אֱמֹר אֲלֵהֶם לְדֹרֹתֵיכֶם כָּל אִישׁ אֲשֶׁר יִקְרַב מִכָּל זַרְעֲכֶם אֶל הַקֳּדָשִׁים אֲשֶׁר יַקְדִּישׁוּ בְנֵי יִשְׂרָאֵל לַיהוָה וְטֻמְאָתוֹ עָלָיו וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִלְּפָנַי אֲנִי יְהוָה` (ויקרא כב:ג)
`אִישׁ אִישׁ מִזֶּרַע אַהֲרֹן וְהוּא צָרוּעַ אוֹ זָב בַּקֳּדָשִׁים לֹא יֹאכַל עַד אֲשֶׁר יִטְהָר וְהַנֹּגֵעַ בְּכָל טְמֵא נֶפֶשׁ אוֹ אִישׁ אֲשֶׁר תֵּצֵא מִמֶּנּוּ שִׁכְבַת זָרַע` (ויקרא כב:ד)
`נֶפֶשׁ אֲשֶׁר תִּגַּע בּוֹ וְטָמְאָה עַד הָעָרֶב וְלֹא יֹאכַל מִן הַקֳּדָשִׁים כִּי אִם רָחַץ בְּשָׂרוֹ בַּמָּיִם` (ויקרא כב:ו)
`וְכֹהֵן כִּי יִקְנֶה נֶפֶשׁ קִנְיַן כַּסְפּוֹ הוּא יֹאכַל בּוֹ וִילִיד בֵּיתוֹ הֵם יֹאכְלוּ בְלַחְמוֹ` (ויקרא כב:יא)
`אַךְ בֶּעָשׂוֹר לַחֹדֶשׁ הַשְּׁבִיעִי הַזֶּה יוֹם הַכִּפֻּרִים הוּא מִקְרָא קֹדֶשׁ יִהְיֶה לָכֶם וְעִנִּיתֶם אֶת נַפְשֹׁתֵיכֶם וְהִקְרַבְתֶּם אִשֶּׁה לַיהוָה` (ויקרא כג:כז)
`כִּי כָל הַנֶּפֶשׁ אֲשֶׁר לֹא תְעֻנֶּה בְּעֶצֶם הַיּוֹם הַזֶּה וְנִכְרְתָה מֵעַמֶּיהָ` (ויקרא כג:כט)
`וְכָל הַנֶּפֶשׁ אֲשֶׁר תַּעֲשֶׂה כָּל מְלָאכָה בְּעֶצֶם הַיּוֹם הַזֶּה וְהַאֲבַדְתִּי אֶת הַנֶּפֶשׁ הַהִוא מִקֶּרֶב עַמָּהּ` (ויקרא כג:ל)
`שַׁבַּת שַׁבָּתוֹן הוּא לָכֶם וְעִנִּיתֶם אֶת נַפְשֹׁתֵיכֶם בְּתִשְׁעָה לַחֹדֶשׁ בָּעֶרֶב מֵעֶרֶב עַד עֶרֶב תִּשְׁבְּתוּ שַׁבַּתְּכֶם` (ויקרא כג:לב)
`וְאִישׁ כִּי יַכֶּה כָּל נֶפֶשׁ אָדָם מוֹת יוּמָת` (ויקרא כד:יז)
`וּמַכֵּה נֶפֶשׁ בְּהֵמָה יְשַׁלְּמֶנָּה נֶפֶשׁ תַּחַת נָפֶשׁ` (ויקרא כד:יח)
`וְנָתַתִּי מִשְׁכָּנִי בְּתוֹכְכֶם וְלֹא תִגְעַל נַפְשִׁי אֶתְכֶם` (ויקרא כו:יא)
`וְאִם בְּחֻקֹּתַי תִּמְאָסוּ וְאִם אֶת מִשְׁפָּטַי תִּגְעַל נַפְשְׁכֶם לְבִלְתִּי עֲשׂוֹת אֶת כָּל מִצְוֺתַי לְהַפְרְכֶם אֶת בְּרִיתִי` (ויקרא כו:טו)
`אַף אֲנִי אֶעֱשֶׂה זֹּאת לָכֶם וְהִפְקַדְתִּי עֲלֵיכֶם בֶּהָלָה אֶת הַשַּׁחֶפֶת וְאֶת הַקַּדַּחַת מְכַלּוֹת עֵינַיִם וּמְדִיבֹת נָפֶשׁ וּזְרַעְתֶּם לָרִיק זַרְעֲכֶם וַאֲכָלֻהוּ אֹיְבֵיכֶם` (ויקרא כו:טז)
`וְהִשְׁמַדְתִּי אֶת בָּמֹתֵיכֶם וְהִכְרַתִּי אֶת חַמָּנֵיכֶם וְנָתַתִּי אֶת פִּגְרֵיכֶם עַל פִּגְרֵי גִּלּוּלֵיכֶם וְגָעֲלָה נַפְשִׁי אֶתְכֶם` (ויקרא כו:ל)
`וְהָאָרֶץ תֵּעָזֵב מֵהֶם וְתִרֶץ אֶת שַׁבְּתֹתֶיהָ בָּהְשַׁמָּה מֵהֶם וְהֵם יִרְצוּ אֶת עֲוֺנָם יַעַן וּבְיַעַן בְּמִשְׁפָּטַי מָאָסוּ וְאֶת חֻקֹּתַי גָּעֲלָה נַפְשָׁם` (ויקרא כו:מג)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אִישׁ כִּי יַפְלִא נֶדֶר בְּעֶרְכְּךָ נְפָשֹׁת לַיהוָה` (ויקרא כז:ב)
`צַו אֶת בְּנֵי יִשְׂרָאֵל וִישַׁלְּחוּ מִן הַמַּחֲנֶה כָּל צָרוּעַ וְכָל זָב וְכֹל טָמֵא לָנָפֶשׁ` (במדבר ה:ב)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל אִישׁ אוֹ אִשָּׁה כִּי יַעֲשׂוּ מִכָּל חַטֹּאת הָאָדָם לִמְעֹל מַעַל בַּיהוָה וְאָשְׁמָה הַנֶּפֶשׁ הַהִוא` (במדבר ה:ו)
`כָּל יְמֵי הַזִּירוֹ לַיהוָה עַל נֶפֶשׁ מֵת לֹא יָבֹא` (במדבר ו:ו)
`וְעָשָׂה הַכֹּהֵן אֶחָד לְחַטָּאת וְאֶחָד לְעֹלָה וְכִפֶּר עָלָיו מֵאֲשֶׁר חָטָא עַל הַנָּפֶשׁ וְקִדַּשׁ אֶת רֹאשׁוֹ בַּיּוֹם הַהוּא` (במדבר ו:יא)
`וַיְהִי אֲנָשִׁים אֲשֶׁר הָיוּ טְמֵאִים לְנֶפֶשׁ אָדָם וְלֹא יָכְלוּ לַעֲשֹׂת הַפֶּסַח בַּיּוֹם הַהוּא וַיִּקְרְבוּ לִפְנֵי מֹשֶׁה וְלִפְנֵי אַהֲרֹן בַּיּוֹם הַהוּא` (במדבר ט:ו)
`וַיֹּאמְרוּ הָאֲנָשִׁים הָהֵמָּה אֵלָיו אֲנַחְנוּ טְמֵאִים לְנֶפֶשׁ אָדָם לָמָּה נִגָּרַע לְבִלְתִּי הַקְרִב אֶת קָרְבַּן יְהוָה בְּמֹעֲדוֹ בְּתוֹךְ בְּנֵי יִשְׂרָאֵל` (במדבר ט:ז)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל לֵאמֹר אִישׁ אִישׁ כִּי יִהְיֶה טָמֵא לָנֶפֶשׁ אוֹ בְדֶרֶךְ רְחֹקָהׄ לָכֶם אוֹ לְדֹרֹתֵיכֶם וְעָשָׂה פֶסַח לַיהוָה` (במדבר ט:י)
`וְהָאִישׁ אֲשֶׁר הוּא טָהוֹר וּבְדֶרֶךְ לֹא הָיָה וְחָדַל לַעֲשׂוֹת הַפֶּסַח וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ כִּי קָרְבַּן יְהוָה לֹא הִקְרִיב בְּמֹעֲדוֹ חֶטְאוֹ יִשָּׂא הָאִישׁ הַהוּא` (במדבר ט:יג)
`וְעַתָּה נַפְשֵׁנוּ יְבֵשָׁה אֵין כֹּל בִּלְתִּי אֶל הַמָּן עֵינֵינוּ` (במדבר יא:ו)
`וְאִם נֶפֶשׁ אַחַת תֶּחֱטָא בִשְׁגָגָה וְהִקְרִיבָה עֵז בַּת שְׁנָתָהּ לְחַטָּאת` (במדבר טו:כז)
`וְכִפֶּר הַכֹּהֵן עַל הַנֶּפֶשׁ הַשֹּׁגֶגֶת בְּחֶטְאָה בִשְׁגָגָה לִפְנֵי יְהוָה לְכַפֵּר עָלָיו וְנִסְלַח לוֹ` (במדבר טו:כח)
`וְהַנֶּפֶשׁ אֲשֶׁר תַּעֲשֶׂה בְּיָד רָמָה מִן הָאֶזְרָח וּמִן הַגֵּר אֶת יְהוָה הוּא מְגַדֵּף וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִקֶּרֶב עַמָּהּ` (במדבר טו:ל)
`כִּי דְבַר יְהוָה בָּזָה וְאֶת מִצְוָתוֹ הֵפַר הִכָּרֵת תִּכָּרֵת הַנֶּפֶשׁ הַהִוא עֲוֺנָה בָהּ` (במדבר טו:לא)
`אֵת מַחְתּוֹת הַחַטָּאִים הָאֵלֶּה בְּנַפְשֹׁתָם וְעָשׂוּ אֹתָם רִקֻּעֵי פַחִים צִפּוּי לַמִּזְבֵּחַ כִּי הִקְרִיבֻם לִפְנֵי יְהוָה וַיִּקְדָּשׁוּ וְיִהְיוּ לְאוֹת לִבְנֵי יִשְׂרָאֵל` (במדבר טז:לח)
`הַנֹּגֵעַ בְּמֵת לְכָל נֶפֶשׁ אָדָם וְטָמֵא שִׁבְעַת יָמִים` (במדבר יט:יא)
`כָּל הַנֹּגֵעַ בְּמֵת בְּנֶפֶשׁ הָאָדָם אֲשֶׁר יָמוּת וְלֹא יִתְחַטָּא אֶת מִשְׁכַּן יְהוָה טִמֵּא וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִיִּשְׂרָאֵל כִּי מֵי נִדָּה לֹא זֹרַק עָלָיו טָמֵא יִהְיֶה עוֹד טֻמְאָתוֹ בוֹ` (במדבר יט:יג)
`וְלָקַח אֵזוֹב וְטָבַל בַּמַּיִם אִישׁ טָהוֹר וְהִזָּה עַל הָאֹהֶל וְעַל כָּל הַכֵּלִים וְעַל הַנְּפָשׁוֹת אֲשֶׁר הָיוּ שָׁם וְעַל הַנֹּגֵעַ בַּעֶצֶם אוֹ בֶחָלָל אוֹ בַמֵּת אוֹ בַקָּבֶר` (במדבר יט:יח)
`וְאִישׁ אֲשֶׁר יִטְמָא וְלֹא יִתְחַטָּא וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִתּוֹךְ הַקָּהָל כִּי אֶת מִקְדַּשׁ יְהוָה טִמֵּא מֵי נִדָּה לֹא זֹרַק עָלָיו טָמֵא הוּא` (במדבר יט:כ)
`וְכֹל אֲשֶׁר יִגַּע בּוֹ הַטָּמֵא יִטְמָא וְהַנֶּפֶשׁ הַנֹּגַעַת תִּטְמָא עַד הָעָרֶב` (במדבר יט:כב)
`וַיִּסְעוּ מֵהֹר הָהָר דֶּרֶךְ יַם סוּף לִסְבֹב אֶת אֶרֶץ אֱדוֹם וַתִּקְצַר נֶפֶשׁ הָעָם בַּדָּרֶךְ` (במדבר כא:ד)
`וַיְדַבֵּר הָעָם בֵּאלֹהִים וּבְמֹשֶׁה לָמָה הֶעֱלִיתֻנוּ מִמִּצְרַיִם לָמוּת בַּמִּדְבָּר כִּי אֵין לֶחֶם וְאֵין מַיִם וְנַפְשֵׁנוּ קָצָה בַּלֶּחֶם הַקְּלֹקֵל` (במדבר כא:ה)
`מִי מָנָה עֲפַר יַעֲקֹב וּמִסְפָּר אֶת רֹבַע יִשְׂרָאֵל תָּמֹת נַפְשִׁי מוֹת יְשָׁרִים וּתְהִי אַחֲרִיתִי כָּמֹהוּ` (במדבר כג:י)
`וּבֶעָשׂוֹר לַחֹדֶשׁ הַשְּׁבִיעִי הַזֶּה מִקְרָא קֹדֶשׁ יִהְיֶה לָכֶם וְעִנִּיתֶם אֶת נַפְשֹׁתֵיכֶם כָּל מְלָאכָה לֹא תַעֲשׂוּ` (במדבר כט:ז)
`אִישׁ כִּי יִדֹּר נֶדֶר לַיהוָה אוֹ הִשָּׁבַע שְׁבֻעָה לֶאְסֹר אִסָּר עַל נַפְשׁוֹ לֹא יַחֵל דְּבָרוֹ כְּכָל הַיֹּצֵא מִפִּיו יַעֲשֶׂה` (במדבר ל:ב)
`וְשָׁמַע אָבִיהָ אֶת נִדְרָהּ וֶאֱסָרָהּ אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ וְהֶחֱרִישׁ לָהּ אָבִיהָ וְקָמוּ כָּל נְדָרֶיהָ וְכָל אִסָּר אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ יָקוּם` (במדבר ל:ד)
`וְאִם הֵנִיא אָבִיהָ אֹתָהּ בְּיוֹם שָׁמְעוֹ כָּל נְדָרֶיהָ וֶאֱסָרֶיהָ אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ לֹא יָקוּם וַיהוָה יִסְלַח לָהּ כִּי הֵנִיא אָבִיהָ אֹתָהּ` (במדבר ל:ה)
`וְאִם הָיוֹ תִהְיֶה לְאִישׁ וּנְדָרֶיהָ עָלֶיהָ אוֹ מִבְטָא שְׂפָתֶיהָ אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ` (במדבר ל:ו)
`וְשָׁמַע אִישָׁהּ בְּיוֹם שָׁמְעוֹ וְהֶחֱרִישׁ לָהּ וְקָמוּ נְדָרֶיהָ וֶאֱסָרֶהָ אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ יָקֻמוּ` (במדבר ל:ז)
`וְאִם בְּיוֹם שְׁמֹעַ אִישָׁהּ יָנִיא אוֹתָהּ וְהֵפֵר אֶת נִדְרָהּ אֲשֶׁר עָלֶיהָ וְאֵת מִבְטָא שְׂפָתֶיהָ אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ וַיהוָה יִסְלַח לָהּ` (במדבר ל:ח)
`וְנֵדֶר אַלְמָנָה וּגְרוּשָׁה כֹּל אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ יָקוּם עָלֶיהָ` (במדבר ל:ט)
`וְאִם בֵּית אִישָׁהּ נָדָרָה אוֹ אָסְרָה אִסָּר עַל נַפְשָׁהּ בִּשְׁבֻעָה` (במדבר ל:י)
`וְשָׁמַע אִישָׁהּ וְהֶחֱרִשׁ לָהּ לֹא הֵנִיא אֹתָהּ וְקָמוּ כָּל נְדָרֶיהָ וְכָל אִסָּר אֲשֶׁר אָסְרָה עַל נַפְשָׁהּ יָקוּם` (במדבר ל:יא)
`וְאִם הָפֵר יָפֵר אֹתָם אִישָׁהּ בְּיוֹם שָׁמְעוֹ כָּל מוֹצָא שְׂפָתֶיהָ לִנְדָרֶיהָ וּלְאִסַּר נַפְשָׁהּ לֹא יָקוּם אִישָׁהּ הֲפֵרָם וַיהוָה יִסְלַח לָהּ` (במדבר ל:יב)
`כָּל נֵדֶר וְכָל שְׁבֻעַת אִסָּר לְעַנֹּת נָפֶשׁ אִישָׁהּ יְקִימֶנּוּ וְאִישָׁהּ יְפֵרֶנּוּ` (במדבר ל:יג)
`וְאַתֶּם חֲנוּ מִחוּץ לַמַּחֲנֶה שִׁבְעַת יָמִים כֹּל הֹרֵג נֶפֶשׁ וְכֹל נֹגֵעַ בֶּחָלָל תִּתְחַטְּאוּ בַּיּוֹם הַשְּׁלִישִׁי וּבַיּוֹם הַשְּׁבִיעִי אַתֶּם וּשְׁבִיכֶם` (במדבר לא:יט)
`וַהֲרֵמֹתָ מֶכֶס לַיהוָה מֵאֵת אַנְשֵׁי הַמִּלְחָמָה הַיֹּצְאִים לַצָּבָא אֶחָד נֶפֶשׁ מֵחֲמֵשׁ הַמֵּאוֹת מִן הָאָדָם וּמִן הַבָּקָר וּמִן הַחֲמֹרִים וּמִן הַצֹּאן` (במדבר לא:כח)
`וְנֶפֶשׁ אָדָם מִן הַנָּשִׁים אֲשֶׁר לֹא יָדְעוּ מִשְׁכַּב זָכָר כָּל נֶפֶשׁ שְׁנַיִם וּשְׁלֹשִׁים אָלֶף` (במדבר לא:לה)
`וְנֶפֶשׁ אָדָם שִׁשָּׁה עָשָׂר אָלֶף וּמִכְסָם לַיהוָה שְׁנַיִם וּשְׁלֹשִׁים נָפֶשׁ` (במדבר לא:מ)
`וְנֶפֶשׁ אָדָם שִׁשָּׁה עָשָׂר אָלֶף` (במדבר לא:מו)
`וַנַּקְרֵב אֶת קָרְבַּן יְהוָה אִישׁ אֲשֶׁר מָצָא כְלִי זָהָב אֶצְעָדָה וְצָמִיד טַבַּעַת עָגִיל וְכוּמָז לְכַפֵּר עַל נַפְשֹׁתֵינוּ לִפְנֵי יְהוָה` (במדבר לא:נ)
`וְהִקְרִיתֶם לָכֶם עָרִים עָרֵי מִקְלָט תִּהְיֶינָה לָכֶם וְנָס שָׁמָּה רֹצֵחַ מַכֵּה נֶפֶשׁ בִּשְׁגָגָה` (במדבר לה:יא)
`לִבְנֵי יִשְׂרָאֵל וְלַגֵּר וְלַתּוֹשָׁב בְּתוֹכָם תִּהְיֶינָה שֵׁשׁ הֶעָרִים הָאֵלֶּה לְמִקְלָט לָנוּס שָׁמָּה כָּל מַכֵּה נֶפֶשׁ בִּשְׁגָגָה` (במדבר לה:טו)
`כָּל מַכֵּה נֶפֶשׁ לְפִי עֵדִים יִרְצַח אֶת הָרֹצֵחַ וְעֵד אֶחָד לֹא יַעֲנֶה בְנֶפֶשׁ לָמוּת` (במדבר לה:ל)
`וְלֹא תִקְחוּ כֹפֶר לְנֶפֶשׁ רֹצֵחַ אֲשֶׁר הוּא רָשָׁע לָמוּת כִּי מוֹת יוּמָת` (במדבר לה:לא)
`רַק הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד פֶּן תִּשְׁכַּח אֶת הַדְּבָרִים אֲשֶׁר רָאוּ עֵינֶיךָ וּפֶן יָסוּרוּ מִלְּבָבְךָ כֹּל יְמֵי חַיֶּיךָ וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ` (דברים ד:ט)
`וְנִשְׁמַרְתֶּם מְאֹד לְנַפְשֹׁתֵיכֶם כִּי לֹא רְאִיתֶם כָּל תְּמוּנָה בְּיוֹם דִּבֶּר יְהוָה אֲלֵיכֶם בְּחֹרֵב מִתּוֹךְ הָאֵשׁ` (דברים ד:טו)
`וּבִקַּשְׁתֶּם מִשָּׁם אֶת יְהוָה אֱלֹהֶיךָ וּמָצָאתָ כִּי תִדְרְשֶׁנּוּ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים ד:כט)
`וְאָהַבְתָּ אֵת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ וּבְכָל מְאֹדֶךָ` (דברים ו:ה)
`וְעַתָּה יִשְׂרָאֵל מָה יְהוָה אֱלֹהֶיךָ שֹׁאֵל מֵעִמָּךְ כִּי אִם לְיִרְאָה אֶת יְהוָה אֱלֹהֶיךָ לָלֶכֶת בְּכָל דְּרָכָיו וּלְאַהֲבָה אֹתוֹ וְלַעֲבֹד אֶת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים י:יב)
`בְּשִׁבְעִים נֶפֶשׁ יָרְדוּ אֲבֹתֶיךָ מִצְרָיְמָהּ וְעַתָּה שָׂמְךָ יְהוָה אֱלֹהֶיךָ כְּכוֹכְבֵי הַשָּׁמַיִם לָרֹב` (דברים י:כב)
`וְהָיָה אִם שָׁמֹעַ תִּשְׁמְעוּ אֶל מִצְוֺתַי אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם לְאַהֲבָה אֶת יְהוָה אֱלֹהֵיכֶם וּלְעָבְדוֹ בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם` (דברים יא:יג)
`וְשַׂמְתֶּם אֶת דְּבָרַי אֵלֶּה עַל לְבַבְכֶם וְעַל נַפְשְׁכֶם וּקְשַׁרְתֶּם אֹתָם לְאוֹת עַל יֶדְכֶם וְהָיוּ לְטוֹטָפֹת בֵּין עֵינֵיכֶם` (דברים יא:יח)
`רַק בְּכָל אַוַּת נַפְשְׁךָ תִּזְבַּח וְאָכַלְתָּ בָשָׂר כְּבִרְכַּת יְהוָה אֱלֹהֶיךָ אֲשֶׁר נָתַן לְךָ בְּכָל שְׁעָרֶיךָ הַטָּמֵא וְהַטָּהוֹר יֹאכְלֶנּוּ כַּצְּבִי וְכָאַיָּל` (דברים יב:טו)
`כִּי יַרְחִיב יְהוָה אֱלֹהֶיךָ אֶת גְּבוּלְךָ כַּאֲשֶׁר דִּבֶּר לָךְ וְאָמַרְתָּ אֹכְלָה בָשָׂר כִּי תְאַוֶּה נַפְשְׁךָ לֶאֱכֹל בָּשָׂר בְּכָל אַוַּת נַפְשְׁךָ תֹּאכַל בָּשָׂר` (דברים יב:כ)
`כִּי יִרְחַק מִמְּךָ הַמָּקוֹם אֲשֶׁר יִבְחַר יְהוָה אֱלֹהֶיךָ לָשׂוּם שְׁמוֹ שָׁם וְזָבַחְתָּ מִבְּקָרְךָ וּמִצֹּאנְךָ אֲשֶׁר נָתַן יְהוָה לְךָ כַּאֲשֶׁר צִוִּיתִךָ וְאָכַלְתָּ בִּשְׁעָרֶיךָ בְּכֹל אַוַּת נַפְשֶׁךָ` (דברים יב:כא)
`רַק חֲזַק לְבִלְתִּי אֲכֹל הַדָּם כִּי הַדָּם הוּא הַנָּפֶשׁ וְלֹא תֹאכַל הַנֶּפֶשׁ עִם הַבָּשָׂר` (דברים יב:כג)
`לֹא תִשְׁמַע אֶל דִּבְרֵי הַנָּבִיא הַהוּא אוֹ אֶל חוֹלֵם הַחֲלוֹם הַהוּא כִּי מְנַסֶּה יְהוָה אֱלֹהֵיכֶם אֶתְכֶם לָדַעַת הֲיִשְׁכֶם אֹהֲבִים אֶת יְהוָה אֱלֹהֵיכֶם בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם` (דברים יג:ג)
`כִּי יְסִיתְךָ אָחִיךָ בֶן אִמֶּךָ אוֹ בִנְךָ אוֹ בִתְּךָ אוֹ אֵשֶׁת חֵיקֶךָ אוֹ רֵעֲךָ אֲשֶׁר כְּנַפְשְׁךָ בַּסֵּתֶר לֵאמֹר נֵלְכָה וְנַעַבְדָה אֱלֹהִים אֲחֵרִים אֲשֶׁר לֹא יָדַעְתָּ אַתָּה וַאֲבֹתֶיךָ` (דברים יג:ו)
`וְנָתַתָּה הַכֶּסֶף בְּכֹל אֲשֶׁר תְּאַוֶּה נַפְשְׁךָ בַּבָּקָר וּבַצֹּאן וּבַיַּיִן וּבַשֵּׁכָר וּבְכֹל אֲשֶׁר תִּשְׁאָלְךָ נַפְשֶׁךָ וְאָכַלְתָּ שָּׁם לִפְנֵי יְהוָה אֱלֹהֶיךָ וְשָׂמַחְתָּ אַתָּה וּבֵיתֶךָ` (דברים יד:כו)
`וְכִי יָבֹא הַלֵּוִי מֵאַחַד שְׁעָרֶיךָ מִכָּל יִשְׂרָאֵל אֲשֶׁר הוּא גָּר שָׁם וּבָא בְּכָל אַוַּת נַפְשׁוֹ אֶל הַמָּקוֹם אֲשֶׁר יִבְחַר יְהוָה` (דברים יח:ו)
`פֶּן יִרְדֹּף גֹּאֵל הַדָּם אַחֲרֵי הָרֹצֵחַ כִּי יֵחַם לְבָבוֹ וְהִשִּׂיגוֹ כִּי יִרְבֶּה הַדֶּרֶךְ וְהִכָּהוּ נָפֶשׁ וְלוֹ אֵין מִשְׁפַּט מָוֶת כִּי לֹא שֹׂנֵא הוּא לוֹ מִתְּמוֹל שִׁלְשׁוֹם` (דברים יט:ו)
`וְכִי יִהְיֶה אִישׁ שֹׂנֵא לְרֵעֵהוּ וְאָרַב לוֹ וְקָם עָלָיו וְהִכָּהוּ נֶפֶשׁ וָמֵת וְנָס אֶל אַחַת הֶעָרִים הָאֵל` (דברים יט:יא)
`וְלֹא תָחוֹס עֵינֶךָ נֶפֶשׁ בְּנֶפֶשׁ עַיִן בְּעַיִן שֵׁן בְּשֵׁן יָד בְּיָד רֶגֶל בְּרָגֶל` (דברים יט:כא)
`וְהָיָה אִם לֹא חָפַצְתָּ בָּהּ וְשִׁלַּחְתָּהּ לְנַפְשָׁהּ וּמָכֹר לֹא תִמְכְּרֶנָּה בַּכָּסֶף לֹא תִתְעַמֵּר בָּהּ תַּחַת אֲשֶׁר עִנִּיתָהּ` (דברים כא:יד)
`וְלַנַּעַר לֹא תַעֲשֶׂה דָבָר אֵין לַנַּעַר חֵטְא מָוֶת כִּי כַּאֲשֶׁר יָקוּם אִישׁ עַל רֵעֵהוּ וּרְצָחוֹ נֶפֶשׁ כֵּן הַדָּבָר הַזֶּה` (דברים כב:כו)
`כִּי תָבֹא בְּכֶרֶם רֵעֶךָ וְאָכַלְתָּ עֲנָבִים כְּנַפְשְׁךָ שָׂבְעֶךָ וְאֶל כֶּלְיְךָ לֹא תִתֵּן` (דברים כג:כד)
`לֹא יַחֲבֹל רֵחַיִם וָרָכֶב כִּי נֶפֶשׁ הוּא חֹבֵל` (דברים כד:ו)
`כִּי יִמָּצֵא אִישׁ גֹּנֵב נֶפֶשׁ מֵאֶחָיו מִבְּנֵי יִשְׂרָאֵל וְהִתְעַמֶּר בּוֹ וּמְכָרוֹ וּמֵת הַגַּנָּב הַהוּא וּבִעַרְתָּ הָרָע מִקִּרְבֶּךָ` (דברים כד:ז)
`בְּיוֹמוֹ תִתֵּן שְׂכָרוֹ וְלֹא תָבוֹא עָלָיו הַשֶּׁמֶשׁ כִּי עָנִי הוּא וְאֵלָיו הוּא נֹשֵׂא אֶת נַפְשׁוֹ וְלֹא יִקְרָא עָלֶיךָ אֶל יְהוָה וְהָיָה בְךָ חֵטְא` (דברים כד:טו)
`הַיּוֹם הַזֶּה יְהוָה אֱלֹהֶיךָ מְצַוְּךָ לַעֲשׂוֹת אֶת הַחֻקִּים הָאֵלֶּה וְאֶת הַמִּשְׁפָּטִים וְשָׁמַרְתָּ וְעָשִׂיתָ אוֹתָם בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים כו:טז)
`אָרוּר לֹקֵחַ שֹׁחַד לְהַכּוֹת נֶפֶשׁ דָּם נָקִי וְאָמַר כָּל הָעָם אָמֵן` (דברים כז:כה)
`וּבַגּוֹיִם הָהֵם לֹא תַרְגִּיעַ וְלֹא יִהְיֶה מָנוֹחַ לְכַף רַגְלֶךָ וְנָתַן יְהוָה לְךָ שָׁם לֵב רַגָּז וְכִלְיוֹן עֵינַיִם וְדַאֲבוֹן נָפֶשׁ` (דברים כח:סה)
`וְשַׁבְתָּ עַד יְהוָה אֱלֹהֶיךָ וְשָׁמַעְתָּ בְקֹלוֹ כְּכֹל אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם אַתָּה וּבָנֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים ל:ב)
`וּמָל יְהוָה אֱלֹהֶיךָ אֶת לְבָבְךָ וְאֶת לְבַב זַרְעֶךָ לְאַהֲבָה אֶת יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ לְמַעַן חַיֶּיךָ` (דברים ל:ו)
`כִּי תִשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ לִשְׁמֹר מִצְוֺתָיו וְחֻקֹּתָיו הַכְּתוּבָה בְּסֵפֶר הַתּוֹרָה הַזֶּה כִּי תָשׁוּב אֶל יְהוָה אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשֶׁךָ` (דברים ל:י)
`וְהַחֲיִתֶם אֶת אָבִי וְאֶת אִמִּי וְאֶת אַחַי וְאֶת אַחוֹתַי וְאֵת כָּל אֲשֶׁר לָהֶם וְהִצַּלְתֶּם אֶת נַפְשֹׁתֵינוּ מִמָּוֶת` (יהושע ב:יג)
`וַיֹּאמְרוּ לָהּ הָאֲנָשִׁים נַפְשֵׁנוּ תַחְתֵּיכֶם לָמוּת אִם לֹא תַגִּידוּ אֶת דְּבָרֵנוּ זֶה וְהָיָה בְּתֵת יְהוָה לָנוּ אֶת הָאָרֶץ וְעָשִׂינוּ עִמָּךְ חֶסֶד וֶאֱמֶת` (יהושע ב:יד)
`וַיַּעֲנוּ אֶת יְהוֹשֻׁעַ וַיֹּאמְרוּ כִּי הֻגֵּד הֻגַּד לַעֲבָדֶיךָ אֵת אֲשֶׁר צִוָּה יְהוָה אֱלֹהֶיךָ אֶת מֹשֶׁה עַבְדּוֹ לָתֵת לָכֶם אֶת כָּל הָאָרֶץ וּלְהַשְׁמִיד אֶת כָּל יֹשְׁבֵי הָאָרֶץ מִפְּנֵיכֶם וַנִּירָא מְאֹד לְנַפְשֹׁתֵינוּ מִפְּנֵיכֶם וַנַּעֲשֵׂה אֶת הַדָּבָר הַזֶּה` (יהושע ט:כד)
`וְאֶת מַקֵּדָה לָכַד יְהוֹשֻׁעַ בַּיּוֹם הַהוּא וַיַּכֶּהָ לְפִי חֶרֶב וְאֶת מַלְכָּהּ הֶחֱרִם אוֹתָם וְאֶת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ לֹא הִשְׁאִיר שָׂרִיד וַיַּעַשׂ לְמֶלֶךְ מַקֵּדָה כַּאֲשֶׁר עָשָׂה לְמֶלֶךְ יְרִיחוֹ` (יהושע י:כח)
`וַיִּתֵּן יְהוָה גַּם אוֹתָהּ בְּיַד יִשְׂרָאֵל וְאֶת מַלְכָּהּ וַיַּכֶּהָ לְפִי חֶרֶב וְאֶת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ לֹא הִשְׁאִיר בָּהּ שָׂרִיד וַיַּעַשׂ לְמַלְכָּהּ כַּאֲשֶׁר עָשָׂה לְמֶלֶךְ יְרִיחוֹ` (יהושע י:ל)
`וַיִּתֵּן יְהוָה אֶת לָכִישׁ בְּיַד יִשְׂרָאֵל וַיִּלְכְּדָהּ בַּיּוֹם הַשֵּׁנִי וַיַּכֶּהָ לְפִי חֶרֶב וְאֶת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ כְּכֹל אֲשֶׁר עָשָׂה לְלִבְנָה` (יהושע י:לב)
`וַיִּלְכְּדוּהָ בַּיּוֹם הַהוּא וַיַּכּוּהָ לְפִי חֶרֶב וְאֵת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ בַּיּוֹם הַהוּא הֶחֱרִים כְּכֹל אֲשֶׁר עָשָׂה לְלָכִישׁ` (יהושע י:לה)
`וַיִּלְכְּדוּהָ וַיַּכּוּהָ לְפִי חֶרֶב וְאֶת מַלְכָּהּ וְאֶת כָּל עָרֶיהָ וְאֶת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ לֹא הִשְׁאִיר שָׂרִיד כְּכֹל אֲשֶׁר עָשָׂה לְעֶגְלוֹן וַיַּחֲרֵם אוֹתָהּ וְאֶת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ` (יהושע י:לז)
`וַיִּלְכְּדָהּ וְאֶת מַלְכָּהּ וְאֶת כָּל עָרֶיהָ וַיַּכּוּם לְפִי חֶרֶב וַיַּחֲרִימוּ אֶת כָּל נֶפֶשׁ אֲשֶׁר בָּהּ לֹא הִשְׁאִיר שָׂרִיד כַּאֲשֶׁר עָשָׂה לְחֶבְרוֹן כֵּן עָשָׂה לִדְבִרָה וּלְמַלְכָּהּ וְכַאֲשֶׁר עָשָׂה לְלִבְנָה וּלְמַלְכָּהּ` (יהושע י:לט)
`וַיַּכּוּ אֶת כָּל הַנֶּפֶשׁ אֲשֶׁר בָּהּ לְפִי חֶרֶב הַחֲרֵם לֹא נוֹתַר כָּל נְשָׁמָה וְאֶת חָצוֹר שָׂרַף בָּאֵשׁ` (יהושע יא:יא)
`לָנוּס שָׁמָּה רוֹצֵחַ מַכֵּה נֶפֶשׁ בִּשְׁגָגָה בִּבְלִי דָעַת וְהָיוּ לָכֶם לְמִקְלָט מִגֹּאֵל הַדָּם` (יהושע כ:ג)
`אֵלֶּה הָיוּ עָרֵי הַמּוּעָדָה לְכֹל בְּנֵי יִשְׂרָאֵל וְלַגֵּר הַגָּר בְּתוֹכָם לָנוּס שָׁמָּה כָּל מַכֵּה נֶפֶשׁ בִּשְׁגָגָה וְלֹא יָמוּת בְּיַד גֹּאֵל הַדָּם עַד עָמְדוֹ לִפְנֵי הָעֵדָה` (יהושע כ:ט)
`רַק שִׁמְרוּ מְאֹד לַעֲשׂוֹת אֶת הַמִּצְוָה וְאֶת הַתּוֹרָה אֲשֶׁר צִוָּה אֶתְכֶם מֹשֶׁה עֶבֶד יְהוָה לְאַהֲבָה אֶת יְהוָה אֱלֹהֵיכֶם וְלָלֶכֶת בְּכָל דְּרָכָיו וְלִשְׁמֹר מִצְוֺתָיו וּלְדָבְקָה בוֹ וּלְעָבְדוֹ בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם` (יהושע כב:ה)
`וְנִשְׁמַרְתֶּם מְאֹד לְנַפְשֹׁתֵיכֶם לְאַהֲבָה אֶת יְהוָה אֱלֹהֵיכֶם` (יהושע כג:יא)
`וְהִנֵּה אָנֹכִי הוֹלֵךְ הַיּוֹם בְּדֶרֶךְ כָּל הָאָרֶץ וִידַעְתֶּם בְּכָל לְבַבְכֶם וּבְכָל נַפְשְׁכֶם כִּי לֹא נָפַל דָּבָר אֶחָד מִכֹּל הַדְּבָרִים הַטּוֹבִים אֲשֶׁר דִּבֶּר יְהוָה אֱלֹהֵיכֶם עֲלֵיכֶם הַכֹּל בָּאוּ לָכֶם לֹא נָפַל מִמֶּנּוּ דָּבָר אֶחָד` (יהושע כג:יד)
`זְבֻלוּן עַם חֵרֵף נַפְשׁוֹ לָמוּת וְנַפְתָּלִי עַל מְרוֹמֵי שָׂדֶה` (שופטים ה:יח)
`נַחַל קִישׁוֹן גְּרָפָם נַחַל קְדוּמִים נַחַל קִישׁוֹן תִּדְרְכִי נַפְשִׁי עֹז` (שופטים ה:כא)
`אֲשֶׁר נִלְחַם אָבִי עֲלֵיכֶם וַיַּשְׁלֵךְ אֶת נַפְשׁוֹ מִנֶּגֶד וַיַּצֵּל אֶתְכֶם מִיַּד מִדְיָן` (שופטים ט:יז)
`וַיָּסִירוּ אֶת אֱלֹהֵי הַנֵּכָר מִקִּרְבָּם וַיַּעַבְדוּ אֶת יְהוָה וַתִּקְצַר נַפְשׁוֹ בַּעֲמַל יִשְׂרָאֵל` (שופטים י:טז)
`וָאֶרְאֶה כִּי אֵינְךָ מוֹשִׁיע וָאָשִׂימָה נַפְשִׁי בְכַפִּי וָאֶעְבְּרָה אֶל בְּנֵי עַמּוֹן וַיִּתְּנֵם יְהוָה בְּיָדִי וְלָמָה עֲלִיתֶם אֵלַי הַיּוֹם הַזֶּה לְהִלָּחֶם בִּי` (שופטים יב:ג)
`וַיְהִי כִּי הֵצִיקָה לּוֹ בִדְבָרֶיהָ כָּל הַיָּמִים וַתְּאַלֲצֵהוּ וַתִּקְצַר נַפְשׁוֹ לָמוּת` (שופטים טז:טז)
`וַיֹּאמֶר שִׁמְשׁוֹן תָּמוֹת נַפְשִׁי עִם פְּלִשְׁתִּים וַיֵּט בְּכֹחַ וַיִּפֹּל הַבַּיִת עַל הַסְּרָנִים וְעַל כָּל הָעָם אֲשֶׁר בּוֹ וַיִּהְיוּ הַמֵּתִים אֲשֶׁר הֵמִית בְּמוֹתוֹ רַבִּים מֵאֲשֶׁר הֵמִית בְּחַיָּיו` (שופטים טז:ל)
`וַיֹּאמְרוּ אֵלָיו בְּנֵי דָן אַל תַּשְׁמַע קוֹלְךָ עִמָּנוּ פֶּן יִפְגְּעוּ בָכֶם אֲנָשִׁים מָרֵי נֶפֶשׁ וְאָסַפְתָּה נַפְשְׁךָ וְנֶפֶשׁ בֵּיתֶךָ` (שופטים יח:כה)
`וְהִיא מָרַת נָפֶשׁ וַתִּתְפַּלֵּל עַל יְהוָה וּבָכֹה תִבְכֶּה` (שמואל-א א:י)
`וַתַּעַן חַנָּה וַתֹּאמֶר לֹא אֲדֹנִי אִשָּׁה קְשַׁת רוּחַ אָנֹכִי וְיַיִן וְשֵׁכָר לֹא שָׁתִיתִי וָאֶשְׁפֹּךְ אֶת נַפְשִׁי לִפְנֵי יְהוָה` (שמואל-א א:טו)
`וַתֹּאמֶר בִּי אֲדֹנִי חֵי נַפְשְׁךָ אֲדֹנִי אֲנִי הָאִשָּׁה הַנִּצֶּבֶת עִמְּכָה בָּזֶה לְהִתְפַּלֵּל אֶל יְהוָה` (שמואל-א א:כו)
`וַיֹּאמֶר אֵלָיו הָאִישׁ קַטֵּר יַקְטִירוּן כַּיּוֹם הַחֵלֶב וְקַח לְךָ כַּאֲשֶׁר תְּאַוֶּה נַפְשֶׁךָ וְאָמַר לוֹ כִּי עַתָּה תִתֵּן וְאִם לֹא לָקַחְתִּי בְחָזְקָה` (שמואל-א ב:טז)
`וְאִישׁ לֹא אַכְרִית לְךָ מֵעִם מִזְבְּחִי לְכַלּוֹת אֶת עֵינֶיךָ וְלַאֲדִיב אֶת נַפְשֶׁךָ וְכָל מַרְבִּית בֵּיתְךָ יָמוּתוּ אֲנָשִׁים` (שמואל-א ב:לג)
`וַהֲקִימֹתִי לִי כֹּהֵן נֶאֱמָן כַּאֲשֶׁר בִּלְבָבִי וּבְנַפְשִׁי יַעֲשֶׂה וּבָנִיתִי לוֹ בַּיִת נֶאֱמָן וְהִתְהַלֵּךְ לִפְנֵי מְשִׁיחִי כָּל הַיָּמִים` (שמואל-א ב:לה)
`וְכִרְאוֹת שָׁאוּל אֶת דָּוִד יֹצֵא לִקְרַאת הַפְּלִשְׁתִּי אָמַר אֶל אַבְנֵר שַׂר הַצָּבָא בֶּן מִי זֶה הַנַּעַר אַבְנֵר וַיֹּאמֶר אַבְנֵר חֵי נַפְשְׁךָ הַמֶּלֶךְ אִם יָדָעְתִּי` (שמואל-א יז:נה)
`וַיְהִי כְּכַלֹּתוֹ לְדַבֵּר אֶל שָׁאוּל וְנֶפֶשׁ יְהוֹנָתָן נִקְשְׁרָה בְּנֶפֶשׁ דָּוִד וַיֶּאֱהָבוֹ יְהוֹנָתָן כְּנַפְשׁוֹ` (שמואל-א יח:א)
`וַיִּכְרֹת יְהוֹנָתָן וְדָוִד בְּרִית בְּאַהֲבָתוֹ אֹתוֹ כְּנַפְשׁוֹ` (שמואל-א יח:ג)
`וַיָּשֶׂם אֶת נַפְשׁוֹ בְכַפּוֹ וַיַּךְ אֶת הַפְּלִשְׁתִּי וַיַּעַשׂ יְהוָה תְּשׁוּעָה גְדוֹלָה לְכָל יִשְׂרָאֵל רָאִיתָ וַתִּשְׂמָח וְלָמָּה תֶחֱטָא בְּדָם נָקִי לְהָמִית אֶת דָּוִד חִנָּם` (שמואל-א יט:ה)
`וַיִּשְׁלַח שָׁאוּל מַלְאָכִים אֶל בֵּית דָּוִד לְשָׁמְרוֹ וְלַהֲמִיתוֹ בַּבֹּקֶר וַתַּגֵּד לְדָוִד מִיכַל אִשְׁתּוֹ לֵאמֹר אִם אֵינְךָ מְמַלֵּט אֶת נַפְשְׁךָ הַלַּיְלָה מָחָר אַתָּה מוּמָת` (שמואל-א יט:יא)
`וַיִּבְרַח דָּוִד מִנֹּוִות בָּרָמָה וַיָּבֹא וַיֹּאמֶר לִפְנֵי יְהוֹנָתָן מֶה עָשִׂיתִי מֶה עֲוֺנִי וּמֶה חַטָּאתִי לִפְנֵי אָבִיךָ כִּי מְבַקֵּשׁ אֶת נַפְשִׁי` (שמואל-א כ:א)
`וַיִּשָּׁבַע עוֹד דָּוִד וַיֹּאמֶר יָדֹעַ יָדַע אָבִיךָ כִּי מָצָאתִי חֵן בְּעֵינֶיךָ וַיֹּאמֶר אַל יֵדַע זֹאת יְהוֹנָתָן פֶּן יֵעָצֵב וְאוּלָם חַי יְהוָה וְחֵי נַפְשֶׁךָ כִּי כְפֶשַׂע בֵּינִי וּבֵין הַמָּוֶת` (שמואל-א כ:ג)
`וַיֹּאמֶר יְהוֹנָתָן אֶל דָּוִד מַה תֹּאמַר נַפְשְׁךָ וְאֶעֱשֶׂה לָּךְ` (שמואל-א כ:ד)
`וַיּוֹסֶף יְהוֹנָתָן לְהַשְׁבִּיעַ אֶת דָּוִד בְּאַהֲבָתוֹ אֹתוֹ כִּי אַהֲבַת נַפְשׁוֹ אֲהֵבוֹ` (שמואל-א כ:יז)
`וַיִּתְקַבְּצוּ אֵלָיו כָּל אִישׁ מָצוֹק וְכָל אִישׁ אֲשֶׁר לוֹ נֹשֶׁא וְכָל אִישׁ מַר נֶפֶשׁ וַיְהִי עֲלֵיהֶם לְשָׂר וַיִּהְיוּ עִמּוֹ כְּאַרְבַּע מֵאוֹת אִישׁ` (שמואל-א כב:ב)
`וַיֹּאמֶר דָּוִד לְאֶבְיָתָר יָדַעְתִּי בַּיּוֹם הַהוּא כִּי שָׁם דּוֹיֵג הָאֲדֹמִי כִּי הַגֵּד יַגִּיד לְשָׁאוּל אָנֹכִי סַבֹּתִי בְּכָל נֶפֶשׁ בֵּית אָבִיךָ` (שמואל-א כב:כב)
`שְׁבָה אִתִּי אַל תִּירָא כִּי אֲשֶׁר יְבַקֵּשׁ אֶת נַפְשִׁי יְבַקֵּשׁ אֶת נַפְשֶׁךָ כִּי מִשְׁמֶרֶת אַתָּה עִמָּדִי` (שמואל-א כב:כג)
`וַיַּרְא דָוִד כִּי יָצָא שָׁאוּל לְבַקֵּשׁ אֶת נַפְשׁוֹ וְדָוִד בְּמִדְבַּר זִיף בַּחֹרְשָׁה` (שמואל-א כג:טו)
`וְעַתָּה לְכָל אַוַּת נַפְשְׁךָ הַמֶּלֶךְ לָרֶדֶת רֵד וְלָנוּ הַסְגִּירוֹ בְּיַד הַמֶּלֶךְ` (שמואל-א כג:כ)
`וְאָבִי רְאֵה גַּם רְאֵה אֶת כְּנַף מְעִילְךָ בְּיָדִי כִּי בְּכָרְתִי אֶת כְּנַף מְעִילְךָ וְלֹא הֲרַגְתִּיךָ דַּע וּרְאֵה כִּי אֵין בְּיָדִי רָעָה וָפֶשַׁע וְלֹא חָטָאתִי לָךְ וְאַתָּה צֹדֶה אֶת נַפְשִׁי לְקַחְתָּהּ` (שמואל-א כד:יא)
`וְעַתָּה אֲדֹנִי חַי יְהוָה וְחֵי נַפְשְׁךָ אֲשֶׁר מְנָעֲךָ יְהוָה מִבּוֹא בְדָמִים וְהוֹשֵׁעַ יָדְךָ לָךְ וְעַתָּה יִהְיוּ כְנָבָל אֹיְבֶיךָ וְהַמְבַקְשִׁים אֶל אֲדֹנִי רָעָה` (שמואל-א כה:כו)
`וַיָּקָם אָדָם לִרְדָפְךָ וּלְבַקֵּשׁ אֶת נַפְשֶׁךָ וְהָיְתָה נֶפֶשׁ אֲדֹנִי צְרוּרָה בִּצְרוֹר הַחַיִּים אֵת יְהוָה אֱלֹהֶיךָ וְאֵת נֶפֶשׁ אֹיְבֶיךָ יְקַלְּעֶנָּה בְּתוֹךְ כַּף הַקָּלַע` (שמואל-א כה:כט)
`וַיֹּאמֶר שָׁאוּל חָטָאתִי שׁוּב בְּנִי דָוִד כִּי לֹא אָרַע לְךָ עוֹד תַּחַת אֲשֶׁר יָקְרָה נַפְשִׁי בְּעֵינֶיךָ הַיּוֹם הַזֶּה הִנֵּה הִסְכַּלְתִּי וָאֶשְׁגֶּה הַרְבֵּה מְאֹד` (שמואל-א כו:כא)
`וְהִנֵּה כַּאֲשֶׁר גָּדְלָה נַפְשְׁךָ הַיּוֹם הַזֶּה בְּעֵינָי כֵּן תִּגְדַּל נַפְשִׁי בְּעֵינֵי יְהוָה וְיַצִּלֵנִי מִכָּל צָרָה` (שמואל-א כו:כד)
`וַתֹּאמֶר הָאִשָּׁה אֵלָיו הִנֵּה אַתָּה יָדַעְתָּ אֵת אֲשֶׁר עָשָׂה שָׁאוּל אֲשֶׁר הִכְרִית אֶת הָאֹבוֹת וְאֶת הַיִּדְּעֹנִי מִן הָאָרֶץ וְלָמָה אַתָּה מִתְנַקֵּשׁ בְּנַפְשִׁי לַהֲמִיתֵנִי` (שמואל-א כח:ט)
`וַתָּבוֹא הָאִשָּׁה אֶל שָׁאוּל וַתֵּרֶא כִּי נִבְהַל מְאֹד וַתֹּאמֶר אֵלָיו הִנֵּה שָׁמְעָה שִׁפְחָתְךָ בְּקוֹלֶךָ וָאָשִׂים נַפְשִׁי בְּכַפִּי וָאֶשְׁמַע אֶת דְּבָרֶיךָ אֲשֶׁר דִּבַּרְתָּ אֵלָי` (שמואל-א כח:כא)
`וַתֵּצֶר לְדָוִד מְאֹד כִּי אָמְרוּ הָעָם לְסָקְלוֹ כִּי מָרָה נֶפֶשׁ כָּל הָעָם אִישׁ עַל בְּנוֹ וְעַל בְּנֹתָיו וַיִּתְחַזֵּק דָּוִד בַּיהוָה אֱלֹהָיו` (שמואל-א ל:ו)
`וַיֹּאמֶר אֵלַי עֲמָד נָא עָלַי וּמֹתְתֵנִי כִּי אֲחָזַנִי הַשָּׁבָץ כִּי כָל עוֹד נַפְשִׁי בִּי` (שמואל-ב א:ט)
`וַיֹּאמֶר אַבְנֵר אֶל דָּוִד אָקוּמָה וְאֵלֵכָה וְאֶקְבְּצָה אֶל אֲדֹנִי הַמֶּלֶךְ אֶת כָּל יִשְׂרָאֵל וְיִכְרְתוּ אִתְּךָ בְּרִית וּמָלַכְתָּ בְּכֹל אֲשֶׁר תְּאַוֶּה נַפְשֶׁךָ וַיְּשַׁלַּח דָּוִד אֶת אַבְנֵר וַיֵּלֶךְ בְּשָׁלוֹם` (שמואל-ב ג:כא)
`וַיָּבִאוּ אֶת רֹאשׁ אִישׁ בֹּשֶׁת אֶל דָּוִד חֶבְרוֹן וַיֹּאמְרוּ אֶל הַמֶּלֶךְ הִנֵּה רֹאשׁ אִישׁ בֹּשֶׁת בֶּן שָׁאוּל אֹיִבְךָ אֲשֶׁר בִּקֵּשׁ אֶת נַפְשֶׁךָ וַיִּתֵּן יְהוָה לַאדֹנִי הַמֶּלֶךְ נְקָמוֹת הַיּוֹם הַזֶּה מִשָּׁאוּל וּמִזַּרְעוֹ` (שמואל-ב ד:ח)
`וַיַּעַן דָּוִד אֶת רֵכָב וְאֶת בַּעֲנָה אָחִיו בְּנֵי רִמּוֹן הַבְּאֵרֹתִי וַיֹּאמֶר לָהֶם חַי יְהוָה אֲשֶׁר פָּדָה אֶת נַפְשִׁי מִכָּל צָרָה` (שמואל-ב ד:ט)
`וַיֹּאמֶר דָּוִד בַּיּוֹם הַהוּא כָּל מַכֵּה יְבֻסִי וְיִגַּע בַּצִּנּוֹר וְאֶת הַפִּסְחִים וְאֶת הַעִוְרִים שָׂנְאוּ נֶפֶשׁ דָּוִד עַל כֵּן יֹאמְרוּ עִוֵּר וּפִסֵּחַ לֹא יָבוֹא אֶל הַבָּיִת` (שמואל-ב ה:ח)
`וַיֹּאמֶר אוּרִיָּה אֶל דָּוִד הָאָרוֹן וְיִשְׂרָאֵל וִיהוּדָה יֹשְׁבִים בַּסֻּכּוֹת וַאדֹנִי יוֹאָב וְעַבְדֵי אֲדֹנִי עַל פְּנֵי הַשָּׂדֶה חֹנִים וַאֲנִי אָבוֹא אֶל בֵּיתִי לֶאֱכֹל וְלִשְׁתּוֹת וְלִשְׁכַּב עִם אִשְׁתִּי חַיֶּךָ וְחֵי נַפְשֶׁךָ אִם אֶעֱשֶׂה אֶת הַדָּבָר הַזֶּה` (שמואל-ב יא:יא)
`וְהִנֵּה קָמָה כָל הַמִּשְׁפָּחָה עַל שִׁפְחָתֶךָ וַיֹּאמְרוּ תְּנִי אֶת מַכֵּה אָחִיו וּנְמִתֵהוּ בְּנֶפֶשׁ אָחִיו אֲשֶׁר הָרָג וְנַשְׁמִידָה גַּם אֶת הַיּוֹרֵשׁ וְכִבּוּ אֶת גַּחַלְתִּי אֲשֶׁר נִשְׁאָרָה לְבִלְתִּי שׂוּם לְאִישִׁי שֵׁם וּשְׁאֵרִית עַל פְּנֵי הָאֲדָמָה` (שמואל-ב יד:ז)
`כִּי מוֹת נָמוּת וְכַמַּיִם הַנִּגָּרִים אַרְצָה אֲשֶׁר לֹא יֵאָסֵפוּ וְלֹא יִשָּׂא אֱלֹהִים נֶפֶשׁ וְחָשַׁב מַחֲשָׁבוֹת לְבִלְתִּי יִדַּח מִמֶּנּוּ נִדָּח` (שמואל-ב יד:יד)
`וַיֹּאמֶר הַמֶּלֶךְ הֲיַד יוֹאָב אִתָּךְ בְּכָל זֹאת וַתַּעַן הָאִשָּׁה וַתֹּאמֶר חֵי נַפְשְׁךָ אֲדֹנִי הַמֶּלֶךְ אִם אִשׁ לְהֵמִין וּלְהַשְׂמִיל מִכֹּל אֲשֶׁר דִּבֶּר אֲדֹנִי הַמֶּלֶךְ כִּי עַבְדְּךָ יוֹאָב הוּא צִוָּנִי וְהוּא שָׂם בְּפִי שִׁפְחָתְךָ אֵת כָּל הַדְּבָרִים הָאֵלֶּה` (שמואל-ב יד:יט)
`וַיֹּאמֶר דָּוִד אֶל אֲבִישַׁי וְאֶל כָּל עֲבָדָיו הִנֵּה בְנִי אֲשֶׁר יָצָא מִמֵּעַי מְבַקֵּשׁ אֶת נַפְשִׁי וְאַף כִּי עַתָּה בֶּן הַיְמִינִי הַנִּחוּ לוֹ וִיקַלֵּל כִּי אָמַר לוֹ יְהוָה` (שמואל-ב טז:יא)
`וַיֹּאמֶר חוּשַׁי אַתָּה יָדַעְתָּ אֶת אָבִיךָ וְאֶת אֲנָשָׁיו כִּי גִבֹּרִים הֵמָּה וּמָרֵי נֶפֶשׁ הֵמָּה כְּדֹב שַׁכּוּל בַּשָּׂדֶה וְאָבִיךָ אִישׁ מִלְחָמָה וְלֹא יָלִין אֶת הָעָם` (שמואל-ב יז:ח)
`אוֹ עָשִׂיתִי בְנַפְשׁוֹ שֶׁקֶר וְכָל דָּבָר לֹא יִכָּחֵד מִן הַמֶּלֶךְ וְאַתָּה תִּתְיַצֵּב מִנֶּגֶד` (שמואל-ב יח:יג)
`וַיָּבֹא יוֹאָב אֶל הַמֶּלֶךְ הַבָּיִת וַיֹּאמֶר הֹבַשְׁתָּ הַיּוֹם אֶת פְּנֵי כָל עֲבָדֶיךָ הַמְמַלְּטִים אֶת נַפְשְׁךָ הַיּוֹם וְאֵת נֶפֶשׁ בָּנֶיךָ וּבְנֹתֶיךָ וְנֶפֶשׁ נָשֶׁיךָ וְנֶפֶשׁ פִּלַגְשֶׁיךָ` (שמואל-ב יט:ה)
`וַיֹּאמֶר חָלִילָה לִּי יְהוָה מֵעֲשֹׂתִי זֹאת הֲדַם הָאֲנָשִׁים הַהֹלְכִים בְּנַפְשׁוֹתָם וְלֹא אָבָה לִשְׁתּוֹתָם אֵלֶּה עָשׂוּ שְׁלֹשֶׁת הַגִּבֹּרִים` (שמואל-ב כג:יז)
`וְעַתָּה לְכִי אִיעָצֵךְ נָא עֵצָה וּמַלְּטִי אֶת נַפְשֵׁךְ וְאֶת נֶפֶשׁ בְּנֵךְ שְׁלֹמֹה` (מלכים-א א:יב)
`וַיִּשָּׁבַע הַמֶּלֶךְ וַיֹּאמַר חַי יְהוָה אֲשֶׁר פָּדָה אֶת נַפְשִׁי מִכָּל צָרָה` (מלכים-א א:כט)
`לְמַעַן יָקִים יְהוָה אֶת דְּבָרוֹ אֲשֶׁר דִּבֶּר עָלַי לֵאמֹר אִם יִשְׁמְרוּ בָנֶיךָ אֶת דַּרְכָּם לָלֶכֶת לְפָנַי בֶּאֱמֶת בְּכָל לְבָבָם וּבְכָל נַפְשָׁם לֵאמֹר לֹא יִכָּרֵת לְךָ אִישׁ מֵעַל כִּסֵּא יִשְׂרָאֵל` (מלכים-א ב:ד)
`וַיִּשָּׁבַע הַמֶּלֶךְ שְׁלֹמֹה בַּיהוָה לֵאמֹר כֹּה יַעֲשֶׂה לִּי אֱלֹהִים וְכֹה יוֹסִיף כִּי בְנַפְשׁוֹ דִּבֶּר אֲדֹנִיָּהוּ אֶת הַדָּבָר הַזֶּה` (מלכים-א ב:כג)
`וַיֹּאמֶר אֱלֹהִים אֵלָיו יַעַן אֲשֶׁר שָׁאַלְתָּ אֶת הַדָּבָר הַזֶּה וְלֹא שָׁאַלְתָּ לְּךָ יָמִים רַבִּים וְלֹא שָׁאַלְתָּ לְּךָ עֹשֶׁר וְלֹא שָׁאַלְתָּ נֶפֶשׁ אֹיְבֶיךָ וְשָׁאַלְתָּ לְּךָ הָבִין לִשְׁמֹעַ מִשְׁפָּט` (מלכים-א ג:יא)
`וְשָׁבוּ אֵלֶיךָ בְּכָל לְבָבָם וּבְכָל נַפְשָׁם בְּאֶרֶץ אֹיְבֵיהֶם אֲשֶׁר שָׁבוּ אֹתָם וְהִתְפַּלְלוּ אֵלֶיךָ דֶּרֶךְ אַרְצָם אֲשֶׁר נָתַתָּה לַאֲבוֹתָם הָעִיר אֲשֶׁר בָּחַרְתָּ וְהַבַּיִת אֲשֶׁר בָּנִיתָ לִשְׁמֶךָ` (מלכים-א ח:מח)
`וְאֹתְךָ אֶקַּח וּמָלַכְתָּ בְּכֹל אֲשֶׁר תְּאַוֶּה נַפְשֶׁךָ וְהָיִיתָ מֶּלֶךְ עַל יִשְׂרָאֵל` (מלכים-א יא:לז)
`וַיִּתְמֹדֵד עַל הַיֶּלֶד שָׁלֹשׁ פְּעָמִים וַיִּקְרָא אֶל יְהוָה וַיֹּאמַר יְהוָה אֱלֹהָי תָּשָׁב נָא נֶפֶשׁ הַיֶּלֶד הַזֶּה עַל קִרְבּוֹ` (מלכים-א יז:כא)
`וַיִּשְׁמַע יְהוָה בְּקוֹל אֵלִיָּהוּ וַתָּשָׁב נֶפֶשׁ הַיֶּלֶד עַל קִרְבּוֹ וַיֶּחִי` (מלכים-א יז:כב)
`וַתִּשְׁלַח אִיזֶבֶל מַלְאָךְ אֶל אֵלִיָּהוּ לֵאמֹר כֹּה יַעֲשׂוּן אֱלֹהִים וְכֹה יוֹסִפוּן כִּי כָעֵת מָחָר אָשִׂים אֶת נַפְשְׁךָ כְּנֶפֶשׁ אַחַד מֵהֶם` (מלכים-א יט:ב)
`וַיַּרְא וַיָּקָם וַיֵּלֶךְ אֶל נַפְשׁוֹ וַיָּבֹא בְּאֵר שֶׁבַע אֲשֶׁר לִיהוּדָה וַיַּנַּח אֶת נַעֲרוֹ שָׁם` (מלכים-א יט:ג)
`וְהוּא הָלַךְ בַּמִּדְבָּר דֶּרֶךְ יוֹם וַיָּבֹא וַיֵּשֶׁב תַּחַת רֹתֶם אֶחָת וַיִּשְׁאַל אֶת נַפְשׁוֹ לָמוּת וַיֹּאמֶר רַב עַתָּה יְהוָה קַח נַפְשִׁי כִּי לֹא טוֹב אָנֹכִי מֵאֲבֹתָי` (מלכים-א יט:ד)
`וַיֹּאמֶר קַנֹּא קִנֵּאתִי לַיהוָה אֱלֹהֵי צְבָאוֹת כִּי עָזְבוּ בְרִיתְךָ בְּנֵי יִשְׂרָאֵל אֶת מִזְבְּחֹתֶיךָ הָרָסוּ וְאֶת נְבִיאֶיךָ הָרְגוּ בֶחָרֶב וָאִוָּתֵר אֲנִי לְבַדִּי וַיְבַקְשׁוּ אֶת נַפְשִׁי לְקַחְתָּהּ` (מלכים-א יט:י)
`וַיֹּאמֶר קַנֹּא קִנֵּאתִי לַיהוָה אֱלֹהֵי צְבָאוֹת כִּי עָזְבוּ בְרִיתְךָ בְּנֵי יִשְׂרָאֵל אֶת מִזְבְּחֹתֶיךָ הָרָסוּ וְאֶת נְבִיאֶיךָ הָרְגוּ בֶחָרֶב וָאִוָּתֵר אֲנִי לְבַדִּי וַיְבַקְשׁוּ אֶת נַפְשִׁי לְקַחְתָּהּ` (מלכים-א יט:יד)
`וַיֹּאמְרוּ אֵלָיו עֲבָדָיו הִנֵּה נָא שָׁמַעְנוּ כִּי מַלְכֵי בֵּית יִשְׂרָאֵל כִּי מַלְכֵי חֶסֶד הֵם נָשִׂימָה נָּא שַׂקִּים בְּמָתְנֵינוּ וַחֲבָלִים בְּרֹאשֵׁנוּ וְנֵצֵא אֶל מֶלֶךְ יִשְׂרָאֵל אוּלַי יְחַיֶּה אֶת נַפְשֶׁךָ` (מלכים-א כ:לא)
`וַיַּחְגְּרוּ שַׂקִּים בְּמָתְנֵיהֶם וַחֲבָלִים בְּרָאשֵׁיהֶם וַיָּבֹאוּ אֶל מֶלֶךְ יִשְׂרָאֵל וַיֹּאמְרוּ עַבְדְּךָ בֶן הֲדַד אָמַר תְּחִי נָא נַפְשִׁי וַיֹּאמֶר הַעוֹדֶנּוּ חַי אָחִי הוּא` (מלכים-א כ:לב)
`וַיְהִי הַמֶּלֶךְ עֹבֵר וְהוּא צָעַק אֶל הַמֶּלֶךְ וַיֹּאמֶר עַבְדְּךָ יָצָא בְקֶרֶב הַמִּלְחָמָה וְהִנֵּה אִישׁ סָר וַיָּבֵא אֵלַי אִישׁ וַיֹּאמֶר שְׁמֹר אֶת הָאִישׁ הַזֶּה אִם הִפָּקֵד יִפָּקֵד וְהָיְתָה נַפְשְׁךָ תַּחַת נַפְשׁוֹ אוֹ כִכַּר כֶּסֶף תִּשְׁקוֹל` (מלכים-א כ:לט)
`וַיֹּאמֶר אֵלָיו כֹּה אָמַר יְהוָה יַעַן שִׁלַּחְתָּ אֶת אִישׁ חֶרְמִי מִיָּד וְהָיְתָה נַפְשְׁךָ תַּחַת נַפְשׁוֹ וְעַמְּךָ תַּחַת עַמּוֹ` (מלכים-א כ:מב)
`וַיָּשָׁב וַיִּשְׁלַח שַׂר חֲמִשִּׁים שְׁלִשִׁים וַחֲמִשָּׁיו וַיַּעַל וַיָּבֹא שַׂר הַחֲמִשִּׁים הַשְּׁלִישִׁי וַיִּכְרַע עַל בִּרְכָּיו לְנֶגֶד אֵלִיָּהוּ וַיִּתְחַנֵּן אֵלָיו וַיְדַבֵּר אֵלָיו אִישׁ הָאֱלֹהִים תִּיקַר נָא נַפְשִׁי וְנֶפֶשׁ עֲבָדֶיךָ אֵלֶּה חֲמִשִּׁים בְּעֵינֶיךָ` (מלכים-ב א:יג)
`הִנֵּה יָרְדָה אֵשׁ מִן הַשָּׁמַיִם וַתֹּאכַל אֶת שְׁנֵי שָׂרֵי הַחֲמִשִּׁים הָרִאשֹׁנִים וְאֶת חֲמִשֵּׁיהֶם וְעַתָּה תִּיקַר נַפְשִׁי בְּעֵינֶיךָ` (מלכים-ב א:יד)
`וַיֹּאמֶר אֵלִיָּהוּ אֶל אֱלִישָׁע שֵׁב נָא פֹה כִּי יְהוָה שְׁלָחַנִי עַד בֵּית אֵל וַיֹּאמֶר אֱלִישָׁע חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיֵּרְדוּ בֵּית אֵל` (מלכים-ב ב:ב)
`וַיֹּאמֶר לוֹ אֵלִיָּהוּ אֱלִישָׁע שֵׁב נָא פֹה כִּי יְהוָה שְׁלָחַנִי יְרִיחוֹ וַיֹּאמֶר חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיָּבֹאוּ יְרִיחוֹ` (מלכים-ב ב:ד)
`וַיֹּאמֶר לוֹ אֵלִיָּהוּ שֵׁב נָא פֹה כִּי יְהוָה שְׁלָחַנִי הַיַּרְדֵּנָה וַיֹּאמֶר חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיֵּלְכוּ שְׁנֵיהֶם` (מלכים-ב ב:ו)
`וַתָּבֹא אֶל אִישׁ הָאֱלֹהִים אֶל הָהָר וַתַּחֲזֵק בְּרַגְלָיו וַיִּגַּשׁ גֵּיחֲזִי לְהָדְפָהּ וַיֹּאמֶר אִישׁ הָאֱלֹהִים הַרְפֵּה לָהּ כִּי נַפְשָׁהּ מָרָה לָהּ וַיהוָה הֶעְלִים מִמֶּנִּי וְלֹא הִגִּיד לִי` (מלכים-ב ד:כז)
`וַתֹּאמֶר אֵם הַנַּעַר חַי יְהוָה וְחֵי נַפְשְׁךָ אִם אֶעֶזְבֶךָּ וַיָּקָם וַיֵּלֶךְ אַחֲרֶיהָ` (מלכים-ב ד:ל)
`וַיָּקוּמוּ וַיָּנוּסוּ בַנֶּשֶׁף וַיַּעַזְבוּ אֶת אָהֳלֵיהֶם וְאֶת סוּסֵיהֶם וְאֶת חֲמֹרֵיהֶם הַמַּחֲנֶה כַּאֲשֶׁר הִיא וַיָּנֻסוּ אֶל נַפְשָׁם` (מלכים-ב ז:ז)
`וַיָּשָׁב יְהוֹרָם הַמֶּלֶךְ לְהִתְרַפֵּא בִיְזְרְעֶאל מִן הַמַּכִּים אֲשֶׁר יַכֻּהוּ אֲרַמִּים בְּהִלָּחֲמוֹ אֶת חֲזָאֵל מֶלֶךְ אֲרָם וַיֹּאמֶר יֵהוּא אִם יֵשׁ נַפְשְׁכֶם אַל יֵצֵא פָלִיט מִן הָעִיר לָלֶכֶת לַגִּיד בְּיִזְרְעֶאל` (מלכים-ב ט:טו)
`וַיָּבֹאוּ לַעֲשׂוֹת זְבָחִים וְעֹלוֹת וְיֵהוּא שָׂם לוֹ בַחוּץ שְׁמֹנִים אִישׁ וַיֹּאמֶר הָאִישׁ אֲשֶׁר יִמָּלֵט מִן הָאֲנָשִׁים אֲשֶׁר אֲנִי מֵבִיא עַל יְדֵיכֶם נַפְשׁוֹ תַּחַת נַפְשׁוֹ` (מלכים-ב י:כד)
`וַיֹּאמֶר יְהוֹאָשׁ אֶל הַכֹּהֲנִים כֹּל כֶּסֶף הַקֳּדָשִׁים אֲשֶׁר יוּבָא בֵית יְהוָה כֶּסֶף עוֹבֵר אִישׁ כֶּסֶף נַפְשׁוֹת עֶרְכּוֹ כָּל כֶּסֶף אֲשֶׁר יַעֲלֶה עַל לֶב אִישׁ לְהָבִיא בֵּית יְהוָה` (מלכים-ב יב:ד)
`וַיַּעֲמֹד הַמֶּלֶךְ עַל הָעַמּוּד וַיִּכְרֹת אֶת הַבְּרִית לִפְנֵי יְהוָה לָלֶכֶת אַחַר יְהוָה וְלִשְׁמֹר מִצְוֺתָיו וְאֶת עֵדְוֺתָיו וְאֶת חֻקֹּתָיו בְּכָל לֵב וּבְכָל נֶפֶשׁ לְהָקִים אֶת דִּבְרֵי הַבְּרִית הַזֹּאת הַכְּתֻבִים עַל הַסֵּפֶר הַזֶּה וַיַּעֲמֹד כָּל הָעָם בַּבְּרִית` (מלכים-ב כג:ג)
`וְכָמֹהוּ לֹא הָיָה לְפָנָיו מֶלֶךְ אֲשֶׁר שָׁב אֶל יְהוָה בְּכָל לְבָבוֹ וּבְכָל נַפְשׁוֹ וּבְכָל מְאֹדוֹ כְּכֹל תּוֹרַת מֹשֶׁה וְאַחֲרָיו לֹא קָם כָּמֹהוּ` (מלכים-ב כג:כה)
`חָדְשֵׁיכֶם וּמוֹעֲדֵיכֶם שָׂנְאָה נַפְשִׁי הָיוּ עָלַי לָטֹרַח נִלְאֵיתִי נְשֹׂא` (ישעיהו א:יד)
`הַכָּרַת פְּנֵיהֶם עָנְתָה בָּם וְחַטָּאתָם כִּסְדֹם הִגִּידוּ לֹא כִחֵדוּ אוֹי לְנַפְשָׁם כִּי גָמְלוּ לָהֶם רָעָה` (ישעיהו ג:ט)
`הַפְּאֵרִים וְהַצְּעָדוֹת וְהַקִּשֻּׁרִים וּבָתֵּי הַנֶּפֶשׁ וְהַלְּחָשִׁים` (ישעיהו ג:כ)
`לָכֵן הִרְחִיבָה שְּׁאוֹל נַפְשָׁהּ וּפָעֲרָה פִיהָ לִבְלִי חֹק וְיָרַד הֲדָרָהּ וַהֲמוֹנָהּ וּשְׁאוֹנָהּ וְעָלֵז בָּהּ` (ישעיהו ה:יד)
`וּכְבוֹד יַעְרוֹ וְכַרְמִלּוֹ מִנֶּפֶשׁ וְעַד בָּשָׂר יְכַלֶּה וְהָיָה כִּמְסֹס נֹסֵס` (ישעיהו י:יח)
`וַתִּזְעַק חֶשְׁבּוֹן וְאֶלְעָלֵה עַד יַהַץ נִשְׁמַע קוֹלָם עַל כֵּן חֲלֻצֵי מוֹאָב יָרִיעוּ נַפְשׁוֹ יָרְעָה לּוֹ` (ישעיהו טו:ד)
`וְהָיוּ שָׁתֹתֶיהָ מְדֻכָּאִים כָּל עֹשֵׂי שֶׂכֶר אַגְמֵי נָפֶשׁ` (ישעיהו יט:י)
`אַף אֹרַח מִשְׁפָּטֶיךָ יְהוָה קִוִּינוּךָ לְשִׁמְךָ וּלְזִכְרְךָ תַּאֲוַת נָפֶשׁ` (ישעיהו כו:ח)
`נַפְשִׁי אִוִּיתִיךָ בַּלַּיְלָה אַף רוּחִי בְקִרְבִּי אֲשַׁחֲרֶךָּ כִּי כַּאֲשֶׁר מִשְׁפָּטֶיךָ לָאָרֶץ צֶדֶק לָמְדוּ יֹשְׁבֵי תֵבֵל` (ישעיהו כו:ט)
`וְהָיָה כַּאֲשֶׁר יַחֲלֹם הָרָעֵב וְהִנֵּה אוֹכֵל וְהֵקִיץ וְרֵיקָה נַפְשׁוֹ וְכַאֲשֶׁר יַחֲלֹם הַצָּמֵא וְהִנֵּה שֹׁתֶה וְהֵקִיץ וְהִנֵּה עָיֵף וְנַפְשׁוֹ שׁוֹקֵקָה כֵּן יִהְיֶה הֲמוֹן כָּל הַגּוֹיִם הַצֹּבְאִים עַל הַר צִיּוֹן` (ישעיהו כט:ח)
`כִּי נָבָל נְבָלָה יְדַבֵּר וְלִבּוֹ יַעֲשֶׂה אָוֶן לַעֲשׂוֹת חֹנֶף וּלְדַבֵּר אֶל יְהוָה תּוֹעָה לְהָרִיק נֶפֶשׁ רָעֵב וּמַשְׁקֶה צָמֵא יַחְסִיר` (ישעיהו לב:ו)
`מָה אֲדַבֵּר וְאָמַר לִי וְהוּא עָשָׂה אֶדַּדֶּה כָל שְׁנוֹתַי עַל מַר נַפְשִׁי` (ישעיהו לח:טו)
`הִנֵּה לְשָׁלוֹם מַר לִי מָר וְאַתָּה חָשַׁקְתָּ נַפְשִׁי מִשַּׁחַת בְּלִי כִּי הִשְׁלַכְתָּ אַחֲרֵי גֵוְךָ כָּל חֲטָאָי` (ישעיהו לח:יז)
`הֵן עַבְדִּי אֶתְמָךְ בּוֹ בְּחִירִי רָצְתָה נַפְשִׁי נָתַתִּי רוּחִי עָלָיו מִשְׁפָּט לַגּוֹיִם יוֹצִיא` (ישעיהו מב:א)
`מֵאֲשֶׁר יָקַרְתָּ בְעֵינַי נִכְבַּדְתָּ וַאֲנִי אֲהַבְתִּיךָ וְאֶתֵּן אָדָם תַּחְתֶּיךָ וּלְאֻמִּים תַּחַת נַפְשֶׁךָ` (ישעיהו מג:ד)
`רֹעֶה אֵפֶר לֵב הוּתַל הִטָּהוּ וְלֹא יַצִּיל אֶת נַפְשׁוֹ וְלֹא יֹאמַר הֲלוֹא שֶׁקֶר בִּימִינִי` (ישעיהו מד:כ)
`קָרְסוּ כָרְעוּ יַחְדָּו לֹא יָכְלוּ מַלֵּט מַשָּׂא וְנַפְשָׁם בַּשְּׁבִי הָלָכָה` (ישעיהו מו:ב)
`הִנֵּה הָיוּ כְקַשׁ אֵשׁ שְׂרָפָתַם לֹא יַצִּילוּ אֶת נַפְשָׁם מִיַּד לֶהָבָה אֵין גַּחֶלֶת לַחְמָם אוּר לָשֶׁבֶת נֶגְדּוֹ` (ישעיהו מז:יד)
`כֹּה אָמַר יְהוָה גֹּאֵל יִשְׂרָאֵל קְדוֹשׁוֹ לִבְזֹה נֶפֶשׁ לִמְתָעֵב גּוֹי לְעֶבֶד מֹשְׁלִים מְלָכִים יִרְאוּ וָקָמוּ שָׂרִים וְיִשְׁתַּחֲוּוּ לְמַעַן יְהוָה אֲשֶׁר נֶאֱמָן קְדֹשׁ יִשְׂרָאֵל וַיִּבְחָרֶךָּ` (ישעיהו מט:ז)
`וְשַׂמְתִּיהָ בְּיַד מוֹגַיִךְ אֲשֶׁר אָמְרוּ לְנַפְשֵׁךְ שְׁחִי וְנַעֲבֹרָה וַתָּשִׂימִי כָאָרֶץ גֵּוֵךְ וְכַחוּץ לַעֹבְרִים` (ישעיהו נא:כג)
`וַיהוָה חָפֵץ דַּכְּאוֹ הֶחֱלִי אִם תָּשִׂים אָשָׁם נַפְשׁוֹ יִרְאֶה זֶרַע יַאֲרִיךְ יָמִים וְחֵפֶץ יְהוָה בְּיָדוֹ יִצְלָח` (ישעיהו נג:י)
`מֵעֲמַל נַפְשׁוֹ יִרְאֶה יִשְׂבָּע בְּדַעְתּוֹ יַצְדִּיק צַדִּיק עַבְדִּי לָרַבִּים וַעֲוֺנֹתָם הוּא יִסְבֹּל` (ישעיהו נג:יא)
`לָכֵן אֲחַלֶּק לוֹ בָרַבִּים וְאֶת עֲצוּמִים יְחַלֵּק שָׁלָל תַּחַת אֲשֶׁר הֶעֱרָה לַמָּוֶת נַפְשׁוֹ וְאֶת פֹּשְׁעִים נִמְנָה וְהוּא חֵטְא רַבִּים נָשָׂא וְלַפֹּשְׁעִים יַפְגִּיעַ` (ישעיהו נג:יב)
`לָמָּה תִשְׁקְלוּ כֶסֶף בְּלוֹא לֶחֶם וִיגִיעֲכֶם בְּלוֹא לְשָׂבְעָה שִׁמְעוּ שָׁמוֹעַ אֵלַי וְאִכְלוּ טוֹב וְתִתְעַנַּג בַּדֶּשֶׁן נַפְשְׁכֶם` (ישעיהו נה:ב)
`הַטּוּ אָזְנְכֶם וּלְכוּ אֵלַי שִׁמְעוּ וּתְחִי נַפְשְׁכֶם וְאֶכְרְתָה לָכֶם בְּרִית עוֹלָם חַסְדֵי דָוִד הַנֶּאֱמָנִים` (ישעיהו נה:ג)
`וְהַכְּלָבִים עַזֵּי נֶפֶשׁ לֹא יָדְעוּ שָׂבְעָה וְהֵמָּה רֹעִים לֹא יָדְעוּ הָבִין כֻּלָּם לְדַרְכָּם פָּנוּ אִישׁ לְבִצְעוֹ מִקָּצֵהוּ` (ישעיהו נו:יא)
`לָמָּה צַּמְנוּ וְלֹא רָאִיתָ עִנִּינוּ נַפְשֵׁנוּ וְלֹא תֵדָע הֵן בְּיוֹם צֹמְכֶם תִּמְצְאוּ חֵפֶץ וְכָל עַצְּבֵיכֶם תִּנְגֹּשׂוּ` (ישעיהו נח:ג)
`הֲכָזֶה יִהְיֶה צוֹם אֶבְחָרֵהוּ יוֹם עַנּוֹת אָדָם נַפְשׁוֹ הֲלָכֹף כְּאַגְמֹן רֹאשׁוֹ וְשַׂק וָאֵפֶר יַצִּיעַ הֲלָזֶה תִּקְרָא צוֹם וְיוֹם רָצוֹן לַיהוָה` (ישעיהו נח:ה)
`וְתָפֵק לָרָעֵב נַפְשֶׁךָ וְנֶפֶשׁ נַעֲנָה תַּשְׂבִּיעַ וְזָרַח בַּחֹשֶׁךְ אוֹרֶךָ וַאֲפֵלָתְךָ כַּצָּהֳרָיִם` (ישעיהו נח:י)
`וְנָחֲךָ יְהוָה תָּמִיד וְהִשְׂבִּיעַ בְּצַחְצָחוֹת נַפְשֶׁךָ וְעַצְמֹתֶיךָ יַחֲלִיץ וְהָיִיתָ כְּגַן רָוֶה וּכְמוֹצָא מַיִם אֲשֶׁר לֹא יְכַזְּבוּ מֵימָיו` (ישעיהו נח:יא)
`שׂוֹשׂ אָשִׂישׂ בַּיהוָה תָּגֵל נַפְשִׁי בֵּאלֹהַי כִּי הִלְבִּישַׁנִי בִּגְדֵי יֶשַׁע מְעִיל צְדָקָה יְעָטָנִי כֶּחָתָן יְכַהֵן פְּאֵר וְכַכַּלָּה תַּעְדֶּה כֵלֶיהָ` (ישעיהו סא:י)
`שׁוֹחֵט הַשּׁוֹר מַכֵּה אִישׁ זוֹבֵחַ הַשֶּׂה עֹרֵף כֶּלֶב מַעֲלֵה מִנְחָה דַּם חֲזִיר מַזְכִּיר לְבֹנָה מְבָרֵךְ אָוֶן גַּם הֵמָּה בָּחֲרוּ בְּדַרְכֵיהֶם וּבְשִׁקּוּצֵיהֶם נַפְשָׁם חָפֵצָה` (ישעיהו סו:ג)
`פֶּרֶה לִמֻּד מִדְבָּר בְּאַוַּת נַפְשׁוֹ שָׁאֲפָה רוּחַ תַּאֲנָתָהּ מִי יְשִׁיבֶנָּה כָּל מְבַקְשֶׁיהָ לֹא יִיעָפוּ בְּחָדְשָׁהּ יִמְצָאוּנְהָ` (ירמיהו ב:כד)
`גַּם בִּכְנָפַיִךְ נִמְצְאוּ דַּם נַפְשׁוֹת אֶבְיוֹנִים נְקִיִּים לֹא בַמַּחְתֶּרֶת מְצָאתִים כִּי עַל כָּל אֵלֶּה` (ירמיהו ב:לד)
`וַיֹּאמֶר יְהוָה אֵלַי צִדְּקָה נַפְשָׁהּ מְשֻׁבָה יִשְׂרָאֵל מִבֹּגֵדָה יְהוּדָה` (ירמיהו ג:יא)
`וָאֹמַר אֲהָהּ אֲדֹנָי יְהוִה אָכֵן הַשֵּׁא הִשֵּׁאתָ לָעָם הַזֶּה וְלִירוּשָׁלִַם לֵאמֹר שָׁלוֹם יִהְיֶה לָכֶם וְנָגְעָה חֶרֶב עַד הַנָּפֶשׁ` (ירמיהו ד:י)
`מֵעַי מֵעַי אָחוּלָה קִירוֹת לִבִּי הֹמֶה לִּי לִבִּי לֹא אַחֲרִישׁ כִּי קוֹל שׁוֹפָר שָׁמַעְתִּי נַפְשִׁי תְּרוּעַת מִלְחָמָה` (ירמיהו ד:יט)
`וְאַתִּי שָׁדוּד מַה תַּעֲשִׂי כִּי תִלְבְּשִׁי שָׁנִי כִּי תַעְדִּי עֲדִי זָהָב כִּי תִקְרְעִי בַפּוּךְ עֵינַיִךְ לַשָּׁוְא תִּתְיַפִּי מָאֲסוּ בָךְ עֹגְבִים נַפְשֵׁךְ יְבַקֵּשׁוּ` (ירמיהו ד:ל)
`כִּי קוֹל כְּחוֹלָה שָׁמַעְתִּי צָרָה כְּמַבְכִּירָה קוֹל בַּת צִיּוֹן תִּתְיַפֵּחַ תְּפָרֵשׂ כַּפֶּיהָ אוֹי נָא לִי כִּי עָיְפָה נַפְשִׁי לְהֹרְגִים` (ירמיהו ד:לא)
`הַעַל אֵלֶּה לוֹא אֶפְקֹד נְאֻם יְהֹוָה וְאִם בְּגוֹי אֲשֶׁר כָּזֶה לֹא תִתְנַקֵּם נַפְשִׁי` (ירמיהו ה:ט)
`הַעַל אֵלֶּה לֹא אֶפְקֹד נְאֻם יְהֹוָה אִם בְּגוֹי אֲשֶׁר כָּזֶה לֹא תִתְנַקֵּם נַפְשִׁי` (ירמיהו ה:כט)
`הִוָּסְרִי יְרוּשָׁלִַם פֶּן תֵּקַע נַפְשִׁי מִמֵּךְ פֶּן אֲשִׂימֵךְ שְׁמָמָה אֶרֶץ לוֹא נוֹשָׁבָה` (ירמיהו ו:ח)
`כֹּה אָמַר יְהוָה עִמְדוּ עַל דְּרָכִים וּרְאוּ וְשַׁאֲלוּ לִנְתִבוֹת עוֹלָם אֵי זֶה דֶרֶךְ הַטּוֹב וּלְכוּ בָהּ וּמִצְאוּ מַרְגּוֹעַ לְנַפְשְׁכֶם וַיֹּאמְרוּ לֹא נֵלֵךְ` (ירמיהו ו:טז)
`הַעַל אֵלֶּה לֹא אֶפְקָד בָּם נְאֻם יְהוָה אִם בְּגוֹי אֲשֶׁר כָּזֶה לֹא תִתְנַקֵּם נַפְשִׁי` (ירמיהו ט:ט)
`לָכֵן כֹּה אָמַר יְהוָה עַל אַנְשֵׁי עֲנָתוֹת הַמְבַקְשִׁים אֶת נַפְשְׁךָ לֵאמֹר לֹא תִנָּבֵא בְּשֵׁם יְהוָה וְלֹא תָמוּת בְּיָדֵנוּ` (ירמיהו יא:כא)
`עָזַבְתִּי אֶת בֵּיתִי נָטַשְׁתִּי אֶת נַחֲלָתִי נָתַתִּי אֶת יְדִדוּת נַפְשִׁי בְּכַף אֹיְבֶיהָ` (ירמיהו יב:ז)
`וְאִם לֹא תִשְׁמָעוּהָ בְּמִסְתָּרִים תִּבְכֶּה נַפְשִׁי מִפְּנֵי גֵוָה וְדָמֹעַ תִּדְמַע וְתֵרַד עֵינִי דִּמְעָה כִּי נִשְׁבָּה עֵדֶר יְהוָה` (ירמיהו יג:יז)
`הֲמָאֹס מָאַסְתָּ אֶת יְהוּדָה אִם בְּצִיּוֹן גָּעֲלָה נַפְשֶׁךָ מַדּוּעַ הִכִּיתָנוּ וְאֵין לָנוּ מַרְפֵּא קַוֵּה לְשָׁלוֹם וְאֵין טוֹב וּלְעֵת מַרְפֵּא וְהִנֵּה בְעָתָה` (ירמיהו יד:יט)
`וַיֹּאמֶר יְהוָה אֵלַי אִם יַעֲמֹד מֹשֶׁה וּשְׁמוּאֵל לְפָנַי אֵין נַפְשִׁי אֶל הָעָם הַזֶּה שַׁלַּח מֵעַל פָּנַי וְיֵצֵאוּ` (ירמיהו טו:א)
`אֻמְלְלָה יֹלֶדֶת הַשִּׁבְעָה נָפְחָה נַפְשָׁהּ בָּאָה שִׁמְשָׁהּ בְּעֹד יוֹמָם בּוֹשָׁה וְחָפֵרָה וּשְׁאֵרִיתָם לַחֶרֶב אֶתֵּן לִפְנֵי אֹיְבֵיהֶם נְאֻם יְהוָה` (ירמיהו טו:ט)
`כֹּה אָמַר יְהוָה הִשָּׁמְרוּ בְּנַפְשׁוֹתֵיכֶם וְאַל תִּשְׂאוּ מַשָּׂא בְּיוֹם הַשַּׁבָּת וַהֲבֵאתֶם בְּשַׁעֲרֵי יְרוּשָׁלִָם` (ירמיהו יז:כא)
`הַיְשֻׁלַּם תַּחַת טוֹבָה רָעָה כִּי כָרוּ שׁוּחָה לְנַפְשִׁי זְכֹר עָמְדִי לְפָנֶיךָ לְדַבֵּר עֲלֵיהֶם טוֹבָה לְהָשִׁיב אֶת חֲמָתְךָ מֵהֶם` (ירמיהו יח:כ)
`וּבַקֹּתִי אֶת עֲצַת יְהוּדָה וִירוּשָׁלִַם בַּמָּקוֹם הַזֶּה וְהִפַּלְתִּים בַּחֶרֶב לִפְנֵי אֹיְבֵיהֶם וּבְיַד מְבַקְשֵׁי נַפְשָׁם וְנָתַתִּי אֶת נִבְלָתָם לְמַאֲכָל לְעוֹף הַשָּׁמַיִם וּלְבֶהֱמַת הָאָרֶץ` (ירמיהו יט:ז)
`וְהַאֲכַלְתִּים אֶת בְּשַׂר בְּנֵיהֶם וְאֵת בְּשַׂר בְּנֹתֵיהֶם וְאִישׁ בְּשַׂר רֵעֵהוּ יֹאכֵלוּ בְּמָצוֹר וּבְמָצוֹק אֲשֶׁר יָצִיקוּ לָהֶם אֹיְבֵיהֶם וּמְבַקְשֵׁי נַפְשָׁם` (ירמיהו יט:ט)
`שִׁירוּ לַיהוָה הַלְלוּ אֶת יְהוָה כִּי הִצִּיל אֶת נֶפֶשׁ אֶבְיוֹן מִיַּד מְרֵעִים` (ירמיהו כ:יג)
`וְאַחֲרֵי כֵן נְאֻם יְהוָה אֶתֵּן אֶת צִדְקִיָּהוּ מֶלֶךְ יְהוּדָה וְאֶת עֲבָדָיו וְאֶת הָעָם וְאֶת הַנִּשְׁאָרִים בָּעִיר הַזֹּאת מִן הַדֶּבֶר מִן הַחֶרֶב וּמִן הָרָעָב בְּיַד נְבוּכַדְרֶאצַּר מֶלֶךְ בָּבֶל וּבְיַד אֹיְבֵיהֶם וּבְיַד מְבַקְשֵׁי נַפְשָׁם וְהִכָּם לְפִי חֶרֶב לֹא יָחוּס עֲלֵיהֶם וְלֹא יַחְמֹל וְלֹא יְרַחֵם` (ירמיהו כא:ז)
`הַיֹּשֵׁב בָּעִיר הַזֹּאת יָמוּת בַּחֶרֶב וּבָרָעָב וּבַדָּבֶר וְהַיּוֹצֵא וְנָפַל עַל הַכַּשְׂדִּים הַצָּרִים עֲלֵיכֶם יִחְיֶה וְהָיְתָה לּוֹ נַפְשׁוֹ לְשָׁלָל` (ירמיהו כא:ט)
`וּנְתַתִּיךָ בְּיַד מְבַקְשֵׁי נַפְשֶׁךָ וּבְיַד אֲשֶׁר אַתָּה יָגוֹר מִפְּנֵיהֶם וּבְיַד נְבוּכַדְרֶאצַּר מֶלֶךְ בָּבֶל וּבְיַד הַכַּשְׂדִּים` (ירמיהו כב:כה)
`וְעַל הָאָרֶץ אֲשֶׁר הֵם מְנַשְּׂאִים אֶת נַפְשָׁם לָשׁוּב שָׁם שָׁמָּה לֹא יָשׁוּבוּ` (ירמיהו כב:כז)
`הֶהָמֵת הֱמִתֻהוּ חִזְקִיָּהוּ מֶלֶךְ יְהוּדָה וְכָל יְהוּדָה הֲלֹא יָרֵא אֶת יְהוָה וַיְחַל אֶת פְּנֵי יְהוָה וַיִּנָּחֶם יְהוָה אֶל הָרָעָה אֲשֶׁר דִּבֶּר עֲלֵיהֶם וַאֲנַחְנוּ עֹשִׂים רָעָה גְדוֹלָה עַל נַפְשׁוֹתֵינוּ` (ירמיהו כו:יט)
`וּבָאוּ וְרִנְּנוּ בִמְרוֹם צִיּוֹן וְנָהֲרוּ אֶל טוּב יְהוָה עַל דָּגָן וְעַל תִּירֹשׁ וְעַל יִצְהָר וְעַל בְּנֵי צֹאן וּבָקָר וְהָיְתָה נַפְשָׁם כְּגַן רָוֶה וְלֹא יוֹסִיפוּ לְדַאֲבָה עוֹד` (ירמיהו לא:יב)
`וְרִוֵּיתִי נֶפֶשׁ הַכֹּהֲנִים דָּשֶׁן וְעַמִּי אֶת טוּבִי יִשְׂבָּעוּ נְאֻם יְהוָה` (ירמיהו לא:יד)

`וְנָתַתִּי אוֹתָם בְּיַד אֹיְבֵיהֶם וּבְיַד מְבַקְשֵׁי נַפְשָׁם וְהָיְתָה נִבְלָתָם לְמַאֲכָל לְעוֹף הַשָּׁמַיִם וּלְבֶהֱמַת הָאָרֶץ` (ירמיהו לד:כ)
`וְאֶת צִדְקִיָּהוּ מֶלֶךְ יְהוּדָה וְאֶת שָׂרָיו אֶתֵּן בְּיַד אֹיְבֵיהֶם וּבְיַד מְבַקְשֵׁי נַפְשָׁם וּבְיַד חֵיל מֶלֶךְ בָּבֶל הָעֹלִים מֵעֲלֵיכֶם` (ירמיהו לד:כא)
`כֹּה אָמַר יְהוָה אַל תַּשִּׁאוּ נַפְשֹׁתֵיכֶם לֵאמֹר הָלֹךְ יֵלְכוּ מֵעָלֵינוּ הַכַּשְׂדִּים כִּי לֹא יֵלֵכוּ` (ירמיהו לז:ט)
`כֹּה אָמַר יְהוָה הַיֹּשֵׁב בָּעִיר הַזֹּאת יָמוּת בַּחֶרֶב בָּרָעָב וּבַדָּבֶר וְהַיֹּצֵא אֶל הַכַּשְׂדִּים יִחְיֶה וְהָיְתָה לּוֹ נַפְשׁוֹ לְשָׁלָל וָחָי` (ירמיהו לח:ב)
`וַיִּשָּׁבַע הַמֶּלֶךְ צִדְקִיָּהוּ אֶל יִרְמְיָהוּ בַּסֵּתֶר לֵאמֹר חַי יְהוָה את אֲשֶׁר עָשָׂה לָנוּ אֶת הַנֶּפֶשׁ הַזֹּאת אִם אֲמִיתֶךָ וְאִם אֶתֶּנְךָ בְּיַד הָאֲנָשִׁים הָאֵלֶּה אֲשֶׁר מְבַקְשִׁים אֶת נַפְשֶׁךָ` (ירמיהו לח:טז)
`וַיֹּאמֶר יִרְמְיָהוּ אֶל צִדְקִיָּהוּ כֹּה אָמַר יְהוָה אֱלֹהֵי צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל אִם יָצֹא תֵצֵא אֶל שָׂרֵי מֶלֶךְ בָּבֶל וְחָיְתָה נַפְשֶׁךָ וְהָעִיר הַזֹּאת לֹא תִשָּׂרֵף בָּאֵשׁ וְחָיִתָה אַתָּה וּבֵיתֶךָ` (ירמיהו לח:יז)
`וַיֹּאמֶר יִרְמְיָהוּ לֹא יִתֵּנוּ שְׁמַע נָא בְּקוֹל יְהוָה לַאֲשֶׁר אֲנִי דֹּבֵר אֵלֶיךָ וְיִיטַב לְךָ וּתְחִי נַפְשֶׁךָ` (ירמיהו לח:כ)
`כִּי מַלֵּט אֲמַלֶּטְךָ וּבַחֶרֶב לֹא תִפֹּל וְהָיְתָה לְךָ נַפְשְׁךָ לְשָׁלָל כִּי בָטַחְתָּ בִּי נְאֻם יְהוָה` (ירמיהו לט:יח)
`וַיֹּאמְרוּ אֵלָיו הֲיָדֹעַ תֵּדַע כִּי בַּעֲלִיס מֶלֶךְ בְּנֵי עַמּוֹן שָׁלַח אֶת יִשְׁמָעֵאל בֶּן נְתַנְיָה לְהַכֹּתְךָ נָפֶשׁ וְלֹא הֶאֱמִין לָהֶם גְּדַלְיָהוּ בֶּן אֲחִיקָם` (ירמיהו מ:יד)
`וְיוֹחָנָן בֶּן קָרֵחַ אָמַר אֶל גְּדַלְיָהוּ בַסֵּתֶר בַּמִּצְפָּה לֵאמֹר אֵלְכָה נָּא וְאַכֶּה אֶת יִשְׁמָעֵאל בֶּן נְתַנְיָה וְאִישׁ לֹא יֵדָע לָמָּה יַכֶּכָּה נֶּפֶשׁ וְנָפֹצוּ כָּל יְהוּדָה הַנִּקְבָּצִים אֵלֶיךָ וְאָבְדָה שְׁאֵרִית יְהוּדָה` (ירמיהו מ:טו)
`כִּי הִתְעֵתֶים בְּנַפְשׁוֹתֵיכֶם כִּי אַתֶּם שְׁלַחְתֶּם אֹתִי אֶל יְהוָה אֱלֹהֵיכֶם לֵאמֹר הִתְפַּלֵּל בַּעֲדֵנוּ אֶל יְהוָה אֱלֹהֵינוּ וּכְכֹל אֲשֶׁר יֹאמַר יְהוָה אֱלֹהֵינוּ כֵּן הַגֶּד לָנוּ וְעָשִׂינוּ` (ירמיהו מב:כ)
`אֶת הַגְּבָרִים וְאֶת הַנָּשִׁים וְאֶת הַטַּף וְאֶת בְּנוֹת הַמֶּלֶךְ וְאֵת כָּל הַנֶּפֶשׁ אֲשֶׁר הִנִּיחַ נְבוּזַרְאֲדָן רַב טַבָּחִים אֶת גְּדַלְיָהוּ בֶּן אֲחִיקָם בֶּן שָׁפָן וְאֵת יִרְמְיָהוּ הַנָּבִיא וְאֶת בָּרוּךְ בֶּן נֵרִיָּהוּ` (ירמיהו מג:ו)
`וְעַתָּה כֹּה אָמַר יְהוָה אֱלֹהֵי צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל לָמָה אַתֶּם עֹשִׂים רָעָה גְדוֹלָה אֶל נַפְשֹׁתֵכֶם לְהַכְרִית לָכֶם אִישׁ וְאִשָּׁה עוֹלֵל וְיוֹנֵק מִתּוֹךְ יְהוּדָה לְבִלְתִּי הוֹתִיר לָכֶם שְׁאֵרִית` (ירמיהו מד:ז)
`וְלֹא יִהְיֶה פָּלִיט וְשָׂרִיד לִשְׁאֵרִית יְהוּדָה הַבָּאִים לָגוּר שָׁם בְּאֶרֶץ מִצְרָיִם וְלָשׁוּב אֶרֶץ יְהוּדָה אֲשֶׁר הֵמָּה מְנַשְּׂאִים אֶת נַפְשָׁם לָשׁוּב לָשֶׁבֶת שָׁם כִּי לֹא יָשׁוּבוּ כִּי אִם פְּלֵטִים` (ירמיהו מד:יד)
`כֹּה אָמַר יְהוָה הִנְנִי נֹתֵן אֶת פַּרְעֹה חָפְרַע מֶלֶךְ מִצְרַיִם בְּיַד אֹיְבָיו וּבְיַד מְבַקְשֵׁי נַפְשׁוֹ כַּאֲשֶׁר נָתַתִּי אֶת צִדְקִיָּהוּ מֶלֶךְ יְהוּדָה בְּיַד נְבוּכַדְרֶאצַּר מֶלֶךְ בָּבֶל אֹיְבוֹ וּמְבַקֵּשׁ נַפְשׁוֹ` (ירמיהו מד:ל)
`וְאַתָּה תְּבַקֶּשׁ לְךָ גְדֹלוֹת אַל תְּבַקֵּשׁ כִּי הִנְנִי מֵבִיא רָעָה עַל כָּל בָּשָׂר נְאֻם יְהוָה וְנָתַתִּי לְךָ אֶת נַפְשְׁךָ לְשָׁלָל עַל כָּל הַמְּקֹמוֹת אֲשֶׁר תֵּלֶךְ שָׁם` (ירמיהו מה:ה)
`וּנְתַתִּים בְּיַד מְבַקְשֵׁי נַפְשָׁם וּבְיַד נְבוּכַדְרֶאצַּר מֶלֶךְ בָּבֶל וּבְיַד עֲבָדָיו וְאַחֲרֵי כֵן תִּשְׁכֹּן כִּימֵי קֶדֶם נְאֻם יְהוָה` (ירמיהו מו:כו)
`נֻסוּ מַלְּטוּ נַפְשְׁכֶם וְתִהְיֶינָה כַּעֲרוֹעֵר בַּמִּדְבָּר` (ירמיהו מח:ו)
`וְהַחְתַּתִּי אֶת עֵילָם לִפְנֵי אֹיְבֵיהֶם וְלִפְנֵי מְבַקְשֵׁי נַפְשָׁם וְהֵבֵאתִי עֲלֵיהֶם רָעָה אֶת חֲרוֹן אַפִּי נְאֻם יְהוָה וְשִׁלַּחְתִּי אַחֲרֵיהֶם אֶת הַחֶרֶב עַד כַּלּוֹתִי אוֹתָם` (ירמיהו מט:לז)
`וְשֹׁבַבְתִּי אֶת יִשְׂרָאֵל אֶל נָוֵהוּ וְרָעָה הַכַּרְמֶל וְהַבָּשָׁן וּבְהַר אֶפְרַיִם וְהַגִּלְעָד תִּשְׂבַּע נַפְשׁוֹ` (ירמיהו נ:יט)
`נֻסוּ מִתּוֹךְ בָּבֶל וּמַלְּטוּ אִישׁ נַפְשׁוֹ אַל תִּדַּמּוּ בַּעֲוֺנָהּ כִּי עֵת נְקָמָה הִיא לַיהוָה גְּמוּל הוּא מְשַׁלֵּם לָהּ` (ירמיהו נא:ו)
`נִשְׁבַּע יְהוָה צְבָאוֹת בְּנַפְשׁוֹ כִּי אִם מִלֵּאתִיךְ אָדָם כַּיֶּלֶק וְעָנוּ עָלַיִךְ הֵידָד` (ירמיהו נא:יד)
`צְאוּ מִתּוֹכָהּ עַמִּי וּמַלְּטוּ אִישׁ אֶת נַפְשׁוֹ מֵחֲרוֹן אַף יְהוָה` (ירמיהו נא:מה)
`בִּשְׁנַת שְׁמוֹנֶה עֶשְׂרֵה לִנְבוּכַדְרֶאצַּר מִירוּשָׁלִַם נֶפֶשׁ שְׁמֹנֶה מֵאוֹת שְׁלֹשִׁים וּשְׁנָיִם` (ירמיהו נב:כט)

`וְאַתָּה כִּי הִזְהַרְתָּ רָשָׁע וְלֹא שָׁב מֵרִשְׁעוֹ וּמִדַּרְכּוֹ הָרְשָׁעָה הוּא בַּעֲוֺנוֹ יָמוּת וְאַתָּה אֶת נַפְשְׁךָ הִצַּלְתָּ` (יחזקאל ג:יט)
`וְאַתָּה כִּי הִזְהַרְתּוֹ צַדִּיק לְבִלְתִּי חֲטֹא צַדִּיק וְהוּא לֹא חָטָא חָיוֹ יִחְיֶה כִּי נִזְהָר וְאַתָּה אֶת נַפְשְׁךָ הִצַּלְתָּ` (יחזקאל ג:כא)
`וָאֹמַר אֲהָהּ אֲדֹנָי יְהוִה הִנֵּה נַפְשִׁי לֹא מְטֻמָּאָה וּנְבֵלָה וּטְרֵפָה לֹא אָכַלְתִּי מִנְּעוּרַי וְעַד עַתָּה וְלֹא בָא בְּפִי בְּשַׂר פִּגּוּל` (יחזקאל ד:יד)
`כַּסְפָּם בַּחוּצוֹת יַשְׁלִיכוּ וּזְהָבָם לְנִדָּה יִהְיֶה כַּסְפָּם וּזְהָבָם לֹא יוּכַל לְהַצִּילָם בְּיוֹם עֶבְרַת יְהוָה נַפְשָׁם לֹא יְשַׂבֵּעוּ וּמֵעֵיהֶם לֹא יְמַלֵּאוּ כִּי מִכְשׁוֹל עֲוֺנָם הָיָה` (יחזקאל ז:יט)
`וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יְהוִה הוֹי לִמְתַפְּרוֹת כְּסָתוֹת עַל כָּל אַצִּילֵי יָדַי וְעֹשׂוֹת הַמִּסְפָּחוֹת עַל רֹאשׁ כָּל קוֹמָה לְצוֹדֵד נְפָשׁוֹת הַנְּפָשׁוֹת תְּצוֹדֵדְנָה לְעַמִּי וּנְפָשׁוֹת לָכֶנָה תְחַיֶּינָה` (יחזקאל יג:יח)
`וַתְּחַלֶּלְנָה אֹתִי אֶל עַמִּי בְּשַׁעֲלֵי שְׂעֹרִים וּבִפְתוֹתֵי לֶחֶם לְהָמִית נְפָשׁוֹת אֲשֶׁר לֹא תְמוּתֶנָה וּלְחַיּוֹת נְפָשׁוֹת אֲשֶׁר לֹא תִחְיֶינָה בְּכַזֶּבְכֶם לְעַמִּי שֹׁמְעֵי כָזָב` (יחזקאל יג:יט)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה הִנְנִי אֶל כִּסְּתוֹתֵיכֶנָה אֲשֶׁר אַתֵּנָה מְצֹדְדוֹת שָׁם אֶת הַנְּפָשׁוֹת לְפֹרְחוֹת וְקָרַעְתִּי אֹתָם מֵעַל זְרוֹעֹתֵיכֶם וְשִׁלַּחְתִּי אֶת הַנְּפָשׁוֹת אֲשֶׁר אַתֶּם מְצֹדְדוֹת אֶת נְפָשִׁים לְפֹרְחֹת` (יחזקאל יג:כ)
`וְהָיוּ שְׁלֹשֶׁת הָאֲנָשִׁים הָאֵלֶּה בְּתוֹכָהּ נֹחַ דָּנִאֵל וְאִיּוֹב הֵמָּה בְצִדְקָתָם יְנַצְּלוּ נַפְשָׁם נְאֻם אֲדֹנָי יְהוִה` (יחזקאל יד:יד)
`וְנֹחַ דָּנִאֵל וְאִיּוֹב בְּתוֹכָהּ חַי אָנִי נְאֻם אֲדֹנָי יְהוִה אִם בֵּן אִם בַּת יַצִּילוּ הֵמָּה בְצִדְקָתָם יַצִּילוּ נַפְשָׁם` (יחזקאל יד:כ)
`לֹא חָסָה עָלַיִךְ עַיִן לַעֲשׂוֹת לָךְ אַחַת מֵאֵלֶּה לְחֻמְלָה עָלָיִךְ וַתֻּשְׁלְכִי אֶל פְּנֵי הַשָּׂדֶה בְּגֹעַל נַפְשֵׁךְ בְּיוֹם הֻלֶּדֶת אֹתָךְ` (יחזקאל טז:ה)
`וְהִנֵּה נָטִיתִי יָדִי עָלַיִךְ וָאֶגְרַע חֻקֵּךְ וָאֶתְּנֵךְ בְּנֶפֶשׁ שֹׂנְאוֹתַיִךְ בְּנוֹת פְּלִשְׁתִּים הַנִּכְלָמוֹת מִדַּרְכֵּךְ זִמָּה` (יחזקאל טז:כז)
`וְלֹא בְחַיִל גָּדוֹל וּבְקָהָל רָב יַעֲשֶׂה אוֹתוֹ פַרְעֹה בַּמִּלְחָמָה בִּשְׁפֹּךְ סֹלְלָה וּבִבְנוֹת דָּיֵק לְהַכְרִית נְפָשׁוֹת רַבּוֹת` (יחזקאל יז:יז)
`הֵן כָּל הַנְּפָשׁוֹת לִי הֵנָּה כְּנֶפֶשׁ הָאָב וּכְנֶפֶשׁ הַבֵּן לִי הֵנָּה הַנֶּפֶשׁ הַחֹטֵאת הִיא תָמוּת` (יחזקאל יח:ד)
`הַנֶּפֶשׁ הַחֹטֵאת הִיא תָמוּת בֵּן לֹא יִשָּׂא בַּעֲוֺן הָאָב וְאָב לֹא יִשָּׂא בַּעֲוֺן הַבֵּן צִדְקַת הַצַּדִּיק עָלָיו תִּהְיֶה וְרִשְׁעַת רָשָׁע עָלָיו תִּהְיֶה` (יחזקאל יח:כ)
`וּבְשׁוּב רָשָׁע מֵרִשְׁעָתוֹ אֲשֶׁר עָשָׂה וַיַּעַשׂ מִשְׁפָּט וּצְדָקָה הוּא אֶת נַפְשׁוֹ יְחַיֶּה` (יחזקאל יח:כז)
`קֶשֶׁר נְבִיאֶיהָ בְּתוֹכָהּ כַּאֲרִי שׁוֹאֵג טֹרֵף טָרֶף נֶפֶשׁ אָכָלוּ חֹסֶן וִיקָר יִקָּחוּ אַלְמְנוֹתֶיהָ הִרְבּוּ בְתוֹכָהּ` (יחזקאל כב:כה)
`שָׂרֶיהָ בְקִרְבָּהּ כִּזְאֵבִים טֹרְפֵי טָרֶף לִשְׁפָּךְ דָּם לְאַבֵּד נְפָשׁוֹת לְמַעַן בְּצֹעַ בָּצַע` (יחזקאל כב:כז)
`וַיָּבֹאוּ אֵלֶיהָ בְנֵי בָבֶל לְמִשְׁכַּב דֹּדִים וַיְטַמְּאוּ אוֹתָהּ בְּתַזְנוּתָם וַתִּטְמָא בָם וַתֵּקַע נַפְשָׁהּ מֵהֶם` (יחזקאל כג:יז)
`וַתְּגַל תַּזְנוּתֶיהָ וַתְּגַל אֶת עֶרְוָתָהּ וַתֵּקַע נַפְשִׁי מֵעָלֶיהָ כַּאֲשֶׁר נָקְעָה נַפְשִׁי מֵעַל אֲחוֹתָהּ` (יחזקאל כג:יח)
`לָכֵן אָהֳלִיבָה כֹּה אָמַר אֲדֹנָי יְהוִה הִנְנִי מֵעִיר אֶת מְאַהֲבַיִךְ עָלַיִךְ אֵת אֲשֶׁר נָקְעָה נַפְשֵׁךְ מֵהֶם וַהֲבֵאתִים עָלַיִךְ מִסָּבִיב` (יחזקאל כג:כב)
`כִּי כֹה אָמַר אֲדֹנָי יְהוִה הִנְנִי נֹתְנָךְ בְּיַד אֲשֶׁר שָׂנֵאת בְּיַד אֲשֶׁר נָקְעָה נַפְשֵׁךְ מֵהֶם` (יחזקאל כג:כח)
`אֱמֹר לְבֵית יִשְׂרָאֵל כֹּה אָמַר אֲדֹנָי יְהוִה הִנְנִי מְחַלֵּל אֶת מִקְדָּשִׁי גְּאוֹן עֻזְּכֶם מַחְמַד עֵינֵיכֶם וּמַחְמַל נַפְשְׁכֶם וּבְנֵיכֶם וּבְנוֹתֵיכֶם אֲשֶׁר עֲזַבְתֶּם בַּחֶרֶב יִפֹּלוּ` (יחזקאל כד:כא)
`וְאַתָּה בֶן אָדָם הֲלוֹא בְּיוֹם קַחְתִּי מֵהֶם אֶת מָעוּזָּם מְשׂוֹשׂ תִּפְאַרְתָּם אֶת מַחְמַד עֵינֵיהֶם וְאֶת מַשָּׂא נַפְשָׁם בְּנֵיהֶם וּבְנוֹתֵיהֶם` (יחזקאל כד:כה)
`כִּי כֹה אָמַר אֲדֹנָי יְהוִה יַעַן מַחְאֲךָ יָד וְרַקְעֲךָ בְּרָגֶל וַתִּשְׂמַח בְּכָל שָׁאטְךָ בְּנֶפֶשׁ אֶל אַדְמַת יִשְׂרָאֵל` (יחזקאל כה:ו)
`כֹּה אָמַר אֲדֹנָי יְהוִה יַעַן עֲשׂוֹת פְּלִשְׁתִּים בִּנְקָמָה וַיִּנָּקְמוּ נָקָם בִּשְׁאָט בְּנֶפֶשׁ לְמַשְׁחִית אֵיבַת עוֹלָם` (יחזקאל כה:טו)
`יָוָן תֻּבַל וָמֶשֶׁךְ הֵמָּה רֹכְלָיִךְ בְּנֶפֶשׁ אָדָם וּכְלֵי נְחֹשֶׁת נָתְנוּ מַעֲרָבֵךְ` (יחזקאל כז:יג)
`וְהִקְרִיחוּ אֵלַיִךְ קָרְחָה וְחָגְרוּ שַׂקִּים וּבָכוּ אֵלַיִךְ בְּמַר נֶפֶשׁ מִסְפֵּד מָר` (יחזקאל כז:לא)
`וַהֲשִׁמּוֹתִי עָלֶיךָ עַמִּים רַבִּים וּמַלְכֵיהֶם יִשְׂעֲרוּ עָלֶיךָ שַׂעַר בְּעוֹפְפִי חַרְבִּי עַל פְּנֵיהֶם וְחָרְדוּ לִרְגָעִים אִישׁ לְנַפְשׁוֹ בְּיוֹם מַפַּלְתֶּךָ` (יחזקאל לב:י)
`אֵת קוֹל הַשּׁוֹפָר שָׁמַע וְלֹא נִזְהָר דָּמוֹ בּוֹ יִהְיֶה וְהוּא נִזְהָר נַפְשׁוֹ מִלֵּט` (יחזקאל לג:ה)
`וְהַצֹּפֶה כִּי יִרְאֶה אֶת הַחֶרֶב בָּאָה וְלֹא תָקַע בַּשּׁוֹפָר וְהָעָם לֹא נִזְהָר וַתָּבוֹא חֶרֶב וַתִּקַּח מֵהֶם נָפֶשׁ הוּא בַּעֲוֺנוֹ נִלְקָח וְדָמוֹ מִיַּד הַצֹּפֶה אֶדְרֹשׁ` (יחזקאל לג:ו)
`וְאַתָּה כִּי הִזְהַרְתָּ רָשָׁע מִדַּרְכּוֹ לָשׁוּב מִמֶּנָּה וְלֹא שָׁב מִדַּרְכּוֹ הוּא בַּעֲוֺנוֹ יָמוּת וְאַתָּה נַפְשְׁךָ הִצַּלְתָּ` (יחזקאל לג:ט)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה אִם לֹא בְּאֵשׁ קִנְאָתִי דִבַּרְתִּי עַל שְׁאֵרִית הַגּוֹיִם וְעַל אֱדוֹם כֻּלָּא אֲשֶׁר נָתְנוּ אֶת אַרְצִי לָהֶם לְמוֹרָשָׁה בְּשִׂמְחַת כָּל לֵבָב בִּשְׁאָט נֶפֶשׁ לְמַעַן מִגְרָשָׁהּ לָבַז` (יחזקאל לו:ה)
`וְהָיָה כָל נֶפֶשׁ חַיָּה אֲ‍שֶׁר יִשְׁרֹץ אֶל כָּל אֲשֶׁר יָבוֹא שָׁם נַחֲלַיִם יִחְיֶה וְהָיָה הַדָּגָה רַבָּה מְאֹד כִּי בָאוּ שָׁמָּה הַמַּיִם הָאֵלֶּה וְיֵרָפְאוּ וָחָי כֹּל אֲשֶׁר יָבוֹא שָׁמָּה הַנָּחַל` (יחזקאל מז:ט)
`חַטַּאת עַמִּי יֹאכֵלוּ וְאֶל עֲוֺנָם יִשְׂאוּ נַפְשׁוֹ` (הושע ד:ח)
`לֹא יִסְּכוּ לַיהוָה יַיִן וְלֹא יֶעֶרְבוּ לוֹ זִבְחֵיהֶם כְּלֶחֶם אוֹנִים לָהֶם כָּל אֹכְלָיו יִטַמָּאוּ כִּי לַחְמָם לְנַפְשָׁם לֹא יָבוֹא בֵּית יְהוָה` (הושע ט:ד)
`וְאָבַד מָנוֹס מִקָּל וְחָזָק לֹא יְאַמֵּץ כֹּחוֹ וְגִבּוֹר לֹא יְמַלֵּט נַפְשׁוֹ` (עמוס ב:יד)
`וְתֹפֵשׂ הַקֶּשֶׁת לֹא יַעֲמֹד וְקַל בְּרַגְלָיו לֹא יְמַלֵּט וְרֹכֵב הַסּוּס לֹא יְמַלֵּט נַפְשׁוֹ` (עמוס ב:טו)
`נִשְׁבַּע אֲדֹנָי יְהוִה בְּנַפְשׁוֹ נְאֻם יְהוָה אֱלֹהֵי צְבָאוֹת מְתָאֵב אָנֹכִי אֶת גְּאוֹן יַעֲקֹב וְאַרְמְנֹתָיו שָׂנֵאתִי וְהִסְגַּרְתִּי עִיר וּמְלֹאָהּ` (עמוס ו:ח)
`וַיִּקְרְאוּ אֶל יְהוָה וַיֹּאמְרוּ אָנָּה יְהוָה אַל נָא נֹאבְדָה בְּנֶפֶשׁ הָאִישׁ הַזֶּה וְאַל תִּתֵּן עָלֵינוּ דָּם נָקִיא כִּי אַתָּה יְהוָה כַּאֲשֶׁר חָפַצְתָּ עָשִׂיתָ` (יונה א:יד)
`אֲפָפוּנִי מַיִם עַד נֶפֶשׁ תְּהוֹם יְסֹבְבֵנִי סוּף חָבוּשׁ לְרֹאשִׁי` (יונה ב:ה)
`בְּהִתְעַטֵּף עָלַי נַפְשִׁי אֶת יְהוָה זָכָרְתִּי וַתָּבוֹא אֵלֶיךָ תְּפִלָּתִי אֶל הֵיכַל קָדְשֶׁךָ` (יונה ב:ז)
`וְעַתָּה יְהוָה קַח נָא אֶת נַפְשִׁי מִמֶּנִּי כִּי טוֹב מוֹתִי מֵחַיָּי` (יונה ד:ג)
`וַיְהִי כִּזְרֹחַ הַשֶּׁמֶשׁ וַיְמַן אֱלֹהִים רוּחַ קָדִים חֲרִישִׁית וַתַּךְ הַשֶּׁמֶשׁ עַל רֹאשׁ יוֹנָה וַיִּתְעַלָּף וַיִּשְׁאַל אֶת נַפְשׁוֹ לָמוּת וַיֹּאמֶר טוֹב מוֹתִי מֵחַיָּי` (יונה ד:ח)
`הֲיִרְצֶה יְהוָה בְּאַלְפֵי אֵילִים בְּרִבְבוֹת נַחֲלֵי שָׁמֶן הַאֶתֵּן בְּכוֹרִי פִּשְׁעִי פְּרִי בִטְנִי חַטַּאת נַפְשִׁי` (מיכה ו:ז)
`אַלְלַי לִי כִּי הָיִיתִי כְּאָסְפֵּי קַיִץ כְּעֹלְלֹת בָּצִיר אֵין אֶשְׁכּוֹל לֶאֱכוֹל בִּכּוּרָה אִוְּתָה נַפְשִׁי` (מיכה ז:א)
`עַל הָרַע כַּפַּיִם לְהֵיטִיב הַשַּׂר שֹׁאֵל וְהַשֹּׁפֵט בַּשִׁלּוּם וְהַגָּדוֹל דֹּבֵר הַוַּת נַפְשׁוֹ הוּא וַיְעַבְּתוּהָ` (מיכה ז:ג)
`הִנֵּה עֻפְּלָה לֹא יָשְׁרָה נַפְשׁוֹ בּוֹ וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה` (חבקוק ב:ד)
`וְאַף כִּי הַיַּיִן בּוֹגֵד גֶּבֶר יָהִיר וְלֹא יִנְוֶה אֲשֶׁר הִרְחִיב כִּשְׁאוֹל נַפְשׁוֹ וְהוּא כַמָּוֶת וְלֹא יִשְׂבָּע וַיֶּאֱסֹף אֵלָיו כָּל הַגּוֹיִם וַיִּקְבֹּץ אֵלָיו כָּל הָעַמִּים` (חבקוק ב:ה)
`יָעַצְתָּ בֹּשֶׁת לְבֵיתֶךָ קְצוֹת עַמִּים רַבִּים וְחוֹטֵא נַפְשֶׁךָ` (חבקוק ב:י)
`וַיֹּאמֶר חַגַּי אִם יִגַּע טְמֵא נֶפֶשׁ בְּכָל אֵלֶּה הֲיִטְמָא וַיַּעֲנוּ הַכֹּהֲנִים וַיֹּאמְרוּ יִטְמָא` (חגי ב:יג)
`וָאַכְחִד אֶת שְׁלֹשֶׁת הָרֹעִים בְּיֶרַח אֶחָד וַתִּקְצַר נַפְשִׁי בָּהֶם וְגַם נַפְשָׁם בָּחֲלָה בִי` (זכריה יא:ח)
`וַיִּשְׁבּוּ מִקְנֵיהֶם גְּמַלֵּיהֶם חֲמִשִּׁים אֶלֶף וְצֹאן מָאתַיִם וַחֲמִשִּׁים אֶלֶף וַחֲמוֹרִים אַלְפָּיִם וְנֶפֶשׁ אָדָם מֵאָה אָלֶף` (דברי-הימים-א ה:כא)
`וַיֹּאמֶר חָלִילָה לִּי מֵאֱלֹהַי מֵעֲשׂוֹת זֹאת הֲדַם הָאֲנָשִׁים הָאֵלֶּה אֶשְׁתֶּה בְנַפְשׁוֹתָם כִּי בְנַפְשׁוֹתָם הֱבִיאוּם וְלֹא אָבָה לִשְׁתּוֹתָם אֵלֶּה עָשׂוּ שְׁלֹשֶׁת הַגִּבּוֹרִים` (דברי-הימים-א יא:יט)
`עַתָּה תְּנוּ לְבַבְכֶם וְנַפְשְׁכֶם לִדְרוֹשׁ לַיהוָה אֱלֹהֵיכֶם וְקוּמוּ וּבְנוּ אֶת מִקְדַּשׁ יְהוָה הָאֱלֹהִים לְהָבִיא אֶת אֲרוֹן בְּרִית יְהוָה וּכְלֵי קֹדֶשׁ הָאֱלֹהִים לַבַּיִת הַנִּבְנֶה לְשֵׁם יְהוָה` (דברי-הימים-א כב:יט)
`וְאַתָּה שְׁלֹמֹה בְנִי דַּע אֶת אֱלֹהֵי אָבִיךָ וְעָבְדֵהוּ בְּלֵב שָׁלֵם וּבְנֶפֶשׁ חֲפֵצָה כִּי כָל לְבָבוֹת דּוֹרֵשׁ יְהוָה וְכָל יֵצֶר מַחֲשָׁבוֹת מֵבִין אִם תִּדְרְשֶׁנּוּ יִמָּצֵא לָךְ וְאִם תַּעַזְבֶנּוּ יַזְנִיחֲךָ לָעַד` (דברי-הימים-א כח:ט)
`וַיֹּאמֶר אֱלֹהִים לִשְׁלֹמֹה יַעַן אֲשֶׁר הָיְתָה זֹאת עִם לְבָבֶךָ וְלֹא שָׁאַלְתָּ עֹשֶׁר נְכָסִים וְכָבוֹד וְאֵת נֶפֶשׁ שֹׂנְאֶיךָ וְגַם יָמִים רַבִּים לֹא שָׁאָלְתָּ וַתִּשְׁאַל לְךָ חָכְמָה וּמַדָּע אֲשֶׁר תִּשְׁפּוֹט אֶת עַמִּי אֲשֶׁר הִמְלַכְתִּיךָ עָלָיו` (דברי-הימים-ב א:יא)
`וְשָׁבוּ אֵלֶיךָ בְּכָל לִבָּם וּבְכָל נַפְשָׁם בְּאֶרֶץ שִׁבְיָם אֲשֶׁר שָׁבוּ אֹתָם וְהִתְפַּלְלוּ דֶּרֶךְ אַרְצָם אֲשֶׁר נָתַתָּה לַאֲבוֹתָם וְהָעִיר אֲשֶׁר בָּחַרְתָּ וְלַבַּיִת אֲשֶׁר בָּנִיתִי לִשְׁמֶךָ` (דברי-הימים-ב ו:לח)
`וַיָּבֹאוּ בַבְּרִית לִדְרוֹשׁ אֶת יְהוָה אֱלֹהֵי אֲבוֹתֵיהֶם בְּכָל לְבָבָם וּבְכָל נַפְשָׁם` (דברי-הימים-ב טו:יב)
`וַיַּעֲמֹד הַמֶּלֶךְ עַל עָמְדוֹ וַיִּכְרֹת אֶת הַבְּרִית לִפְנֵי יְהוָה לָלֶכֶת אַחֲרֵי יְהוָה וְלִשְׁמוֹר אֶת מִצְוֺתָיו וְעֵדְוֺתָיו וְחֻקָּיו בְּכָל לְבָבוֹ וּבְכָל נַפְשׁוֹ לַעֲשׂוֹת אֶת דִּבְרֵי הַבְּרִית הַכְּתוּבִים עַל הַסֵּפֶר הַזֶּה` (דברי-הימים-ב לד:לא)
`רַבִּים אֹמְרִים לְנַפְשִׁי אֵין יְשׁוּעָתָה לּוֹ בֵאלֹהִים סֶלָה` (תהילים ג:ב)
`וְנַפְשִׁי נִבְהֲלָה מְאֹד וְאַתְּ יְהוָה עַד מָתָי` (תהילים ו:ג)
`שׁוּבָה יְהוָה חַלְּצָה נַפְשִׁי הוֹשִׁיעֵנִי לְמַעַן חַסְדֶּךָ` (תהילים ו:ד)
`פֶּן יִטְרֹף כְּאַרְיֵה נַפְשִׁי פֹּרֵק וְאֵין מַצִּיל` (תהילים ז:ב)
`יִרַדֹּף אוֹיֵב נַפְשִׁי וְיַשֵּׂג וְיִרְמֹס לָאָרֶץ חַיָּי וּכְבוֹדִי לֶעָפָר יַשְׁכֵּן סֶלָה` (תהילים ז:ה)
`כִּי הִלֵּל רָשָׁע עַל תַּאֲוַת נַפְשׁוֹ וּבֹצֵעַ בֵּרֵךְ נִאֵץ יְהוָה` (תהילים י:ג)
`לַמְנַצֵּחַ לְדָוִד בַּיהוָה חָסִיתִי אֵיךְ תֹּאמְרוּ לְנַפְשִׁי נוּדוּ הַרְכֶם צִפּוֹר` (תהילים יא:א)
`יְהוָה צַדִּיק יִבְחָן וְרָשָׁע וְאֹהֵב חָמָס שָׂנְאָה נַפְשׁוֹ` (תהילים יא:ה)
`עַד אָנָה אָשִׁית עֵצוֹת בְּנַפְשִׁי יָגוֹן בִּלְבָבִי יוֹמָם עַד אָנָה יָרוּם אֹיְבִי עָלָי` (תהילים יג:ב)
`כִּי לֹא תַעֲזֹב נַפְשִׁי לִשְׁאוֹל לֹא תִתֵּן חֲסִידְךָ לִרְאוֹת שָׁחַת` (תהילים טז:י)
`מִפְּנֵי רְשָׁעִים זוּ שַׁדּוּנִי אֹיְבַי בְּנֶפֶשׁ יַקִּיפוּ עָלָי` (תהילים יז:ט)
`קוּמָה יְהוָה קַדְּמָה פָנָיו הַכְרִיעֵהוּ פַּלְּטָה נַפְשִׁי מֵרָשָׁע חַרְבֶּךָ` (תהילים יז:יג)
`תּוֹרַת יְהוָה תְּמִימָה מְשִׁיבַת נָפֶשׁ עֵדוּת יְהוָה נֶאֱמָנָה מַחְכִּימַת פֶּתִי` (תהילים יט:ז)
`הַצִּילָה מֵחֶרֶב נַפְשִׁי מִיַּד כֶּלֶב יְחִידָתִי` (תהילים כב:כ)
`אָכְלוּ וַיִּשְׁתַּחֲוּוּ כָּל דִּשְׁנֵי אֶרֶץ לְפָנָיו יִכְרְעוּ כָּל יוֹרְדֵי עָפָר וְנַפְשׁוֹ לֹא חִיָּה` (תהילים כב:כט)
`נַפְשִׁי יְשׁוֹבֵב יַנְחֵנִי בְמַעְגְּלֵי צֶדֶק לְמַעַן שְׁמוֹ` (תהילים כג:ג)
`נְקִי כַפַּיִם וּבַר לֵבָב אֲשֶׁר לֹא נָשָׂא לַשָּׁוְא נַפְשִׁי וְלֹא נִשְׁבַּע לְמִרְמָה` (תהילים כד:ד)
`לְדָוִד אֵלֶיךָ יְהוָה נַפְשִׁי אֶשָּׂא` (תהילים כה:א)
`נַפְשׁוֹ בְּטוֹב תָּלִין וְזַרְעוֹ יִירַשׁ אָרֶץ` (תהילים כה:יג)
`שָׁמְרָה נַפְשִׁי וְהַצִּילֵנִי אַל אֵבוֹשׁ כִּי חָסִיתִי בָךְ` (תהילים כה:כ)
`אַל תֶּאֱסֹף עִם חַטָּאִים נַפְשִׁי וְעִם אַנְשֵׁי דָמִים חַיָּי` (תהילים כו:ט)
`אַל תִּתְּנֵנִי בְּנֶפֶשׁ צָרָי כִּי קָמוּ בִי עֵדֵי שֶׁקֶר וִיפֵחַ חָמָס` (תהילים כז:יב)
`יְהוָה הֶעֱלִיתָ מִן שְׁאוֹל נַפְשִׁי חִיִּיתַנִי מִיּוֹרְדֵי בוֹר` (תהילים ל:ג)
`אָגִילָה וְאֶשְׂמְחָה בְּחַסְדֶּךָ אֲשֶׁר רָאִיתָ אֶת עָנְיִי יָדַעְתָּ בְּצָרוֹת נַפְשִׁי` (תהילים לא:ז)
`חָנֵּנִי יְהוָה כִּי צַר לִי עָשְׁשָׁה בְכַעַס עֵינִי נַפְשִׁי וּבִטְנִי` (תהילים לא:ט)
`כִּי שָׁמַעְתִּי דִּבַּת רַבִּים מָגוֹר מִסָּבִיב בְּהִוָּסְדָם יַחַד עָלַי לָקַחַת נַפְשִׁי זָמָמוּ` (תהילים לא:יג)
`לְהַצִּיל מִמָּוֶת נַפְשָׁם וּלְחַיּוֹתָם בָּרָעָב` (תהילים לג:יט)
`נַפְשֵׁנוּ חִכְּתָה לַיהוָה עֶזְרֵנוּ וּמָגִנֵּנוּ הוּא` (תהילים לג:כ)
`בַּיהוָה תִּתְהַלֵּל נַפְשִׁי יִשְׁמְעוּ עֲנָוִים וְיִשְׂמָחוּ` (תהילים לד:ב)
`פּוֹדֶה יְהוָה נֶפֶשׁ עֲבָדָיו וְלֹא יֶאְשְׁמוּ כָּל הַחֹסִים בּוֹ` (תהילים לד:כב)
`וְהָרֵק חֲנִית וּסְגֹר לִקְרַאת רֹדְפָי אֱמֹר לְנַפְשִׁי יְשֻׁעָתֵךְ אָנִי` (תהילים לה:ג)
`יֵבֹשׁוּ וְיִכָּלְמוּ מְבַקְשֵׁי נַפְשִׁי יִסֹּגוּ אָחוֹר וְיַחְפְּרוּ חֹשְׁבֵי רָעָתִי` (תהילים לה:ד)
`כִּי חִנָּם טָמְנוּ לִי שַׁחַת רִשְׁתָּם חִנָּם חָפְרוּ לְנַפְשִׁי` (תהילים לה:ז)
`וְנַפְשִׁי תָּגִיל בַּיהוָה תָּשִׂישׂ בִּישׁוּעָתוֹ` (תהילים לה:ט)
`יְשַׁלְּמוּנִי רָעָה תַּחַת טוֹבָה שְׁכוֹל לְנַפְשִׁי` (תהילים לה:יב)
`וַאֲנִי בַּחֲלוֹתָם לְבוּשִׁי שָׂק עִנֵּיתִי בַצּוֹם נַפְשִׁי וּתְפִלָּתִי עַל חֵיקִי תָשׁוּב` (תהילים לה:יג)
`אֲדֹנָי כַּמָּה תִּרְאֶה הָשִׁיבָה נַפְשִׁי מִשֹּׁאֵיהֶם מִכְּפִירִים יְחִידָתִי` (תהילים לה:יז)
`אַל יֹאמְרוּ בְלִבָּם הֶאָח נַפְשֵׁנוּ אַל יֹאמְרוּ בִּלַּעֲנוּהוּ` (תהילים לה:כה)
`וַיְנַקְשׁוּ מְבַקְשֵׁי נַפְשִׁי וְדֹרְשֵׁי רָעָתִי דִּבְּרוּ הַוּוֹת וּמִרְמוֹת כָּל הַיּוֹם יֶהְגּוּ` (תהילים לח:יב)
`יֵבֹשׁוּ וְיַחְפְּרוּ יַחַד מְבַקְשֵׁי נַפְשִׁי לִסְפּוֹתָהּ יִסֹּגוּ אָחוֹר וְיִכָּלְמוּ חֲפֵצֵי רָעָתִי` (תהילים מ:יד)
`יְהוָה יִשְׁמְרֵהוּ וִיחַיֵּהוּ יֶאְשֹּׁר בָּאָרֶץ וְאַל תִּתְּנֵהוּ בְּנֶפֶשׁ אֹיְבָיו` (תהילים מא:ב)
`אֲ‍נִי אָמַרְתִּי יְהוָה חָנֵּנִי רְפָאָה נַפְשִׁי כִּי חָטָאתִי לָךְ` (תהילים מא:ד)
`לַמְנַצֵּחַ מַשְׂכִּיל לִבְנֵי קֹרַח כְּאַיָּל תַּעֲרֹג עַל אֲפִיקֵי מָיִם כֵּן נַפְשִׁי תַעֲרֹג אֵלֶיךָ אֱלֹהִים` (תהילים מב:א)
`צָמְאָה נַפְשִׁי לֵאלֹהִים לְאֵל חָי מָתַי אָבוֹא וְאֵרָאֶה פְּנֵי אֱלֹהִים` (תהילים מב:ב)
`אֵלֶּה אֶזְכְּרָה וְאֶשְׁפְּכָה עָלַי נַפְשִׁי כִּי אֶעֱבֹר בַּסָּךְ אֶדַּדֵּם עַד בֵּית אֱלֹהִים בְּקוֹל רִנָּה וְתוֹדָה הָמוֹן חוֹגֵג` (תהילים מב:ד)
`מַה תִּשְׁתּוֹחֲחִי נַפְשִׁי וַתֶּהֱמִי עָלָי הוֹחִילִי לֵאלֹהִים כִּי עוֹד אוֹדֶנּוּ יְשׁוּעוֹת פָּנָיו` (תהילים מב:ה)
`אֱ‍לֹהַי עָלַי נַפְשִׁי תִשְׁתּוֹחָח עַל כֵּן אֶזְכָּרְךָ מֵאֶרֶץ יַרְדֵּן וְחֶרְמוֹנִים מֵהַר מִצְעָר` (תהילים מב:ו)
`מַה תִּשְׁתּוֹחֲחִי נַפְשִׁי וּמַה תֶּהֱמִי עָלָי הוֹחִילִי לֵאלֹהִים כִּי עוֹד אוֹדֶנּוּ יְשׁוּעֹת פָּנַי וֵאלֹהָי` (תהילים מב:יא)
`מַה תִּשְׁתּוֹחֲחִי נַפְשִׁי וּמַה תֶּהֱמִי עָלָי הוֹחִילִי לֵאלֹהִים כִּי עוֹד אוֹדֶנּוּ יְשׁוּעֹת פָּנַי וֵאלֹהָי` (תהילים מג:ה)
`כִּי שָׁחָה לֶעָפָר נַפְשֵׁנוּ דָּבְקָה לָאָרֶץ בִּטְנֵנוּ` (תהילים מד:כה)
`וְיֵקַר פִּדְיוֹן נַפְשָׁם וְחָדַל לְעוֹלָם` (תהילים מט:ח)
`אַךְ אֱלֹהִים יִפְדֶּה נַפְשִׁי מִיַּד שְׁאוֹל כִּי יִקָּחֵנִי סֶלָה` (תהילים מט:טו)
`כִּי נַפְשׁוֹ בְּחַיָּיו יְבָרֵךְ וְיוֹדֻךָ כִּי תֵיטִיב לָךְ` (תהילים מט:יח)
`כִּי זָרִים קָמוּ עָלַי וְעָרִיצִים בִּקְשׁוּ נַפְשִׁי לֹא שָׂמוּ אֱלֹהִים לְנֶגְדָּם סֶלָה` (תהילים נד:ג)
`הִנֵּה אֱלֹהִים עֹזֵר לִי אֲדֹנָי בְּסֹמְכֵי נַפְשִׁי` (תהילים נד:ד)
`פָּדָה בְשָׁלוֹם נַפְשִׁי מִקֲּרָב לִי כִּי בְרַבִּים הָיוּ עִמָּדִי` (תהילים נה:יח)
`יָגוּרוּ יַצְפִּינוּ הֵמָּה עֲקֵבַי יִשְׁמֹרוּ כַּאֲשֶׁר קִוּוּ נַפְשִׁי` (תהילים נו:ו)
`כִּי הִצַּלְתָּ נַפְשִׁי מִמָּוֶת הֲלֹא רַגְלַי מִדֶּחִי לְהִתְהַלֵּךְ לִפְנֵי אֱלֹהִים בְּאוֹר הַחַיִּים` (תהילים נו:יג)
`לַמְנַצֵּחַ אַל תַּשְׁחֵת לְדָוִד מִכְתָּם בְּבָרְחוֹ מִפְּנֵי שָׁאוּל בַּמְּעָרָה חָנֵּנִי אֱלֹהִים חָנֵּנִי כִּי בְךָ חָסָיָה נַפְשִׁי וּבְצֵל כְּנָפֶיךָ אֶחְסֶה עַד יַעֲבֹר הַוּוֹת` (תהילים נז:א)
`נַפְשִׁי בְּתוֹךְ לְבָאִם אֶשְׁכְּבָה לֹהֲטִים בְּנֵי אָדָם שִׁנֵּיהֶם חֲנִית וְחִצִּים וּלְשׁוֹנָם חֶרֶב חַדָּה` (תהילים נז:ד)
`רֶשֶׁת הֵכִינוּ לִפְעָמַי כָּפַף נַפְשִׁי כָּרוּ לְפָנַי שִׁיחָה נָפְלוּ בְתוֹכָהּ סֶלָה` (תהילים נז:ו)
`כִּי הִנֵּה אָרְבוּ לְנַפְשִׁי יָגוּרוּ עָלַי עַזִים לֹא פִשְׁעִי וְלֹא חַטָּאתִי יְהוָה` (תהילים נט:ג)
`לַמְנַצֵּחַ עַל יְדוּתוּן מִזְמוֹר לְדָוִד אַךְ אֶל אֱלֹהִים דּוּמִיָּה נַפְשִׁי מִמֶּנּוּ יְשׁוּעָתִי` (תהילים סב:א)
`אַךְ לֵאלֹהִים דּוֹמִּי נַפְשִׁי כִּי מִמֶּנּוּ תִּקְוָתִי` (תהילים סב:ה)
`מִזְמוֹר לְדָוִד בִּהְיוֹתוֹ בְּמִדְבַּר יְהוּדָה אֱלֹהִים אֵלִי אַתָּה אֲ‍שַׁחֲרֶךָּ צָמְאָה לְךָ נַפְשִׁי כָּמַהּ לְךָ בְשָׂרִי בְּאֶרֶץ צִיָּה וְעָיֵף בְּלִי מָיִם` (תהילים סג:א)
`כְּמוֹ חֵלֶב וָדֶשֶׁן תִּשְׂבַּע נַפְשִׁי וְשִׂפְתֵי רְנָנוֹת יְהַלֶּל פִּי` (תהילים סג:ה)
`דָּבְקָה נַפְשִׁי אַחֲרֶיךָ בִּי תָּמְכָה יְמִינֶךָ` (תהילים סג:ח)
`וְהֵמָּה לְשׁוֹאָה יְבַקְשׁוּ נַפְשִׁי יָבֹאוּ בְּתַחְתִּיּוֹת הָאָרֶץ` (תהילים סג:ט)
`הַשָּׂם נַפְשֵׁנוּ בַּחַיִּים וְלֹא נָתַן לַמּוֹט רַגְלֵנוּ` (תהילים סו:ט)
`לְכוּ שִׁמְעוּ וַאֲסַפְּרָה כָּל יִרְאֵי אֱלֹהִים אֲשֶׁר עָשָׂה לְנַפְשִׁי` (תהילים סו:טז)
`לַמְנַצֵּחַ עַל שׁוֹשַׁנִּים לְדָוִד הוֹשִׁיעֵנִי אֱלֹהִים כִּי בָאוּ מַיִם עַד נָפֶשׁ` (תהילים סט:א)
`וָאֶבְכֶּה בַצּוֹם נַפְשִׁי וַתְּהִי לַחֲרָפוֹת לִי` (תהילים סט:י)
`קָרְבָה אֶל נַפְשִׁי גְאָלָהּ לְמַעַן אֹיְבַי פְּדֵנִי` (תהילים סט:יח)
`יֵבֹשׁוּ וְיַחְפְּרוּ מְבַקְשֵׁי נַפְשִׁי יִסֹּגוּ אָחוֹר וְיִכָּלְמוּ חֲפֵצֵי רָעָתִי` (תהילים ע:ב)
`כִּי אָמְרוּ אוֹיְבַי לִי וְשֹׁמְרֵי נַפְשִׁי נוֹעֲצוּ יַחְדָּו` (תהילים עא:י)
`יֵבֹשׁוּ יִכְלוּ שֹׂטְנֵי נַפְשִׁי יַעֲטוּ חֶרְפָּה וּכְלִמָּה מְבַקְשֵׁי רָעָתִי` (תהילים עא:יג)
`תְּרַנֵּנָּה שְׂפָתַי כִּי אֲזַמְּרָה לָּךְ וְנַפְשִׁי אֲשֶׁר פָּדִיתָ` (תהילים עא:כג)
`יָחֹס עַל דַּל וְאֶבְיוֹן וְנַפְשׁוֹת אֶבְיוֹנִים יוֹשִׁיעַ` (תהילים עב:יג)
`מִתּוֹךְ וּמֵחָמָס יִגְאַל נַפְשָׁם וְיֵיקַר דָּמָם בְּעֵינָיו` (תהילים עב:יד)
`אַל תִּתֵּן לְחַיַּת נֶפֶשׁ תּוֹרֶךָ חַיַּת עֲנִיֶּיךָ אַל תִּשְׁכַּח לָנֶצַח` (תהילים עד:יט)
`בְּיוֹם צָרָתִי אֲדֹנָי דָּרָשְׁתִּי יָדִי לַיְלָה נִגְּרָה וְלֹא תָפוּג מֵאֲנָה הִנָּחֵם נַפְשִׁי` (תהילים עז:ב)
`וַיְנַסּוּ אֵל בִּלְבָבָם לִשְׁאָל אֹכֶל לְנַפְשָׁם` (תהילים עח:יח)
`יְפַלֵּס נָתִיב לְאַפּוֹ לֹא חָשַׂךְ מִמָּוֶת נַפְשָׁם וְחַיָּתָם לַדֶּבֶר הִסְגִּיר` (תהילים עח:נ)
`נִכְסְפָה וְגַם כָּלְתָה נַפְשִׁי לְחַצְרוֹת יְהוָה לִבִּי וּבְשָׂרִי יְרַנְּנוּ אֶל אֵל חָי` (תהילים פד:ב)
`שָׁמְרָה נַפְשִׁי כִּי חָסִיד אָנִי הוֹשַׁע עַבְדְּךָ אַתָּה אֱלֹהַי הַבּוֹטֵחַ אֵלֶיךָ` (תהילים פו:ב)
`שַׂמֵּחַ נֶפֶשׁ עַבְדֶּךָ כִּי אֵלֶיךָ אֲדֹנָי נַפְשִׁי אֶשָּׂא` (תהילים פו:ד)
`כִּי חַסְדְּךָ גָּדוֹל עָלָי וְהִצַּלְתָּ נַפְשִׁי מִשְּׁאוֹל תַּחְתִּיָּה` (תהילים פו:יג)
`אֱלֹהִים זֵדִים קָמוּ עָלַי וַעֲדַת עָרִיצִים בִּקְשׁוּ נַפְשִׁי וְלֹא שָׂמוּךָ לְנֶגְדָּם` (תהילים פו:יד)
`כִּי שָׂבְעָה בְרָעוֹת נַפְשִׁי וְחַיַּי לִשְׁאוֹל הִגִּיעוּ` (תהילים פח:ג)
`לָמָה יְהוָה תִּזְנַח נַפְשִׁי תַּסְתִּיר פָּנֶיךָ מִמֶּנִּי` (תהילים פח:יד)
`מִי גֶבֶר יִחְיֶה וְלֹא יִרְאֶה מָּוֶת יְמַלֵּט נַפְשׁוֹ מִיַּד שְׁאוֹל סֶלָה` (תהילים פט:מח)
`לוּלֵי יְהוָה עֶזְרָתָה לִּי כִּמְעַט שָׁכְנָה דוּמָה נַפְשִׁי` (תהילים צד:יז)
`בְּרֹב שַׂרְעַפַּי בְּקִרְבִּי תַּנְחוּמֶיךָ יְשַׁעַשְׁעוּ נַפְשִׁי` (תהילים צד:יט)
`יָגוֹדּוּ עַל נֶפֶשׁ צַדִּיק וְדָם נָקִי יַרְשִׁיעוּ` (תהילים צד:כא)
`אֹהֲבֵי יְהוָה שִׂנְאוּ רָע שֹׁמֵר נַפְשׁוֹת חֲסִידָיו מִיַּד רְשָׁעִים יַצִּילֵם` (תהילים צז:י)
`לְדָוִד בָּרֲכִי נַפְשִׁי אֶת יְהוָה וְכָל קְרָבַי אֶת שֵׁם קָדְשׁוֹ` (תהילים קג:א)
`בָּרֲכִי נַפְשִׁי אֶת יְהוָה וְאַל תִּשְׁכְּחִי כָּל גְּמוּלָיו` (תהילים קג:ב)
`בָּרֲכוּ יְהוָה כָּל מַעֲשָׂיו בְּכָל מְקֹמוֹת מֶמְשַׁלְתּוֹ בָּרֲכִי נַפְשִׁי אֶת יְהוָה` (תהילים קג:כב)
`בָּרֲכִי נַפְשִׁי אֶת יְהוָה יְהוָה אֱלֹהַי גָּדַלְתָּ מְּאֹד הוֹד וְהָדָר לָבָשְׁתָּ` (תהילים קד:א)
`יִתַּמּוּ חַטָּאִים מִן הָאָרֶץ וּרְשָׁעִים עוֹד אֵינָם בָּרֲכִי נַפְשִׁי אֶת יְהוָה הַלְלוּ יָהּ` (תהילים קד:לה)
`עִנּוּ בַכֶּבֶל רַגְלָיו בַּרְזֶל בָּאָה נַפְשׁוֹ` (תהילים קה:יח)
`לֶאְסֹר שָׂרָיו בְּנַפְשׁוֹ וּזְקֵנָיו יְחַכֵּם` (תהילים קה:כב)
`וַיִּתֵּן לָהֶם שֶׁאֱלָתָם וַיְשַׁלַּח רָזוֹן בְּנַפְשָׁם` (תהילים קו:טו)
`רְעֵבִים גַּם צְמֵאִים נַפְשָׁם בָּהֶם תִּתְעַטָּף` (תהילים קז:ה)
`כִּי הִשְׂבִּיעַ נֶפֶשׁ שֹׁקֵקָה וְנֶפֶשׁ רְעֵבָה מִלֵּא טוֹב` (תהילים קז:ט)
`כָּל אֹכֶל תְּתַעֵב נַפְשָׁם וַיַּגִּיעוּ עַד שַׁעֲרֵי מָוֶת` (תהילים קז:יח)
`יַעֲלוּ שָׁמַיִם יֵרְדוּ תְהוֹמוֹת נַפְשָׁם בְּרָעָה תִתְמוֹגָג` (תהילים קז:כו)
`זֹאת פְּעֻלַּת שֹׂטְנַי מֵאֵת יְהוָה וְהַדֹּבְרִים רָע עַל נַפְשִׁי` (תהילים קט:כ)
`כִּי יַעֲמֹד לִימִין אֶבְיוֹן לְהוֹשִׁיעַ מִשֹּׁפְטֵי נַפְשׁוֹ` (תהילים קט:לא)
`וּבְשֵׁם יְהוָה אֶקְרָא אָנָּה יְהוָה מַלְּטָה נַפְשִׁי` (תהילים קיו:ד)
`שׁוּבִי נַפְשִׁי לִמְנוּחָיְכִי כִּי יְהוָה גָּמַל עָלָיְכִי` (תהילים קיו:ז)
`כִּי חִלַּצְתָּ נַפְשִׁי מִמָּוֶת אֶת עֵינִי מִן דִּמְעָה אֶת רַגְלִי מִדֶּחִי` (תהילים קיו:ח)
`גָּרְסָה נַפְשִׁי לְתַאֲבָה אֶל מִשְׁפָּטֶיךָ בְכָל עֵת` (תהילים קיט:כ)
`דָּבְקָה לֶעָפָר נַפְשִׁי חַיֵּנִי כִּדְבָרֶךָ` (תהילים קיט:כה)
`דָּלְפָה נַפְשִׁי מִתּוּגָה קַיְּמֵנִי כִּדְבָרֶךָ` (תהילים קיט:כח)
`כָּלְתָה לִתְשׁוּעָתְךָ נַפְשִׁי לִדְבָרְךָ יִחָלְתִּי` (תהילים קיט:פא)
`נַפְשִׁי בְכַפִּי תָמִיד וְתוֹרָתְךָ לֹא שָׁכָחְתִּי` (תהילים קיט:קט)
`פְּלָאוֹת עֵדְוֺתֶיךָ עַל כֵּן נְצָרָתַם נַפְשִׁי` (תהילים קיט:קכט)
`שָׁמְרָה נַפְשִׁי עֵדֹתֶיךָ וָאֹהֲבֵם מְאֹד` (תהילים קיט:קסז)
`תְּחִי נַפְשִׁי וּתְהַלְלֶךָּ וּמִשְׁפָּטֶךָ יַעֲזְרֻנִי` (תהילים קיט:קעה)
`יְהוָה הַצִּילָה נַפְשִׁי מִשְּׂפַת שֶׁקֶר מִלָּשׁוֹן רְמִיָּה` (תהילים קכ:ב)
`רַבַּת שָׁכְנָה לָּהּ נַפְשִׁי עִם שׂוֹנֵא שָׁלוֹם` (תהילים קכ:ו)
`יְהוָה יִשְׁמָרְךָ מִכָּל רָע יִשְׁמֹר אֶת נַפְשֶׁךָ` (תהילים קכא:ז)
`רַבַּת שָׂבְעָה לָּהּ נַפְשֵׁנוּ הַלַּעַג הַשַּׁאֲנַנִּים הַבּוּז לִגְאֵיוֹנִים` (תהילים קכג:ד)
`אֲזַי הַמַּיִם שְׁטָפוּנוּ נַחְלָה עָבַר עַל נַפְשֵׁנוּ` (תהילים קכד:ד)
`אֲזַי עָבַר עַל נַפְשֵׁנוּ הַמַּיִם הַזֵּידוֹנִים` (תהילים קכד:ה)
`נַפְשֵׁנוּ כְּצִפּוֹר נִמְלְטָה מִפַּח יוֹקְשִׁים הַפַּח נִשְׁבָּר וַאֲנַחְנוּ נִמְלָטְנוּ` (תהילים קכד:ז)
`קִוִּיתִי יְהוָה קִוְּתָה נַפְשִׁי וְלִדְבָרוֹ הוֹחָלְתִּי` (תהילים קל:ה)
`נַפְשִׁי לַאדֹנָי מִשֹּׁמְרִים לַבֹּקֶר שֹׁמְרִים לַבֹּקֶר` (תהילים קל:ו)
`אִם לֹא שִׁוִּיתִי וְדוֹמַמְתִּי נַפְשִׁי כְּגָמֻל עֲלֵי אִמּוֹ כַּגָּמֻל עָלַי נַפְשִׁי` (תהילים קלא:ב)
`בְּיוֹם קָרָאתִי וַתַּעֲנֵנִי תַּרְהִבֵנִי בְנַפְשִׁי עֹז` (תהילים קלח:ג)
`אוֹדְךָ עַל כִּי נוֹרָאוֹת נִפְלֵיתִי נִפְלָאִים מַעֲשֶׂיךָ וְנַפְשִׁי יֹדַעַת מְאֹד` (תהילים קלט:יד)
`כִּי אֵלֶיךָ יְהֹוִה אֲדֹנָי עֵינָי בְּכָה חָסִיתִי אַל תְּעַר נַפְשִׁי` (תהילים קמא:ח)
`הַבֵּיט יָמִין וּרְאֵה וְאֵין לִי מַכִּיר אָבַד מָנוֹס מִמֶּנִּי אֵין דּוֹרֵשׁ לְנַפְשִׁי` (תהילים קמב:ד)
`הוֹצִיאָה מִמַּסְגֵּר נַפְשִׁי לְהוֹדוֹת אֶת שְׁמֶךָ בִּי יַכְתִּרוּ צַדִּיקִים כִּי תִגְמֹל עָלָי` (תהילים קמב:ז)
`כִּי רָדַף אוֹיֵב נַפְשִׁי דִּכָּא לָאָרֶץ חַיָּתִי הוֹשִׁיבַנִי בְמַחֲשַׁכִּים כְּמֵתֵי עוֹלָם` (תהילים קמג:ג)
`פֵּרַשְׂתִּי יָדַי אֵלֶיךָ נַפְשִׁי כְּאֶרֶץ עֲיֵפָה לְךָ סֶלָה` (תהילים קמג:ו)
`הַשְׁמִיעֵנִי בַבֹּקֶר חַסְדֶּךָ כִּי בְךָ בָטָחְתִּי הוֹדִיעֵנִי דֶּרֶךְ זוּ אֵלֵךְ כִּי אֵלֶיךָ נָשָׂאתִי נַפְשִׁי` (תהילים קמג:ח)
`לְמַעַן שִׁמְךָ יְהוָה תְּחַיֵּנִי בְּצִדְקָתְךָ תוֹצִיא מִצָּרָה נַפְשִׁי` (תהילים קמג:יא)
`וּבְחַסְדְּךָ תַּצְמִית אֹיְבָי וְהַאֲבַדְתָּ כָּל צֹרֲרֵי נַפְשִׁי כִּי אֲנִי עַבְדֶּךָ` (תהילים קמג:יב)
`הַלְלוּ יָהּ הַלְלִי נַפְשִׁי אֶת יְהוָה` (תהילים קמו:א)
`וַיַּעַן הַשָּׂטָן אֶת יְהוָה וַיֹּאמַר עוֹר בְּעַד עוֹר וְכֹל אֲשֶׁר לָאִישׁ יִתֵּן בְּעַד נַפְשׁוֹ` (איוב ב:ד)
`וַיֹּאמֶר יְהוָה אֶל הַשָּׂטָן הִנּוֹ בְיָדֶךָ אַךְ אֶת נַפְשׁוֹ שְׁמֹר` (איוב ב:ו)
`לָמָּה יִתֵּן לְעָמֵל אוֹר וְחַיִּים לְמָרֵי נָפֶשׁ` (איוב ג:כ)
`מֵאֲנָה לִנְגּוֹעַ נַפְשִׁי הֵמָּה כִּדְוֵי לַחְמִי` (איוב ו:ז)
`מַה כֹּחִי כִי אֲיַחֵל וּמַה קִּצִּי כִּי אַאֲרִיךְ נַפְשִׁי` (איוב ו:יא)
`גַּם אֲנִי לֹא אֶחֱשָׂךְ פִּי אֲ‍דַבְּרָה בְּצַר רוּחִי אָשִׂיחָה בְּמַר נַפְשִׁי` (איוב ז:יא)
`וַתִּבְחַר מַחֲנָק נַפְשִׁי מָוֶת מֵעַצְמוֹתָי` (איוב ז:טו)
`תָּם אָנִי לֹא אֵדַע נַפְשִׁי אֶמְאַס חַיָּי` (איוב ט:כא)
`נָקְטָה נַפְשִׁי בְּחַיָּי אֶעֶזְבָה עָלַי שִׂיחִי אֲדַבְּרָה בְּמַר נַפְשִׁי` (איוב י:א)
`וְעֵינֵי רְשָׁעִים תִּכְלֶינָה וּמָנוֹס אָבַד מִנְהֶם וְתִקְוָתָם מַפַּח נָפֶשׁ` (איוב יא:כ)
`אֲשֶׁר בְּיָדוֹ נֶפֶשׁ כָּל חָי וְרוּחַ כָּל בְּשַׂר אִישׁ` (איוב יב:י)
`עַל מָה אֶשָּׂא בְשָׂרִי בְשִׁנָּי וְנַפְשִׁי אָשִׂים בְּכַפִּי` (איוב יג:יד)
`אַךְ בְּשָׂרוֹ עָלָיו יִכְאָב וְנַפְשׁוֹ עָלָיו תֶּאֱבָל` (איוב יד:כב)
`גַּם אָנֹכִי כָּכֶם אֲדַבֵּרָה לוּ יֵשׁ נַפְשְׁכֶם תַּחַת נַפְשִׁי אַחְבִּירָה עֲלֵיכֶם בְּמִלִּים וְאָנִיעָה עֲלֵיכֶם בְּמוֹ רֹאשִׁי` (איוב טז:ד)
`טֹרֵף נַפְשׁוֹ בְּאַפּוֹ הַלְמַעַנְךָ תֵּעָזַב אָרֶץ וְיֶעְתַּק צוּר מִמְּקֹמוֹ` (איוב יח:ד)
`עַד אָנָה תּוֹגְיוּן נַפְשִׁי וּתְדַכְּאוּנַנִי בְמִלִּים` (איוב יט:ב)
`וְזֶה יָמוּת בְּנֶפֶשׁ מָרָה וְלֹא אָכַל בַּטּוֹבָה` (איוב כא:כה)
`וְהוּא בְאֶחָד וּמִי יְשִׁיבֶנּוּ וְנַפְשׁוֹ אִוְּתָה וַיָּעַשׂ` (איוב כג:יג)
`מֵעִיר מְתִים יִנְאָקוּ וְנֶפֶשׁ חֲלָלִים תְּשַׁוֵּעַ וֶאֱלוֹהַּ לֹא יָשִׂים תִּפְלָה` (איוב כד:יב)
`חַי אֵל הֵסִיר מִשְׁפָּטִי וְשַׁדַּי הֵמַר נַפְשִׁי` (איוב כז:ב)
`כִּי מַה תִּקְוַת חָנֵף כִּי יִבְצָע כִּי יֵשֶׁל אֱלוֹהַּ נַפְשׁוֹ` (איוב כז:ח)
`וְעַתָּה עָלַי תִּשְׁתַּפֵּךְ נַפְשִׁי יֹאחֲזוּנִי יְמֵי עֹנִי` (איוב ל:טז)
`אִם לֹא בָכִיתִי לִקְשֵׁה יוֹם עָגְמָה נַפְשִׁי לָאֶבְיוֹן` (איוב ל:כה)
`וְלֹא נָתַתִּי לַחֲטֹא חִכִּי לִשְׁאֹל בְּאָלָה נַפְשׁוֹ` (איוב לא:ל)
`אִם כֹּחָהּ אָכַלְתִּי בְלִי כָסֶף וְנֶפֶשׁ בְּעָלֶיהָ הִפָּחְתִּי` (איוב לא:לט)
`וַיִּחַר אַף אֱלִיהוּא בֶן בַּרַכְאֵל הַבּוּזִי מִמִּשְׁפַּחַת רָם בְּאִיּוֹב חָרָה אַפּוֹ עַל צַדְּקוֹ נַפְשׁוֹ מֵאֱלֹהִים` (איוב לב:ב)
`יַחְשֹׂךְ נַפְשׁוֹ מִנִּי שָׁחַת וְחַיָּתוֹ מֵעֲבֹר בַּשָּׁלַח` (איוב לג:יח)
`וְזִהֲמַתּוּ חַיָּתוֹ לָחֶם וְנַפְשׁוֹ מַאֲכַל תַּאֲוָה` (איוב לג:כ)
`וַתִּקְרַב לַשַּׁחַת נַפְשׁוֹ וְחַיָּתוֹ לַמְמִתִים` (איוב לג:כב)
`פָּדָה נַפְשִׁי מֵעֲבֹר בַּשָּׁחַת וְחַיָּתִי בָּאוֹר תִּרְאֶה` (איוב לג:כח)
`לְהָשִׁיב נַפְשׁוֹ מִנִּי שָׁחַת לֵאוֹר בְּאוֹר הַחַיִּים` (איוב לג:ל)
`תָּמֹת בַּנֹּעַר נַפְשָׁם וְחַיָּתָם בַּקְּדֵשִׁים` (איוב לו:יד)
`נַפְשׁוֹ גֶּחָלִים תְּלַהֵט וְלַהַב מִפִּיו יֵצֵא` (איוב מא:כא)
`וְהֵם לְדָמָם יֶאֱרֹבוּ יִצְפְּנוּ לְנַפְשֹׁתָם` (משלי א:יח)
`כֵּן אָרְחוֹת כָּל בֹּצֵעַ בָּצַע אֶת נֶפֶשׁ בְּעָלָיו יִקָּח` (משלי א:יט)
`כִּי תָבוֹא חָכְמָה בְלִבֶּךָ וְדַעַת לְנַפְשְׁךָ יִנְעָם` (משלי ב:י)
`וְיִהְיוּ חַיִּים לְנַפְשֶׁךָ וְחֵן לְגַרְגְּרֹתֶיךָ` (משלי ג:כב)
`שֶׁשׁ הֵנָּה שָׂנֵא יְהוָה וְשֶׁבַע תּוֹעֲבוֹת נַפְשׁוֹ` (משלי ו:טז)
`כִּי בְעַד אִשָּׁה זוֹנָה עַד כִּכַּר לָחֶם וְאֵשֶׁת אִישׁ נֶפֶשׁ יְקָרָה תָצוּד` (משלי ו:כו)
`לֹא יָבוּזוּ לַגַּנָּב כִּי יִגְנוֹב לְמַלֵּא נַפְשׁוֹ כִּי יִרְעָב` (משלי ו:ל)
`נֹאֵף אִשָּׁה חֲסַר לֵב מַשְׁחִית נַפְשׁוֹ הוּא יַעֲשֶׂנָּה` (משלי ו:לב)
`עַד יְפַלַּח חֵץ כְּבֵדוֹ כְּמַהֵר צִפּוֹר אֶל פָּח וְלֹא יָדַע כִּי בְנַפְשׁוֹ הוּא` (משלי ז:כג)
`וְחֹטְאִי חֹמֵס נַפְשׁוֹ כָּל מְשַׂנְאַי אָהֲבוּ מָוֶת` (משלי ח:לו)
`לֹא יַרְעִיב יְהוָה נֶפֶשׁ צַדִּיק וְהַוַּת רְשָׁעִים יֶהְדֹּף` (משלי י:ג)
`גֹּמֵל נַפְשׁוֹ אִישׁ חָסֶד וְעֹכֵר שְׁאֵרוֹ אַכְזָרִי` (משלי יא:יז)
`נֶפֶשׁ בְּרָכָה תְדֻשָּׁן וּמַרְוֶה גַּם הוּא יוֹרֶא` (משלי יא:כה)
`פְּרִי צַדִּיק עֵץ חַיִּים וְלֹקֵחַ נְפָשׂוֹת חָכָם` (משלי יא:ל)
`יוֹדֵעַ צַדִּיק נֶפֶשׁ בְּהֶמְתּוֹ וְרַחֲמֵי רְשָׁעִים אַכְזָרִי` (משלי יב:י)
`מִפְּרִי פִי אִישׁ יֹאכַל טוֹב וְנֶפֶשׁ בֹּגְדִים חָמָס` (משלי יג:ב)
`נֹצֵר פִּיו שֹׁמֵר נַפְשׁוֹ פֹּשֵׂק שְׂפָתָיו מְחִתָּה לוֹ` (משלי יג:ג)
`מִתְאַוָּה וָאַיִן נַפְשׁוֹ עָצֵל וְנֶפֶשׁ חָרֻצִים תְּדֻשָּׁן` (משלי יג:ד)
`כֹּפֶר נֶפֶשׁ אִישׁ עָשְׁרוֹ וְרָשׁ לֹא שָׁמַע גְּעָרָה` (משלי יג:ח)
`תַּאֲוָה נִהְיָה תֶּעֱרַב לְנָפֶשׁ וְתוֹעֲבַת כְּסִילִים סוּר מֵרָע` (משלי יג:יט)
`צַדִּיק אֹכֵל לְשֹׂבַע נַפְשׁוֹ וּבֶטֶן רְשָׁעִים תֶּחְסָר` (משלי יג:כה)
`לֵב יוֹדֵעַ מָרַּת נַפְשׁוֹ וּבְשִׂמְחָתוֹ לֹא יִתְעָרַב זָר` (משלי יד:י)
`מַצִּיל נְפָשׁוֹת עֵד אֱמֶת וְיָפִחַ כְּזָבִים מִרְמָה` (משלי יד:כה)
`פּוֹרֵעַ מוּסָר מוֹאֵס נַפְשׁוֹ וְשׁוֹמֵעַ תּוֹכַחַת קוֹנֶה לֵּב` (משלי טו:לב)
`מְסִלַּת יְשָׁרִים סוּר מֵרָע שֹׁמֵר נַפְשׁוֹ נֹצֵר דַּרְכּוֹ` (משלי טז:יז)
`צוּף דְּבַשׁ אִמְרֵי נֹעַם מָתוֹק לַנֶּפֶשׁ וּמַרְפֵּא לָעָצֶם` (משלי טז:כד)
`נֶפֶשׁ עָמֵל עָמְלָה לּוֹ כִּי אָכַף עָלָיו פִּיהוּ` (משלי טז:כו)
`פִּי כְסִיל מְחִתָּה לוֹ וּשְׂפָתָיו מוֹקֵשׁ נַפְשׁוֹ` (משלי יח:ז)
`גַּם בְּלֹא דַעַת נֶפֶשׁ לֹא טוֹב וְאָץ בְּרַגְלַיִם חוֹטֵא` (משלי יט:ב)
`קֹנֶה לֵּב אֹהֵב נַפְשׁוֹ שֹׁמֵר תְּבוּנָה לִמְצֹא טוֹב` (משלי יט:ח)
`עַצְלָה תַּפִּיל תַּרְדֵּמָה וְנֶפֶשׁ רְמִיָּה תִרְעָב` (משלי יט:טו)
`שֹׁמֵר מִצְוָה שֹׁמֵר נַפְשׁוֹ בּוֹזֵה דְרָכָיו יוּמָת` (משלי יט:טז)
`יַסֵּר בִּנְךָ כִּי יֵשׁ תִּקְוָה וְאֶל הֲמִיתוֹ אַל תִּשָּׂא נַפְשֶׁךָ` (משלי יט:יח)
`נַהַם כַּכְּפִיר אֵימַת מֶלֶךְ מִתְעַבְּרוֹ חוֹטֵא נַפְשׁוֹ` (משלי כ:ב)
`נֶפֶשׁ רָשָׁע אִוְּתָה רָע לֹא יֻחַן בְּעֵינָיו רֵעֵהוּ` (משלי כא:י)
`שֹׁמֵר פִּיו וּלְשׁוֹנוֹ שֹׁמֵר מִצָּרוֹת נַפְשׁוֹ` (משלי כא:כג)
`צִנִּים פַּחִים בְּדֶרֶךְ עִקֵּשׁ שׁוֹמֵר נַפְשׁוֹ יִרְחַק מֵהֶם` (משלי כב:ה)
`כִּי יְהוָה יָרִיב רִיבָם וְקָבַע אֶת קֹבְעֵיהֶם נָפֶשׁ` (משלי כב:כג)
`פֶּן תֶּאֱלַף אָרְחָתוֹ וְלָקַחְתָּ מוֹקֵשׁ לְנַפְשֶׁךָ` (משלי כב:כה)
`וְשַׂמְתָּ שַׂכִּין בְּלֹעֶךָ אִם בַּעַל נֶפֶשׁ אָתָּה` (משלי כג:ב)
`כִּי כְּמוֹ שָׁעַר בְּנַפְשׁוֹ כֶּן הוּא אֱכֹל וּשְׁתֵה יֹאמַר לָךְ וְלִבּוֹ בַּל עִמָּךְ` (משלי כג:ז)
`אַתָּה בַּשֵּׁבֶט תַּכֶּנּוּ וְנַפְשׁוֹ מִשְּׁאוֹל תַּצִּיל` (משלי כג:יד)
`כִּי תֹאמַר הֵן לֹא יָדַעְנוּ זֶה הֲ‍לֹא תֹכֵן לִבּוֹת הוּא יָבִין וְנֹצֵר נַפְשְׁךָ הוּא יֵדָע וְהֵשִׁיב לְאָדָם כְּפָעֳלוֹ` (משלי כד:יב)
`כֵּן דְּעֶה חָכְמָה לְנַפְשֶׁךָ אִם מָצָאתָ וְיֵשׁ אַחֲרִית וְתִקְוָתְךָ לֹא תִכָּרֵת` (משלי כד:יד)
`כְּצִנַּת שֶׁלֶג בְּיוֹם קָצִיר צִיר נֶאֱמָן לְשֹׁלְחָיו וְנֶפֶשׁ אֲדֹנָיו יָשִׁיב` (משלי כה:יג)
`מַיִם קָרִים עַל נֶפֶשׁ עֲיֵפָה וּשְׁמוּעָה טוֹבָה מֵאֶרֶץ מֶרְחָק` (משלי כה:כה)
`נֶפֶשׁ שְׂבֵעָה תָּבוּס נֹפֶת וְנֶפֶשׁ רְעֵבָה כָּל מַר מָתוֹק` (משלי כז:ז)
`שֶׁמֶן וּקְטֹרֶת יְשַׂמַּח לֵב וּמֶתֶק רֵעֵהוּ מֵעֲצַת נָפֶשׁ` (משלי כז:ט)
`אָדָם עָשֻׁק בְּדַם נָפֶשׁ עַד בּוֹר יָנוּס אַל יִתְמְכוּ בוֹ` (משלי כח:יז)
`רְחַב נֶפֶשׁ יְגָרֶה מָדוֹן וּבוֹטֵחַ עַל יְהוָה יְדֻשָּׁן` (משלי כח:כה)
`אַנְשֵׁי דָמִים יִשְׂנְאוּ תָם וִישָׁרִים יְבַקְשׁוּ נַפְשׁוֹ` (משלי כט:י)
`יַסֵּר בִּנְךָ וִינִיחֶךָ וְיִתֵּן מַעֲדַנִּים לְנַפְשֶׁךָ` (משלי כט:יז)
`חוֹלֵק עִם גַּנָּב שׂוֹנֵא נַפְשׁוֹ אָלָה יִשְׁמַע וְלֹא יַגִּיד` (משלי כט:כד)
`תְּנוּ שֵׁכָר לְאוֹבֵד וְיַיִן לְמָרֵי נָפֶשׁ` (משלי לא:ו)
`וְהָיָה לָךְ לְמֵשִׁיב נֶפֶשׁ וּלְכַלְכֵּל אֶת שֵׂיבָתֵךְ כִּי כַלָּתֵךְ אֲ‍שֶׁר אֲהֵבַתֶךְ יְלָדַתּוּ אֲשֶׁר הִיא טוֹבָה לָךְ מִשִּׁבְעָה בָּנִים` (רות ד:טו)
`הַגִּידָה לִּי שֶׁאָהֲבָה נַפְשִׁי אֵיכָה תִרְעֶה אֵיכָה תַּרְבִּיץ בַּצָּהֳרָיִם שַׁלָּמָה אֶהְיֶה כְּעֹטְיָה עַל עֶדְרֵי חֲבֵרֶיךָ` (שיר-השירים א:ז)
`עַל מִשְׁכָּבִי בַּלֵּילוֹת בִּקַּשְׁתִּי אֵת שֶׁאָהֲבָה נַפְשִׁי בִּקַּשְׁתִּיו וְלֹא מְצָאתִיו` (שיר-השירים ג:א)
`אָקוּמָה נָּא וַאֲסוֹבְבָה בָעִיר בַּשְּׁוָקִים וּבָרְחֹבוֹת אֲבַקְשָׁה אֵת שֶׁאָהֲבָה נַפְשִׁי בִּקַּשְׁתִּיו וְלֹא מְצָאתִיו` (שיר-השירים ג:ב)
`מְצָאוּנִי הַשֹּׁמְרִים הַסֹּבְבִים בָּעִיר אֵת שֶׁאָהֲבָה נַפְשִׁי רְאִיתֶם` (שיר-השירים ג:ג)
`כִּמְעַט שֶׁעָבַרְתִּי מֵהֶם עַד שֶׁמָּצָאתִי אֵת שֶׁאָהֲבָה נַפְשִׁי אֲחַזְתִּיו וְלֹא אַרְפֶּנּוּ עַד שֶׁהֲבֵיאתִיו אֶל בֵּית אִמִּי וְאֶל חֶדֶר הוֹרָתִי` (שיר-השירים ג:ד)
`פָּתַחְתִּי אֲנִי לְדוֹדִי וְדוֹדִי חָמַק עָבָר נַפְשִׁי יָצְאָה בְדַבְּרוֹ בִּקַּשְׁתִּיהוּ וְלֹא מְצָאתִיהוּ קְרָאתִיו וְלֹא עָנָנִי` (שיר-השירים ה:ו)
`לֹא יָדַעְתִּי נַפְשִׁי שָׂמַתְנִי מַרְכְּבוֹת עַמִּי נָדִיב` (שיר-השירים ו:יב)
`אֵין טוֹב בָּאָדָם שֶׁיֹּאכַל וְשָׁתָה וְהֶרְאָה אֶת נַפְשׁוֹ טוֹב בַּעֲמָלוֹ גַּם זֹה רָאִיתִי אָנִי כִּי מִיַּד הָאֱלֹהִים הִיא` (קהלת ב:כד)
`יֵשׁ אֶחָד וְאֵין שֵׁנִי גַּם בֵּן וָאָח אֵין לוֹ וְאֵין קֵץ לְכָל עֲמָלוֹ גַּם עֵינָיו לֹא תִשְׂבַּע עֹשֶׁר וּלְמִי אֲנִי עָמֵל וּמְחַסֵּר אֶת נַפְשִׁי מִטּוֹבָה גַּם זֶה הֶבֶל וְעִנְיַן רָע הוּא` (קהלת ד:ח)
`אִישׁ אֲשֶׁר יִתֶּן לוֹ הָאֱלֹהִים עֹשֶׁר וּנְכָסִים וְכָבוֹד וְאֵינֶנּוּ חָסֵר לְנַפְשׁוֹ מִכֹּל אֲשֶׁר יִתְאַוֶּה וְלֹא יַשְׁלִיטֶנּוּ הָאֱלֹהִים לֶאֱכֹל מִמֶּנּוּ כִּי אִישׁ נָכְרִי יֹאכֲלֶנּוּ זֶה הֶבֶל וָחֳלִי רָע הוּא` (קהלת ו:ב)
`אִם יוֹלִיד אִישׁ מֵאָה וְשָׁנִים רַבּוֹת יִחְיֶה וְרַב שֶׁיִּהְיוּ יְמֵי שָׁנָיו וְנַפְשׁוֹ לֹא תִשְׂבַּע מִן הַטּוֹבָה וְגַם קְבוּרָה לֹא הָיְתָה לּוֹ אָמַרְתִּי טוֹב מִמֶּנּוּ הַנָּפֶל` (קהלת ו:ג)
`כָּל עֲמַל הָאָדָם לְפִיהוּ וְגַם הַנֶּפֶשׁ לֹא תִמָּלֵא` (קהלת ו:ז)
`טוֹב מַרְאֵה עֵינַיִם מֵהֲלָךְ נָפֶשׁ גַּם זֶה הֶבֶל וּרְעוּת רוּחַ` (קהלת ו:ט)
`אֲשֶׁר עוֹד בִּקְשָׁה נַפְשִׁי וְלֹא מָצָאתִי אָדָם אֶחָד מֵאֶלֶף מָצָאתִי וְאִשָּׁה בְכָל אֵלֶּה לֹא מָצָאתִי` (קהלת ז:כח)
`כָּל עַמָּהּ נֶאֱנָחִים מְבַקְּשִׁים לֶחֶם נָתְנוּ מַחֲמוֹדֵּיהֶם בְּאֹכֶל לְהָשִׁיב נָפֶשׁ רְאֵה יְהוָה וְהַבִּיטָה כִּי הָיִיתִי זוֹלֵלָה` (איכה א:יא)
`עַל אֵלֶּה אֲנִי בוֹכִיָּה עֵינִי עֵינִי יֹרְדָה מַּיִם כִּי רָחַק מִמֶּנִּי מְנַחֵם מֵשִׁיב נַפְשִׁי הָיוּ בָנַי שׁוֹמֵמִים כִּי גָבַר אוֹיֵב` (איכה א:טז)
`קָרָאתִי לַמְאַהֲבַי הֵמָּה רִמּוּנִי כֹּהֲנַי וּזְקֵנַי בָּעִיר גָּוָעוּ כִּי בִקְשׁוּ אֹכֶל לָמוֹ וְיָשִׁיבוּ אֶת נַפְשָׁם` (איכה א:יט)
`לְאִמֹּתָם יֹאמְרוּ אַיֵּה דָּגָן וָיָיִן בְּהִתְעַטְּפָם כֶּחָלָל בִּרְחֹבוֹת עִיר בְּהִשְׁתַּפֵּךְ נַפְשָׁם אֶל חֵיק אִמֹּתָם` (איכה ב:יב)
`קוּמִי רֹנִּי בַלַּיִל לְרֹאשׁ אַשְׁמֻרוֹת שִׁפְכִי כַמַּיִם לִבֵּךְ נֹכַח פְּנֵי אֲדֹנָי שְׂאִי אֵלָיו כַּפַּיִךְ עַל נֶפֶשׁ עוֹלָלַיִךְ הָעֲטוּפִים בְּרָעָב בְּרֹאשׁ כָּל חוּצוֹת` (איכה ב:יט)
`וַתִּזְנַח מִשָּׁלוֹם נַפְשִׁי נָשִׁיתִי טוֹבָה` (איכה ג:יז)
`זָכוֹר תִּזְכּוֹר וְתָשִׁיחַ עָלַי נַפְשִׁי` (איכה ג:כ)
`חֶלְקִי יְהוָה אָמְרָה נַפְשִׁי עַל כֵּן אוֹחִיל לוֹ` (איכה ג:כד)
`טוֹב יְהוָה לְקוָֹו לְנֶפֶשׁ תִּדְרְשֶׁנּוּ` (איכה ג:כה)
`עֵינִי עוֹלְלָה לְנַפְשִׁי מִכֹּל בְּנוֹת עִירִי` (איכה ג:נא)
`רַבְתָּ אֲדֹנָי רִיבֵי נַפְשִׁי גָּאַלְתָּ חַיָּי` (איכה ג:נח)
`בְּנַפְשֵׁנוּ נָבִיא לַחְמֵנוּ מִפְּנֵי חֶרֶב הַמִּדְבָּר` (איכה ה:ט)
`וַיֹּאמֶר מָרְדֳּכַי לְהָשִׁיב אֶל אֶסְתֵּר אַל תְּדַמִּי בְנַפְשֵׁךְ לְהִמָּלֵט בֵּית הַמֶּלֶךְ מִכָּל הַיְּהוּדִים` (אסתר ד:יג)
`וַתַּעַן אֶסְתֵּר הַמַּלְכָּה וַתֹּאמַר אִם מָצָאתִי חֵן בְּעֵינֶיךָ הַמֶּלֶךְ וְאִם עַל הַמֶּלֶךְ טוֹב תִּנָּתֶן לִי נַפְשִׁי בִּשְׁאֵלָתִי וְעַמִּי בְּבַקָּשָׁתִי` (אסתר ז:ג)
`וְהַמֶּלֶךְ קָם בַּחֲמָתוֹ מִמִּשְׁתֵּה הַיַּיִן אֶל גִּנַּת הַבִּיתָן וְהָמָן עָמַד לְבַקֵּשׁ עַל נַפְשׁוֹ מֵאֶסְתֵּר הַמַּלְכָּה כִּי רָאָה כִּי כָלְתָה אֵלָיו הָרָעָה מֵאֵת הַמֶּלֶךְ` (אסתר ז:ז)
`אֲשֶׁר נָתַן הַמֶּלֶךְ לַיְּהוּדִים אֲשֶׁר בְּכָל עִיר וָעִיר לְהִקָּהֵל וְלַעֲמֹד עַל נַפְשָׁם לְהַשְׁמִיד וְלַהֲרֹג וּלְאַבֵּד אֶת כָּל חֵיל עַם וּמְדִינָה הַצָּרִים אֹתָם טַף וְנָשִׁים וּשְׁלָלָם לָבוֹז` (אסתר ח:יא)
`וּשְׁאָר הַיְּהוּדִים אֲשֶׁר בִּמְדִינוֹת הַמֶּלֶךְ נִקְהֲלוּ וְעָמֹד עַל נַפְשָׁם וְנוֹחַ מֵאֹיְבֵיהֶם וְהָרֹג בְּשֹׂנְאֵיהֶם חֲמִשָּׁה וְשִׁבְעִים אָלֶף וּבַבִּזָּה לֹא שָׁלְחוּ אֶת יָדָם` (אסתר ט:טז)
`לְקַיֵּם אֵת יְמֵי הַפֻּרִים הָאֵלֶּה בִּזְמַנֵּיהֶם כַּאֲשֶׁר קִיַּם עֲלֵיהֶם מָרְדֳּכַי הַיְּהוּדִי וְאֶסְתֵּר הַמַּלְכָּה וְכַאֲשֶׁר קִיְּמוּ עַל נַפְשָׁם וְעַל זַרְעָם דִּבְרֵי הַצֹּמוֹת וְזַעֲקָתָם` (אסתר ט:לא)
