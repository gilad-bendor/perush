`וַיַּפֵּל יְהוָה אֱלֹהִים תַּרְדֵּמָה עַל הָאָדָם וַיִּישָׁן וַיִּקַּח אַחַת מִצַּלְעֹתָיו וַיִּסְגֹּר בָּשָׂר תַּחְתֶּנָּה` (בראשית ב:כא)
`וַיֹּאמֶר הָאָדָם זֹאת הַפַּעַם עֶצֶם מֵעֲצָמַי וּבָשָׂר מִבְּשָׂרִי לְזֹאת יִקָּרֵא אִשָּׁה כִּי מֵאִישׁ לֻקֳחָה זֹּאת` (בראשית ב:כג)
`עַל כֵּן יַעֲזָב אִישׁ אֶת אָבִיו וְאֶת אִמּוֹ וְדָבַק בְּאִשְׁתּוֹ וְהָיוּ לְבָשָׂר אֶחָד` (בראשית ב:כד)
`וַיֹּאמֶר יְהוָה לֹא יָדוֹן רוּחִי בָאָדָם לְעֹלָם בְּשַׁגַּם הוּא בָשָׂר וְהָיוּ יָמָיו מֵאָה וְעֶשְׂרִים שָׁנָה` (בראשית ו:ג)
`וַיַּרְא אֱלֹהִים אֶת הָאָרֶץ וְהִנֵּה נִשְׁחָתָה כִּי הִשְׁחִית כָּל בָּשָׂר אֶת דַּרְכּוֹ עַל הָאָרֶץ` (בראשית ו:יב)
`וַיֹּאמֶר אֱלֹהִים לְנֹחַ קֵץ כָּל בָּשָׂר בָּא לְפָנַי כִּי מָלְאָה הָאָרֶץ חָמָס מִפְּנֵיהֶם וְהִנְנִי מַשְׁחִיתָם אֶת הָאָרֶץ` (בראשית ו:יג)
`וַאֲנִי הִנְנִי מֵבִיא אֶת הַמַּבּוּל מַיִם עַל הָאָרֶץ לְשַׁחֵת כָּל בָּשָׂר אֲשֶׁר בּוֹ רוּחַ חַיִּים מִתַּחַת הַשָּׁמָיִם כֹּל אֲשֶׁר בָּאָרֶץ יִגְוָע` (בראשית ו:יז)
`וּמִכָּל הָחַי מִכָּל בָּשָׂר שְׁנַיִם מִכֹּל תָּבִיא אֶל הַתֵּבָה לְהַחֲיֹת אִתָּךְ זָכָר וּנְקֵבָה יִהְיוּ` (בראשית ו:יט)
`וַיָּבֹאוּ אֶל נֹחַ אֶל הַתֵּבָה שְׁנַיִם שְׁנַיִם מִכָּל הַבָּשָׂר אֲשֶׁר בּוֹ רוּחַ חַיִּים` (בראשית ז:טו)
`וְהַבָּאִים זָכָר וּנְקֵבָה מִכָּל בָּשָׂר בָּאוּ כַּאֲשֶׁר צִוָּה אֹתוֹ אֱלֹהִים וַיִּסְגֹּר יְהוָה בַּעֲדוֹ` (בראשית ז:טז)
`וַיִּגְוַע כָּל בָּשָׂר הָרֹמֵשׂ עַל הָאָרֶץ בָּעוֹף וּבַבְּהֵמָה וּבַחַיָּה וּבְכָל הַשֶּׁרֶץ הַשֹּׁרֵץ עַל הָאָרֶץ וְכֹל הָאָדָם` (בראשית ז:כא)
`כָּל הַחַיָּה אֲשֶׁר אִתְּךָ מִכָּל בָּשָׂר בָּעוֹף וּבַבְּהֵמָה וּבְכָל הָרֶמֶשׂ הָרֹמֵשׂ עַל הָאָרֶץ הוֹצֵא אִתָּךְ וְשָׁרְצוּ בָאָרֶץ וּפָרוּ וְרָבוּ עַל הָאָרֶץ` (בראשית ח:יז)
`אַךְ בָּשָׂר בְּנַפְשׁוֹ דָמוֹ לֹא תֹאכֵלוּ` (בראשית ט:ד)
`וַהֲקִמֹתִי אֶת בְּרִיתִי אִתְּכֶם וְלֹא יִכָּרֵת כָּל בָּשָׂר עוֹד מִמֵּי הַמַּבּוּל וְלֹא יִהְיֶה עוֹד מַבּוּל לְשַׁחֵת הָאָרֶץ` (בראשית ט:יא)
`וְזָכַרְתִּי אֶת בְּרִיתִי אֲשֶׁר בֵּינִי וּבֵינֵיכֶם וּבֵין כָּל נֶפֶשׁ חַיָּה בְּכָל בָּשָׂר וְלֹא יִהְיֶה עוֹד הַמַּיִם לְמַבּוּל לְשַׁחֵת כָּל בָּשָׂר` (בראשית ט:טו)
`וְהָיְתָה הַקֶּשֶׁת בֶּעָנָן וּרְאִיתִיהָ לִזְכֹּר בְּרִית עוֹלָם בֵּין אֱלֹהִים וּבֵין כָּל נֶפֶשׁ חַיָּה בְּכָל בָּשָׂר אֲשֶׁר עַל הָאָרֶץ` (בראשית ט:טז)
`וַיֹּאמֶר אֱלֹהִים אֶל נֹחַ זֹאת אוֹת הַבְּרִית אֲשֶׁר הֲקִמֹתִי בֵּינִי וּבֵין כָּל בָּשָׂר אֲשֶׁר עַל הָאָרֶץ` (בראשית ט:יז)
`וּנְמַלְתֶּם אֵת בְּשַׂר עָרְלַתְכֶם וְהָיָה לְאוֹת בְּרִית בֵּינִי וּבֵינֵיכֶם` (בראשית יז:יא)
`הִמּוֹל יִמּוֹל יְלִיד בֵּיתְךָ וּמִקְנַת כַּסְפֶּךָ וְהָיְתָה בְרִיתִי בִּבְשַׂרְכֶם לִבְרִית עוֹלָם` (בראשית יז:יג)
`וְעָרֵל זָכָר אֲשֶׁר לֹא יִמּוֹל אֶת בְּשַׂר עָרְלָתוֹ וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ אֶת בְּרִיתִי הֵפַר` (בראשית יז:יד)
`וַיִּקַּח אַבְרָהָם אֶת יִשְׁמָעֵאל בְּנוֹ וְאֵת כָּל יְלִידֵי בֵיתוֹ וְאֵת כָּל מִקְנַת כַּסְפּוֹ כָּל זָכָר בְּאַנְשֵׁי בֵּית אַבְרָהָם וַיָּמָל אֶת בְּשַׂר עָרְלָתָם בְּעֶצֶם הַיּוֹם הַזֶּה כַּאֲשֶׁר דִּבֶּר אִתּוֹ אֱלֹהִים` (בראשית יז:כג)
`וְאַבְרָהָם בֶּן תִּשְׁעִים וָתֵשַׁע שָׁנָה בְּהִמֹּלוֹ בְּשַׂר עָרְלָתוֹ` (בראשית יז:כד)
`וְיִשְׁמָעֵאל בְּנוֹ בֶּן שְׁלֹשׁ עֶשְׂרֵה שָׁנָה בְּהִמֹּלוֹ אֵת בְּשַׂר עָרְלָתוֹ` (בראשית יז:כה)
`וַיֹּאמֶר לוֹ לָבָן אַךְ עַצְמִי וּבְשָׂרִי אָתָּה וַיֵּשֶׁב עִמּוֹ חֹדֶשׁ יָמִים` (בראשית כט:יד)
`לְכוּ וְנִמְכְּרֶנּוּ לַיִּשְׁמְעֵאלִים וְיָדֵנוּ אַל תְּהִי בוֹ כִּי אָחִינוּ בְשָׂרֵנוּ הוּא וַיִּשְׁמְעוּ אֶחָיו` (בראשית לז:כז)
`בְּעוֹד שְׁלֹשֶׁת יָמִים יִשָּׂא פַרְעֹה אֶת רֹאשְׁךָ מֵעָלֶיךָ וְתָלָה אוֹתְךָ עַל עֵץ וְאָכַל הָעוֹף אֶת בְּשָׂרְךָ מֵעָלֶיךָ` (בראשית מ:יט)
`וְהִנֵּה מִן הַיְאֹר עֹלֹת שֶׁבַע פָּרוֹת יְפוֹת מַרְאֶה וּבְרִיאֹת בָּשָׂר וַתִּרְעֶינָה בָּאָחוּ` (בראשית מא:ב)
`וְהִנֵּה שֶׁבַע פָּרוֹת אֲחֵרוֹת עֹלוֹת אַחֲרֵיהֶן מִן הַיְאֹר רָעוֹת מַרְאֶה וְדַקּוֹת בָּשָׂר וַתַּעֲמֹדְנָה אֵצֶל הַפָּרוֹת עַל שְׂפַת הַיְאֹר` (בראשית מא:ג)
`וַתֹּאכַלְנָה הַפָּרוֹת רָעוֹת הַמַּרְאֶה וְדַקֹּת הַבָּשָׂר אֵת שֶׁבַע הַפָּרוֹת יְפֹת הַמַּרְאֶה וְהַבְּרִיאֹת וַיִּיקַץ פַּרְעֹה` (בראשית מא:ד)
`וְהִנֵּה מִן הַיְאֹר עֹלֹת שֶׁבַע פָּרוֹת בְּרִיאוֹת בָּשָׂר וִיפֹת תֹּאַר וַתִּרְעֶינָה בָּאָחוּ` (בראשית מא:יח)
`וְהִנֵּה שֶׁבַע פָּרוֹת אֲחֵרוֹת עֹלוֹת אַחֲרֵיהֶן דַּלּוֹת וְרָעוֹת תֹּאַר מְאֹד וְרַקּוֹת בָּשָׂר לֹא רָאִיתִי כָהֵנָּה בְּכָל אֶרֶץ מִצְרַיִם לָרֹעַ` (בראשית מא:יט)
`וַיֹּאמֶר הָשֵׁב יָדְךָ אֶל חֵיקֶךָ וַיָּשֶׁב יָדוֹ אֶל חֵיקוֹ וַיּוֹצִאָהּ מֵחֵיקוֹ וְהִנֵּה שָׁבָה כִּבְשָׂרוֹ` (שמות ד:ז)
`וְאָכְלוּ אֶת הַבָּשָׂר בַּלַּיְלָה הַזֶּה צְלִי אֵשׁ וּמַצּוֹת עַל מְרֹרִים יֹאכְלֻהוּ` (שמות יב:ח)
`בְּבַיִת אֶחָד יֵאָכֵל לֹא תוֹצִיא מִן הַבַּיִת מִן הַבָּשָׂר חוּצָה וְעֶצֶם לֹא תִשְׁבְּרוּ בוֹ` (שמות יב:מו)
`וַיֹּאמְרוּ אֲלֵהֶם בְּנֵי יִשְׂרָאֵל מִי יִתֵּן מוּתֵנוּ בְיַד יְהוָה בְּאֶרֶץ מִצְרַיִם בְּשִׁבְתֵּנוּ עַל סִיר הַבָּשָׂר בְּאָכְלֵנוּ לֶחֶם לָשֹׂבַע כִּי הוֹצֵאתֶם אֹתָנוּ אֶל הַמִּדְבָּר הַזֶּה לְהָמִית אֶת כָּל הַקָּהָל הַזֶּה בָּרָעָב` (שמות טז:ג)
`וַיֹּאמֶר מֹשֶׁה בְּתֵת יְהוָה לָכֶם בָּעֶרֶב בָּשָׂר לֶאֱכֹל וְלֶחֶם בַּבֹּקֶר לִשְׂבֹּעַ בִּשְׁמֹעַ יְהוָה אֶת תְּלֻנֹּתֵיכֶם אֲשֶׁר אַתֶּם מַלִּינִם עָלָיו וְנַחְנוּ מָה לֹא עָלֵינוּ תְלֻנֹּתֵיכֶם כִּי עַל יְהוָה` (שמות טז:ח)
`שָׁמַעְתִּי אֶת תְּלוּנֹּת בְּנֵי יִשְׂרָאֵל דַּבֵּר אֲלֵהֶם לֵאמֹר בֵּין הָעַרְבַּיִם תֹּאכְלוּ בָשָׂר וּבַבֹּקֶר תִּשְׂבְּעוּ לָחֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם` (שמות טז:יב)
`וְכִי יִגַּח שׁוֹר אֶת אִישׁ אוֹ אֶת אִשָּׁה וָמֵת סָקוֹל יִסָּקֵל הַשּׁוֹר וְלֹא יֵאָכֵל אֶת בְּשָׂרוֹ וּבַעַל הַשּׁוֹר נָקִי` (שמות כא:כח)
`וְאַנְשֵׁי קֹדֶשׁ תִּהְיוּן לִי וּבָשָׂר בַּשָּׂדֶה טְרֵפָה לֹא תֹאכֵלוּ לַכֶּלֶב תַּשְׁלִכוּן אֹתוֹ` (שמות כב:לא)
`וַעֲשֵׂה לָהֶם מִכְנְסֵי בָד לְכַסּוֹת בְּשַׂר עֶרְוָה מִמָּתְנַיִם וְעַד יְרֵכַיִם יִהְיוּ` (שמות כח:מב)
`וְאֶת בְּשַׂר הַפָּר וְאֶת עֹרוֹ וְאֶת פִּרְשׁוֹ תִּשְׂרֹף בָּאֵשׁ מִחוּץ לַמַּחֲנֶה חַטָּאת הוּא` (שמות כט:יד)
`וְאֵת אֵיל הַמִּלֻּאִים תִּקָּח וּבִשַּׁלְתָּ אֶת בְּשָׂרוֹ בְּמָקֹם קָדֹשׁ` (שמות כט:לא)
`וְאָכַל אַהֲרֹן וּבָנָיו אֶת בְּשַׂר הָאַיִל וְאֶת הַלֶּחֶם אֲשֶׁר בַּסָּל פֶּתַח אֹהֶל מוֹעֵד` (שמות כט:לב)
`וְאִם יִוָּתֵר מִבְּשַׂר הַמִּלֻּאִים וּמִן הַלֶּחֶם עַד הַבֹּקֶר וְשָׂרַפְתָּ אֶת הַנּוֹתָר בָּאֵשׁ לֹא יֵאָכֵל כִּי קֹדֶשׁ הוּא` (שמות כט:לד)
`עַל בְּשַׂר אָדָם לֹא יִיסָךְ וּבְמַתְכֻּנְתּוֹ לֹא תַעֲשׂוּ כָּמֹהוּ קֹדֶשׁ הוּא קֹדֶשׁ יִהְיֶה לָכֶם` (שמות ל:לב)
`וְאֶת עוֹר הַפָּר וְאֶת כָּל בְּשָׂרוֹ עַל רֹאשׁוֹ וְעַל כְּרָעָיו וְקִרְבּוֹ וּפִרְשׁוֹ` (ויקרא ד:יא)
`וְלָבַשׁ הַכֹּהֵן מִדּוֹ בַד וּמִכְנְסֵי בַד יִלְבַּשׁ עַל בְּשָׂרוֹ וְהֵרִים אֶת הַדֶּשֶׁן אֲשֶׁר תֹּאכַל הָאֵשׁ אֶת הָעֹלָה עַל הַמִּזְבֵּחַ וְשָׂמוֹ אֵצֶל הַמִּזְבֵּחַ` (ויקרא ו:י)
`כֹּל אֲשֶׁר יִגַּע בִּבְשָׂרָהּ יִקְדָּשׁ וַאֲשֶׁר יִזֶּה מִדָּמָהּ עַל הַבֶּגֶד אֲשֶׁר יִזֶּה עָלֶיהָ תְּכַבֵּס בְּמָקוֹם קָדֹשׁ` (ויקרא ו:כז)
`וּבְשַׂר זֶבַח תּוֹדַת שְׁלָמָיו בְּיוֹם קָרְבָּנוֹ יֵאָכֵל לֹא יַנִּיחַ מִמֶּנּוּ עַד בֹּקֶר` (ויקרא ז:טו)
`וְהַנּוֹתָר מִבְּשַׂר הַזָּבַח בַּיּוֹם הַשְּׁלִישִׁי בָּאֵשׁ יִשָּׂרֵף` (ויקרא ז:יז)
`וְאִם הֵאָכֹל יֵאָכֵל מִבְּשַׂר זֶבַח שְׁלָמָיו בַּיּוֹם הַשְּׁלִישִׁי לֹא יֵרָצֶה הַמַּקְרִיב אֹתוֹ לֹא יֵחָשֵׁב לוֹ פִּגּוּל יִהְיֶה וְהַנֶּפֶשׁ הָאֹכֶלֶת מִמֶּנּוּ עֲוֺנָהּ תִּשָּׂא` (ויקרא ז:יח)
`וְהַבָּשָׂר אֲשֶׁר יִגַּע בְּכָל טָמֵא לֹא יֵאָכֵל בָּאֵשׁ יִשָּׂרֵף וְהַבָּשָׂר כָּל טָהוֹר יֹאכַל בָּשָׂר` (ויקרא ז:יט)
`וְהַנֶּפֶשׁ אֲשֶׁר תֹּאכַל בָּשָׂר מִזֶּבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה וְטֻמְאָתוֹ עָלָיו וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא ז:כ)
`וְנֶפֶשׁ כִּי תִגַּע בְּכָל טָמֵא בְּטֻמְאַת אָדָם אוֹ בִּבְהֵמָה טְמֵאָה אוֹ בְּכָל שֶׁקֶץ טָמֵא וְאָכַל מִבְּשַׂר זֶבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא ז:כא)
`וְאֶת הַפָּר וְאֶת עֹרוֹ וְאֶת בְּשָׂרוֹ וְאֶת פִּרְשׁוֹ שָׂרַף בָּאֵשׁ מִחוּץ לַמַּחֲנֶה כַּאֲשֶׁר צִוָּה יְהוָה אֶת מֹשֶׁה` (ויקרא ח:יז)
`וַיֹּאמֶר מֹשֶׁה אֶל אַהֲרֹן וְאֶל בָּנָיו בַּשְּׁלוּ אֶת הַבָּשָׂר פֶּתַח אֹהֶל מוֹעֵד וְשָׁם תֹּאכְלוּ אֹתוֹ וְאֶת הַלֶּחֶם אֲשֶׁר בְּסַל הַמִּלֻּאִים כַּאֲשֶׁר צִוֵּיתִי לֵאמֹר אַהֲרֹן וּבָנָיו יֹאכְלֻהוּ` (ויקרא ח:לא)
`וְהַנּוֹתָר בַּבָּשָׂר וּבַלָּחֶם בָּאֵשׁ תִּשְׂרֹפוּ` (ויקרא ח:לב)
`וְאֶת הַבָּשָׂר וְאֶת הָעוֹר שָׂרַף בָּאֵשׁ מִחוּץ לַמַּחֲנֶה` (ויקרא ט:יא)
`מִבְּשָׂרָם לֹא תֹאכֵלוּ וּבְנִבְלָתָם לֹא תִגָּעוּ טְמֵאִים הֵם לָכֶם` (ויקרא יא:ח)
`וְשֶׁקֶץ יִהְיוּ לָכֶם מִבְּשָׂרָם לֹא תֹאכֵלוּ וְאֶת נִבְלָתָם תְּשַׁקֵּצוּ` (ויקרא יא:יא)
`וּבַיּוֹם הַשְּׁמִינִי יִמּוֹל בְּשַׂר עָרְלָתוֹ` (ויקרא יב:ג)
`אָדָם כִּי יִהְיֶה בְעוֹר בְּשָׂרוֹ שְׂאֵת אוֹ סַפַּחַת אוֹ בַהֶרֶת וְהָיָה בְעוֹר בְּשָׂרוֹ לְנֶגַע צָרָעַת וְהוּבָא אֶל אַהֲרֹן הַכֹּהֵן אוֹ אֶל אַחַד מִבָּנָיו הַכֹּהֲנִים` (ויקרא יג:ב)
`וְרָאָה הַכֹּהֵן אֶת הַנֶּגַע בְּעוֹר הַבָּשָׂר וְשֵׂעָר בַּנֶּגַע הָפַךְ לָבָן וּמַרְאֵה הַנֶּגַע עָמֹק מֵעוֹר בְּשָׂרוֹ נֶגַע צָרַעַת הוּא וְרָאָהוּ הַכֹּהֵן וְטִמֵּא אֹתוֹ` (ויקרא יג:ג)
`וְאִם בַּהֶרֶת לְבָנָה הִוא בְּעוֹר בְּשָׂרוֹ וְעָמֹק אֵין מַרְאֶהָ מִן הָעוֹר וּשְׂעָרָה לֹא הָפַךְ לָבָן וְהִסְגִּיר הַכֹּהֵן אֶת הַנֶּגַע שִׁבְעַת יָמִים` (ויקרא יג:ד)
`וְרָאָה הַכֹּהֵן וְהִנֵּה שְׂאֵת לְבָנָה בָּעוֹר וְהִיא הָפְכָה שֵׂעָר לָבָן וּמִחְיַת בָּשָׂר חַי בַּשְׂאֵת` (ויקרא יג:י)
`צָרַעַת נוֹשֶׁנֶת הִוא בְּעוֹר בְּשָׂרוֹ וְטִמְּאוֹ הַכֹּהֵן לֹא יַסְגִּרֶנּוּ כִּי טָמֵא הוּא` (ויקרא יג:יא)
`וְרָאָה הַכֹּהֵן וְהִנֵּה כִסְּתָה הַצָּרַעַת אֶת כָּל בְּשָׂרוֹ וְטִהַר אֶת הַנָּגַע כֻּלּוֹ הָפַךְ לָבָן טָהוֹר הוּא` (ויקרא יג:יג)
`וּבְיוֹם הֵרָאוֹת בּוֹ בָּשָׂר חַי יִטְמָא` (ויקרא יג:יד)
`וְרָאָה הַכֹּהֵן אֶת הַבָּשָׂר הַחַי וְטִמְּאוֹ הַבָּשָׂר הַחַי טָמֵא הוּא צָרַעַת הוּא` (ויקרא יג:טו)
`אוֹ כִי יָשׁוּב הַבָּשָׂר הַחַי וְנֶהְפַּךְ לְלָבָן וּבָא אֶל הַכֹּהֵן` (ויקרא יג:טז)
`וּבָשָׂר כִּי יִהְיֶה בוֹ בְעֹרוֹ שְׁחִין וְנִרְפָּא` (ויקרא יג:יח)
`אוֹ בָשָׂר כִּי יִהְיֶה בְעֹרוֹ מִכְוַת אֵשׁ וְהָיְתָה מִחְיַת הַמִּכְוָה בַּהֶרֶת לְבָנָה אֲדַמְדֶּמֶת אוֹ לְבָנָה` (ויקרא יג:כד)
`וְאִישׁ אוֹ אִשָּׁה כִּי יִהְיֶה בְעוֹר בְּשָׂרָם בֶּהָרֹת בֶּהָרֹת לְבָנֹת` (ויקרא יג:לח)
`וְרָאָה הַכֹּהֵן וְהִנֵּה בְעוֹר בְּשָׂרָם בֶּהָרֹת כֵּהוֹת לְבָנֹת בֹּהַק הוּא פָּרַח בָּעוֹר טָהוֹר הוּא` (ויקרא יג:לט)
`וְרָאָה אֹתוֹ הַכֹּהֵן וְהִנֵּה שְׂאֵת הַנֶּגַע לְבָנָה אֲדַמְדֶּמֶת בְּקָרַחְתּוֹ אוֹ בְגַבַּחְתּוֹ כְּמַרְאֵה צָרַעַת עוֹר בָּשָׂר` (ויקרא יג:מג)
`וְהָיָה בַיּוֹם הַשְּׁבִיעִי יְגַלַּח אֶת כָּל שְׂעָרוֹ אֶת רֹאשׁוֹ וְאֶת זְקָנוֹ וְאֵת גַּבֹּת עֵינָיו וְאֶת כָּל שְׂעָרוֹ יְגַלֵּחַ וְכִבֶּס אֶת בְּגָדָיו וְרָחַץ אֶת בְּשָׂרוֹ בַּמַּיִם וְטָהֵר` (ויקרא יד:ט)
`דַּבְּרוּ אֶל בְּנֵי יִשְׂרָאֵל וַאֲמַרְתֶּם אֲלֵהֶם אִישׁ אִישׁ כִּי יִהְיֶה זָב מִבְּשָׂרוֹ זוֹבוֹ טָמֵא הוּא` (ויקרא טו:ב)
`וְזֹאת תִּהְיֶה טֻמְאָתוֹ בְּזוֹבוֹ רָר בְּשָׂרוֹ אֶת זוֹבוֹ אוֹ הֶחְתִּים בְּשָׂרוֹ מִזּוֹבוֹ טֻמְאָתוֹ הִוא` (ויקרא טו:ג)
`וְהַנֹּגֵעַ בִּבְשַׂר הַזָּב יְכַבֵּס בְּגָדָיו וְרָחַץ בַּמַּיִם וְטָמֵא עַד הָעָרֶב` (ויקרא טו:ז)
`וְכִי יִטְהַר הַזָּב מִזּוֹבוֹ וְסָפַר לוֹ שִׁבְעַת יָמִים לְטָהֳרָתוֹ וְכִבֶּס בְּגָדָיו וְרָחַץ בְּשָׂרוֹ בְּמַיִם חַיִּים וְטָהֵר` (ויקרא טו:יג)
`וְאִישׁ כִּי תֵצֵא מִמֶּנּוּ שִׁכְבַת זָרַע וְרָחַץ בַּמַּיִם אֶת כָּל בְּשָׂרוֹ וְטָמֵא עַד הָעָרֶב` (ויקרא טו:טז)
`וְאִשָּׁה כִּי תִהְיֶה זָבָה דָּם יִהְיֶה זֹבָהּ בִּבְשָׂרָהּ שִׁבְעַת יָמִים תִּהְיֶה בְנִדָּתָהּ וְכָל הַנֹּגֵעַ בָּהּ יִטְמָא עַד הָעָרֶב` (ויקרא טו:יט)
`כְּתֹנֶת בַּד קֹדֶשׁ יִלְבָּשׁ וּמִכְנְסֵי בַד יִהְיוּ עַל בְּשָׂרוֹ וּבְאַבְנֵט בַּד יַחְגֹּר וּבְמִצְנֶפֶת בַּד יִצְנֹף בִּגְדֵי קֹדֶשׁ הֵם וְרָחַץ בַּמַּיִם אֶת בְּשָׂרוֹ וּלְבֵשָׁם` (ויקרא טז:ד)
`וְרָחַץ אֶת בְּשָׂרוֹ בַמַּיִם בְּמָקוֹם קָדוֹשׁ וְלָבַשׁ אֶת בְּגָדָיו וְיָצָא וְעָשָׂה אֶת עֹלָתוֹ וְאֶת עֹלַת הָעָם וְכִפֶּר בַּעֲדוֹ וּבְעַד הָעָם` (ויקרא טז:כד)
`וְהַמְשַׁלֵּחַ אֶת הַשָּׂעִיר לַעֲזָאזֵל יְכַבֵּס בְּגָדָיו וְרָחַץ אֶת בְּשָׂרוֹ בַּמָּיִם וְאַחֲרֵי כֵן יָבוֹא אֶל הַמַּחֲנֶה` (ויקרא טז:כו)
`וְאֵת פַּר הַחַטָּאת וְאֵת שְׂעִיר הַחַטָּאת אֲשֶׁר הוּבָא אֶת דָּמָם לְכַפֵּר בַּקֹּדֶשׁ יוֹצִיא אֶל מִחוּץ לַמַּחֲנֶה וְשָׂרְפוּ בָאֵשׁ אֶת עֹרֹתָם וְאֶת בְּשָׂרָם וְאֶת פִּרְשָׁם` (ויקרא טז:כז)
`וְהַשֹּׂרֵף אֹתָם יְכַבֵּס בְּגָדָיו וְרָחַץ אֶת בְּשָׂרוֹ בַּמָּיִם וְאַחֲרֵי כֵן יָבוֹא אֶל הַמַּחֲנֶה` (ויקרא טז:כח)
`כִּי נֶפֶשׁ הַבָּשָׂר בַּדָּם הִוא וַאֲנִי נְתַתִּיו לָכֶם עַל הַמִּזְבֵּחַ לְכַפֵּר עַל נַפְשֹׁתֵיכֶם כִּי הַדָּם הוּא בַּנֶּפֶשׁ יְכַפֵּר` (ויקרא יז:יא)
`כִּי נֶפֶשׁ כָּל בָּשָׂר דָּמוֹ בְנַפְשׁוֹ הוּא וָאֹמַר לִבְנֵי יִשְׂרָאֵל דַּם כָּל בָּשָׂר לֹא תֹאכֵלוּ כִּי נֶפֶשׁ כָּל בָּשָׂר דָּמוֹ הִוא כָּל אֹכְלָיו יִכָּרֵת` (ויקרא יז:יד)
`וְאִם לֹא יְכַבֵּס וּבְשָׂרוֹ לֹא יִרְחָץ וְנָשָׂא עֲוֺנוֹ` (ויקרא יז:טז)
`אִישׁ אִישׁ אֶל כָּל שְׁאֵר בְּשָׂרוֹ לֹא תִקְרְבוּ לְגַלּוֹת עֶרְוָה אֲנִי יְהוָה` (ויקרא יח:ו)
`וְשֶׂרֶט לָנֶפֶשׁ לֹא תִתְּנוּ בִּבְשַׂרְכֶם וּכְתֹבֶת קַעֲקַע לֹא תִתְּנוּ בָּכֶם אֲנִי יְהוָה` (ויקרא יט:כח)
`לֹא יִקְרְחָה קָרְחָה בְּרֹאשָׁם וּפְאַת זְקָנָם לֹא יְגַלֵּחוּ וּבִבְשָׂרָם לֹא יִשְׂרְטוּ שָׂרָטֶת` (ויקרא כא:ה)
`נֶפֶשׁ אֲשֶׁר תִּגַּע בּוֹ וְטָמְאָה עַד הָעָרֶב וְלֹא יֹאכַל מִן הַקֳּדָשִׁים כִּי אִם רָחַץ בְּשָׂרוֹ בַּמָּיִם` (ויקרא כב:ו)
`אוֹ דֹדוֹ אוֹ בֶן דֹּדוֹ יִגְאָלֶנּוּ אוֹ מִשְּׁאֵר בְּשָׂרוֹ מִמִּשְׁפַּחְתּוֹ יִגְאָלֶנּוּ אוֹ הִשִּׂיגָה יָדוֹ וְנִגְאָל` (ויקרא כה:מט)
`וַאֲכַלְתֶּם בְּשַׂר בְּנֵיכֶם וּבְשַׂר בְּנֹתֵיכֶם תֹּאכֵלוּ` (ויקרא כו:כט)
`וְכֹה תַעֲשֶׂה לָהֶם לְטַהֲרָם הַזֵּה עֲלֵיהֶם מֵי חַטָּאת וְהֶעֱבִירוּ תַעַר עַל כָּל בְּשָׂרָם וְכִבְּסוּ בִגְדֵיהֶם וְהִטֶּהָרוּ` (במדבר ח:ז)
`וְהָאסַפְסֻף אֲשֶׁר בְּקִרְבּוֹ הִתְאַוּוּ תַּאֲוָה וַיָּשֻׁבוּ וַיִּבְכּוּ גַּם בְּנֵי יִשְׂרָאֵל וַיֹּאמְרוּ מִי יַאֲכִלֵנוּ בָּשָׂר` (במדבר יא:ד)
`מֵאַיִן לִי בָּשָׂר לָתֵת לְכָל הָעָם הַזֶּה כִּי יִבְכּוּ עָלַי לֵאמֹר תְּנָה לָּנוּ בָשָׂר וְנֹאכֵלָה` (במדבר יא:יג)
`וְאֶל הָעָם תֹּאמַר הִתְקַדְּשׁוּ לְמָחָר וַאֲכַלְתֶּם בָּשָׂר כִּי בְּכִיתֶם בְּאָזְנֵי יְהוָה לֵאמֹר מִי יַאֲכִלֵנוּ בָּשָׂר כִּי טוֹב לָנוּ בְּמִצְרָיִם וְנָתַן יְהוָה לָכֶם בָּשָׂר וַאֲכַלְתֶּם` (במדבר יא:יח)
`וַיֹּאמֶר מֹשֶׁה שֵׁשׁ מֵאוֹת אֶלֶף רַגְלִי הָעָם אֲשֶׁר אָנֹכִי בְּקִרְבּוֹ וְאַתָּה אָמַרְתָּ בָּשָׂר אֶתֵּן לָהֶם וְאָכְלוּ חֹדֶשׁ יָמִים` (במדבר יא:כא)
`הַבָּשָׂר עוֹדֶנּוּ בֵּין שִׁנֵּיהֶם טֶרֶם יִכָּרֵת וְאַף יְהוָה חָרָה בָעָם וַיַּךְ יְהוָה בָּעָם מַכָּה רַבָּה מְאֹד` (במדבר יא:לג)
`אַל נָא תְהִי כַּמֵּת אֲשֶׁר בְּצֵאתוֹ מֵרֶחֶם אִמּוֹ וַיֵּאָכֵל חֲצִי בְשָׂרוֹ` (במדבר יב:יב)
`וַיִּפְּלוּ עַל פְּנֵיהֶם וַיֹּאמְרוּ אֵל אֱלֹהֵי הָרוּחֹת לְכָל בָּשָׂר הָאִישׁ אֶחָד יֶחֱטָא וְעַל כָּל הָעֵדָה תִּקְצֹף` (במדבר טז:כב)
`כָּל פֶּטֶר רֶחֶם לְכָל בָּשָׂר אֲשֶׁר יַקְרִיבוּ לַיהוָה בָּאָדָם וּבַבְּהֵמָה יִהְיֶה לָּךְ אַךְ פָּדֹה תִפְדֶּה אֵת בְּכוֹר הָאָדָם וְאֵת בְּכוֹר הַבְּהֵמָה הַטְּמֵאָה תִּפְדֶּה` (במדבר יח:טו)
`וּבְשָׂרָם יִהְיֶה לָּךְ כַּחֲזֵה הַתְּנוּפָה וּכְשׁוֹק הַיָּמִין לְךָ יִהְיֶה` (במדבר יח:יח)
`וְשָׂרַף אֶת הַפָּרָה לְעֵינָיו אֶת עֹרָהּ וְאֶת בְּשָׂרָהּ וְאֶת דָּמָהּ עַל פִּרְשָׁהּ יִשְׂרֹף` (במדבר יט:ה)
`וְכִבֶּס בְּגָדָיו הַכֹּהֵן וְרָחַץ בְּשָׂרוֹ בַּמַּיִם וְאַחַר יָבוֹא אֶל הַמַּחֲנֶה וְטָמֵא הַכֹּהֵן עַד הָעָרֶב` (במדבר יט:ז)
`וְהַשֹּׂרֵף אֹתָהּ יְכַבֵּס בְּגָדָיו בַּמַּיִם וְרָחַץ בְּשָׂרוֹ בַּמָּיִם וְטָמֵא עַד הָעָרֶב` (במדבר יט:ח)
`יִפְקֹד יְהוָה אֱלֹהֵי הָרוּחֹת לְכָל בָּשָׂר אִישׁ עַל הָעֵדָה` (במדבר כז:טז)
`כִּי מִי כָל בָּשָׂר אֲשֶׁר שָׁמַע קוֹל אֱלֹהִים חַיִּים מְדַבֵּר מִתּוֹךְ הָאֵשׁ כָּמֹנוּ וַיֶּחִי` (דברים ה:כו)
`רַק בְּכָל אַוַּת נַפְשְׁךָ תִּזְבַּח וְאָכַלְתָּ בָשָׂר כְּבִרְכַּת יְהוָה אֱלֹהֶיךָ אֲשֶׁר נָתַן לְךָ בְּכָל שְׁעָרֶיךָ הַטָּמֵא וְהַטָּהוֹר יֹאכְלֶנּוּ כַּצְּבִי וְכָאַיָּל` (דברים יב:טו)
`כִּי יַרְחִיב יְהוָה אֱלֹהֶיךָ אֶת גְּבוּלְךָ כַּאֲשֶׁר דִּבֶּר לָךְ וְאָמַרְתָּ אֹכְלָה בָשָׂר כִּי תְאַוֶּה נַפְשְׁךָ לֶאֱכֹל בָּשָׂר בְּכָל אַוַּת נַפְשְׁךָ תֹּאכַל בָּשָׂר` (דברים יב:כ)
`רַק חֲזַק לְבִלְתִּי אֲכֹל הַדָּם כִּי הַדָּם הוּא הַנָּפֶשׁ וְלֹא תֹאכַל הַנֶּפֶשׁ עִם הַבָּשָׂר` (דברים יב:כג)
`וְעָשִׂיתָ עֹלֹתֶיךָ הַבָּשָׂר וְהַדָּם עַל מִזְבַּח יְהוָה אֱלֹהֶיךָ וְדַם זְבָחֶיךָ יִשָּׁפֵךְ עַל מִזְבַּח יְהוָה אֱלֹהֶיךָ וְהַבָּשָׂר תֹּאכֵל` (דברים יב:כז)
`וְאֶת הַחֲזִיר כִּי מַפְרִיס פַּרְסָה הוּא וְלֹא גֵרָה טָמֵא הוּא לָכֶם מִבְּשָׂרָם לֹא תֹאכֵלוּ וּבְנִבְלָתָם לֹא תִגָּעוּ` (דברים יד:ח)
`וְלֹא יֵרָאֶה לְךָ שְׂאֹר בְּכָל גְּבֻלְךָ שִׁבְעַת יָמִים וְלֹא יָלִין מִן הַבָּשָׂר אֲשֶׁר תִּזְבַּח בָּעֶרֶב בַּיּוֹם הָרִאשׁוֹן לַבֹּקֶר` (דברים טז:ד)
`וְאָכַלְתָּ פְרִי בִטְנְךָ בְּשַׂר בָּנֶיךָ וּבְנֹתֶיךָ אֲשֶׁר נָתַן לְךָ יְהוָה אֱלֹהֶיךָ בְּמָצוֹר וּבְמָצוֹק אֲשֶׁר יָצִיק לְךָ אֹיְבֶךָ` (דברים כח:נג)
`מִתֵּת לְאַחַד מֵהֶם מִבְּשַׂר בָּנָיו אֲשֶׁר יֹאכֵל מִבְּלִי הִשְׁאִיר לוֹ כֹּל בְּמָצוֹר וּבְמָצוֹק אֲשֶׁר יָצִיק לְךָ אֹיִבְךָ בְּכָל שְׁעָרֶיךָ` (דברים כח:נה)
`אַשְׁכִּיר חִצַּי מִדָּם וְחַרְבִּי תֹּאכַל בָּשָׂר מִדַּם חָלָל וְשִׁבְיָה מֵרֹאשׁ פַּרְעוֹת אוֹיֵב` (דברים לב:מב)
`וְגִדְעוֹן בָּא וַיַּעַשׂ גְּדִי עִזִּים וְאֵיפַת קֶמַח מַצּוֹת הַבָּשָׂר שָׂם בַּסַּל וְהַמָּרַק שָׂם בַּפָּרוּר וַיּוֹצֵא אֵלָיו אֶל תַּחַת הָאֵלָה וַיַּגַּשׁ` (שופטים ו:יט)
`וַיֹּאמֶר אֵלָיו מַלְאַךְ הָאֱלֹהִים קַח אֶת הַבָּשָׂר וְאֶת הַמַּצּוֹת וְהַנַּח אֶל הַסֶּלַע הַלָּז וְאֶת הַמָּרַק שְׁפוֹךְ וַיַּעַשׂ כֵּן` (שופטים ו:כ)
`וַיִּשְׁלַח מַלְאַךְ יְהוָה אֶת קְצֵה הַמִּשְׁעֶנֶת אֲשֶׁר בְּיָדוֹ וַיִּגַּע בַּבָּשָׂר וּבַמַּצּוֹת וַתַּעַל הָאֵשׁ מִן הַצּוּר וַתֹּאכַל אֶת הַבָּשָׂר וְאֶת הַמַּצּוֹת וּמַלְאַךְ יְהוָה הָלַךְ מֵעֵינָיו` (שופטים ו:כא)
`וַיֹּאמֶר גִּדְעוֹן לָכֵן בְּתֵת יְהוָה אֶת זֶבַח וְאֶת צַלְמֻנָּע בְּיָדִי וְדַשְׁתִּי אֶת בְּשַׂרְכֶם אֶת קוֹצֵי הַמִּדְבָּר וְאֶת הַבַּרְקֳנִים` (שופטים ח:ז)
`דַּבְּרוּ נָא בְּאָזְנֵי כָל בַּעֲלֵי שְׁכֶם מַה טּוֹב לָכֶם הַמְשֹׁל בָּכֶם שִׁבְעִים אִישׁ כֹּל בְּנֵי יְרֻבַּעַל אִם מְשֹׁל בָּכֶם אִישׁ אֶחָד וּזְכַרְתֶּם כִּי עַצְמֵכֶם וּבְשַׂרְכֶם אָנִי` (שופטים ט:ב)
`וּמִשְׁפַּט הַכֹּהֲנִים אֶת הָעָם כָּל אִישׁ זֹבֵחַ זֶבַח וּבָא נַעַר הַכֹּהֵן כְּבַשֵּׁל הַבָּשָׂר וְהַמַּזְלֵג שְׁלֹשׁ הַשִּׁנַּיִם בְּיָדוֹ` (שמואל-א ב:יג)
`גַּם בְּטֶרֶם יַקְטִרוּן אֶת הַחֵלֶב וּבָא נַעַר הַכֹּהֵן וְאָמַר לָאִישׁ הַזֹּבֵחַ תְּנָה בָשָׂר לִצְלוֹת לַכֹּהֵן וְלֹא יִקַּח מִמְּךָ בָּשָׂר מְבֻשָּׁל כִּי אִם חָי` (שמואל-א ב:טו)
`וַיֹּאמֶר הַפְּלִשְׁתִּי אֶל דָּוִד לְכָה אֵלַי וְאֶתְּנָה אֶת בְּשָׂרְךָ לְעוֹף הַשָּׁמַיִם וּלְבֶהֱמַת הַשָּׂדֶה` (שמואל-א יז:מד)
`וַיָּבֹאוּ כָּל שִׁבְטֵי יִשְׂרָאֵל אֶל דָּוִד חֶבְרוֹנָה וַיֹּאמְרוּ לֵאמֹר הִנְנוּ עַצְמְךָ וּבְשָׂרְךָ אֲנָחְנוּ` (שמואל-ב ה:א)
`אַחַי אַתֶּם עַצְמִי וּבְשָׂרִי אַתֶּם וְלָמָּה תִהְיוּ אַחֲרֹנִים לְהָשִׁיב אֶת הַמֶּלֶךְ` (שמואל-ב יט:יב)
`וְלַעֲמָשָׂא תֹּמְרוּ הֲלוֹא עַצְמִי וּבְשָׂרִי אָתָּה כֹּה יַעֲשֶׂה לִּי אֱלֹהִים וְכֹה יוֹסִיף אִם לֹא שַׂר צָבָא תִּהְיֶה לְפָנַי כָּל הַיָּמִים תַּחַת יוֹאָב` (שמואל-ב יט:יג)
`וְהָעֹרְבִים מְבִיאִים לוֹ לֶחֶם וּבָשָׂר בַּבֹּקֶר וְלֶחֶם וּבָשָׂר בָּעָרֶב וּמִן הַנַּחַל יִשְׁתֶּה` (מלכים-א יז:ו)
`וַיָּשָׁב מֵאַחֲרָיו וַיִּקַּח אֶת צֶמֶד הַבָּקָר וַיִּזְבָּחֵהוּ וּבִכְלִי הַבָּקָר בִּשְּׁלָם הַבָּשָׂר וַיִּתֵּן לָעָם וַיֹּאכֵלוּ וַיָּקָם וַיֵּלֶךְ אַחֲרֵי אֵלִיָּהוּ וַיְשָׁרְתֵהוּ` (מלכים-א יט:כא)
`וַיְהִי כִשְׁמֹעַ אַחְאָב אֶת הַדְּבָרִים הָאֵלֶּה וַיִּקְרַע בְּגָדָיו וַיָּשֶׂם שַׂק עַל בְּשָׂרוֹ וַיָּצוֹם וַיִּשְׁכַּב בַּשָּׂק וַיְהַלֵּךְ אַט` (מלכים-א כא:כז)
`וַיַּעַל וַיִּשְׁכַּב עַל הַיֶּלֶד וַיָּשֶׂם פִּיו עַל פִּיו וְעֵינָיו עַל עֵינָיו וְכַפָּיו עַל כַּפּוֹ וַיִּגְהַר עָלָיו וַיָּחָם בְּשַׂר הַיָּלֶד` (מלכים-ב ד:לד)
`וַיִּשְׁלַח אֵלָיו אֱלִישָׁע מַלְאָךְ לֵאמֹר הָלוֹךְ וְרָחַצְתָּ שֶׁבַע פְּעָמִים בַּיַּרְדֵּן וְיָשֹׁב בְּשָׂרְךָ לְךָ וּטְהָר` (מלכים-ב ה:י)
`וַיֵּרֶד וַיִּטְבֹּל בַּיַּרְדֵּן שֶׁבַע פְּעָמִים כִּדְבַר אִישׁ הָאֱלֹהִים וַיָּשָׁב בְּשָׂרוֹ כִּבְשַׂר נַעַר קָטֹן וַיִּטְהָר` (מלכים-ב ה:יד)
`וַיְהִי כִשְׁמֹעַ הַמֶּלֶךְ אֶת דִּבְרֵי הָאִשָּׁה וַיִּקְרַע אֶת בְּגָדָיו וְהוּא עֹבֵר עַל הַחֹמָה וַיַּרְא הָעָם וְהִנֵּה הַשַּׂק עַל בְּשָׂרוֹ מִבָּיִת` (מלכים-ב ו:ל)
`וַיָּשֻׁבוּ וַיַּגִּידוּ לוֹ וַיֹּאמֶר דְּבַר יְהוָה הוּא אֲשֶׁר דִּבֶּר בְּיַד עַבְדּוֹ אֵלִיָּהוּ הַתִּשְׁבִּי לֵאמֹר בְּחֵלֶק יִזְרְעֶאל יֹאכְלוּ הַכְּלָבִים אֶת בְּשַׂר אִיזָבֶל` (מלכים-ב ט:לו)
`וַיִּגְזֹר עַל יָמִין וְרָעֵב וַיֹּאכַל עַל שְׂמֹאול וְלֹא שָׂבֵעוּ אִישׁ בְּשַׂר זְרֹעוֹ יֹאכֵלוּ` (ישעיהו ט:כ)
`וּכְבוֹד יַעְרוֹ וְכַרְמִלּוֹ מִנֶּפֶשׁ וְעַד בָּשָׂר יְכַלֶּה וְהָיָה כִּמְסֹס נֹסֵס` (ישעיהו י:יח)
`וְהָיָה בַּיּוֹם הַהוּא יִדַּל כְּבוֹד יַעֲקֹב וּמִשְׁמַן בְּשָׂרוֹ יֵרָזֶה` (ישעיהו יז:ד)
`וְהִנֵּה שָׂשׂוֹן וְשִׂמְחָה הָרֹג בָּקָר וְשָׁחֹט צֹאן אָכֹל בָּשָׂר וְשָׁתוֹת יָיִן אָכוֹל וְשָׁתוֹ כִּי מָחָר נָמוּת` (ישעיהו כב:יג)
`וּמִצְרַיִם אָדָם וְלֹא אֵל וְסוּסֵיהֶם בָּשָׂר וְלֹא רוּחַ וַיהוָה יַטֶּה יָדוֹ וְכָשַׁל עוֹזֵר וְנָפַל עָזֻר וְיַחְדָּו כֻּלָּם יִכְלָיוּן` (ישעיהו לא:ג)
`וְנִגְלָה כְּבוֹד יְהוָה וְרָאוּ כָל בָּשָׂר יַחְדָּו כִּי פִּי יְהוָה דִּבֵּר` (ישעיהו מ:ה)
`קוֹל אֹמֵר קְרָא וְאָמַר מָה אֶקְרָא כָּל הַבָּשָׂר חָצִיר וְכָל חַסְדּוֹ כְּצִיץ הַשָּׂדֶה` (ישעיהו מ:ו)
`חֶצְיוֹ שָׂרַף בְּמוֹ אֵשׁ עַל חֶצְיוֹ בָּשָׂר יֹאכֵל יִצְלֶה צָלִי וְיִשְׂבָּע אַף יָחֹם וְיֹאמַר הֶאָח חַמּוֹתִי רָאִיתִי אוּר` (ישעיהו מד:טז)
`וְלֹא יָשִׁיב אֶל לִבּוֹ וְלֹא דַעַת וְלֹא תְבוּנָה לֵאמֹר חֶצְיוֹ שָׂרַפְתִּי בְמוֹ אֵשׁ וְאַף אָפִיתִי עַל גֶּחָלָיו לֶחֶם אֶצְלֶה בָשָׂר וְאֹכֵל וְיִתְרוֹ לְתוֹעֵבָה אֶעֱשֶׂה לְבוּל עֵץ אֶסְגּוֹד` (ישעיהו מד:יט)
`וְהַאֲכַלְתִּי אֶת מוֹנַיִךְ אֶת בְּשָׂרָם וְכֶעָסִיס דָּמָם יִשְׁכָּרוּן וְיָדְעוּ כָל בָּשָׂר כִּי אֲנִי יְהוָה מוֹשִׁיעֵךְ וְגֹאֲלֵךְ אֲבִיר יַעֲקֹב` (ישעיהו מט:כו)
`הֲלוֹא פָרֹס לָרָעֵב לַחְמֶךָ וַעֲנִיִּים מְרוּדִים תָּבִיא בָיִת כִּי תִרְאֶה עָרֹם וְכִסִּיתוֹ וּמִבְּשָׂרְךָ לֹא תִתְעַלָּם` (ישעיהו נח:ז)
`הַיֹּשְׁבִים בַּקְּבָרִים וּבַנְּצוּרִים יָלִינוּ הָאֹכְלִים בְּשַׂר הַחֲזִיר וּפְרַק פִּגֻּלִים כְּלֵיהֶם` (ישעיהו סה:ד)
`כִּי בָאֵשׁ יְהוָה נִשְׁפָּט וּבְחַרְבּוֹ אֶת כָּל בָּשָׂר וְרַבּוּ חַלְלֵי יְהוָה` (ישעיהו סו:טז)
`הַמִּתְקַדְּשִׁים וְהַמִּטַּהֲרִים אֶל הַגַּנּוֹת אַחַר אֶחָד בַּתָּוֶךְ אֹכְלֵי בְּשַׂר הַחֲזִיר וְהַשֶּׁקֶץ וְהָעַכְבָּר יַחְדָּו יָסֻפוּ נְאֻם יְהוָה` (ישעיהו סו:יז)
`וְהָיָה מִדֵּי חֹדֶשׁ בְּחָדְשׁוֹ וּמִדֵּי שַׁבָּת בְּשַׁבַּתּוֹ יָבוֹא כָל בָּשָׂר לְהִשְׁתַּחֲוֺת לְפָנַי אָמַר יְהוָה` (ישעיהו סו:כג)
`וְיָצְאוּ וְרָאוּ בְּפִגְרֵי הָאֲנָשִׁים הַפֹּשְׁעִים בִּי כִּי תוֹלַעְתָּם לֹא תָמוּת וְאִשָּׁם לֹא תִכְבֶּה וְהָיוּ דֵרָאוֹן לְכָל בָּשָׂר` (ישעיהו סו:כד)
`כֹּה אָמַר יְהוָה צְבָאוֹת אֱלֹהֵי יִשְׂרָאֵל עֹלוֹתֵיכֶם סְפוּ עַל זִבְחֵיכֶם וְאִכְלוּ בָשָׂר` (ירמיהו ז:כא)
`מֶה לִידִידִי בְּבֵיתִי עֲשׂוֹתָהּ הַמְזִמָּתָה הָרַבִּים וּבְשַׂר קֹדֶשׁ יַעַבְרוּ מֵעָלָיִךְ כִּי רָעָתֵכִי אָז תַּעֲלֹזִי` (ירמיהו יא:טו)
`עַל כָּל שְׁפָיִם בַּמִּדְבָּר בָּאוּ שֹׁדְדִים כִּי חֶרֶב לַיהוָה אֹכְלָה מִקְצֵה אֶרֶץ וְעַד קְצֵה הָאָרֶץ אֵין שָׁלוֹם לְכָל בָּשָׂר` (ירמיהו יב:יב)
`כֹּה אָמַר יְהוָה אָרוּר הַגֶּבֶר אֲשֶׁר יִבְטַח בָּאָדָם וְשָׂם בָּשָׂר זְרֹעוֹ וּמִן יְהוָה יָסוּר לִבּוֹ` (ירמיהו יז:ה)
`וְהַאֲכַלְתִּים אֶת בְּשַׂר בְּנֵיהֶם וְאֵת בְּשַׂר בְּנֹתֵיהֶם וְאִישׁ בְּשַׂר רֵעֵהוּ יֹאכֵלוּ בְּמָצוֹר וּבְמָצוֹק אֲשֶׁר יָצִיקוּ לָהֶם אֹיְבֵיהֶם וּמְבַקְשֵׁי נַפְשָׁם` (ירמיהו יט:ט)
`בָּא שָׁאוֹן עַד קְצֵה הָאָרֶץ כִּי רִיב לַיהוָה בַּגּוֹיִם נִשְׁפָּט הוּא לְכָל בָּשָׂר הָרְשָׁעִים נְתָנָם לַחֶרֶב נְאֻם יְהוָה` (ירמיהו כה:לא)
`הִנֵּה אֲנִי יְהוָה אֱלֹהֵי כָּל בָּשָׂר הֲ‍מִמֶּנִּי יִפָּלֵא כָּל דָּבָר` (ירמיהו לב:כז)
`וְאַתָּה תְּבַקֶּשׁ לְךָ גְדֹלוֹת אַל תְּבַקֵּשׁ כִּי הִנְנִי מֵבִיא רָעָה עַל כָּל בָּשָׂר נְאֻם יְהוָה וְנָתַתִּי לְךָ אֶת נַפְשְׁךָ לְשָׁלָל עַל כָּל הַמְּקֹמוֹת אֲשֶׁר תֵּלֶךְ שָׁם` (ירמיהו מה:ה)
`וָאֹמַר אֲהָהּ אֲדֹנָי יְהוִה הִנֵּה נַפְשִׁי לֹא מְטֻמָּאָה וּנְבֵלָה וּטְרֵפָה לֹא אָכַלְתִּי מִנְּעוּרַי וְעַד עַתָּה וְלֹא בָא בְּפִי בְּשַׂר פִּגּוּל` (יחזקאל ד:יד)
`וְכָל בְּשָׂרָם וְגַבֵּהֶם וִידֵיהֶם וְכַנְפֵיהֶם וְהָאוֹפַנִּים מְלֵאִים עֵינַיִם סָבִיב לְאַרְבַּעְתָּם אוֹפַנֵּיהֶם` (יחזקאל י:יב)
`הָאֹמְרִים לֹא בְקָרוֹב בְּנוֹת בָּתִּים הִיא הַסִּיר וַאֲנַחְנוּ הַבָּשָׂר` (יחזקאל יא:ג)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה חַלְלֵיכֶם אֲשֶׁר שַׂמְתֶּם בְּתוֹכָהּ הֵמָּה הַבָּשָׂר וְהִיא הַסִּיר וְאֶתְכֶם הוֹצִיא מִתּוֹכָהּ` (יחזקאל יא:ז)
`הִיא לֹא תִהְיֶה לָכֶם לְסִיר וְאַתֶּם תִּהְיוּ בְתוֹכָהּ לְבָשָׂר אֶל גְּבוּל יִשְׂרָאֵל אֶשְׁפֹּט אֶתְכֶם` (יחזקאל יא:יא)
`וְנָתַתִּי לָהֶם לֵב אֶחָד וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי לֵב הָאֶבֶן מִבְּשָׂרָם וְנָתַתִּי לָהֶם לֵב בָּשָׂר` (יחזקאל יא:יט)
`וַתִּזְנִי אֶל בְּנֵי מִצְרַיִם שְׁכֵנַיִךְ גִּדְלֵי בָשָׂר וַתַּרְבִּי אֶת תַּזְנֻתֵךְ לְהַכְעִיסֵנִי` (יחזקאל טז:כו)
`וְרָאוּ כָּל בָּשָׂר כִּי אֲנִי יְהוָה בִּעַרְתִּיהָ לֹא תִּכְבֶּה` (יחזקאל כ:מח)
`יַעַן אֲשֶׁר הִכְרַתִּי מִמֵּךְ צַדִּיק וְרָשָׁע לָכֵן תֵּצֵא חַרְבִּי מִתַּעְרָהּ אֶל כָּל בָּשָׂר מִנֶּגֶב צָפוֹן` (יחזקאל כא:ד)
`וְיָדְעוּ כָּל בָּשָׂר כִּי אֲנִי יְהוָה הוֹצֵאתִי חַרְבִּי מִתַּעְרָהּ לֹא תָשׁוּב עוֹד` (יחזקאל כא:ה)
`וַתַּעְגְּבָה עַל פִּלַגְשֵׁיהֶם אֲשֶׁר בְּשַׂר חֲמוֹרִים בְּשָׂרָם וְזִרְמַת סוּסִים זִרְמָתָם` (יחזקאל כג:כ)
`הַרְבֵּה הָעֵצִים הַדְלֵק הָאֵשׁ הָתֵם הַבָּשָׂר וְהַרְקַח הַמֶּרְקָחָה וְהָעֲצָמוֹת יֵחָרוּ` (יחזקאל כד:י)
`וְנָתַתִּי אֶת בְּשָׂרְךָ עַל הֶהָרִים וּמִלֵּאתִי הַגֵּאָיוֹת רָמוּתֶךָ` (יחזקאל לב:ה)
`וְנָתַתִּי לָכֶם לֵב חָדָשׁ וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי אֶת לֵב הָאֶבֶן מִבְּשַׂרְכֶם וְנָתַתִּי לָכֶם לֵב בָּשָׂר` (יחזקאל לו:כו)
`וְנָתַתִּי עֲלֵיכֶם גִּדִים וְהַעֲלֵתִי עֲלֵיכֶם בָּשָׂר וְקָרַמְתִּי עֲלֵיכֶם עוֹר וְנָתַתִּי בָכֶם רוּחַ וִחְיִיתֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה` (יחזקאל לז:ו)
`וְרָאִיתִי וְהִנֵּה עֲלֵיהֶם גִּדִים וּבָשָׂר עָלָה וַיִּקְרַם עֲלֵיהֶם עוֹר מִלְמָעְלָה וְרוּחַ אֵין בָּהֶם` (יחזקאל לז:ח)
`וְאַתָּה בֶן אָדָם כֹּה אָמַר אֲדֹנָי יְהֹוִה אֱמֹר לְצִפּוֹר כָּל כָּנָף וּלְכֹל חַיַּת הַשָּׂדֶה הִקָּבְצוּ וָבֹאוּ הֵאָסְפוּ מִסָּבִיב עַל זִבְחִי אֲשֶׁר אֲנִי זֹבֵחַ לָכֶם זֶבַח גָּדוֹל עַל הָרֵי יִשְׂרָאֵל וַאֲכַלְתֶּם בָּשָׂר וּשְׁתִיתֶם דָּם` (יחזקאל לט:יז)
`בְּשַׂר גִּבּוֹרִים תֹּאכֵלוּ וְדַם נְשִׂיאֵי הָאָרֶץ תִּשְׁתּוּ אֵילִים כָּרִים וְעַתּוּדִים פָּרִים מְרִיאֵי בָשָׁן כֻּלָּם` (יחזקאל לט:יח)
`וְהַשְׁפַתַּיִם טֹפַח אֶחָד מוּכָנִים בַּבַּיִת סָבִיב סָבִיב וְאֶל הַשֻּׁלְחָנוֹת בְּשַׂר הַקָּרְבָן` (יחזקאל מ:מג)
`בַּהֲבִיאֲכֶם בְּנֵי נֵכָר עַרְלֵי לֵב וְעַרְלֵי בָשָׂר לִהְיוֹת בְּמִקְדָּשִׁי לְחַלְּלוֹ אֶת בֵּיתִי בְּהַקְרִיבְכֶם אֶת לַחְמִי חֵלֶב וָדָם וַיָּפֵרוּ אֶת בְּרִיתִי אֶל כָּל תּוֹעֲבוֹתֵיכֶם` (יחזקאל מד:ז)
`כֹּה אָמַר אֲדֹנָי יְהוִה כָּל בֶּן נֵכָר עֶרֶל לֵב וְעֶרֶל בָּשָׂר לֹא יָבוֹא אֶל מִקְדָּשִׁי לְכָל בֶּן נֵכָר אֲשֶׁר בְּתוֹךְ בְּנֵי יִשְׂרָאֵל` (יחזקאל מד:ט)
`זִבְחֵי הַבְהָבַי יִזְבְּחוּ בָשָׂר וַיֹּאכֵלוּ יְהוָה לֹא רָצָם עַתָּה יִזְכֹּר עֲוֺנָם וְיִפְקֹד חַטֹּאותָם הֵמָּה מִצְרַיִם יָשׁוּבוּ` (הושע ח:יג)
`וְהָיָה אַחֲרֵי כֵן אֶשְׁפּוֹךְ אֶת רוּחִי עַל כָּל בָּשָׂר וְנִבְּאוּ בְּנֵיכֶם וּבְנוֹתֵיכֶם זִקְנֵיכֶם חֲלֹמוֹת יַחֲלֹמוּן בַּחוּרֵיכֶם חֶזְיֹנוֹת יִרְאוּ` (יואל ב:כח)
`וַאֲשֶׁר אָכְלוּ שְׁאֵר עַמִּי וְעוֹרָם מֵעֲלֵיהֶם הִפְשִׁיטוּ וְאֶת עַצְמֹתֵיהֶם פִּצֵּחוּ וּפָרְשׂוּ כַּאֲשֶׁר בַּסִּיר וּכְבָשָׂר בְּתוֹךְ קַלָּחַת` (מיכה ג:ג)
`הֵן יִשָּׂא אִישׁ בְּשַׂר קֹדֶשׁ בִּכְנַף בִּגְדוֹ וְנָגַע בִּכְנָפוֹ אֶל הַלֶּחֶם וְאֶל הַנָּזִיד וְאֶל הַיַּיִן וְאֶל שֶׁמֶן וְאֶל כָּל מַאֲכָל הֲיִקְדָּשׁ וַיַּעֲנוּ הַכֹּהֲנִים וַיֹּאמְרוּ לֹא` (חגי ב:יב)
`הַס כָּל בָּשָׂר מִפְּנֵי יְהוָה כִּי נֵעוֹר מִמְּעוֹן קָדְשׁוֹ` (זכריה ב:יג)
`וָאֹמַר לֹא אֶרְעֶה אֶתְכֶם הַמֵּתָה תָמוּת וְהַנִּכְחֶדֶת תִּכָּחֵד וְהַנִּשְׁאָרוֹת תֹּאכַלְנָה אִשָּׁה אֶת בְּשַׂר רְעוּתָהּ` (זכריה יא:ט)
`כִּי הִנֵּה אָנֹכִי מֵקִים רֹעֶה בָּאָרֶץ הַנִּכְחָדוֹת לֹא יִפְקֹד הַנַּעַר לֹא יְבַקֵּשׁ וְהַנִּשְׁבֶּרֶת לֹא יְרַפֵּא הַנִּצָּבָה לֹא יְכַלְכֵּל וּבְשַׂר הַבְּרִיאָה יֹאכַל וּפַרְסֵיהֶן יְפָרֵק` (זכריה יא:טז)
`וְזֹאת תִּהְיֶה הַמַּגֵּפָה אֲשֶׁר יִגֹּף יְהוָה אֶת כָּל הָעַמִּים אֲשֶׁר צָבְאוּ עַל יְרוּשָׁלִָם הָמֵק בְּשָׂרוֹ וְהוּא עֹמֵד עַל רַגְלָיו וְעֵינָיו תִּמַּקְנָה בְחֹרֵיהֶן וּלְשׁוֹנוֹ תִּמַּק בְּפִיהֶם` (זכריה יד:יב)
`וַיִּקָּבְצוּ כָל יִשְׂרָאֵל אֶל דָּוִיד חֶבְרוֹנָה לֵאמֹר הִנֵּה עַצְמְךָ וּבְשָׂרְךָ אֲנָחְנוּ` (דברי-הימים-א יא:א)
`עִמּוֹ זְרוֹעַ בָּשָׂר וְעִמָּנוּ יְהוָה אֱלֹהֵינוּ לְעָזְרֵנוּ וּלְהִלָּחֵם מִלְחֲמֹתֵנוּ וַיִּסָּמְכוּ הָעָם עַל דִּבְרֵי יְחִזְקִיָּהוּ מֶלֶךְ יְהוּדָה` (דברי-הימים-ב לב:ח)
`לָכֵן שָׂמַח לִבִּי וַיָּגֶל כְּבוֹדִי אַף בְּשָׂרִי יִשְׁכֹּן לָבֶטַח` (תהילים טז:ט)
`בִּקְרֹב עָלַי מְרֵעִים לֶאֱכֹל אֶת בְּשָׂרִי צָרַי וְאֹיְבַי לִי הֵמָּה כָשְׁלוּ וְנָפָלוּ` (תהילים כז:ב)
`אֵין מְתֹם בִּבְשָׂרִי מִפְּנֵי זַעְמֶךָ אֵין שָׁלוֹם בַּעֲצָמַי מִפְּנֵי חַטָּאתִי` (תהילים לח:ג)
`כִּי כְסָלַי מָלְאוּ נִקְלֶה וְאֵין מְתֹם בִּבְשָׂרִי` (תהילים לח:ז)
`הַאוֹכַל בְּשַׂר אַבִּירִים וְדַם עַתּוּדִים אֶשְׁתֶּה` (תהילים נ:יג)
`בֵּאלֹהִים אֲהַלֵּל דְּבָרוֹ בֵּאלֹהִים בָּטַחְתִּי לֹא אִירָא מַה יַּעֲשֶׂה בָשָׂר לִי` (תהילים נו:ד)
`מִזְמוֹר לְדָוִד בִּהְיוֹתוֹ בְּמִדְבַּר יְהוּדָה אֱלֹהִים אֵלִי אַתָּה אֲ‍שַׁחֲרֶךָּ צָמְאָה לְךָ נַפְשִׁי כָּמַהּ לְךָ בְשָׂרִי בְּאֶרֶץ צִיָּה וְעָיֵף בְּלִי מָיִם` (תהילים סג:א)
`שֹׁמֵעַ תְּפִלָּה עָדֶיךָ כָּל בָּשָׂר יָבֹאוּ` (תהילים סה:ב)
`וַיִּזְכֹּר כִּי בָשָׂר הֵמָּה רוּחַ הוֹלֵךְ וְלֹא יָשׁוּב` (תהילים עח:לט)
`נָתְנוּ אֶת נִבְלַת עֲבָדֶיךָ מַאֲכָל לְעוֹף הַשָּׁמָיִם בְּשַׂר חֲסִידֶיךָ לְחַיְתוֹ אָרֶץ` (תהילים עט:ב)
`נִכְסְפָה וְגַם כָּלְתָה נַפְשִׁי לְחַצְרוֹת יְהוָה לִבִּי וּבְשָׂרִי יְרַנְּנוּ אֶל אֵל חָי` (תהילים פד:ב)
`מִקּוֹל אַנְחָתִי דָּבְקָה עַצְמִי לִבְשָׂרִי` (תהילים קב:ה)
`בִּרְכַּי כָּשְׁלוּ מִצּוֹם וּבְשָׂרִי כָּחַשׁ מִשָּׁמֶן` (תהילים קט:כד)
`סָמַר מִפַּחְדְּךָ בְשָׂרִי וּמִמִּשְׁפָּטֶיךָ יָרֵאתִי` (תהילים קיט:קכ)
`נֹתֵן לֶחֶם לְכָל בָּשָׂר כִּי לְעוֹלָם חַסְדּוֹ` (תהילים קלו:כה)
`תְּהִלַּת יְהוָה יְדַבֶּר פִּי וִיבָרֵךְ כָּל בָּשָׂר שֵׁם קָדְשׁוֹ לְעוֹלָם וָעֶד` (תהילים קמה:כא)
`אוּלָם שְׁלַח נָא יָדְךָ וְגַע אֶל עַצְמוֹ וְאֶל בְּשָׂרוֹ אִם לֹא אֶל פָּנֶיךָ יְבָרֲכֶךָּ` (איוב ב:ה)
`וְרוּחַ עַל פָּנַי יַחֲלֹף תְּסַמֵּר שַׂעֲרַת בְּשָׂרִי` (איוב ד:טו)
`אִם כֹּחַ אֲבָנִים כֹּחִי אִם בְּשָׂרִי נָחוּשׁ` (איוב ו:יב)
`לָבַשׁ בְּשָׂרִי רִמָּה וְגִישׁ עָפָר עוֹרִי רָגַע וַיִּמָּאֵס` (איוב ז:ה)
`הַעֵינֵי בָשָׂר לָךְ אִם כִּרְאוֹת אֱנוֹשׁ תִּרְאֶה` (איוב י:ד)
`עוֹר וּבָשָׂר תַּלְבִּישֵׁנִי וּבַעֲצָמוֹת וְגִידִים תְּסֹכְכֵנִי` (איוב י:יא)
`אֲשֶׁר בְּיָדוֹ נֶפֶשׁ כָּל חָי וְרוּחַ כָּל בְּשַׂר אִישׁ` (איוב יב:י)
`עַל מָה אֶשָּׂא בְשָׂרִי בְשִׁנָּי וְנַפְשִׁי אָשִׂים בְּכַפִּי` (איוב יג:יד)
`אַךְ בְּשָׂרוֹ עָלָיו יִכְאָב וְנַפְשׁוֹ עָלָיו תֶּאֱבָל` (איוב יד:כב)
`בְּעוֹרִי וּבִבְשָׂרִי דָּבְקָה עַצְמִי וָאֶתְמַלְּטָה בְּעוֹר שִׁנָּי` (איוב יט:כ)
`לָמָּה תִּרְדְּפֻנִי כְמוֹ אֵל וּמִבְּשָׂרִי לֹא תִשְׂבָּעוּ` (איוב יט:כב)
`וְאַחַר עוֹרִי נִקְּפוּ זֹאת וּמִבְּשָׂרִי אֶחֱזֶה אֱלוֹהַּ` (איוב יט:כו)
`וְאִם זָכַרְתִּי וְנִבְהָלְתִּי וְאָחַז בְּשָׂרִי פַּלָּצוּת` (איוב כא:ו)
`אִם לֹא אָמְרוּ מְתֵי אָהֳלִי מִי יִתֵּן מִבְּשָׂרוֹ לֹא נִשְׂבָּע` (איוב לא:לא)
`יִכֶל בְּשָׂרוֹ מֵרֹאִי וּשְׁפִי עַצְמוֹתָיו לֹא רֻאּוּ` (איוב לג:כא)
`רֻטֲפַשׁ בְּשָׂרוֹ מִנֹּעַר יָשׁוּב לִימֵי עֲלוּמָיו` (איוב לג:כה)
`יִגְוַע כָּל בָּשָׂר יָחַד וְאָדָם עַל עָפָר יָשׁוּב` (איוב לד:טו)
`מַפְּלֵי בְשָׂרוֹ דָבֵקוּ יָצוּק עָלָיו בַּל יִמּוֹט` (איוב מא:כג)
`כִּי חַיִּים הֵם לְמֹצְאֵיהֶם וּלְכָל בְּשָׂרוֹ מַרְפֵּא` (משלי ד:כב)
`וְנָהַמְתָּ בְאַחֲרִיתֶךָ בִּכְלוֹת בְּשָׂרְךָ וּשְׁאֵרֶךָ` (משלי ה:יא)
`חַיֵּי בְשָׂרִים לֵב מַרְפֵּא וּרְקַב עֲצָמוֹת קִנְאָה` (משלי יד:ל)
`אַל תְּהִי בְסֹבְאֵי יָיִן בְּזֹלֲלֵי בָשָׂר לָמוֹ` (משלי כג:כ)
`תַּרְתִּי בְלִבִּי לִמְשׁוֹךְ בַּיַּיִן אֶת בְּשָׂרִי וְלִבִּי נֹהֵג בַּחָכְמָה וְלֶאֱחֹז בְּסִכְלוּת עַד אֲשֶׁר אֶרְאֶה אֵי זֶה טוֹב לִבְנֵי הָאָדָם אֲשֶׁר יַעֲשׂוּ תַּחַת הַשָּׁמַיִם מִסְפַּר יְמֵי חַיֵּיהֶם` (קהלת ב:ג)
`הַכְּסִיל חֹבֵק אֶת יָדָיו וְאֹכֵל אֶת בְּשָׂרוֹ` (קהלת ד:ה)
`אַל תִּתֵּן אֶת פִּיךָ לַחֲטִיא אֶת בְּשָׂרֶךָ וְאַל תֹּאמַר לִפְנֵי הַמַּלְאָךְ כִּי שְׁגָגָה הִיא לָמָּה יִקְצֹף הָאֱלֹהִים עַל קוֹלֶךָ וְחִבֵּל אֶת מַעֲשֵׂה יָדֶיךָ` (קהלת ה:ו)
`וְהָסֵר כַּעַס מִלִּבֶּךָ וְהַעֲבֵר רָעָה מִבְּשָׂרֶךָ כִּי הַיַּלְדוּת וְהַשַּׁחֲרוּת הָבֶל` (קהלת יא:י)
`וְיֹתֵר מֵהֵמָּה בְּנִי הִזָּהֵר עֲשׂוֹת סְפָרִים הַרְבֵּה אֵין קֵץ וְלַהַג הַרְבֵּה יְגִעַת בָּשָׂר` (קהלת יב:יב)
`בִּלָּה בְשָׂרִי וְעוֹרִי שִׁבַּר עַצְמוֹתָי` (איכה ג:ד)
`וּמִקְצָת יָמִים עֲשָׂרָה נִרְאָה מַרְאֵיהֶם טוֹב וּבְרִיאֵי בָּשָׂר מִן כָּל הַיְלָדִים הָאֹכְלִים אֵת פַּתְבַּג הַמֶּלֶךְ` (דניאל א:טו)
`לֶחֶם חֲמֻדוֹת לֹא אָכַלְתִּי וּבָשָׂר וָיַיִן לֹא בָא אֶל פִּי וְסוֹךְ לֹא סָכְתִּי עַד מְלֹאת שְׁלֹשֶׁת שָׁבֻעִים יָמִים` (דניאל י:ג)
`וְעַתָּה כִּבְשַׂר אַחֵינוּ בְּשָׂרֵנוּ כִּבְנֵיהֶם בָּנֵינוּ וְהִנֵּה אֲנַחְנוּ כֹבְשִׁים אֶת בָּנֵינוּ וְאֶת בְּנֹתֵינוּ לַעֲבָדִים וְיֵשׁ מִבְּנֹתֵינוּ נִכְבָּשׁוֹת וְאֵין לְאֵל יָדֵנוּ וּשְׂדֹתֵינוּ וּכְרָמֵינוּ לַאֲחֵרִים` (נחמיה ה:ה)
