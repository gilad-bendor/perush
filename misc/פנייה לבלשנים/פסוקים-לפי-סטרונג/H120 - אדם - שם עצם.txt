`וַיֹּאמֶר אֱלֹהִים נַעֲשֶׂה אָדָם בְּצַלְמֵנוּ כִּדְמוּתֵנוּ וְיִרְדּוּ בִדְגַת הַיָּם וּבְעוֹף הַשָּׁמַיִם וּבַבְּהֵמָה וּבְכָל הָאָרֶץ וּבְכָל הָרֶמֶשׂ הָרֹמֵשׂ עַל הָאָרֶץ` (בראשית א:כו)
`וַיִּבְרָא אֱלֹהִים אֶת הָאָדָם בְּצַלְמוֹ בְּצֶלֶם אֱלֹהִים בָּרָא אֹתוֹ זָכָר וּנְקֵבָה בָּרָא אֹתָם` (בראשית א:כז)
`וְכֹל שִׂיחַ הַשָּׂדֶה טֶרֶם יִהְיֶה בָאָרֶץ וְכָל עֵשֶׂב הַשָּׂדֶה טֶרֶם יִצְמָח כִּי לֹא הִמְטִיר יְהוָה אֱלֹהִים עַל הָאָרֶץ וְאָדָם אַיִן לַעֲבֹד אֶת הָאֲדָמָה` (בראשית ב:ה)
`וַיִּיצֶר יְהוָה אֱלֹהִים אֶת הָאָדָם עָפָר מִן הָאֲדָמָה וַיִּפַּח בְּאַפָּיו נִשְׁמַת חַיִּים וַיְהִי הָאָדָם לְנֶפֶשׁ חַיָּה` (בראשית ב:ז)
`וַיִּטַּע יְהוָה אֱלֹהִים גַּן בְעֵדֶן מִקֶּדֶם וַיָּשֶׂם שָׁם אֶת הָאָדָם אֲשֶׁר יָצָר` (בראשית ב:ח)
`וַיִּקַּח יְהוָה אֱלֹהִים אֶת הָאָדָם וַיַּנִּחֵהוּ בְגַן עֵדֶן לְעָבְדָהּ וּלְשָׁמְרָהּ` (בראשית ב:טו)
`וַיְצַו יְהוָה אֱלֹהִים עַל הָאָדָם לֵאמֹר מִכֹּל עֵץ הַגָּן אָכֹל תֹּאכֵל` (בראשית ב:טז)
`וַיֹּאמֶר יְהוָה אֱלֹהִים לֹא טוֹב הֱיוֹת הָאָדָם לְבַדּוֹ אֶעֱשֶׂהּ לּוֹ עֵזֶר כְּנֶגְדּוֹ` (בראשית ב:יח)
`וַיִּצֶר יְהוָה אֱלֹהִים מִן הָאֲדָמָה כָּל חַיַּת הַשָּׂדֶה וְאֵת כָּל עוֹף הַשָּׁמַיִם וַיָּבֵא אֶל הָאָדָם לִרְאוֹת מַה יִּקְרָא לוֹ וְכֹל אֲשֶׁר יִקְרָא לוֹ הָאָדָם נֶפֶשׁ חַיָּה הוּא שְׁמוֹ` (בראשית ב:יט)
`וַיִּקְרָא הָאָדָם שֵׁמוֹת לְכָל הַבְּהֵמָה וּלְעוֹף הַשָּׁמַיִם וּלְכֹל חַיַּת הַשָּׂדֶה וּלְאָדָם לֹא מָצָא עֵזֶר כְּנֶגְדּוֹ` (בראשית ב:כ)
`וַיַּפֵּל יְהוָה אֱלֹהִים תַּרְדֵּמָה עַל הָאָדָם וַיִּישָׁן וַיִּקַּח אַחַת מִצַּלְעֹתָיו וַיִּסְגֹּר בָּשָׂר תַּחְתֶּנָּה` (בראשית ב:כא)
`וַיִּבֶן יְהוָה אֱלֹהִים אֶת הַצֵּלָע אֲשֶׁר לָקַח מִן הָאָדָם לְאִשָּׁה וַיְבִאֶהָ אֶל הָאָדָם` (בראשית ב:כב)
`וַיֹּאמֶר הָאָדָם זֹאת הַפַּעַם עֶצֶם מֵעֲצָמַי וּבָשָׂר מִבְּשָׂרִי לְזֹאת יִקָּרֵא אִשָּׁה כִּי מֵאִישׁ לֻקֳחָה זֹּאת` (בראשית ב:כג)
`וַיִּהְיוּ שְׁנֵיהֶם עֲרוּמִּים הָאָדָם וְאִשְׁתּוֹ וְלֹא יִתְבֹּשָׁשׁוּ` (בראשית ב:כה)
`וַיִּשְׁמְעוּ אֶת קוֹל יְהוָה אֱלֹהִים מִתְהַלֵּךְ בַּגָּן לְרוּחַ הַיּוֹם וַיִּתְחַבֵּא הָאָדָם וְאִשְׁתּוֹ מִפְּנֵי יְהוָה אֱלֹהִים בְּתוֹךְ עֵץ הַגָּן` (בראשית ג:ח)
`וַיִּקְרָא יְהוָה אֱלֹהִים אֶל הָאָדָם וַיֹּאמֶר לוֹ אַיֶּכָּה` (בראשית ג:ט)
`וַיֹּאמֶר הָאָדָם הָאִשָּׁה אֲשֶׁר נָתַתָּה עִמָּדִי הִוא נָתְנָה לִּי מִן הָעֵץ וָאֹכֵל` (בראשית ג:יב)
`וַיִּקְרָא הָאָדָם שֵׁם אִשְׁתּוֹ חַוָּה כִּי הִוא הָיְתָה אֵם כָּל חָי` (בראשית ג:כ)
`וַיֹּאמֶר יְהוָה אֱלֹהִים הֵן הָאָדָם הָיָה כְּאַחַד מִמֶּנּוּ לָדַעַת טוֹב וָרָע וְעַתָּה פֶּן יִשְׁלַח יָדוֹ וְלָקַח גַּם מֵעֵץ הַחַיִּים וְאָכַל וָחַי לְעֹלָם` (בראשית ג:כב)
`וַיְגָרֶשׁ אֶת הָאָדָם וַיַּשְׁכֵּן מִקֶּדֶם לְגַן עֵדֶן אֶת הַכְּרֻבִים וְאֵת לַהַט הַחֶרֶב הַמִּתְהַפֶּכֶת לִשְׁמֹר אֶת דֶּרֶךְ עֵץ הַחַיִּים` (בראשית ג:כד)
`וְהָאָדָם יָדַע אֶת חַוָּה אִשְׁתּוֹ וַתַּהַר וַתֵּלֶד אֶת קַיִן וַתֹּאמֶר קָנִיתִי אִישׁ אֶת יְהוָה` (בראשית ד:א)
`זֶה סֵפֶר תּוֹלְדֹת אָדָם בְּיוֹם בְּרֹא אֱלֹהִים אָדָם בִּדְמוּת אֱלֹהִים עָשָׂה אֹתוֹ` (בראשית ה:א)
`זָכָר וּנְקֵבָה בְּרָאָם וַיְבָרֶךְ אֹתָם וַיִּקְרָא אֶת שְׁמָם אָדָם בְּיוֹם הִבָּרְאָם` (בראשית ה:ב)
`וַיְהִי כִּי הֵחֵל הָאָדָם לָרֹב עַל פְּנֵי הָאֲדָמָה וּבָנוֹת יֻלְּדוּ לָהֶם` (בראשית ו:א)
`וַיִּרְאוּ בְנֵי הָאֱלֹהִים אֶת בְּנוֹת הָאָדָם כִּי טֹבֹת הֵנָּה וַיִּקְחוּ לָהֶם נָשִׁים מִכֹּל אֲשֶׁר בָּחָרוּ` (בראשית ו:ב)
`וַיֹּאמֶר יְהוָה לֹא יָדוֹן רוּחִי בָאָדָם לְעֹלָם בְּשַׁגַּם הוּא בָשָׂר וְהָיוּ יָמָיו מֵאָה וְעֶשְׂרִים שָׁנָה` (בראשית ו:ג)
`הַנְּפִלִים הָיוּ בָאָרֶץ בַּיָּמִים הָהֵם וְגַם אַחֲרֵי כֵן אֲשֶׁר יָבֹאוּ בְּנֵי הָאֱלֹהִים אֶל בְּנוֹת הָאָדָם וְיָלְדוּ לָהֶם הֵמָּה הַגִּבֹּרִים אֲשֶׁר מֵעוֹלָם אַנְשֵׁי הַשֵּׁם` (בראשית ו:ד)
`וַיַּרְא יְהוָה כִּי רַבָּה רָעַת הָאָדָם בָּאָרֶץ וְכָל יֵצֶר מַחְשְׁבֹת לִבּוֹ רַק רַע כָּל הַיּוֹם` (בראשית ו:ה)
`וַיִּנָּחֶם יְהוָה כִּי עָשָׂה אֶת הָאָדָם בָּאָרֶץ וַיִּתְעַצֵּב אֶל לִבּוֹ` (בראשית ו:ו)
`וַיֹּאמֶר יְהוָה אֶמְחֶה אֶת הָאָדָם אֲשֶׁר בָּרָאתִי מֵעַל פְּנֵי הָאֲדָמָה מֵאָדָם עַד בְּהֵמָה עַד רֶמֶשׂ וְעַד עוֹף הַשָּׁמָיִם כִּי נִחַמְתִּי כִּי עֲשִׂיתִם` (בראשית ו:ז)
`וַיִּגְוַע כָּל בָּשָׂר הָרֹמֵשׂ עַל הָאָרֶץ בָּעוֹף וּבַבְּהֵמָה וּבַחַיָּה וּבְכָל הַשֶּׁרֶץ הַשֹּׁרֵץ עַל הָאָרֶץ וְכֹל הָאָדָם` (בראשית ז:כא)
`וַיִּמַח אֶת כָּל הַיְקוּם אֲשֶׁר עַל פְּנֵי הָאֲדָמָה מֵאָדָם עַד בְּהֵמָה עַד רֶמֶשׂ וְעַד עוֹף הַשָּׁמַיִם וַיִּמָּחוּ מִן הָאָרֶץ וַיִשָּׁאֶר אַךְ נֹחַ וַאֲשֶׁר אִתּוֹ בַּתֵּבָה` (בראשית ז:כג)
`וַיָּרַח יְהוָה אֶת רֵיחַ הַנִּיחֹחַ וַיֹּאמֶר יְהוָה אֶל לִבּוֹ לֹא אֹסִף לְקַלֵּל עוֹד אֶת הָאֲדָמָה בַּעֲבוּר הָאָדָם כִּי יֵצֶר לֵב הָאָדָם רַע מִנְּעֻרָיו וְלֹא אֹסִף עוֹד לְהַכּוֹת אֶת כָּל חַי כַּאֲשֶׁר עָשִׂיתִי` (בראשית ח:כא)
`וְאַךְ אֶת דִּמְכֶם לְנַפְשֹׁתֵיכֶם אֶדְרֹשׁ מִיַּד כָּל חַיָּה אֶדְרְשֶׁנּוּ וּמִיַּד הָאָדָם מִיַּד אִישׁ אָחִיו אֶדְרֹשׁ אֶת נֶפֶשׁ הָאָדָם` (בראשית ט:ה)
`שֹׁפֵךְ דַּם הָאָדָם בָּאָדָם דָּמוֹ יִשָּׁפֵךְ כִּי בְּצֶלֶם אֱלֹהִים עָשָׂה אֶת הָאָדָם` (בראשית ט:ו)
`וַיֵּרֶד יְהוָה לִרְאֹת אֶת הָעִיר וְאֶת הַמִּגְדָּל אֲשֶׁר בָּנוּ בְּנֵי הָאָדָם` (בראשית יא:ה)
`וְהוּא יִהְיֶה פֶּרֶא אָדָם יָדוֹ בַכֹּל וְיַד כֹּל בּוֹ וְעַל פְּנֵי כָל אֶחָיו יִשְׁכֹּן` (בראשית טז:יב)
`וַיֹּאמֶר יְהוָה אֵלָיו מִי שָׂם פֶּה לָאָדָם אוֹ מִי יָשׂוּם אִלֵּם אוֹ חֵרֵשׁ אוֹ פִקֵּחַ אוֹ עִוֵּר הֲלֹא אָנֹכִי יְהוָה` (שמות ד:יא)
`וַיַּעֲשׂוּ כֵן וַיֵּט אַהֲרֹן אֶת יָדוֹ בְמַטֵּהוּ וַיַּךְ אֶת עֲפַר הָאָרֶץ וַתְּהִי הַכִּנָּם בָּאָדָם וּבַבְּהֵמָה כָּל עֲפַר הָאָרֶץ הָיָה כִנִּים בְּכָל אֶרֶץ מִצְרָיִם` (שמות ח:יז)
`וַיַּעֲשׂוּ כֵן הַחַרְטֻמִּים בְּלָטֵיהֶם לְהוֹצִיא אֶת הַכִּנִּים וְלֹא יָכֹלוּ וַתְּהִי הַכִּנָּם בָּאָדָם וּבַבְּהֵמָה` (שמות ח:יח)
`וְהָיָה לְאָבָק עַל כָּל אֶרֶץ מִצְרָיִם וְהָיָה עַל הָאָדָם וְעַל הַבְּהֵמָה לִשְׁחִין פֹּרֵחַ אֲבַעְבֻּעֹת בְּכָל אֶרֶץ מִצְרָיִם` (שמות ט:ט)
`וַיִּקְחוּ אֶת פִּיחַ הַכִּבְשָׁן וַיַּעַמְדוּ לִפְנֵי פַרְעֹה וַיִּזְרֹק אֹתוֹ מֹשֶׁה הַשָּׁמָיְמָה וַיְהִי שְׁחִין אֲבַעְבֻּעֹת פֹּרֵחַ בָּאָדָם וּבַבְּהֵמָה` (שמות ט:י)
`וְעַתָּה שְׁלַח הָעֵז אֶת מִקְנְךָ וְאֵת כָּל אֲשֶׁר לְךָ בַּשָּׂדֶה כָּל הָאָדָם וְהַבְּהֵמָה אֲשֶׁר יִמָּצֵא בַשָּׂדֶה וְלֹא יֵאָסֵף הַבַּיְתָה וְיָרַד עֲלֵהֶם הַבָּרָד וָמֵתוּ` (שמות ט:יט)
`וַיֹּאמֶר יְהוָה אֶל מֹשֶׁה נְטֵה אֶת יָדְךָ עַל הַשָּׁמַיִם וִיהִי בָרָד בְּכָל אֶרֶץ מִצְרָיִם עַל הָאָדָם וְעַל הַבְּהֵמָה וְעַל כָּל עֵשֶׂב הַשָּׂדֶה בְּאֶרֶץ מִצְרָיִם` (שמות ט:כב)
`וַיַּךְ הַבָּרָד בְּכָל אֶרֶץ מִצְרַיִם אֵת כָּל אֲשֶׁר בַּשָּׂדֶה מֵאָדָם וְעַד בְּהֵמָה וְאֵת כָּל עֵשֶׂב הַשָּׂדֶה הִכָּה הַבָּרָד וְאֶת כָּל עֵץ הַשָּׂדֶה שִׁבֵּר` (שמות ט:כה)
`וְעָבַרְתִּי בְאֶרֶץ מִצְרַיִם בַּלַּיְלָה הַזֶּה וְהִכֵּיתִי כָל בְּכוֹר בְּאֶרֶץ מִצְרַיִם מֵאָדָם וְעַד בְּהֵמָה וּבְכָל אֱלֹהֵי מִצְרַיִם אֶעֱשֶׂה שְׁפָטִים אֲנִי יְהוָה` (שמות יב:יב)
`קַדֶּשׁ לִי כָל בְּכוֹר פֶּטֶר כָּל רֶחֶם בִּבְנֵי יִשְׂרָאֵל בָּאָדָם וּבַבְּהֵמָה לִי הוּא` (שמות יג:ב)
`וְכָל פֶּטֶר חֲמֹר תִּפְדֶּה בְשֶׂה וְאִם לֹא תִפְדֶּה וַעֲרַפְתּוֹ וְכֹל בְּכוֹר אָדָם בְּבָנֶיךָ תִּפְדֶּה` (שמות יג:יג)
`וַיְהִי כִּי הִקְשָׁה פַרְעֹה לְשַׁלְּחֵנוּ וַיַּהֲרֹג יְהֹוָה כָּל בְּכוֹר בְּאֶרֶץ מִצְרַיִם מִבְּכֹר אָדָם וְעַד בְּכוֹר בְּהֵמָה עַל כֵּן אֲנִי זֹבֵחַ לַיהוָה כָּל פֶּטֶר רֶחֶם הַזְּכָרִים וְכָל בְּכוֹר בָּנַי אֶפְדֶּה` (שמות יג:טו)
`עַל בְּשַׂר אָדָם לֹא יִיסָךְ וּבְמַתְכֻּנְתּוֹ לֹא תַעֲשׂוּ כָּמֹהוּ קֹדֶשׁ הוּא קֹדֶשׁ יִהְיֶה לָכֶם` (שמות ל:לב)
`וַיֹּאמֶר לֹא תוּכַל לִרְאֹת אֶת פָּנָי כִּי לֹא יִרְאַנִי הָאָדָם וָחָי` (שמות לג:כ)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אָדָם כִּי יַקְרִיב מִכֶּם קָרְבָּן לַיהוָה מִן הַבְּהֵמָה מִן הַבָּקָר וּמִן הַצֹּאן תַּקְרִיבוּ אֶת קָרְבַּנְכֶם` (ויקרא א:ב)
`אוֹ כִי יִגַּע בְּטֻמְאַת אָדָם לְכֹל טֻמְאָתוֹ אֲשֶׁר יִטְמָא בָּהּ וְנֶעְלַם מִמֶּנּוּ וְהוּא יָדַע וְאָשֵׁם` (ויקרא ה:ג)
`אוֹ נֶפֶשׁ כִּי תִשָּׁבַע לְבַטֵּא בִשְׂפָתַיִם לְהָרַע אוֹ לְהֵיטִיב לְכֹל אֲשֶׁר יְבַטֵּא הָאָדָם בִּשְׁבֻעָה וְנֶעְלַם מִמֶּנּוּ וְהוּא יָדַע וְאָשֵׁם לְאַחַת מֵאֵלֶּה` (ויקרא ה:ד)
`אוֹ מָצָא אֲבֵדָה וְכִחֶשׁ בָּהּ וְנִשְׁבַּע עַל שָׁקֶר עַל אַחַת מִכֹּל אֲשֶׁר יַעֲשֶׂה הָאָדָם לַחֲטֹא בָהֵנָּה` (ויקרא ו:ג)
`וְנֶפֶשׁ כִּי תִגַּע בְּכָל טָמֵא בְּטֻמְאַת אָדָם אוֹ בִּבְהֵמָה טְמֵאָה אוֹ בְּכָל שֶׁקֶץ טָמֵא וְאָכַל מִבְּשַׂר זֶבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ` (ויקרא ז:כא)
`אָדָם כִּי יִהְיֶה בְעוֹר בְּשָׂרוֹ שְׂאֵת אוֹ סַפַּחַת אוֹ בַהֶרֶת וְהָיָה בְעוֹר בְּשָׂרוֹ לְנֶגַע צָרָעַת וְהוּבָא אֶל אַהֲרֹן הַכֹּהֵן אוֹ אֶל אַחַד מִבָּנָיו הַכֹּהֲנִים` (ויקרא יג:ב)
`נֶגַע צָרַעַת כִּי תִהְיֶה בְּאָדָם וְהוּבָא אֶל הַכֹּהֵן` (ויקרא יג:ט)
`וְכָל אָדָם לֹא יִהְיֶה בְּאֹהֶל מוֹעֵד בְּבֹאוֹ לְכַפֵּר בַּקֹּדֶשׁ עַד צֵאתוֹ וְכִפֶּר בַּעֲדוֹ וּבְעַד בֵּיתוֹ וּבְעַד כָּל קְהַל יִשְׂרָאֵל` (ויקרא טז:יז)
`וּשְׁמַרְתֶּם אֶת חֻקֹּתַי וְאֶת מִשְׁפָּטַי אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם וָחַי בָּהֶם אֲנִי יְהוָה` (ויקרא יח:ה)
`אוֹ אִישׁ אֲשֶׁר יִגַּע בְּכָל שֶׁרֶץ אֲשֶׁר יִטְמָא לוֹ אוֹ בְאָדָם אֲשֶׁר יִטְמָא לוֹ לְכֹל טֻמְאָתוֹ` (ויקרא כב:ה)
`וְאִישׁ כִּי יַכֶּה כָּל נֶפֶשׁ אָדָם מוֹת יוּמָת` (ויקרא כד:יז)
`שֶׁבֶר תַּחַת שֶׁבֶר עַיִן תַּחַת עַיִן שֵׁן תַּחַת שֵׁן כַּאֲשֶׁר יִתֵּן מוּם בָּאָדָם כֵּן יִנָּתֶן בּוֹ` (ויקרא כד:כ)
`וּמַכֵּה בְהֵמָה יְשַׁלְּמֶנָּה וּמַכֵּה אָדָם יוּמָת` (ויקרא כד:כא)
`אַךְ כָּל חֵרֶם אֲשֶׁר יַחֲרִם אִישׁ לַיהוָה מִכָּל אֲשֶׁר לוֹ מֵאָדָם וּבְהֵמָה וּמִשְּׂדֵה אֲחֻזָּתוֹ לֹא יִמָּכֵר וְלֹא יִגָּאֵל כָּל חֵרֶם קֹדֶשׁ קָדָשִׁים הוּא לַיהוָה` (ויקרא כז:כח)
`כָּל חֵרֶם אֲשֶׁר יָחֳרַם מִן הָאָדָם לֹא יִפָּדֶה מוֹת יוּמָת` (ויקרא כז:כט)
`כִּי לִי כָּל בְּכוֹר בְּיוֹם הַכֹּתִי כָל בְּכוֹר בְּאֶרֶץ מִצְרַיִם הִקְדַּשְׁתִּי לִי כָל בְּכוֹר בְּיִשְׂרָאֵל מֵאָדָם עַד בְּהֵמָה לִי יִהְיוּ אֲנִי יְהוָה` (במדבר ג:יג)
`דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל אִישׁ אוֹ אִשָּׁה כִּי יַעֲשׂוּ מִכָּל חַטֹּאת הָאָדָם לִמְעֹל מַעַל בַּיהוָה וְאָשְׁמָה הַנֶּפֶשׁ הַהִוא` (במדבר ה:ו)
`כִּי לִי כָל בְּכוֹר בִּבְנֵי יִשְׂרָאֵל בָּאָדָם וּבַבְּהֵמָה בְּיוֹם הַכֹּתִי כָל בְּכוֹר בְּאֶרֶץ מִצְרַיִם הִקְדַּשְׁתִּי אֹתָם לִי` (במדבר ח:יז)
`וַיְהִי אֲנָשִׁים אֲשֶׁר הָיוּ טְמֵאִים לְנֶפֶשׁ אָדָם וְלֹא יָכְלוּ לַעֲשֹׂת הַפֶּסַח בַּיּוֹם הַהוּא וַיִּקְרְבוּ לִפְנֵי מֹשֶׁה וְלִפְנֵי אַהֲרֹן בַּיּוֹם הַהוּא` (במדבר ט:ו)
`וַיֹּאמְרוּ הָאֲנָשִׁים הָהֵמָּה אֵלָיו אֲנַחְנוּ טְמֵאִים לְנֶפֶשׁ אָדָם לָמָּה נִגָּרַע לְבִלְתִּי הַקְרִב אֶת קָרְבַּן יְהוָה בְּמֹעֲדוֹ בְּתוֹךְ בְּנֵי יִשְׂרָאֵל` (במדבר ט:ז)
`וְהָאִישׁ מֹשֶׁה עָנָו מְאֹד מִכֹּל הָאָדָם אֲשֶׁר עַל פְּנֵי הָאֲדָמָה` (במדבר יב:ג)
`אִם כְּמוֹת כָּל הָאָדָם יְמֻתוּן אֵלֶּה וּפְקֻדַּת כָּל הָאָדָם יִפָּקֵד עֲלֵיהֶם לֹא יְהוָה שְׁלָחָנִי` (במדבר טז:כט)
`וַתִּפְתַּח הָאָרֶץ אֶת פִּיהָ וַתִּבְלַע אֹתָם וְאֶת בָּתֵּיהֶם וְאֵת כָּל הָאָדָם אֲשֶׁר לְקֹרַח וְאֵת כָּל הָרֲכוּשׁ` (במדבר טז:לב)
`כָּל פֶּטֶר רֶחֶם לְכָל בָּשָׂר אֲשֶׁר יַקְרִיבוּ לַיהוָה בָּאָדָם וּבַבְּהֵמָה יִהְיֶה לָּךְ אַךְ פָּדֹה תִפְדֶּה אֵת בְּכוֹר הָאָדָם וְאֵת בְּכוֹר הַבְּהֵמָה הַטְּמֵאָה תִּפְדֶּה` (במדבר יח:טו)
`הַנֹּגֵעַ בְּמֵת לְכָל נֶפֶשׁ אָדָם וְטָמֵא שִׁבְעַת יָמִים` (במדבר יט:יא)
`כָּל הַנֹּגֵעַ בְּמֵת בְּנֶפֶשׁ הָאָדָם אֲשֶׁר יָמוּת וְלֹא יִתְחַטָּא אֶת מִשְׁכַּן יְהוָה טִמֵּא וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִיִּשְׂרָאֵל כִּי מֵי נִדָּה לֹא זֹרַק עָלָיו טָמֵא יִהְיֶה עוֹד טֻמְאָתוֹ בוֹ` (במדבר יט:יג)
`זֹאת הַתּוֹרָה אָדָם כִּי יָמוּת בְּאֹהֶל כָּל הַבָּא אֶל הָאֹהֶל וְכָל אֲשֶׁר בָּאֹהֶל יִטְמָא שִׁבְעַת יָמִים` (במדבר יט:יד)
`וְכֹל אֲשֶׁר יִגַּע עַל פְּנֵי הַשָּׂדֶה בַּחֲלַל חֶרֶב אוֹ בְמֵת אוֹ בְעֶצֶם אָדָם אוֹ בְקָבֶר יִטְמָא שִׁבְעַת יָמִים` (במדבר יט:טז)
`לֹא אִישׁ אֵל וִיכַזֵּב וּבֶן אָדָם וְיִתְנֶחָם הַהוּא אָמַר וְלֹא יַעֲשֶׂה וְדִבֶּר וְלֹא יְקִימֶנָּה` (במדבר כג:יט)
`וַיִּקְחוּ אֶת כָּל הַשָּׁלָל וְאֵת כָּל הַמַּלְקוֹחַ בָּאָדָם וּבַבְּהֵמָה` (במדבר לא:יא)
`שָׂא אֵת רֹאשׁ מַלְקוֹחַ הַשְּׁבִי בָּאָדָם וּבַבְּהֵמָה אַתָּה וְאֶלְעָזָר הַכֹּהֵן וְרָאשֵׁי אֲבוֹת הָעֵדָה` (במדבר לא:כו)
`וַהֲרֵמֹתָ מֶכֶס לַיהוָה מֵאֵת אַנְשֵׁי הַמִּלְחָמָה הַיֹּצְאִים לַצָּבָא אֶחָד נֶפֶשׁ מֵחֲמֵשׁ הַמֵּאוֹת מִן הָאָדָם וּמִן הַבָּקָר וּמִן הַחֲמֹרִים וּמִן הַצֹּאן` (במדבר לא:כח)
`וּמִמַּחֲצִת בְּנֵי יִשְׂרָאֵל תִּקַּח אֶחָד אָחֻז מִן הַחֲמִשִּׁים מִן הָאָדָם מִן הַבָּקָר מִן הַחֲמֹרִים וּמִן הַצֹּאן מִכָּל הַבְּהֵמָה וְנָתַתָּה אֹתָם לַלְוִיִּם שֹׁמְרֵי מִשְׁמֶרֶת מִשְׁכַּן יְהוָה` (במדבר לא:ל)
`וְנֶפֶשׁ אָדָם מִן הַנָּשִׁים אֲשֶׁר לֹא יָדְעוּ מִשְׁכַּב זָכָר כָּל נֶפֶשׁ שְׁנַיִם וּשְׁלֹשִׁים אָלֶף` (במדבר לא:לה)
`וְנֶפֶשׁ אָדָם שִׁשָּׁה עָשָׂר אָלֶף וּמִכְסָם לַיהוָה שְׁנַיִם וּשְׁלֹשִׁים נָפֶשׁ` (במדבר לא:מ)
`וְנֶפֶשׁ אָדָם שִׁשָּׁה עָשָׂר אָלֶף` (במדבר לא:מו)

`עֵינֵי גַּבְהוּת אָדָם שָׁפֵל וְשַׁח רוּם אֲנָשִׁים וְנִשְׂגַּב יְהוָה לְבַדּוֹ בַּיּוֹם הַהוּא` (ישעיהו ב:יא)
`וְשַׁח גַּבְהוּת הָאָדָם וְשָׁפֵל רוּם אֲנָשִׁים וְנִשְׂגַּב יְהוָה לְבַדּוֹ בַּיּוֹם הַהוּא` (ישעיהו ב:יז)
`בַּיּוֹם הַהוּא יַשְׁלִיךְ הָאָדָם אֵת אֱלִילֵי כַסְפּוֹ וְאֵת אֱלִילֵי זְהָבוֹ אֲשֶׁר עָשׂוּ לוֹ לְהִשְׁתַּחֲוֺת לַחְפֹּר פֵּרוֹת וְלָעֲטַלֵּפִים` (ישעיהו ב:כ)
`חִדְלוּ לָכֶם מִן הָאָדָם אֲשֶׁר נְשָׁמָה בְּאַפּוֹ כִּי בַמֶּה נֶחְשָׁב הוּא` (ישעיהו ב:כב)
`וַיִּשַּׁח אָדָם וַיִּשְׁפַּל אִישׁ וְעֵינֵי גְבֹהִים תִּשְׁפַּלְנָה` (ישעיהו ה:טו)
`וָאֹמַר עַד מָתַי אֲדֹנָי וַיֹּאמֶר עַד אֲשֶׁר אִם שָׁאוּ עָרִים מֵאֵין יוֹשֵׁב וּבָתִּים מֵאֵין אָדָם וְהָאֲדָמָה תִּשָּׁאֶה שְׁמָמָה` (ישעיהו ו:יא)
`וְרִחַק יְהוָה אֶת הָאָדָם וְרַבָּה הָעֲזוּבָה בְּקֶרֶב הָאָרֶץ` (ישעיהו ו:יב)
`אוֹקִיר אֱנוֹשׁ מִפָּז וְאָדָם מִכֶּתֶם אוֹפִיר` (ישעיהו יג:יב)
`בַּיּוֹם הַהוּא יִשְׁעֶה הָאָדָם עַל עֹשֵׂהוּ וְעֵינָיו אֶל קְדוֹשׁ יִשְׂרָאֵל תִּרְאֶינָה` (ישעיהו יז:ז)
`וְעֵילָם נָשָׂא אַשְׁפָּה בְּרֶכֶב אָדָם פָּרָשִׁים וְקִיר עֵרָה מָגֵן` (ישעיהו כב:ו)
`וְיָסְפוּ עֲנָוִים בַּיהוָה שִׂמְחָה וְאֶבְיוֹנֵי אָדָם בִּקְדוֹשׁ יִשְׂרָאֵל יָגִילוּ` (ישעיהו כט:יט)
`מַחֲטִיאֵי אָדָם בְּדָבָר וְלַמּוֹכִיחַ בַּשַּׁעַר יְקֹשׁוּן וַיַּטּוּ בַתֹּהוּ צַדִּיק` (ישעיהו כט:כא)
`וּמִצְרַיִם אָדָם וְלֹא אֵל וְסוּסֵיהֶם בָּשָׂר וְלֹא רוּחַ וַיהוָה יַטֶּה יָדוֹ וְכָשַׁל עוֹזֵר וְנָפַל עָזֻר וְיַחְדָּו כֻּלָּם יִכְלָיוּן` (ישעיהו לא:ג)
`וְנָפַל אַשּׁוּר בְּחֶרֶב לֹא אִישׁ וְחֶרֶב לֹא אָדָם תֹּאכֲלֶנּוּ וְנָס לוֹ מִפְּנֵי חֶרֶב וּבַחוּרָיו לָמַס יִהְיוּ` (ישעיהו לא:ח)
`וְנָתֹן אֶת אֱלֹהֵיהֶם בָּאֵשׁ כִּי לֹא אֱלֹהִים הֵמָּה כִּי אִם מַעֲשֵׂה יְדֵי אָדָם עֵץ וָאֶבֶן וַיְאַבְּדוּם` (ישעיהו לז:יט)
`אָמַרְתִּי לֹא אֶרְאֶה יָהּ יָהּ בְּאֶרֶץ הַחַיִּים לֹא אַבִּיט אָדָם עוֹד עִם יוֹשְׁבֵי חָדֶל` (ישעיהו לח:יא)
`מֵאֲשֶׁר יָקַרְתָּ בְעֵינַי נִכְבַּדְתָּ וַאֲנִי אֲהַבְתִּיךָ וְאֶתֵּן אָדָם תַּחְתֶּיךָ וּלְאֻמִּים תַּחַת נַפְשֶׁךָ` (ישעיהו מג:ד)
`הֵן כָּל חֲבֵרָיו יֵבֹשׁוּ וְחָרָשִׁים הֵמָּה מֵאָדָם יִתְקַבְּצוּ כֻלָּם יַעֲמֹדוּ יִפְחֲדוּ יֵבֹשׁוּ יָחַד` (ישעיהו מד:יא)
`חָרַשׁ עֵצִים נָטָה קָו יְתָאֲרֵהוּ בַשֶּׂרֶד יַעֲשֵׂהוּ בַּמַּקְצֻעוֹת וּבַמְּחוּגָה יְתָאֳרֵהוּ וַיַּעֲשֵׂהוּ כְּתַבְנִית אִישׁ כְּתִפְאֶרֶת אָדָם לָשֶׁבֶת בָּיִת` (ישעיהו מד:יג)
`וְהָיָה לְאָדָם לְבָעֵר וַיִּקַּח מֵהֶם וַיָּחָם אַף יַשִּׂיק וְאָפָה לָחֶם אַף יִפְעַל אֵל וַיִּשְׁתָּחוּ עָשָׂהוּ פֶסֶל וַיִּסְגָּד לָמוֹ` (ישעיהו מד:טו)
`אָנֹכִי עָשִׂיתִי אֶרֶץ וְאָדָם עָלֶיהָ בָרָאתִי אֲנִי יָדַי נָטוּ שָׁמַיִם וְכָל צְבָאָם צִוֵּיתִי` (ישעיהו מה:יב)
`תִּגָּל עֶרְוָתֵךְ גַּם תֵּרָאֶה חֶרְפָּתֵךְ נָקָם אֶקָּח וְלֹא אֶפְגַּע אָדָם` (ישעיהו מז:ג)
`אָנֹכִי אָנֹכִי הוּא מְנַחֶמְכֶם מִי אַתְּ וַתִּירְאִי מֵאֱנוֹשׁ יָמוּת וּמִבֶּן אָדָם חָצִיר יִנָּתֵן` (ישעיהו נא:יב)
`כַּאֲשֶׁר שָׁמְמוּ עָלֶיךָ רַבִּים כֵּן מִשְׁחַת מֵאִישׁ מַרְאֵהוּ וְתֹאֲרוֹ מִבְּנֵי אָדָם` (ישעיהו נב:יד)
`אַשְׁרֵי אֱנוֹשׁ יַעֲשֶׂה זֹּאת וּבֶן אָדָם יַחֲזִיק בָּהּ שֹׁמֵר שַׁבָּת מֵחַלְּלוֹ וְשֹׁמֵר יָדוֹ מֵעֲשׂוֹת כָּל רָע` (ישעיהו נו:ב)
`הֲכָזֶה יִהְיֶה צוֹם אֶבְחָרֵהוּ יוֹם עַנּוֹת אָדָם נַפְשׁוֹ הֲלָכֹף כְּאַגְמֹן רֹאשׁוֹ וְשַׂק וָאֵפֶר יַצִּיעַ הֲלָזֶה תִּקְרָא צוֹם וְיוֹם רָצוֹן לַיהוָה` (ישעיהו נח:ה)
`וְלֹא אָמְרוּ אַיֵּה יְהוָה הַמַּעֲלֶה אֹתָנוּ מֵאֶרֶץ מִצְרָיִם הַמּוֹלִיךְ אֹתָנוּ בַּמִּדְבָּר בְּאֶרֶץ עֲרָבָה וְשׁוּחָה בְּאֶרֶץ צִיָּה וְצַלְמָוֶת בְּאֶרֶץ לֹא עָבַר בָּהּ אִישׁ וְלֹא יָשַׁב אָדָם שָׁם` (ירמיהו ב:ו)
`רָאִיתִי וְהִנֵּה אֵין הָאָדָם וְכָל עוֹף הַשָּׁמַיִם נָדָדוּ` (ירמיהו ד:כה)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהֹוִה הִנֵּה אַפִּי וַחֲמָתִי נִתֶּכֶת אֶל הַמָּקוֹם הַזֶּה עַל הָאָדָם וְעַל הַבְּהֵמָה וְעַל עֵץ הַשָּׂדֶה וְעַל פְּרִי הָאֲדָמָה וּבָעֲרָה וְלֹא תִכְבֶּה` (ירמיהו ז:כ)
`דַּבֵּר כֹּה נְאֻם יְהוָה וְנָפְלָה נִבְלַת הָאָדָם כְּדֹמֶן עַל פְּנֵי הַשָּׂדֶה וּכְעָמִיר מֵאַחֲרֵי הַקֹּצֵר וְאֵין מְאַסֵּף` (ירמיהו ט:כב)
`נִבְעַר כָּל אָדָם מִדַּעַת הֹבִישׁ כָּל צוֹרֵף מִפָּסֶל כִּי שֶׁקֶר נִסְכּוֹ וְלֹא רוּחַ בָּם` (ירמיהו י:יד)
`יָדַעְתִּי יְהוָה כִּי לֹא לָאָדָם דַּרְכּוֹ לֹא לְאִישׁ הֹלֵךְ וְהָכִין אֶת צַעֲדוֹ` (ירמיהו י:כג)
`הֲיַעֲשֶׂה לּוֹ אָדָם אֱלֹהִים וְהֵמָּה לֹא אֱלֹהִים` (ירמיהו טז:כ)
`כֹּה אָמַר יְהוָה אָרוּר הַגֶּבֶר אֲשֶׁר יִבְטַח בָּאָדָם וְשָׂם בָּשָׂר זְרֹעוֹ וּמִן יְהוָה יָסוּר לִבּוֹ` (ירמיהו יז:ה)
`וְהִכֵּיתִי אֶת יוֹשְׁבֵי הָעִיר הַזֹּאת וְאֶת הָאָדָם וְאֶת הַבְּהֵמָה בְּדֶבֶר גָּדוֹל יָמֻתוּ` (ירמיהו כא:ו)
`אָנֹכִי עָשִׂיתִי אֶת הָאָרֶץ אֶת הָאָדָם וְאֶת הַבְּהֵמָה אֲשֶׁר עַל פְּנֵי הָאָרֶץ בְּכֹחִי הַגָּדוֹל וּבִזְרוֹעִי הַנְּטוּיָה וּנְתַתִּיהָ לַאֲשֶׁר יָשַׁר בְּעֵינָי` (ירמיהו כז:ה)
`הִנֵּה יָמִים בָּאִים נְאֻם יְהוָה וְזָרַעְתִּי אֶת בֵּית יִשְׂרָאֵל וְאֶת בֵּית יְהוּדָה זֶרַע אָדָם וְזֶרַע בְּהֵמָה` (ירמיהו לא:כז)
`כִּי אִם אִישׁ בַּעֲוֺנוֹ יָמוּת כָּל הָאָדָם הָאֹכֵל הַבֹּסֶר תִּקְהֶינָה שִׁנָּיו` (ירמיהו לא:ל)
`גְּדֹל הָעֵצָה וְרַב הָעֲלִילִיָּה אֲשֶׁר עֵינֶיךָ פְקֻחוֹת עַל כָּל דַּרְכֵי בְּנֵי אָדָם לָתֵת לְאִישׁ כִּדְרָכָיו וְכִפְרִי מַעֲלָלָיו` (ירמיהו לב:יט)
`אֲשֶׁר שַׂמְתָּ אֹתוֹת וּמֹפְתִים בְּאֶרֶץ מִצְרַיִם עַד הַיּוֹם הַזֶּה וּבְיִשְׂרָאֵל וּבָאָדָם וַתַּעֲשֶׂה לְּךָ שֵׁם כַּיּוֹם הַזֶּה` (ירמיהו לב:כ)
`וְנִקְנָה הַשָּׂדֶה בָּאָרֶץ הַזֹּאת אֲשֶׁר אַתֶּם אֹמְרִים שְׁמָמָה הִיא מֵאֵין אָדָם וּבְהֵמָה נִתְּנָה בְּיַד הַכַּשְׂדִּים` (ירמיהו לב:מג)
`בָּאִים לְהִלָּחֵם אֶת הַכַּשְׂדִּים וּלְמַלְאָם אֶת פִּגְרֵי הָאָדָם אֲשֶׁר הִכֵּיתִי בְאַפִּי וּבַחֲמָתִי וַאֲשֶׁר הִסְתַּרְתִּי פָנַי מֵהָעִיר הַזֹּאת עַל כָּל רָעָתָם` (ירמיהו לג:ה)
`כֹּה אָמַר יְהוָה עוֹד יִשָּׁמַע בַּמָּקוֹם הַזֶּה אֲשֶׁר אַתֶּם אֹמְרִים חָרֵב הוּא מֵאֵין אָדָם וּמֵאֵין בְּהֵמָה בְּעָרֵי יְהוּדָה וּבְחֻצוֹת יְרוּשָׁלִַם הַנְשַׁמּוֹת מֵאֵין אָדָם וּמֵאֵין יוֹשֵׁב וּמֵאֵין בְּהֵמָה` (ירמיהו לג:י)
`כֹּה אָמַר יְהוָה צְבָאוֹת עוֹד יִהְיֶה בַּמָּקוֹם הַזֶּה הֶחָרֵב מֵאֵין אָדָם וְעַד בְּהֵמָה וּבְכָל עָרָיו נְוֵה רֹעִים מַרְבִּצִים צֹאן` (ירמיהו לג:יב)
`וְעַל יְהוֹיָקִים מֶלֶךְ יְהוּדָה תֹּאמַר כֹּה אָמַר יְהוָה אַתָּה שָׂרַפְתָּ אֶת הַמְּגִלָּה הַזֹּאת לֵאמֹר מַדּוּעַ כָּתַבְתָּ עָלֶיהָ לֵאמֹר בֹּא יָבוֹא מֶלֶךְ בָּבֶל וְהִשְׁחִית אֶת הָאָרֶץ הַזֹּאת וְהִשְׁבִּית מִמֶּנָּה אָדָם וּבְהֵמָה` (ירמיהו לו:כט)
`כֹּה אָמַר יְהוָה הִנֵּה מַיִם עֹלִים מִצָּפוֹן וְהָיוּ לְנַחַל שׁוֹטֵף וְיִשְׁטְפוּ אֶרֶץ וּמְלוֹאָהּ עִיר וְיֹשְׁבֵי בָהּ וְזָעֲקוּ הָאָדָם וְהֵילִל כֹּל יוֹשֵׁב הָאָרֶץ` (ירמיהו מז:ב)
`כִּי הִנֵּה קָטֹן נְתַתִּיךָ בַּגּוֹיִם בָּזוּי בָּאָדָם` (ירמיהו מט:טו)
`כְּמַהְפֵּכַת סְדֹם וַעֲמֹרָה וּשְׁכֵנֶיהָ אָמַר יְהוָה לֹא יֵשֵׁב שָׁם אִישׁ וְלֹא יָגוּר בָּהּ בֶּן אָדָם` (ירמיהו מט:יח)
`וְהָיְתָה חָצוֹר לִמְעוֹן תַּנִּים שְׁמָמָה עַד עוֹלָם לֹא יֵשֵׁב שָׁם אִישׁ וְלֹא יָגוּר בָּהּ בֶּן אָדָם` (ירמיהו מט:לג)
`כִּי עָלָה עָלֶיהָ גּוֹי מִצָּפוֹן הוּא יָשִׁית אֶת אַרְצָהּ לְשַׁמָּה וְלֹא יִהְיֶה יוֹשֵׁב בָּהּ מֵאָדָם וְעַד בְּהֵמָה נָדוּ הָלָכוּ` (ירמיהו נ:ג)
`כְּמַהְפֵּכַת אֱלֹהִים אֶת סְדֹם וְאֶת עֲמֹרָה וְאֶת שְׁכֵנֶיהָ נְאֻם יְהוָה לֹא יֵשֵׁב שָׁם אִישׁ וְלֹא יָגוּר בָּהּ בֶּן אָדָם` (ירמיהו נ:מ)
`נִשְׁבַּע יְהוָה צְבָאוֹת בְּנַפְשׁוֹ כִּי אִם מִלֵּאתִיךְ אָדָם כַּיֶּלֶק וְעָנוּ עָלַיִךְ הֵידָד` (ירמיהו נא:יד)
`נִבְעַר כָּל אָדָם מִדַּעַת הֹבִישׁ כָּל צֹרֵף מִפָּסֶל כִּי שֶׁקֶר נִסְכּוֹ וְלֹא רוּחַ בָּם` (ירמיהו נא:יז)
`הָיוּ עָרֶיהָ לְשַׁמָּה אֶרֶץ צִיָּה וַעֲרָבָה אֶרֶץ לֹא יֵשֵׁב בָּהֵן כָּל אִישׁ וְלֹא יַעֲבֹר בָּהֵן בֶּן אָדָם` (ירמיהו נא:מג)
`וְאָמַרְתָּ יְהוָה אַתָּה דִבַּרְתָּ אֶל הַמָּקוֹם הַזֶּה לְהַכְרִיתוֹ לְבִלְתִּי הֱיוֹת בּוֹ יוֹשֵׁב לְמֵאָדָם וְעַד בְּהֵמָה כִּי שִׁמְמוֹת עוֹלָם תִּהְיֶה` (ירמיהו נא:סב)
`וּמִתּוֹכָהּ דְּמוּת אַרְבַּע חַיּוֹת וְזֶה מַרְאֵיהֶן דְּמוּת אָדָם לָהֵנָּה` (יחזקאל א:ה)
`וְיָדוֹ אָדָם מִתַּחַת כַּנְפֵיהֶם עַל אַרְבַּעַת רִבְעֵיהֶם וּפְנֵיהֶם וְכַנְפֵיהֶם לְאַרְבַּעְתָּם` (יחזקאל א:ח)
`וּדְמוּת פְּנֵיהֶם פְּנֵי אָדָם וּפְנֵי אַרְיֵה אֶל הַיָּמִין לְאַרְבַּעְתָּם וּפְנֵי שׁוֹר מֵהַשְּׂמֹאול לְאַרְבַּעְתָּן וּפְנֵי נֶשֶׁר לְאַרְבַּעְתָּן` (יחזקאל א:י)
`וּמִמַּעַל לָרָקִיעַ אֲשֶׁר עַל רֹאשָׁם כְּמַרְאֵה אֶבֶן סַפִּיר דְּמוּת כִּסֵּא וְעַל דְּמוּת הַכִּסֵּא דְּמוּת כְּמַרְאֵה אָדָם עָלָיו מִלְמָעְלָה` (יחזקאל א:כו)
`וַיֹּאמֶר אֵלָי בֶּן אָדָם עֲמֹד עַל רַגְלֶיךָ וַאֲדַבֵּר אֹתָךְ` (יחזקאל ב:א)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם שׁוֹלֵחַ אֲנִי אוֹתְךָ אֶל בְּנֵי יִשְׂרָאֵל אֶל גּוֹיִם הַמּוֹרְדִים אֲשֶׁר מָרְדוּ בִי הֵמָּה וַאֲבוֹתָם פָּשְׁעוּ בִי עַד עֶצֶם הַיּוֹם הַזֶּה` (יחזקאל ב:ג)
`וְאַתָּה בֶן אָדָם אַל תִּירָא מֵהֶם וּמִדִּבְרֵיהֶם אַל תִּירָא כִּי סָרָבִים וְסַלּוֹנִים אוֹתָךְ וְאֶל עַקְרַבִּים אַתָּה יוֹשֵׁב מִדִּבְרֵיהֶם אַל תִּירָא וּמִפְּנֵיהֶם אַל תֵּחָת כִּי בֵּית מְרִי הֵמָּה` (יחזקאל ב:ו)
`וְאַתָּה בֶן אָדָם שְׁמַע אֵת אֲשֶׁר אֲנִי מְדַבֵּר אֵלֶיךָ אַל תְּהִי מֶרִי כְּבֵית הַמֶּרִי פְּצֵה פִיךָ וֶאֱכֹל אֵת אֲשֶׁר אֲנִי נֹתֵן אֵלֶיךָ` (יחזקאל ב:ח)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם אֵת אֲשֶׁר תִּמְצָא אֱכוֹל אֱכוֹל אֶת הַמְּגִלָּה הַזֹּאת וְלֵךְ דַּבֵּר אֶל בֵּית יִשְׂרָאֵל` (יחזקאל ג:א)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם בִּטְנְךָ תַאֲכֵל וּמֵעֶיךָ תְמַלֵּא אֵת הַמְּגִלָּה הַזֹּאת אֲשֶׁר אֲנִי נֹתֵן אֵלֶיךָ וָאֹכְלָה וַתְּהִי בְּפִי כִּדְבַשׁ לְמָתוֹק` (יחזקאל ג:ג)
`וַיֹּאמֶר אֵלָי בֶּן אָדָם לֶךְ בֹּא אֶל בֵּית יִשְׂרָאֵל וְדִבַּרְתָּ בִדְבָרַי אֲלֵיהֶם` (יחזקאל ג:ד)
`וַיֹּאמֶר אֵלָי בֶּן אָדָם אֶת כָּל דְּבָרַי אֲשֶׁר אֲדַבֵּר אֵלֶיךָ קַח בִּלְבָבְךָ וּבְאָזְנֶיךָ שְׁמָע` (יחזקאל ג:י)
`בֶּן אָדָם צֹפֶה נְתַתִּיךָ לְבֵית יִשְׂרָאֵל וְשָׁמַעְתָּ מִפִּי דָּבָר וְהִזְהַרְתָּ אוֹתָם מִמֶּנִּי` (יחזקאל ג:יז)
`וְאַתָּה בֶן אָדָם הִנֵּה נָתְנוּ עָלֶיךָ עֲבוֹתִים וַאֲסָרוּךָ בָּהֶם וְלֹא תֵצֵא בְּתוֹכָם` (יחזקאל ג:כה)
`וְאַתָּה בֶן אָדָם קַח לְךָ לְבֵנָה וְנָתַתָּה אוֹתָהּ לְפָנֶיךָ וְחַקּוֹתָ עָלֶיהָ עִיר אֶת יְרוּשָׁלִָם` (יחזקאל ד:א)
`וְעֻגַת שְׂעֹרִים תֹּאכֲלֶנָּה וְהִיא בְּגֶלְלֵי צֵאַת הָאָדָם תְּעֻגֶנָה לְעֵינֵיהֶם` (יחזקאל ד:יב)
`וַיֹּאמֶר אֵלַי רְאֵה נָתַתִּי לְךָ אֶת צְפוּעֵי הַבָּקָר תַּחַת גֶּלְלֵי הָאָדָם וְעָשִׂיתָ אֶת לַחְמְךָ עֲלֵיהֶם` (יחזקאל ד:טו)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם הִנְנִי שֹׁבֵר מַטֵּה לֶחֶם בִּירוּשָׁלִַם וְאָכְלוּ לֶחֶם בְּמִשְׁקָל וּבִדְאָגָה וּמַיִם בִּמְשׂוּרָה וּבְשִׁמָּמוֹן יִשְׁתּוּ` (יחזקאל ד:טז)
`וְאַתָּה בֶן אָדָם קַח לְךָ חֶרֶב חַדָּה תַּעַר הַגַּלָּבִים תִּקָּחֶנָּה לָּךְ וְהַעֲבַרְתָּ עַל רֹאשְׁךָ וְעַל זְקָנֶךָ וְלָקַחְתָּ לְךָ מֹאזְנֵי מִשְׁקָל וְחִלַּקְתָּם` (יחזקאל ה:א)
`בֶּן אָדָם שִׂים פָּנֶיךָ אֶל הָרֵי יִשְׂרָאֵל וְהִנָּבֵא אֲלֵיהֶם` (יחזקאל ו:ב)
`וְאַתָּה בֶן אָדָם כֹּה אָמַר אֲדֹנָי יְהוִה לְאַדְמַת יִשְׂרָאֵל קֵץ בָּא הַקֵּץ עַל אַרְבַּעַת כַּנְפוֹת הָאָרֶץ` (יחזקאל ז:ב)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם שָׂא נָא עֵינֶיךָ דֶּרֶךְ צָפוֹנָה וָאֶשָּׂא עֵינַי דֶּרֶךְ צָפוֹנָה וְהִנֵּה מִצָּפוֹן לְשַׁעַר הַמִּזְבֵּחַ סֵמֶל הַקִּנְאָה הַזֶּה בַּבִּאָה` (יחזקאל ח:ה)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם הֲרֹאֶה אַתָּה מֵהֶם עֹשִׂים תּוֹעֵבוֹת גְּדֹלוֹת אֲשֶׁר בֵּית יִשְׂרָאֵל עֹשִׂים פֹּה לְרָחֳקָה מֵעַל מִקְדָּשִׁי וְעוֹד תָּשׁוּב תִּרְאֶה תּוֹעֵבוֹת גְּדֹלוֹת` (יחזקאל ח:ו)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם חֲתָר נָא בַקִּיר וָאֶחְתֹּר בַּקִּיר וְהִנֵּה פֶּתַח אֶחָד` (יחזקאל ח:ח)
`וַיֹּאמֶר אֵלַי הֲרָאִיתָ בֶן אָדָם אֲשֶׁר זִקְנֵי בֵית יִשְׂרָאֵל עֹשִׂים בַּחֹשֶׁךְ אִישׁ בְּחַדְרֵי מַשְׂכִּיתוֹ כִּי אֹמְרִים אֵין יְהוָה רֹאֶה אֹתָנוּ עָזַב יְהוָה אֶת הָאָרֶץ` (יחזקאל ח:יב)
`וַיֹּאמֶר אֵלַי הֲרָאִיתָ בֶן אָדָם עוֹד תָּשׁוּב תִּרְאֶה תּוֹעֵבוֹת גְּדֹלוֹת מֵאֵלֶּה` (יחזקאל ח:טו)
`וַיֹּאמֶר אֵלַי הֲרָאִיתָ בֶן אָדָם הֲנָקֵל לְבֵית יְהוּדָה מֵעֲשׂוֹת אֶת הַתּוֹעֵבוֹת אֲשֶׁר עָשׂוּ פֹה כִּי מָלְאוּ אֶת הָאָרֶץ חָמָס וַיָּשֻׁבוּ לְהַכְעִיסֵנִי וְהִנָּם שֹׁלְחִים אֶת הַזְּמוֹרָה אֶל אַפָּם` (יחזקאל ח:יז)
`וַיֵּרָא לַכְּרֻבִים תַּבְנִית יַד אָדָם תַּחַת כַּנְפֵיהֶם` (יחזקאל י:ח)
`וְאַרְבָּעָה פָנִים לְאֶחָד פְּנֵי הָאֶחָד פְּנֵי הַכְּרוּב וּפְנֵי הַשֵּׁנִי פְּנֵי אָדָם וְהַשְּׁלִישִׁי פְּנֵי אַרְיֵה וְהָרְבִיעִי פְּנֵי נָשֶׁר` (יחזקאל י:יד)
`אַרְבָּעָה אַרְבָּעָה פָנִים לְאֶחָד וְאַרְבַּע כְּנָפַיִם לְאֶחָד וּדְמוּת יְדֵי אָדָם תַּחַת כַּנְפֵיהֶם` (יחזקאל י:כא)
`וַיֹּאמֶר אֵלָי בֶּן אָדָם אֵלֶּה הָאֲנָשִׁים הַחֹשְׁבִים אָוֶן וְהַיֹּעֲצִים עֲצַת רָע בָּעִיר הַזֹּאת` (יחזקאל יא:ב)
`לָכֵן הִנָּבֵא עֲלֵיהֶם הִנָּבֵא בֶּן אָדָם` (יחזקאל יא:ד)
`בֶּן אָדָם אַחֶיךָ אַחֶיךָ אַנְשֵׁי גְאֻלָּתֶךָ וְכָל בֵּית יִשְׂרָאֵל כֻּלֹּה אֲשֶׁר אָמְרוּ לָהֶם יֹשְׁבֵי יְרוּשָׁלִַם רַחֲקוּ מֵעַל יְהוָה לָנוּ הִיא נִתְּנָה הָאָרֶץ לְמוֹרָשָׁה` (יחזקאל יא:טו)
`בֶּן אָדָם בְּתוֹךְ בֵּית הַמֶּרִי אַתָּה יֹשֵׁב אֲשֶׁר עֵינַיִם לָהֶם לִרְאוֹת וְלֹא רָאוּ אָזְנַיִם לָהֶם לִשְׁמֹעַ וְלֹא שָׁמֵעוּ כִּי בֵּית מְרִי הֵם` (יחזקאל יב:ב)
`וְאַתָּה בֶן אָדָם עֲשֵׂה לְךָ כְּלֵי גוֹלָה וּגְלֵה יוֹמָם לְעֵינֵיהֶם וְגָלִיתָ מִמְּקוֹמְךָ אֶל מָקוֹם אַחֵר לְעֵינֵיהֶם אוּלַי יִרְאוּ כִּי בֵּית מְרִי הֵמָּה` (יחזקאל יב:ג)
`בֶּן אָדָם הֲלֹא אָמְרוּ אֵלֶיךָ בֵּית יִשְׂרָאֵל בֵּית הַמֶּרִי מָה אַתָּה עֹשֶׂה` (יחזקאל יב:ט)
`בֶּן אָדָם לַחְמְךָ בְּרַעַשׁ תֹּאכֵל וּמֵימֶיךָ בְּרָגְזָה וּבִדְאָגָה תִּשְׁתֶּה` (יחזקאל יב:יח)
`בֶּן אָדָם מָה הַמָּשָׁל הַזֶּה לָכֶם עַל אַדְמַת יִשְׂרָאֵל לֵאמֹר יַאַרְכוּ הַיָּמִים וְאָבַד כָּל חָזוֹן` (יחזקאל יב:כב)
`בֶּן אָדָם הִנֵּה בֵית יִשְׂרָאֵל אֹמְרִים הֶחָזוֹן אֲשֶׁר הוּא חֹזֶה לְיָמִים רַבִּים וּלְעִתִּים רְחוֹקוֹת הוּא נִבָּא` (יחזקאל יב:כז)
`בֶּן אָדָם הִנָּבֵא אֶל נְבִיאֵי יִשְׂרָאֵל הַנִּבָּאִים וְאָמַרְתָּ לִנְבִיאֵי מִלִּבָּם שִׁמְעוּ דְּבַר יְהוָה` (יחזקאל יג:ב)
`וְאַתָּה בֶן אָדָם שִׂים פָּנֶיךָ אֶל בְּנוֹת עַמְּךָ הַמִּתְנַבְּאוֹת מִלִּבְּהֶן וְהִנָּבֵא עֲלֵיהֶן` (יחזקאל יג:יז)
`בֶּן אָדָם הָאֲנָשִׁים הָאֵלֶּה הֶעֱלוּ גִלּוּלֵיהֶם עַל לִבָּם וּמִכְשׁוֹל עֲוֺנָם נָתְנוּ נֹכַח פְּנֵיהֶם הַאִדָּרֹשׁ אִדָּרֵשׁ לָהֶם` (יחזקאל יד:ג)
`בֶּן אָדָם אֶרֶץ כִּי תֶחֱטָא לִי לִמְעָל מַעַל וְנָטִיתִי יָדִי עָלֶיהָ וְשָׁבַרְתִּי לָהּ מַטֵּה לָחֶם וְהִשְׁלַחְתִּי בָהּ רָעָב וְהִכְרַתִּי מִמֶּנָּה אָדָם וּבְהֵמָה` (יחזקאל יד:יג)
`אוֹ חֶרֶב אָבִיא עַל הָאָרֶץ הַהִיא וְאָמַרְתִּי חֶרֶב תַּעֲבֹר בָּאָרֶץ וְהִכְרַתִּי מִמֶּנָּה אָדָם וּבְהֵמָה` (יחזקאל יד:יז)
`אוֹ דֶּבֶר אֲשַׁלַּח אֶל הָאָרֶץ הַהִיא וְשָׁפַכְתִּי חֲמָתִי עָלֶיהָ בְּדָם לְהַכְרִית מִמֶּנָּה אָדָם וּבְהֵמָה` (יחזקאל יד:יט)
`כִּי כֹה אָמַר אֲדֹנָי יְהֹוִה אַף כִּי אַרְבַּעַת שְׁפָטַי הָרָעִים חֶרֶב וְרָעָב וְחַיָּה רָעָה וָדֶבֶר שִׁלַּחְתִּי אֶל יְרוּשָׁלִָם לְהַכְרִית מִמֶּנָּה אָדָם וּבְהֵמָה` (יחזקאל יד:כא)
`בֶּן אָדָם מַה יִּהְיֶה עֵץ הַגֶּפֶן מִכָּל עֵץ הַזְּמוֹרָה אֲשֶׁר הָיָה בַּעֲצֵי הַיָּעַר` (יחזקאל טו:ב)
`בֶּן אָדָם הוֹדַע אֶת יְרוּשָׁלִַם אֶת תּוֹעֲבֹתֶיהָ` (יחזקאל טז:ב)
`בֶּן אָדָם חוּד חִידָה וּמְשֹׁל מָשָׁל אֶל בֵּית יִשְׂרָאֵל` (יחזקאל יז:ב)
`וַתַּעַל אֶחָד מִגֻּרֶיהָ כְּפִיר הָיָה וַיִּלְמַד לִטְרָף טֶרֶף אָדָם אָכָל` (יחזקאל יט:ג)
`וַיִּתְהַלֵּךְ בְּתוֹךְ אֲרָיוֹת כְּפִיר הָיָה וַיִּלְמַד לִטְרָף טֶרֶף אָדָם אָכָל` (יחזקאל יט:ו)
`בֶּן אָדָם דַּבֵּר אֶת זִקְנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם כֹּה אָמַר אֲדֹנָי יְהוִה הֲלִדְרֹשׁ אֹתִי אַתֶּם בָּאִים חַי אָנִי אִם אִדָּרֵשׁ לָכֶם נְאֻם אֲדֹנָי יְהוִה` (יחזקאל כ:ג)
`הֲתִשְׁפֹּט אֹתָם הֲתִשְׁפּוֹט בֶּן אָדָם אֶת תּוֹעֲבֹת אֲבוֹתָם הוֹדִיעֵם` (יחזקאל כ:ד)
`וָאֶתֵּן לָהֶם אֶת חֻקּוֹתַי וְאֶת מִשְׁפָּטַי הוֹדַעְתִּי אוֹתָם אֲשֶׁר יַעֲשֶׂה אוֹתָם הָאָדָם וָחַי בָּהֶם` (יחזקאל כ:יא)
`וַיַּמְרוּ בִי בֵית יִשְׂרָאֵל בַּמִּדְבָּר בְּחֻקּוֹתַי לֹא הָלָכוּ וְאֶת מִשְׁפָּטַי מָאָסוּ אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם וָחַי בָּהֶם וְאֶת שַׁבְּתֹתַי חִלְּלוּ מְאֹד וָאֹמַר לִשְׁפֹּךְ חֲמָתִי עֲלֵיהֶם בַּמִּדְבָּר לְכַלּוֹתָם` (יחזקאל כ:יג)
`וַיַּמְרוּ בִי הַבָּנִים בְּחֻקּוֹתַי לֹא הָלָכוּ וְאֶת מִשְׁפָּטַי לֹא שָׁמְרוּ לַעֲשׂוֹת אוֹתָם אֲשֶׁר יַעֲשֶׂה אוֹתָם הָאָדָם וָחַי בָּהֶם אֶת שַׁבְּתוֹתַי חִלֵּלוּ וָאֹמַר לִשְׁפֹּךְ חֲמָתִי עֲלֵיהֶם לְכַלּוֹת אַפִּי בָּם בַּמִּדְבָּר` (יחזקאל כ:כא)
`לָכֵן דַּבֵּר אֶל בֵּית יִשְׂרָאֵל בֶּן אָדָם וְאָמַרְתָּ אֲלֵיהֶם כֹּה אָמַר אֲדֹנָי יְהוִה עוֹד זֹאת גִּדְּפוּ אוֹתִי אֲבוֹתֵיכֶם בְּמַעֲלָם בִּי מָעַל` (יחזקאל כ:כז)
`בֶּן אָדָם שִׂים פָּנֶיךָ דֶּרֶךְ תֵּימָנָה וְהַטֵּף אֶל דָּרוֹם וְהִנָּבֵא אֶל יַעַר הַשָּׂדֶה נֶגֶב` (יחזקאל כ:מו)
`בֶּן אָדָם שִׂים פָּנֶיךָ אֶל יְרוּשָׁלִַם וְהַטֵּף אֶל מִקְדָּשִׁים וְהִנָּבֵא אֶל אַדְמַת יִשְׂרָאֵל` (יחזקאל כא:ב)
`וְאַתָּה בֶן אָדָם הֵאָנַח בְּשִׁבְרוֹן מָתְנַיִם וּבִמְרִירוּת תֵּאָנַח לְעֵינֵיהֶם` (יחזקאל כא:ו)
`בֶּן אָדָם הִנָּבֵא וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי אֱמֹר חֶרֶב חֶרֶב הוּחַדָּה וְגַם מְרוּטָה` (יחזקאל כא:ט)
`זְעַק וְהֵילֵל בֶּן אָדָם כִּי הִיא הָיתָה בְעַמִּי הִיא בְּכָל נְשִׂיאֵי יִשְׂרָאֵל מְגוּרֵי אֶל חֶרֶב הָיוּ אֶת עַמִּי לָכֵן סְפֹק אֶל יָרֵךְ` (יחזקאל כא:יב)
`וְאַתָּה בֶן אָדָם הִנָּבֵא וְהַךְ כַּף אֶל כָּף וְתִכָּפֵל חֶרֶב שְׁלִישִׁתָה חֶרֶב חֲלָלִים הִיא חֶרֶב חָלָל הַגָּדוֹל הַחֹדֶרֶת לָהֶם` (יחזקאל כא:יד)
`וְאַתָּה בֶן אָדָם שִׂים לְךָ שְׁנַיִם דְּרָכִים לָבוֹא חֶרֶב מֶלֶךְ בָּבֶל מֵאֶרֶץ אֶחָד יֵצְאוּ שְׁנֵיהֶם וְיָד בָּרֵא בְּרֹאשׁ דֶּרֶךְ עִיר בָּרֵא` (יחזקאל כא:יט)
`וְאַתָּה בֶן אָדָם הִנָּבֵא וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יְהֹוִה אֶל בְּנֵי עַמּוֹן וְאֶל חֶרְפָּתָם וְאָמַרְתָּ חֶרֶב חֶרֶב פְּתוּחָה לְטֶבַח מְרוּטָה לְהָכִיל לְמַעַן בָּרָק` (יחזקאל כא:כח)
`וְאַתָּה בֶן אָדָם הֲתִשְׁפֹּט הֲתִשְׁפֹּט אֶת עִיר הַדָּמִים וְהוֹדַעְתָּהּ אֵת כָּל תּוֹעֲבוֹתֶיהָ` (יחזקאל כב:ב)
`בֶּן אָדָם הָיוּ לִי בֵית יִשְׂרָאֵל לְסוּג כֻּלָּם נְחֹשֶׁת וּבְדִיל וּבַרְזֶל וְעוֹפֶרֶת בְּתוֹךְ כּוּר סִגִים כֶּסֶף הָיוּ` (יחזקאל כב:יח)
`בֶּן אָדָם אֱמָר לָהּ אַתְּ אֶרֶץ לֹא מְטֹהָרָה הִיא לֹא גֻשְׁמָהּ בְּיוֹם זָעַם` (יחזקאל כב:כד)
`בֶּן אָדָם שְׁתַּיִם נָשִׁים בְּנוֹת אֵם אַחַת הָיוּ` (יחזקאל כג:ב)
`וַיֹּאמֶר יְהוָה אֵלַי בֶּן אָדָם הֲתִשְׁפּוֹט אֶת אָהֳלָה וְאֶת אָהֳלִיבָה וְהַגֵּד לָהֶן אֵת תוֹעֲבוֹתֵיהֶן` (יחזקאל כג:לו)
`וְקוֹל הָמוֹן שָׁלֵו בָהּ וְאֶל אֲנָשִׁים מֵרֹב אָדָם מוּבָאִים סוֹבָאִים מִמִּדְבָּר וַיִּתְּנוּ צְמִידִים אֶל יְדֵיהֶן וַעֲטֶרֶת תִּפְאֶרֶת עַל רָאשֵׁיהֶן` (יחזקאל כג:מב)
`בֶּן אָדָם כְּתוֹב לְךָ אֶת שֵׁם הַיּוֹם אֶת עֶצֶם הַיּוֹם הַזֶּה סָמַךְ מֶלֶךְ בָּבֶל אֶל יְרוּשָׁלִַם בְּעֶצֶם הַיּוֹם הַזֶּה` (יחזקאל כד:ב)
`בֶּן אָדָם הִנְנִי לֹקֵחַ מִמְּךָ אֶת מַחְמַד עֵינֶיךָ בְּמַגֵּפָה וְלֹא תִסְפֹּד וְלֹא תִבְכֶּה וְלוֹא תָבוֹא דִּמְעָתֶךָ` (יחזקאל כד:טז)
`וְאַתָּה בֶן אָדָם הֲלוֹא בְּיוֹם קַחְתִּי מֵהֶם אֶת מָעוּזָּם מְשׂוֹשׂ תִּפְאַרְתָּם אֶת מַחְמַד עֵינֵיהֶם וְאֶת מַשָּׂא נַפְשָׁם בְּנֵיהֶם וּבְנוֹתֵיהֶם` (יחזקאל כד:כה)
`בֶּן אָדָם שִׂים פָּנֶיךָ אֶל בְּנֵי עַמּוֹן וְהִנָּבֵא עֲלֵיהֶם` (יחזקאל כה:ב)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה וְנָטִתִי יָדִי עַל אֱדוֹם וְהִכְרַתִּי מִמֶּנָּה אָדָם וּבְהֵמָה וּנְתַתִּיהָ חָרְבָּה מִתֵּימָן וּדְדָנֶה בַּחֶרֶב יִפֹּלוּ` (יחזקאל כה:יג)
`בֶּן אָדָם יַעַן אֲשֶׁר אָמְרָה צֹּר עַל יְרוּשָׁלִַם הֶאָח נִשְׁבְּרָה דַּלְתוֹת הָעַמִּים נָסֵבָּה אֵלָי אִמָּלְאָה הָחֳרָבָה` (יחזקאל כו:ב)
`וְאַתָּה בֶן אָדָם שָׂא עַל צֹר קִינָה` (יחזקאל כז:ב)
`יָוָן תֻּבַל וָמֶשֶׁךְ הֵמָּה רֹכְלָיִךְ בְּנֶפֶשׁ אָדָם וּכְלֵי נְחֹשֶׁת נָתְנוּ מַעֲרָבֵךְ` (יחזקאל כז:יג)
`בֶּן אָדָם אֱמֹר לִנְגִיד צֹר כֹּה אָמַר אֲדֹנָי יְהֹוִה יַעַן גָּבַהּ לִבְּךָ וַתֹּאמֶר אֵל אָנִי מוֹשַׁב אֱלֹהִים יָשַׁבְתִּי בְּלֵב יַמִּים וְאַתָּה אָדָם וְלֹא אֵל וַתִּתֵּן לִבְּךָ כְּלֵב אֱלֹהִים` (יחזקאל כח:ב)
`הֶאָמֹר תֹּאמַר אֱלֹהִים אָנִי לִפְנֵי הֹרְגֶךָ וְאַתָּה אָדָם וְלֹא אֵל בְּיַד מְחַלְלֶיךָ` (יחזקאל כח:ט)
`בֶּן אָדָם שָׂא קִינָה עַל מֶלֶךְ צוֹר וְאָמַרְתָּ לּוֹ כֹּה אָמַר אֲדֹנָי יְהוִה אַתָּה חוֹתֵם תָּכְנִית מָלֵא חָכְמָה וּכְלִיל יֹפִי` (יחזקאל כח:יב)
`בֶּן אָדָם שִׂים פָּנֶיךָ אֶל צִידוֹן וְהִנָּבֵא עָלֶיהָ` (יחזקאל כח:כא)
`בֶּן אָדָם שִׂים פָּנֶיךָ עַל פַּרְעֹה מֶלֶךְ מִצְרָיִם וְהִנָּבֵא עָלָיו וְעַל מִצְרַיִם כֻּלָּהּ` (יחזקאל כט:ב)
`לָכֵן כֹּה אָמַר אֲדֹנָי יְהוִה הִנְנִי מֵבִיא עָלַיִךְ חָרֶב וְהִכְרַתִּי מִמֵּךְ אָדָם וּבְהֵמָה` (יחזקאל כט:ח)
`לֹא תַעֲבָר בָּהּ רֶגֶל אָדָם וְרֶגֶל בְּהֵמָה לֹא תַעֲבָר בָּהּ וְלֹא תֵשֵׁב אַרְבָּעִים שָׁנָה` (יחזקאל כט:יא)
`בֶּן אָדָם נְבוּכַדְרֶאצַּר מֶלֶךְ בָּבֶל הֶעֱבִיד אֶת חֵילוֹ עֲבֹדָה גְדֹלָה אֶל צֹר כָּל רֹאשׁ מֻקְרָח וְכָל כָּתֵף מְרוּטָה וְשָׂכָר לֹא הָיָה לוֹ וּלְחֵילוֹ מִצֹּר עַל הָעֲבֹדָה אֲשֶׁר עָבַד עָלֶיהָ` (יחזקאל כט:יח)
`בֶּן אָדָם הִנָּבֵא וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יְהוִה הֵילִילוּ הָהּ לַיּוֹם` (יחזקאל ל:ב)
`בֶּן אָדָם אֶת זְרוֹעַ פַּרְעֹה מֶלֶךְ מִצְרַיִם שָׁבָרְתִּי וְהִנֵּה לֹא חֻבְּשָׁה לָתֵת רְפֻאוֹת לָשׂוּם חִתּוּל לְחָבְשָׁהּ לְחָזְקָהּ לִתְפֹּשׂ בֶּחָרֶב` (יחזקאל ל:כא)
`בֶּן אָדָם אֱמֹר אֶל פַּרְעֹה מֶלֶךְ מִצְרַיִם וְאֶל הֲמוֹנוֹ אֶל מִי דָּמִיתָ בְגָדְלֶךָ` (יחזקאל לא:ב)
`לְמַעַן אֲשֶׁר לֹא יִגְבְּהוּ בְקוֹמָתָם כָּל עֲצֵי מַיִם וְלֹא יִתְּנוּ אֶת צַמַּרְתָּם אֶל בֵּין עֲבֹתִים וְלֹא יַעַמְדוּ אֵלֵיהֶם בְּגָבְהָם כָּל שֹׁתֵי מָיִם כִּי כֻלָּם נִתְּנוּ לַמָּוֶת אֶל אֶרֶץ תַּחְתִּית בְּתוֹךְ בְּנֵי אָדָם אֶל יוֹרְדֵי בוֹר` (יחזקאל לא:יד)
`בֶּן אָדָם שָׂא קִינָה עַל פַּרְעֹה מֶלֶךְ מִצְרַיִם וְאָמַרְתָּ אֵלָיו כְּפִיר גּוֹיִם נִדְמֵיתָ וְאַתָּה כַּתַּנִּים בַּיַּמִּים וַתָּגַח בְּנַהֲרוֹתֶיךָ וַתִּדְלַח מַיִם בְּרַגְלֶיךָ וַתִּרְפֹּס נַהֲרוֹתָם` (יחזקאל לב:ב)
`וְהַאֲבַדְתִּי אֶת כָּל בְּהֶמְתָּהּ מֵעַל מַיִם רַבִּים וְלֹא תִדְלָחֵם רֶגֶל אָדָם עוֹד וּפַרְסוֹת בְּהֵמָה לֹא תִדְלָחֵם` (יחזקאל לב:יג)
`בֶּן אָדָם נְהֵה עַל הֲמוֹן מִצְרַיִם וְהוֹרִדֵהוּ אוֹתָהּ וּבְנוֹת גּוֹיִם אַדִּרִם אֶל אֶרֶץ תַּחְתִּיּוֹת אֶת יוֹרְדֵי בוֹר` (יחזקאל לב:יח)
`בֶּן אָדָם דַּבֵּר אֶל בְּנֵי עַמְּךָ וְאָמַרְתָּ אֲלֵיהֶם אֶרֶץ כִּי אָבִיא עָלֶיהָ חָרֶב וְלָקְחוּ עַם הָאָרֶץ אִישׁ אֶחָד מִקְצֵיהֶם וְנָתְנוּ אֹתוֹ לָהֶם לְצֹפֶה` (יחזקאל לג:ב)
`וְאַתָּה בֶן אָדָם צֹפֶה נְתַתִּיךָ לְבֵית יִשְׂרָאֵל וְשָׁמַעְתָּ מִפִּי דָּבָר וְהִזְהַרְתָּ אֹתָם מִמֶּנִּי` (יחזקאל לג:ז)
`וְאַתָּה בֶן אָדָם אֱמֹר אֶל בֵּית יִשְׂרָאֵל כֵּן אֲמַרְתֶּם לֵאמֹר כִּי פְשָׁעֵינוּ וְחַטֹּאתֵינוּ עָלֵינוּ וּבָם אֲנַחְנוּ נְמַקִּים וְאֵיךְ נִחְיֶה` (יחזקאל לג:י)
`וְאַתָּה בֶן אָדָם אֱמֹר אֶל בְּנֵי עַמְּךָ צִדְקַת הַצַּדִּיק לֹא תַצִּילֶנּוּ בְּיוֹם פִּשְׁעוֹ וְרִשְׁעַת הָרָשָׁע לֹא יִכָּשֶׁל בָּהּ בְּיוֹם שׁוּבוֹ מֵרִשְׁעוֹ וְצַדִּיק לֹא יוּכַל לִחְיוֹת בָּהּ בְּיוֹם חֲטֹאתוֹ` (יחזקאל לג:יב)
`בֶּן אָדָם יֹשְׁבֵי הֶחֳרָבוֹת הָאֵלֶּה עַל אַדְמַת יִשְׂרָאֵל אֹמְרִים לֵאמֹר אֶחָד הָיָה אַבְרָהָם וַיִּירַשׁ אֶת הָאָרֶץ וַאֲנַחְנוּ רַבִּים לָנוּ נִתְּנָה הָאָרֶץ לְמוֹרָשָׁה` (יחזקאל לג:כד)
`וְאַתָּה בֶן אָדָם בְּנֵי עַמְּךָ הַנִּדְבָּרִים בְּךָ אֵצֶל הַקִּירוֹת וּבְפִתְחֵי הַבָּתִּים וְדִבֶּר חַד אֶת אַחַד אִישׁ אֶת אָחִיו לֵאמֹר בֹּאוּ נָא וְשִׁמְעוּ מָה הַדָּבָר הַיּוֹצֵא מֵאֵת יְהוָה` (יחזקאל לג:ל)
`בֶּן אָדָם הִנָּבֵא עַל רוֹעֵי יִשְׂרָאֵל הִנָּבֵא וְאָמַרְתָּ אֲלֵיהֶם לָרֹעִים כֹּה אָמַר אֲדֹנָי יְהוִה הוֹי רֹעֵי יִשְׂרָאֵל אֲשֶׁר הָיוּ רֹעִים אוֹתָם הֲלוֹא הַצֹּאן יִרְעוּ הָרֹעִים` (יחזקאל לד:ב)
`וְאַתֵּן צֹאנִי צֹאן מַרְעִיתִי אָדָם אַתֶּם אֲנִי אֱלֹהֵיכֶם נְאֻם אֲדֹנָי יְהוִה` (יחזקאל לד:לא)
`בֶּן אָדָם שִׂים פָּנֶיךָ עַל הַר שֵׂעִיר וְהִנָּבֵא עָלָיו` (יחזקאל לה:ב)
`וְאַתָּה בֶן אָדָם הִנָּבֵא אֶל הָרֵי יִשְׂרָאֵל וְאָמַרְתָּ הָרֵי יִשְׂרָאֵל שִׁמְעוּ דְּבַר יְהוָה` (יחזקאל לו:א)
`וְהִרְבֵּיתִי עֲלֵיכֶם אָדָם כָּל בֵּית יִשְׂרָאֵל כֻּלֹּה וְנֹשְׁבוּ הֶעָרִים וְהֶחֳרָבוֹת תִּבָּנֶינָה` (יחזקאל לו:י)
`וְהִרְבֵּיתִי עֲלֵיכֶם אָדָם וּבְהֵמָה וְרָבוּ וּפָרוּ וְהוֹשַׁבְתִּי אֶתְכֶם כְּקַדְמוֹתֵיכֶם וְהֵטִבֹתִי מֵרִאשֹׁתֵיכֶם וִידַעְתֶּם כִּי אֲנִי יְהוָה` (יחזקאל לו:יא)
`וְהוֹלַכְתִּי עֲלֵיכֶם אָדָם אֶת עַמִּי יִשְׂרָאֵל וִירֵשׁוּךָ וְהָיִיתָ לָהֶם לְנַחֲלָה וְלֹא תוֹסִף עוֹד לְשַׁכְּלָם` (יחזקאל לו:יב)
`כֹּה אָמַר אֲדֹנָי יְהוִה יַעַן אֹמְרִים לָכֶם אֹכֶלֶת אָדָם אָתִּי וּמְשַׁכֶּלֶת גּוֹיֵךְ הָיִית` (יחזקאל לו:יג)
`לָכֵן אָדָם לֹא תֹאכְלִי עוֹד וְגוֹיֵךְ לֹא תְכַשְּׁלִי עוֹד נְאֻם אֲדֹנָי יְהוִה` (יחזקאל לו:יד)
`בֶּן אָדָם בֵּית יִשְׂרָאֵל יֹשְׁבִים עַל אַדְמָתָם וַיְטַמְּאוּ אוֹתָהּ בְּדַרְכָּם וּבַעֲלִילוֹתָם כְּטֻמְאַת הַנִּדָּה הָיְתָה דַרְכָּם לְפָנָי` (יחזקאל לו:יז)
`כֹּה אָמַר אֲדֹנָי יְהוִה עוֹד זֹאת אִדָּרֵשׁ לְבֵית יִשְׂרָאֵל לַעֲשׂוֹת לָהֶם אַרְבֶּה אֹתָם כַּצֹּאן אָדָם` (יחזקאל לו:לז)
`כְּצֹאן קָדָשִׁים כְּצֹאן יְרוּשָׁלִַם בְּמוֹעֲדֶיהָ כֵּן תִּהְיֶינָה הֶעָרִים הֶחֳרֵבוֹת מְלֵאוֹת צֹאן אָדָם וְיָדְעוּ כִּי אֲנִי יְהוָה` (יחזקאל לו:לח)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם הֲתִחְיֶינָה הָעֲצָמוֹת הָאֵלֶּה וָאֹמַר אֲדֹנָי יְהוִה אַתָּה יָדָעְתָּ` (יחזקאל לז:ג)
`וַיֹּאמֶר אֵלַי הִנָּבֵא אֶל הָרוּחַ הִנָּבֵא בֶן אָדָם וְאָמַרְתָּ אֶל הָרוּחַ כֹּה אָמַר אֲדֹנָי יְהוִה מֵאַרְבַּע רוּחוֹת בֹּאִי הָרוּחַ וּפְחִי בַּהֲרוּגִים הָאֵלֶּה וְיִחְיוּ` (יחזקאל לז:ט)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם הָעֲצָמוֹת הָאֵלֶּה כָּל בֵּית יִשְׂרָאֵל הֵמָּה הִנֵּה אֹמְרִים יָבְשׁוּ עַצְמוֹתֵינוּ וְאָבְדָה תִקְוָתֵנוּ נִגְזַרְנוּ לָנוּ` (יחזקאל לז:יא)
`וְאַתָּה בֶן אָדָם קַח לְךָ עֵץ אֶחָד וּכְתֹב עָלָיו לִיהוּדָה וְלִבְנֵי יִשְׂרָאֵל חֲבֵרוֹ וּלְקַח עֵץ אֶחָד וּכְתוֹב עָלָיו לְיוֹסֵף עֵץ אֶפְרַיִם וְכָל בֵּית יִשְׂרָאֵל חֲבֵרוֹ` (יחזקאל לז:טז)
`בֶּן אָדָם שִׂים פָּנֶיךָ אֶל גּוֹג אֶרֶץ הַמָּגוֹג נְשִׂיא רֹאשׁ מֶשֶׁךְ וְתֻבָל וְהִנָּבֵא עָלָיו` (יחזקאל לח:ב)
`לָכֵן הִנָּבֵא בֶן אָדָם וְאָמַרְתָּ לְגוֹג כֹּה אָמַר אֲדֹנָי יְהוִה הֲלוֹא בַּיּוֹם הַהוּא בְּשֶׁבֶת עַמִּי יִשְׂרָאֵל לָבֶטַח תֵּדָע` (יחזקאל לח:יד)
`וְרָעֲשׁוּ מִפָּנַי דְּגֵי הַיָּם וְעוֹף הַשָּׁמַיִם וְחַיַּת הַשָּׂדֶה וְכָל הָרֶמֶשׂ הָרֹמֵשׂ עַל הָאֲדָמָה וְכֹל הָאָדָם אֲשֶׁר עַל פְּנֵי הָאֲדָמָה וְנֶהֶרְסוּ הֶהָרִים וְנָפְלוּ הַמַּדְרֵגוֹת וְכָל חוֹמָה לָאָרֶץ תִּפּוֹל` (יחזקאל לח:כ)
`וְאַתָּה בֶן אָדָם הִנָּבֵא עַל גּוֹג וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יְהוִה הִנְנִי אֵלֶיךָ גּוֹג נְשִׂיא רֹאשׁ מֶשֶׁךְ וְתֻבָל` (יחזקאל לט:א)
`וְעָבְרוּ הָעֹבְרִים בָּאָרֶץ וְרָאָה עֶצֶם אָדָם וּבָנָה אֶצְלוֹ צִיּוּן עַד קָבְרוּ אֹתוֹ הַמְקַבְּרִים אֶל גֵּיא הֲמוֹן גּוֹג` (יחזקאל לט:טו)
`וְאַתָּה בֶן אָדָם כֹּה אָמַר אֲדֹנָי יְהֹוִה אֱמֹר לְצִפּוֹר כָּל כָּנָף וּלְכֹל חַיַּת הַשָּׂדֶה הִקָּבְצוּ וָבֹאוּ הֵאָסְפוּ מִסָּבִיב עַל זִבְחִי אֲשֶׁר אֲנִי זֹבֵחַ לָכֶם זֶבַח גָּדוֹל עַל הָרֵי יִשְׂרָאֵל וַאֲכַלְתֶּם בָּשָׂר וּשְׁתִיתֶם דָּם` (יחזקאל לט:יז)
`וַיְדַבֵּר אֵלַי הָאִישׁ בֶּן אָדָם רְאֵה בְעֵינֶיךָ וּבְאָזְנֶיךָ שְּׁמָע וְשִׂים לִבְּךָ לְכֹל אֲשֶׁר אֲנִי מַרְאֶה אוֹתָךְ כִּי לְמַעַן הַרְאוֹתְכָה הֻבָאתָה הֵנָּה הַגֵּד אֶת כָּל אֲשֶׁר אַתָּה רֹאֶה לְבֵית יִשְׂרָאֵל` (יחזקאל מ:ד)
`וּפְנֵי אָדָם אֶל הַתִּמֹרָה מִפּוֹ וּפְנֵי כְפִיר אֶל הַתִּמֹרָה מִפּוֹ עָשׂוּי אֶל כָּל הַבַּיִת סָבִיב סָבִיב` (יחזקאל מא:יט)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם אֶת מְקוֹם כִּסְאִי וְאֶת מְקוֹם כַּפּוֹת רַגְלַי אֲשֶׁר אֶשְׁכָּן שָׁם בְּתוֹךְ בְּנֵי יִשְׂרָאֵל לְעוֹלָם וְלֹא יְטַמְּאוּ עוֹד בֵּית יִשְׂרָאֵל שֵׁם קָדְשִׁי הֵמָּה וּמַלְכֵיהֶם בִּזְנוּתָם וּבְפִגְרֵי מַלְכֵיהֶם בָּמוֹתָם` (יחזקאל מג:ז)
`אַתָּה בֶן אָדָם הַגֵּד אֶת בֵּית יִשְׂרָאֵל אֶת הַבַּיִת וְיִכָּלְמוּ מֵעֲוֺנוֹתֵיהֶם וּמָדְדוּ אֶת תָּכְנִית` (יחזקאל מג:י)
`וַיֹּאמֶר אֵלַי בֶּן אָדָם כֹּה אָמַר אֲדֹנָי יְהוִה אֵלֶּה חֻקּוֹת הַמִּזְבֵּחַ בְּיוֹם הֵעָשׂוֹתוֹ לְהַעֲלוֹת עָלָיו עוֹלָה וְלִזְרֹק עָלָיו דָּם` (יחזקאל מג:יח)
`וַיֹּאמֶר אֵלַי יְהֹוָה בֶּן אָדָם שִׂים לִבְּךָ וּרְאֵה בְעֵינֶיךָ וּבְאָזְנֶיךָ שְּׁמָע אֵת כָּל אֲשֶׁר אֲנִי מְדַבֵּר אֹתָךְ לְכָל חֻקּוֹת בֵּית יְהוָה וּלְכָל תּוֹרֹתוֹ וְשַׂמְתָּ לִבְּךָ לִמְבוֹא הַבַּיִת בְּכֹל מוֹצָאֵי הַמִּקְדָּשׁ` (יחזקאל מד:ה)
`וְאֶל מֵת אָדָם לֹא יָבוֹא לְטָמְאָה כִּי אִם לְאָב וּלְאֵם וּלְבֵן וּלְבַת לְאָח וּלְאָחוֹת אֲשֶׁר לֹא הָיְתָה לְאִישׁ יִטַּמָּאוּ` (יחזקאל מד:כה)
`וַיֹּאמֶר אֵלַי הֲרָאִיתָ בֶן אָדָם וַיּוֹלִכֵנִי וַיְשִׁבֵנִי שְׂפַת הַנָּחַל` (יחזקאל מז:ו)
`כִּי אִם יְגַדְּלוּ אֶת בְּנֵיהֶם וְשִׁכַּלְתִּים מֵאָדָם כִּי גַם אוֹי לָהֶם בְּשׂוּרִי מֵהֶם` (הושע ט:יב)
`בְּחַבְלֵי אָדָם אֶמְשְׁכֵם בַּעֲבֹתוֹת אַהֲבָה וָאֶהְיֶה לָהֶם כִּמְרִימֵי עֹל עַל לְחֵיהֶם וְאַט אֵלָיו אוֹכִיל` (הושע יא:ד)
`וְעַתָּה יוֹסִפוּ לַחֲטֹא וַיַּעְשׂוּ לָהֶם מַסֵּכָה מִכַּסְפָּם כִּתְבוּנָם עֲצַבִּים מַעֲשֵׂה חָרָשִׁים כֻּלֹּה לָהֶם הֵם אֹמְרִים זֹבְחֵי אָדָם עֲגָלִים יִשָּׁקוּן` (הושע יג:ב)
`הַגֶּפֶן הוֹבִישָׁה וְהַתְּאֵנָה אֻמְלָלָה רִמּוֹן גַּם תָּמָר וְתַפּוּחַ כָּל עֲצֵי הַשָּׂדֶה יָבֵשׁוּ כִּי הֹבִישׁ שָׂשׂוֹן מִן בְּנֵי אָדָם` (יואל א:יב)
`כִּי הִנֵּה יוֹצֵר הָרִים וּבֹרֵא רוּחַ וּמַגִּיד לְאָדָם מַה שֵּׂחוֹ עֹשֵׂה שַׁחַר עֵיפָה וְדֹרֵךְ עַל בָּמֳתֵי אָרֶץ יְהוָה אֱלֹהֵי צְבָאוֹת שְׁמוֹ` (עמוס ד:יג)
`וַיַּזְעֵק וַיֹּאמֶר בְּנִינְוֵה מִטַּעַם הַמֶּלֶךְ וּגְדֹלָיו לֵאמֹר הָאָדָם וְהַבְּהֵמָה הַבָּקָר וְהַצֹּאן אַל יִטְעֲמוּ מְאוּמָה אַל יִרְעוּ וּמַיִם אַל יִשְׁתּוּ` (יונה ג:ז)
`וְיִתְכַּסּוּ שַׂקִּים הָאָדָם וְהַבְּהֵמָה וְיִקְרְאוּ אֶל אֱלֹהִים בְּחָזְקָה וְיָשֻׁבוּ אִישׁ מִדַּרְכּוֹ הָרָעָה וּמִן הֶחָמָס אֲשֶׁר בְּכַפֵּיהֶם` (יונה ג:ח)
`וַאֲנִי לֹא אָחוּס עַל נִינְוֵה הָעִיר הַגְּדוֹלָה אֲשֶׁר יֶשׁ בָּהּ הַרְבֵּה מִשְׁתֵּים עֶשְׂרֵה רִבּוֹ אָדָם אֲשֶׁר לֹא יָדַע בֵּין יְמִינוֹ לִשְׂמֹאלוֹ וּבְהֵמָה רַבָּה` (יונה ד:יא)
`אָסֹף אֶאֱסֹף יַעֲקֹב כֻּלָּךְ קַבֵּץ אֲקַבֵּץ שְׁאֵרִית יִשְׂרָאֵל יַחַד אֲשִׂימֶנּוּ כְּצֹאן בָּצְרָה כְּעֵדֶר בְּתוֹךְ הַדָּבְרוֹ תְּהִימֶנָה מֵאָדָם` (מיכה ב:יב)
`וְהָיָה זֶה שָׁלוֹם אַשּׁוּר כִּי יָבוֹא בְאַרְצֵנוּ וְכִי יִדְרֹךְ בְּאַרְמְנֹתֵינוּ וַהֲקֵמֹנוּ עָלָיו שִׁבְעָה רֹעִים וּשְׁמֹנָה נְסִיכֵי אָדָם` (מיכה ה:ה)
`וְהָיָה שְׁאֵרִית יַעֲקֹב בְּקֶרֶב עַמִּים רַבִּים כְּטַל מֵאֵת יְהוָה כִּרְבִיבִים עֲלֵי עֵשֶׂב אֲשֶׁר לֹא יְקַוֶּה לְאִישׁ וְלֹא יְיַחֵל לִבְנֵי אָדָם` (מיכה ה:ז)
`הִגִּיד לְךָ אָדָם מַה טּוֹב וּמָה יְהוָה דּוֹרֵשׁ מִמְּךָ כִּי אִם עֲשׂוֹת מִשְׁפָּט וְאַהֲבַת חֶסֶד וְהַצְנֵעַ לֶכֶת עִם אֱלֹהֶיךָ` (מיכה ו:ח)
`אָבַד חָסִיד מִן הָאָרֶץ וְיָשָׁר בָּאָדָם אָיִן כֻּלָּם לְדָמִים יֶאֱרֹבוּ אִישׁ אֶת אָחִיהוּ יָצוּדוּ חֵרֶם` (מיכה ז:ב)
`וַתַּעֲשֶׂה אָדָם כִּדְגֵי הַיָּם כְּרֶמֶשׂ לֹא מֹשֵׁל בּוֹ` (חבקוק א:יד)
`כִּי אַתָּה שַׁלּוֹתָ גּוֹיִם רַבִּים יְשָׁלּוּךָ כָּל יֶתֶר עַמִּים מִדְּמֵי אָדָם וַחֲמַס אֶרֶץ קִרְיָה וְכָל יֹשְׁבֵי בָהּ` (חבקוק ב:ח)
`כִּי חֲמַס לְבָנוֹן יְכַסֶּךָּ וְשֹׁד בְּהֵמוֹת יְחִיתַן מִדְּמֵי אָדָם וַחֲמַס אֶרֶץ קִרְיָה וְכָל יֹשְׁבֵי בָהּ` (חבקוק ב:יז)
`אָסֵף אָדָם וּבְהֵמָה אָסֵף עוֹף הַשָּׁמַיִם וּדְגֵי הַיָּם וְהַמַּכְשֵׁלוֹת אֶת הָרְשָׁעִים וְהִכְרַתִּי אֶת הָאָדָם מֵעַל פְּנֵי הָאֲדָמָה נְאֻם יְהוָה` (צפניה א:ג)
`וַהֲצֵרֹתִי לָאָדָם וְהָלְכוּ כַּעִוְרִים כִּי לַיהוָה חָטָאוּ וְשֻׁפַּךְ דָּמָם כֶּעָפָר וּלְחֻמָם כַּגְּלָלִים` (צפניה א:יז)
`וָאֶקְרָא חֹרֶב עַל הָאָרֶץ וְעַל הֶהָרִים וְעַל הַדָּגָן וְעַל הַתִּירוֹשׁ וְעַל הַיִּצְהָר וְעַל אֲשֶׁר תּוֹצִיא הָאֲדָמָה וְעַל הָאָדָם וְעַל הַבְּהֵמָה וְעַל כָּל יְגִיעַ כַּפָּיִם` (חגי א:יא)
`וַיֹּאמֶר אֵלָו רֻץ דַּבֵּר אֶל הַנַּעַר הַלָּז לֵאמֹר פְּרָזוֹת תֵּשֵׁב יְרוּשָׁלִַם מֵרֹב אָדָם וּבְהֵמָה בְּתוֹכָהּ` (זכריה ב:ד)
`כִּי לִפְנֵי הַיָּמִים הָהֵם שְׂכַר הָאָדָם לֹא נִהְיָה וּשְׂכַר הַבְּהֵמָה אֵינֶנָּה וְלַיּוֹצֵא וְלַבָּא אֵין שָׁלוֹם מִן הַצָּר וַאֲשַׁלַּח אֶת כָּל הָאָדָם אִישׁ בְּרֵעֵהוּ` (זכריה ח:י)
`מַשָּׂא דְבַר יְהוָה בְּאֶרֶץ חַדְרָךְ וְדַמֶּשֶׂק מְנֻחָתוֹ כִּי לַיהוָה עֵין אָדָם וְכֹל שִׁבְטֵי יִשְׂרָאֵל` (זכריה ט:א)
`כִּי לֹא אֶחְמוֹל עוֹד עַל יֹשְׁבֵי הָאָרֶץ נְאֻם יְהוָה וְהִנֵּה אָנֹכִי מַמְצִיא אֶת הָאָדָם אִישׁ בְּיַד רֵעֵהוּ וּבְיַד מַלְכּוֹ וְכִתְּתוּ אֶת הָאָרֶץ וְלֹא אַצִּיל מִיָּדָם` (זכריה יא:ו)
`מַשָּׂא דְבַר יְהוָה עַל יִשְׂרָאֵל נְאֻם יְהוָה נֹטֶה שָׁמַיִם וְיֹסֵד אָרֶץ וְיֹצֵר רוּחַ אָדָם בְּקִרְבּוֹ` (זכריה יב:א)
`וְאָמַר לֹא נָבִיא אָנֹכִי אִישׁ עֹבֵד אֲדָמָה אָנֹכִי כִּי אָדָם הִקְנַנִי מִנְּעוּרָי` (זכריה יג:ה)
`הֲיִקְבַּע אָדָם אֱלֹהִים כִּי אַתֶּם קֹבְעִים אֹתִי וַאֲמַרְתֶּם בַּמֶּה קְבַעֲנוּךָ הַמַּעֲשֵׂר וְהַתְּרוּמָה` (מלאכי ג:ח)
`וַיִּשְׁבּוּ מִקְנֵיהֶם גְּמַלֵּיהֶם חֲמִשִּׁים אֶלֶף וְצֹאן מָאתַיִם וַחֲמִשִּׁים אֶלֶף וַחֲמוֹרִים אַלְפָּיִם וְנֶפֶשׁ אָדָם מֵאָה אָלֶף` (דברי-הימים-א ה:כא)
`וַתִּקְטַן זֹאת בְּעֵינֶיךָ אֱלֹהִים וַתְּדַבֵּר עַל בֵּית עַבְדְּךָ לְמֵרָחוֹק וּרְאִיתַנִי כְּתוֹר הָאָדָם הַמַּעֲלָה יְהוָה אֱלֹהִים` (דברי-הימים-א יז:יז)
`וַיֹּאמֶר דָּוִיד אֶל גָּד צַר לִי מְאֹד אֶפְּלָה נָּא בְיַד יְהוָה כִּי רַבִּים רַחֲמָיו מְאֹד וּבְיַד אָדָם אַל אֶפֹּל` (דברי-הימים-א כא:יג)
`וַיֹּאמֶר דָּוִיד הַמֶּלֶךְ לְכָל הַקָּהָל שְׁלֹמֹה בְנִי אֶחָד בָּחַר בּוֹ אֱלֹהִים נַעַר וָרָךְ וְהַמְּלָאכָה גְדוֹלָה כִּי לֹא לְאָדָם הַבִּירָה כִּי לַיהוָה אֱלֹהִים` (דברי-הימים-א כט:א)
`כִּי הַאֻמְנָם יֵשֵׁב אֱלֹהִים אֶת הָאָדָם עַל הָאָרֶץ הִנֵּה שָׁמַיִם וּשְׁמֵי הַשָּׁמַיִם לֹא יְכַלְכְּלוּךָ אַף כִּי הַבַּיִת הַזֶּה אֲשֶׁר בָּנִיתִי` (דברי-הימים-ב ו:יח)
`כָּל תְּפִלָּה כָל תְּחִנָּה אֲשֶׁר יִהְיֶה לְכָל הָאָדָם וּלְכֹל עַמְּךָ יִשְׂרָאֵל אֲשֶׁר יֵדְעוּ אִישׁ נִגְעוֹ וּמַכְאֹבוֹ וּפָרַשׂ כַּפָּיו אֶל הַבַּיִת הַזֶּה` (דברי-הימים-ב ו:כט)
`וְאַתָּה תִּשְׁמַע מִן הַשָּׁמַיִם מְכוֹן שִׁבְתֶּךָ וְסָלַחְתָּ וְנָתַתָּה לָאִישׁ כְּכָל דְּרָכָיו אֲשֶׁר תֵּדַע אֶת לְבָבוֹ כִּי אַתָּה לְבַדְּךָ יָדַעְתָּ אֶת לְבַב בְּנֵי הָאָדָם` (דברי-הימים-ב ו:ל)
`כִּי יֶחֶטְאוּ לָךְ כִּי אֵין אָדָם אֲשֶׁר לֹא יֶחֱטָא וְאָנַפְתָּ בָם וּנְתַתָּם לִפְנֵי אוֹיֵב וְשָׁבוּם שׁוֹבֵיהֶם אֶל אֶרֶץ רְחוֹקָה אוֹ קְרוֹבָה` (דברי-הימים-ב ו:לו)
`וַיֹּאמֶר אֶל הַשֹּׁפְטִים רְאוּ מָה אַתֶּם עֹשִׂים כִּי לֹא לְאָדָם תִּשְׁפְּטוּ כִּי לַיהוָה וְעִמָּכֶם בִּדְבַר מִשְׁפָּט` (דברי-הימים-ב יט:ו)
`וַיְדַבְּרוּ אֶל אֱלֹהֵי יְרוּשָׁלִָם כְּעַל אֱלֹהֵי עַמֵּי הָאָרֶץ מַעֲשֵׂה יְדֵי הָאָדָם` (דברי-הימים-ב לב:יט)
`מָה אֱנוֹשׁ כִּי תִזְכְּרֶנּוּ וּבֶן אָדָם כִּי תִפְקְדֶנּוּ` (תהילים ח:ד)
`יְהוָה בְּהֵיכַל קָדְשׁוֹ יְהוָה בַּשָּׁמַיִם כִּסְאוֹ עֵינָיו יֶחֱזוּ עַפְעַפָּיו יִבְחֲנוּ בְּנֵי אָדָם` (תהילים יא:ד)
`לַמְנַצֵּחַ עַל הַשְּׁמִינִית מִזְמוֹר לְדָוִד הוֹשִׁיעָה יְהוָה כִּי גָמַר חָסִיד כִּי פַסּוּ אֱמוּנִים מִבְּנֵי אָדָם` (תהילים יב:א)
`סָבִיב רְשָׁעִים יִתְהַלָּכוּן כְּרֻם זֻלּוּת לִבְנֵי אָדָם` (תהילים יב:ח)
`יְהוָה מִשָּׁמַיִם הִשְׁקִיף עַל בְּנֵי אָדָם לִרְאוֹת הֲיֵשׁ מַשְׂכִּיל דֹּרֵשׁ אֶת אֱלֹהִים` (תהילים יד:ב)
`לִפְעֻלּוֹת אָדָם בִּדְבַר שְׂפָתֶיךָ אֲנִי שָׁמַרְתִּי אָרְחוֹת פָּרִיץ` (תהילים יז:ד)
`פִּרְיָמוֹ מֵאֶרֶץ תְּאַבֵּד וְזַרְעָם מִבְּנֵי אָדָם` (תהילים כא:י)
`וְאָנֹכִי תוֹלַעַת וְלֹא אִישׁ חֶרְפַּת אָדָם וּבְזוּי עָם` (תהילים כב:ו)
`מָה רַב טוּבְךָ אֲשֶׁר צָפַנְתָּ לִּירֵאֶיךָ פָּעַלְתָּ לַחֹסִים בָּךְ נֶגֶד בְּנֵי אָדָם` (תהילים לא:יט)
`אַשְׁרֵי אָדָם לֹא יַחְשֹׁב יְהוָה לוֹ עָוֺן וְאֵין בְּרוּחוֹ רְמִיָּה` (תהילים לב:ב)
`מִשָּׁמַיִם הִבִּיט יְהוָה רָאָה אֶת כָּל בְּנֵי הָאָדָם` (תהילים לג:יג)
`צִדְקָתְךָ כְּהַרְרֵי אֵל מִשְׁפָּטֶךָ תְּהוֹם רַבָּה אָדָם וּבְהֵמָה תוֹשִׁיעַ יְהוָה` (תהילים לו:ו)
`מַה יָּקָר חַסְדְּךָ אֱלֹהִים וּבְנֵי אָדָם בְּצֵל כְּנָפֶיךָ יֶחֱסָיוּן` (תהילים לו:ז)
`הִנֵּה טְפָחוֹת נָתַתָּה יָמַי וְחֶלְדִּי כְאַיִן נֶגְדֶּךָ אַךְ כָּל הֶבֶל כָּל אָדָם נִצָּב סֶלָה` (תהילים לט:ה)
`בְּתוֹכָחוֹת עַל עָוֺן יִסַּרְתָּ אִישׁ וַתֶּמֶס כָּעָשׁ חֲמוּדוֹ אַךְ הֶבֶל כָּל אָדָם סֶלָה` (תהילים לט:יא)
`יָפְיָפִיתָ מִבְּנֵי אָדָם הוּצַק חֵן בְּשְׂפְתוֹתֶיךָ עַל כֵּן בֵּרַכְךָ אֱלֹהִים לְעוֹלָם` (תהילים מה:ב)
`גַּם בְּנֵי אָדָם גַּם בְּנֵי אִישׁ יַחַד עָשִׁיר וְאֶבְיוֹן` (תהילים מט:ב)
`וְאָדָם בִּיקָר בַּל יָלִין נִמְשַׁל כַּבְּהֵמוֹת נִדְמוּ` (תהילים מט:יב)
`אָדָם בִּיקָר וְלֹא יָבִין נִמְשַׁל כַּבְּהֵמוֹת נִדְמוּ` (תהילים מט:כ)
`אֱ‍לֹהִים מִשָּׁמַיִם הִשְׁקִיף עַל בְּנֵי אָדָם לִרְאוֹת הֲיֵשׁ מַשְׂכִּיל דֹּרֵשׁ אֶת אֱלֹהִים` (תהילים נג:ב)
`בֵּאלֹהִים בָּטַחְתִּי לֹא אִירָא מַה יַּעֲשֶׂה אָדָם לִי` (תהילים נו:יא)
`נַפְשִׁי בְּתוֹךְ לְבָאִם אֶשְׁכְּבָה לֹהֲטִים בְּנֵי אָדָם שִׁנֵּיהֶם חֲנִית וְחִצִּים וּלְשׁוֹנָם חֶרֶב חַדָּה` (תהילים נז:ד)
`לַמְנַצֵּחַ אַל תַּשְׁחֵת לְדָוִד מִכְתָּם הַאֻמְנָם אֵלֶם צֶדֶק תְּדַבֵּרוּן מֵישָׁרִים תִּשְׁפְּטוּ בְּנֵי אָדָם` (תהילים נח:א)
`וְיֹאמַר אָדָם אַךְ פְּרִי לַצַּדִּיק אַךְ יֵשׁ אֱלֹהִים שֹׁפְטִים בָּאָרֶץ` (תהילים נח:יא)
`הָבָה לָּנוּ עֶזְרָת מִצָּר וְשָׁוְא תְּשׁוּעַת אָדָם` (תהילים ס:יא)
`אַךְ הֶבֶל בְּנֵי אָדָם כָּזָב בְּנֵי אִישׁ בְּמֹאזְנַיִם לַעֲלוֹת הֵמָּה מֵהֶבֶל יָחַד` (תהילים סב:ט)
`וַיִּירְאוּ כָּל אָדָם וַיַּגִּידוּ פֹּעַל אֱלֹהִים וּמַעֲשֵׂהוּ הִשְׂכִּילוּ` (תהילים סד:ט)
`לְכוּ וּרְאוּ מִפְעֲלוֹת אֱלֹהִים נוֹרָא עֲלִילָה עַל בְּנֵי אָדָם` (תהילים סו:ה)
`עָלִיתָ לַמָּרוֹם שָׁבִיתָ שֶּׁבִי לָקַחְתָּ מַתָּנוֹת בָּאָדָם וְאַף סוֹרְרִים לִשְׁכֹּן יָהּ אֱלֹהִים` (תהילים סח:יח)
`בַּעֲמַל אֱנוֹשׁ אֵינֵמוֹ וְעִם אָדָם לֹא יְנֻגָּעוּ` (תהילים עג:ה)
`כִּי חֲמַת אָדָם תּוֹדֶךָּ שְׁאֵרִית חֵמֹת תַּחְגֹּר` (תהילים עו:י)
`וַיִּטֹּשׁ מִשְׁכַּן שִׁלוֹ אֹהֶל שִׁכֵּן בָּאָדָם` (תהילים עח:ס)
`תְּהִי יָדְךָ עַל אִישׁ יְמִינֶךָ עַל בֶּן אָדָם אִמַּצְתָּ לָּךְ` (תהילים פ:יז)
`אָכֵן כְּאָדָם תְּמוּתוּן וּכְאַחַד הַשָּׂרִים תִּפֹּלוּ` (תהילים פב:ז)
`אַשְׁרֵי אָדָם עוֹז לוֹ בָךְ מְסִלּוֹת בִּלְבָבָם` (תהילים פד:ה)
`יְהוָה צְבָאוֹת אַשְׁרֵי אָדָם בֹּטֵחַ בָּךְ` (תהילים פד:יב)
`זְכָר אֲנִי מֶה חָלֶד עַל מַה שָּׁוְא בָּרָאתָ כָל בְּנֵי אָדָם` (תהילים פט:מז)
`תָּשֵׁב אֱנוֹשׁ עַד דַּכָּא וַתֹּאמֶר שׁוּבוּ בְנֵי אָדָם` (תהילים צ:ג)
`הֲיֹסֵר גּוֹיִם הֲלֹא יוֹכִיחַ הַמְלַמֵּד אָדָם דָּעַת` (תהילים צד:י)
`יְהוָה יֹדֵעַ מַחְשְׁבוֹת אָדָם כִּי הֵמָּה הָבֶל` (תהילים צד:יא)
`מַצְמִיחַ חָצִיר לַבְּהֵמָה וְעֵשֶׂב לַעֲבֹדַת הָאָדָם לְהוֹצִיא לֶחֶם מִן הָאָרֶץ` (תהילים קד:יד)
`יֵצֵא אָדָם לְפָעֳלוֹ וְלַעֲבֹדָתוֹ עֲדֵי עָרֶב` (תהילים קד:כג)
`לֹא הִנִּיחַ אָדָם לְעָשְׁקָם וַיּוֹכַח עֲלֵיהֶם מְלָכִים` (תהילים קה:יד)
`יוֹדוּ לַיהוָה חַסְדּוֹ וְנִפְלְאוֹתָיו לִבְנֵי אָדָם` (תהילים קז:ח)
`יוֹדוּ לַיהוָה חַסְדּוֹ וְנִפְלְאוֹתָיו לִבְנֵי אָדָם` (תהילים קז:טו)
`יוֹדוּ לַיהוָה חַסְדּוֹ וְנִפְלְאוֹתָיו לִבְנֵי אָדָם` (תהילים קז:כא)
`יוֹדוּ לַיהוָה חַסְדּוֹ וְנִפְלְאוֹתָיו לִבְנֵי אָדָם` (תהילים קז:לא)
`הָבָה לָּנוּ עֶזְרָת מִצָּר וְשָׁוְא תְּשׁוּעַת אָדָם` (תהילים קח:יב)
`עֲ‍צַבֵּיהֶם כֶּסֶף וְזָהָב מַעֲשֵׂה יְדֵי אָדָם` (תהילים קיה:ד)
`הַשָּׁמַיִם שָׁמַיִם לַיהוָה וְהָאָרֶץ נָתַן לִבְנֵי אָדָם` (תהילים קיה:טז)
`אֲנִי אָמַרְתִּי בְחָפְזִי כָּל הָאָדָם כֹּזֵב` (תהילים קיו:יא)
`יְהוָה לִי לֹא אִירָא מַה יַּעֲשֶׂה לִי אָדָם` (תהילים קיח:ו)
`טוֹב לַחֲסוֹת בַּיהוָה מִבְּטֹחַ בָּאָדָם` (תהילים קיח:ח)
`פְּדֵנִי מֵעֹשֶׁק אָדָם וְאֶשְׁמְרָה פִּקּוּדֶיךָ` (תהילים קיט:קלד)
`לוּלֵי יְהוָה שֶׁהָיָה לָנוּ בְּקוּם עָלֵינוּ אָדָם` (תהילים קכד:ב)
`שֶׁהִכָּה בְּכוֹרֵי מִצְרָיִם מֵאָדָם עַד בְּהֵמָה` (תהילים קלה:ח)
`עֲצַבֵּי הַגּוֹיִם כֶּסֶף וְזָהָב מַעֲשֵׂה יְדֵי אָדָם` (תהילים קלה:טו)
`לַמְנַצֵּחַ מִזְמוֹר לְדָוִד חַלְּצֵנִי יְהוָה מֵאָדָם רָע מֵאִישׁ חֲמָסִים תִּנְצְרֵנִי` (תהילים קמ:א)
`יְהוָה מָה אָדָם וַתֵּדָעֵהוּ בֶּן אֱנוֹשׁ וַתְּחַשְּׁבֵהוּ` (תהילים קמד:ג)
`אָדָם לַהֶבֶל דָּמָה יָמָיו כְּצֵל עוֹבֵר` (תהילים קמד:ד)
`לְהוֹדִיעַ לִבְנֵי הָאָדָם גְּבוּרֹתָיו וּכְבוֹד הֲדַר מַלְכוּתוֹ` (תהילים קמה:יב)
`אַל תִּבְטְחוּ בִנְדִיבִים בְּבֶן אָדָם שֶׁאֵין לוֹ תְשׁוּעָה` (תהילים קמו:ג)
`כִּי אָדָם לְעָמָל יוּלָּד וּבְנֵי רֶשֶׁף יַגְבִּיהוּ עוּף` (איוב ה:ז)
`חָטָאתִי מָה אֶפְעַל לָךְ נֹצֵר הָאָדָם לָמָה שַׂמְתַּנִי לְמִפְגָּע לָךְ וָאֶהְיֶה עָלַי לְמַשָּׂא` (איוב ז:כ)
`וְאִישׁ נָבוּב יִלָּבֵב וְעַיִר פֶּרֶא אָדָם יִוָּלֵד` (איוב יא:יב)
`אָדָם יְלוּד אִשָּׁה קְצַר יָמִים וּשְׂבַע רֹגֶז` (איוב יד:א)
`וְגֶבֶר יָמוּת וַיֶּחֱלָשׁ וַיִּגְוַע אָדָם וְאַיּוֹ` (איוב יד:י)
`הֲרִאישׁוֹן אָדָם תִּוָּלֵד וְלִפְנֵי גְבָעוֹת חוֹלָלְתָּ` (איוב טו:ז)
`וְיוֹכַח לְגֶבֶר עִם אֱלוֹהַּ וּבֶן אָדָם לְרֵעֵהוּ` (איוב טז:כא)
`הֲזֹאת יָדַעְתָּ מִנִּי עַד מִנִּי שִׂים אָדָם עֲלֵי אָרֶץ` (איוב כ:ד)
`זֶה חֵלֶק אָדָם רָשָׁע מֵאֱלֹהִים וְנַחֲלַת אִמְרוֹ מֵאֵל` (איוב כ:כט)
`הֶאָנֹכִי לְאָדָם שִׂיחִי וְאִם מַדּוּעַ לֹא תִקְצַר רוּחִי` (איוב כא:ד)
`מָתְקוּ לוֹ רִגְבֵי נָחַל וְאַחֲרָיו כָּל אָדָם יִמְשׁוֹךְ וּלְפָנָיו אֵין מִסְפָּר` (איוב כא:לג)
`אַף כִּי אֱנוֹשׁ רִמָּה וּבֶן אָדָם תּוֹלֵעָה` (איוב כה:ו)
`זֶה חֵלֶק אָדָם רָשָׁע עִם אֵל וְנַחֲלַת עָרִיצִים מִשַּׁדַּי יִקָּחוּ` (איוב כז:יג)
`וַיֹּאמֶר לָאָדָם הֵן יִרְאַת אֲדֹנָי הִיא חָכְמָה וְסוּר מֵרָע בִּינָה` (איוב כח:כח)
`אִם כִּסִּיתִי כְאָדָם פְּשָׁעָי לִטְמוֹן בְּחֻבִּי עֲוֺנִי` (איוב לא:לג)
`אַל נָא אֶשָּׂא פְנֵי אִישׁ וְאֶל אָדָם לֹא אֲכַנֶּה` (איוב לב:כא)
`לְהָסִיר אָדָם מַעֲשֶׂה וְגֵוָה מִגֶּבֶר יְכַסֶּה` (איוב לג:יז)
`אִם יֵשׁ עָלָיו מַלְאָךְ מֵלִיץ אֶחָד מִנִּי אָלֶף לְהַגִּיד לְאָדָם יָשְׁרוֹ` (איוב לג:כג)
`כִּי פֹעַל אָדָם יְשַׁלֶּם לוֹ וּכְאֹרַח אִישׁ יַמְצִאֶנּוּ` (איוב לד:יא)
`יִגְוַע כָּל בָּשָׂר יָחַד וְאָדָם עַל עָפָר יָשׁוּב` (איוב לד:טו)
`וְהוּא יַשְׁקִט וּמִי יַרְשִׁעַ וְיַסְתֵּר פָּנִים וּמִי יְשׁוּרֶנּוּ וְעַל גּוֹי וְעַל אָדָם יָחַד` (איוב לד:כט)
`מִמְּלֹךְ אָדָם חָנֵף מִמֹּקְשֵׁי עָם` (איוב לד:ל)
`לְאִישׁ כָּמוֹךָ רִשְׁעֶךָ וּלְבֶן אָדָם צִדְקָתֶךָ` (איוב לה:ח)
`כָּל אָדָם חָזוּ בוֹ אֱנוֹשׁ יַבִּיט מֵרָחוֹק` (איוב לו:כה)
`אֲשֶׁר יִזְּלוּ שְׁחָקִים יִרְעֲפוּ עֲלֵי אָדָם רָב` (איוב לו:כח)
`בְּיַד כָּל אָדָם יַחְתּוֹם לָדַעַת כָּל אַנְשֵׁי מַעֲשֵׂהוּ` (איוב לז:ז)
`לְהַמְטִיר עַל אֶרֶץ לֹא אִישׁ מִדְבָּר לֹא אָדָם בּוֹ` (איוב לח:כו)
`וּמְצָא חֵן וְשֵׂכֶל טוֹב בְּעֵינֵי אֱלֹהִים וְאָדָם` (משלי ג:ד)
`אַשְׁרֵי אָדָם מָצָא חָכְמָה וְאָדָם יָפִיק תְּבוּנָה` (משלי ג:יג)
`אַל תָּרוֹב עִם אָדָם חִנָּם אִם לֹא גְמָלְךָ רָעָה` (משלי ג:ל)
`אָדָם בְּלִיַּעַל אִישׁ אָוֶן הוֹלֵךְ עִקְּשׁוּת פֶּה` (משלי ו:יב)
`אֲלֵיכֶם אִישִׁים אֶקְרָא וְקוֹלִי אֶל בְּנֵי אָדָם` (משלי ח:ד)
`מְשַׂחֶקֶת בְּתֵבֵל אַרְצוֹ וְשַׁעֲשֻׁעַי אֶת בְּנֵי אָדָם` (משלי ח:לא)
`אַשְׁרֵי אָדָם שֹׁמֵעַ לִי לִשְׁקֹד עַל דַּלְתֹתַי יוֹם יוֹם לִשְׁמֹר מְזוּזֹת פְּתָחָי` (משלי ח:לד)
`בְּמוֹת אָדָם רָשָׁע תֹּאבַד תִּקְוָה וְתוֹחֶלֶת אוֹנִים אָבָדָה` (משלי יא:ז)
`לֹא יִכּוֹן אָדָם בְּרֶשַׁע וְשֹׁרֶשׁ צַדִּיקִים בַּל יִמּוֹט` (משלי יב:ג)
`מִפְּרִי פִי אִישׁ יִשְׂבַּע טוֹב וּגְמוּל יְדֵי אָדָם יָשׁוּב לוֹ` (משלי יב:יד)
`אָדָם עָרוּם כֹּסֶה דָּעַת וְלֵב כְּסִילִים יִקְרָא אִוֶּלֶת` (משלי יב:כג)
`לֹא יַחֲרֹךְ רְמִיָּה צֵידוֹ וְהוֹן אָדָם יָקָר חָרוּץ` (משלי יב:כז)
`שְׁאוֹל וַאֲבַדּוֹן נֶגֶד יְהוָה אַף כִּי לִבּוֹת בְּנֵי אָדָם` (משלי טו:יא)
`בֵּן חָכָם יְשַׂמַּח אָב וּכְסִיל אָדָם בּוֹזֶה אִמּוֹ` (משלי טו:כ)
`לְאָדָם מַעַרְכֵי לֵב וּמֵיְהוָה מַעֲנֵה לָשׁוֹן` (משלי טז:א)
`לֵב אָדָם יְחַשֵּׁב דַּרְכּוֹ וַיהוָה יָכִין צַעֲדוֹ` (משלי טז:ט)
`אָדָם חֲסַר לֵב תּוֹקֵעַ כָּף עֹרֵב עֲרֻבָּה לִפְנֵי רֵעֵהוּ` (משלי יז:יח)
`מַתָּן אָדָם יַרְחִיב לוֹ וְלִפְנֵי גְדֹלִים יַנְחֶנּוּ` (משלי יח:טז)
`אִוֶּלֶת אָדָם תְּסַלֵּף דַּרְכּוֹ וְעַל יְהוָה יִזְעַף לִבּוֹ` (משלי יט:ג)
`שֵׂכֶל אָדָם הֶאֱרִיךְ אַפּוֹ וְתִפאַרְתּוֹ עֲבֹר עַל פָּשַׁע` (משלי יט:יא)
`תַּאֲוַת אָדָם חַסְדּוֹ וְטוֹב רָשׁ מֵאִישׁ כָּזָב` (משלי יט:כב)
`רָב אָדָם יִקְרָא אִישׁ חַסְדּוֹ וְאִישׁ אֱמוּנִים מִי יִמְצָא` (משלי כ:ו)
`מֵיהוָה מִצְעֲדֵי גָבֶר וְאָדָם מַה יָּבִין דַּרְכּוֹ` (משלי כ:כד)
`מוֹקֵשׁ אָדָם יָלַע קֹדֶשׁ וְאַחַר נְדָרִים לְבַקֵּר` (משלי כ:כה)
`נֵר יְהוָה נִשְׁמַת אָדָם חֹפֵשׂ כָּל חַדְרֵי בָטֶן` (משלי כ:כז)
`אָדָם תּוֹעֶה מִדֶּרֶךְ הַשְׂכֵּל בִּקְהַל רְפָאִים יָנוּחַ` (משלי כא:טז)
`אוֹצָר נֶחְמָד וָשֶׁמֶן בִּנְוֵה חָכָם וּכְסִיל אָדָם יְבַלְּעֶנּוּ` (משלי כא:כ)
`אַף הִיא כְּחֶתֶף תֶּאֱרֹב וּבוֹגְדִים בְּאָדָם תּוֹסִף` (משלי כג:כח)
`זִמַּת אִוֶּלֶת חַטָּאת וְתוֹעֲבַת לְאָדָם לֵץ` (משלי כד:ט)
`כִּי תֹאמַר הֵן לֹא יָדַעְנוּ זֶה הֲ‍לֹא תֹכֵן לִבּוֹת הוּא יָבִין וְנֹצֵר נַפְשְׁךָ הוּא יֵדָע וְהֵשִׁיב לְאָדָם כְּפָעֳלוֹ` (משלי כד:יב)
`עַל שְׂדֵה אִישׁ עָצֵל עָבַרְתִּי וְעַל כֶּרֶם אָדָם חֲסַר לֵב` (משלי כד:ל)
`כַּמַּיִם הַפָּנִים לַפָּנִים כֵּן לֵב הָאָדָם לָאָדָם` (משלי כז:יט)
`שְׁאוֹל וַאֲבַדֹּה לֹא תִשְׂבַּעְנָה וְעֵינֵי הָאָדָם לֹא תִשְׂבַּעְנָה` (משלי כז:כ)
`בְּפֶשַׁע אֶרֶץ רַבִּים שָׂרֶיהָ וּבְאָדָם מֵבִין יֹדֵעַ כֵּן יַאֲרִיךְ` (משלי כח:ב)
`בַּעֲלֹץ צַדִּיקִים רַבָּה תִפְאָרֶת וּבְקוּם רְשָׁעִים יְחֻפַּשׂ אָדָם` (משלי כח:יב)
`אַשְׁרֵי אָדָם מְפַחֵד תָּמִיד וּמַקְשֶׁה לִבּוֹ יִפּוֹל בְּרָעָה` (משלי כח:יד)
`אָדָם עָשֻׁק בְּדַם נָפֶשׁ עַד בּוֹר יָנוּס אַל יִתְמְכוּ בוֹ` (משלי כח:יז)
`מוֹכִיחַ אָדָם אַחֲרַי חֵן יִמְצָא מִמַּחֲלִיק לָשׁוֹן` (משלי כח:כג)
`בְּקוּם רְשָׁעִים יִסָּתֵר אָדָם וּבְאָבְדָם יִרְבּוּ צַדִּיקִים` (משלי כח:כח)
`גַּאֲוַת אָדָם תַּשְׁפִּילֶנּוּ וּשְׁפַל רוּחַ יִתְמֹךְ כָּבוֹד` (משלי כט:כג)
`חֶרְדַּת אָדָם יִתֵּן מוֹקֵשׁ וּבוֹטֵחַ בַּיהוָה יְשֻׂגָּב` (משלי כט:כה)
`כִּי בַעַר אָנֹכִי מֵאִישׁ וְלֹא בִינַת אָדָם לִי` (משלי ל:ב)
`דּוֹר חֲרָבוֹת שִׁנָּיו וּמַאֲכָלוֹת מְתַלְּעֹתָיו לֶאֱכֹל עֲנִיִּים מֵאֶרֶץ וְאֶבְיוֹנִים מֵאָדָם` (משלי ל:יד)
`מַה יִּתְרוֹן לָאָדָם בְּכָל עֲמָלוֹ שֶׁיַּעֲמֹל תַּחַת הַשָּׁמֶשׁ` (קהלת א:ג)
`וְנָתַתִּי אֶת לִבִּי לִדְרוֹשׁ וְלָתוּר בַּחָכְמָה עַל כָּל אֲשֶׁר נַעֲשָׂה תַּחַת הַשָּׁמָיִם הוּא עִנְיַן רָע נָתַן אֱלֹהִים לִבְנֵי הָאָדָם לַעֲנוֹת בּוֹ` (קהלת א:יג)
`תַּרְתִּי בְלִבִּי לִמְשׁוֹךְ בַּיַּיִן אֶת בְּשָׂרִי וְלִבִּי נֹהֵג בַּחָכְמָה וְלֶאֱחֹז בְּסִכְלוּת עַד אֲשֶׁר אֶרְאֶה אֵי זֶה טוֹב לִבְנֵי הָאָדָם אֲשֶׁר יַעֲשׂוּ תַּחַת הַשָּׁמַיִם מִסְפַּר יְמֵי חַיֵּיהֶם` (קהלת ב:ג)
`כָּנַסְתִּי לִי גַּם כֶּסֶף וְזָהָב וּסְגֻלַּת מְלָכִים וְהַמְּדִינוֹת עָשִׂיתִי לִי שָׁרִים וְשָׁרוֹת וְתַעֲנוּגֹת בְּנֵי הָאָדָם שִׁדָּה וְשִׁדּוֹת` (קהלת ב:ח)
`וּפָנִיתִי אֲנִי לִרְאוֹת חָכְמָה וְהוֹלֵלוֹת וְסִכְלוּת כִּי מֶה הָאָדָם שֶׁיָּבוֹא אַחֲרֵי הַמֶּלֶךְ אֵת אֲשֶׁר כְּבָר עָשׂוּהוּ` (קהלת ב:יב)
`וְשָׂנֵאתִי אֲנִי אֶת כָּל עֲמָלִי שֶׁאֲנִי עָמֵל תַּחַת הַשָּׁמֶשׁ שֶׁאַנִּיחֶנּוּ לָאָדָם שֶׁיִּהְיֶה אַחֲרָי` (קהלת ב:יח)
`כִּי יֵשׁ אָדָם שֶׁעֲמָלוֹ בְּחָכְמָה וּבְדַעַת וּבְכִשְׁרוֹן וּלְאָדָם שֶׁלֹּא עָמַל בּוֹ יִתְּנֶנּוּ חֶלְקוֹ גַּם זֶה הֶבֶל וְרָעָה רַבָּה` (קהלת ב:כא)
`כִּי מֶה הֹוֶה לָאָדָם בְּכָל עֲמָלוֹ וּבְרַעְיוֹן לִבּוֹ שֶׁהוּא עָמֵל תַּחַת הַשָּׁמֶשׁ` (קהלת ב:כב)
`אֵין טוֹב בָּאָדָם שֶׁיֹּאכַל וְשָׁתָה וְהֶרְאָה אֶת נַפְשׁוֹ טוֹב בַּעֲמָלוֹ גַּם זֹה רָאִיתִי אָנִי כִּי מִיַּד הָאֱלֹהִים הִיא` (קהלת ב:כד)
`כִּי לְאָדָם שֶׁטּוֹב לְפָנָיו נָתַן חָכְמָה וְדַעַת וְשִׂמְחָה וְלַחוֹטֶא נָתַן עִנְיָן לֶאֱסוֹף וְלִכְנוֹס לָתֵת לְטוֹב לִפְנֵי הָאֱלֹהִים גַּם זֶה הֶבֶל וּרְעוּת רוּחַ` (קהלת ב:כו)
`רָאִיתִי אֶת הָעִנְיָן אֲשֶׁר נָתַן אֱלֹהִים לִבְנֵי הָאָדָם לַעֲנוֹת בּוֹ` (קהלת ג:י)
`אֶת הַכֹּל עָשָׂה יָפֶה בְעִתּוֹ גַּם אֶת הָעֹלָם נָתַן בְּלִבָּם מִבְּלִי אֲשֶׁר לֹא יִמְצָא הָאָדָם אֶת הַמַּעֲשֶׂה אֲשֶׁר עָשָׂה הָאֱלֹהִים מֵרֹאשׁ וְעַד סוֹף` (קהלת ג:יא)
`וְגַם כָּל הָאָדָם שֶׁיֹּאכַל וְשָׁתָה וְרָאָה טוֹב בְּכָל עֲמָלוֹ מַתַּת אֱלֹהִים הִיא` (קהלת ג:יג)
`אָמַרְתִּי אֲנִי בְּלִבִּי עַל דִּבְרַת בְּנֵי הָאָדָם לְבָרָם הָאֱלֹהִים וְלִרְאוֹת שְׁהֶם בְּהֵמָה הֵמָּה לָהֶם` (קהלת ג:יח)
`כִּי מִקְרֶה בְנֵי הָאָדָם וּמִקְרֶה הַבְּהֵמָה וּמִקְרֶה אֶחָד לָהֶם כְּמוֹת זֶה כֵּן מוֹת זֶה וְרוּחַ אֶחָד לַכֹּל וּמוֹתַר הָאָדָם מִן הַבְּהֵמָה אָיִן כִּי הַכֹּל הָבֶל` (קהלת ג:יט)
`מִי יוֹדֵעַ רוּחַ בְּנֵי הָאָדָם הָעֹלָה הִיא לְמָעְלָה וְרוּחַ הַבְּהֵמָה הַיֹּרֶדֶת הִיא לְמַטָּה לָאָרֶץ` (קהלת ג:כא)
`וְרָאִיתִי כִּי אֵין טוֹב מֵאֲשֶׁר יִשְׂמַח הָאָדָם בְּמַעֲשָׂיו כִּי הוּא חֶלְקוֹ כִּי מִי יְבִיאֶנּוּ לִרְאוֹת בְּמֶה שֶׁיִּהְיֶה אַחֲרָיו` (קהלת ג:כב)
`גַּם כָּל הָאָדָם אֲשֶׁר נָתַן לוֹ הָאֱלֹהִים עֹשֶׁר וּנְכָסִים וְהִשְׁלִיטוֹ לֶאֱכֹל מִמֶּנּוּ וְלָשֵׂאת אֶת חֶלְקוֹ וְלִשְׂמֹחַ בַּעֲמָלוֹ זֹה מַתַּת אֱלֹהִים הִיא` (קהלת ה:יט)
`יֵשׁ רָעָה אֲשֶׁר רָאִיתִי תַּחַת הַשָּׁמֶשׁ וְרַבָּה הִיא עַל הָאָדָם` (קהלת ו:א)
`כָּל עֲמַל הָאָדָם לְפִיהוּ וְגַם הַנֶּפֶשׁ לֹא תִמָּלֵא` (קהלת ו:ז)
`מַה שֶּׁהָיָה כְּבָר נִקְרָא שְׁמוֹ וְנוֹדָע אֲשֶׁר הוּא אָדָם וְלֹא יוּכַל לָדִין עִם שֶׁהַתְקִיף מִמֶּנּוּ` (קהלת ו:י)
`כִּי יֵשׁ דְּבָרִים הַרְבֵּה מַרְבִּים הָבֶל מַה יֹּתֵר לָאָדָם` (קהלת ו:יא)
`כִּי מִי יוֹדֵעַ מַה טּוֹב לָאָדָם בַּחַיִּים מִסְפַּר יְמֵי חַיֵּי הֶבְלוֹ וְיַעֲשֵׂם כַּצֵּל אֲשֶׁר מִי יַגִּיד לָאָדָם מַה יִּהְיֶה אַחֲרָיו תַּחַת הַשָּׁמֶשׁ` (קהלת ו:יב)
`טוֹב לָלֶכֶת אֶל בֵּית אֵבֶל מִלֶּכֶת אֶל בֵּית מִשְׁתֶּה בַּאֲשֶׁר הוּא סוֹף כָּל הָאָדָם וְהַחַי יִתֵּן אֶל לִבּוֹ` (קהלת ז:ב)
`בְּיוֹם טוֹבָה הֱיֵה בְטוֹב וּבְיוֹם רָעָה רְאֵה גַּם אֶת זֶה לְעֻמַּת זֶה עָשָׂה הָאֱלֹהִים עַל דִּבְרַת שֶׁלֹּא יִמְצָא הָאָדָם אַחֲרָיו מְאוּמָה` (קהלת ז:יד)
`כִּי אָדָם אֵין צַדִּיק בָּאָרֶץ אֲשֶׁר יַעֲשֶׂה טּוֹב וְלֹא יֶחֱטָא` (קהלת ז:כ)
`אֲשֶׁר עוֹד בִּקְשָׁה נַפְשִׁי וְלֹא מָצָאתִי אָדָם אֶחָד מֵאֶלֶף מָצָאתִי וְאִשָּׁה בְכָל אֵלֶּה לֹא מָצָאתִי` (קהלת ז:כח)
`לְבַד רְאֵה זֶה מָצָאתִי אֲשֶׁר עָשָׂה הָאֱלֹהִים אֶת הָאָדָם יָשָׁר וְהֵמָּה בִקְשׁוּ חִשְּׁבֹנוֹת רַבִּים` (קהלת ז:כט)
`מִי כְּהֶחָכָם וּמִי יוֹדֵעַ פֵּשֶׁר דָּבָר חָכְמַת אָדָם תָּאִיר פָּנָיו וְעֹז פָּנָיו יְשֻׁנֶּא` (קהלת ח:א)
`כִּי לְכָל חֵפֶץ יֵשׁ עֵת וּמִשְׁפָּט כִּי רָעַת הָאָדָם רַבָּה עָלָיו` (קהלת ח:ו)
`אֵין אָדָם שַׁלִּיט בָּרוּחַ לִכְלוֹא אֶת הָרוּחַ וְאֵין שִׁלְטוֹן בְּיוֹם הַמָּוֶת וְאֵין מִשְׁלַחַת בַּמִּלְחָמָה וְלֹא יְמַלֵּט רֶשַׁע אֶת בְּעָלָיו` (קהלת ח:ח)
`אֶת כָּל זֶה רָאִיתִי וְנָתוֹן אֶת לִבִּי לְכָל מַעֲשֶׂה אֲשֶׁר נַעֲשָׂה תַּחַת הַשָּׁמֶשׁ עֵת אֲשֶׁר שָׁלַט הָאָדָם בְּאָדָם לְרַע לוֹ` (קהלת ח:ט)
`אֲשֶׁר אֵין נַעֲשָׂה פִתְגָם מַעֲשֵׂה הָרָעָה מְהֵרָה עַל כֵּן מָלֵא לֵב בְּנֵי הָאָדָם בָּהֶם לַעֲשׂוֹת רָע` (קהלת ח:יא)
`וְשִׁבַּחְתִּי אֲנִי אֶת הַשִּׂמְחָה אֲשֶׁר אֵין טוֹב לָאָדָם תַּחַת הַשֶּׁמֶשׁ כִּי אִם לֶאֱכוֹל וְלִשְׁתּוֹת וְלִשְׂמוֹחַ וְהוּא יִלְוֶנּוּ בַעֲמָלוֹ יְמֵי חַיָּיו אֲשֶׁר נָתַן לוֹ הָאֱלֹהִים תַּחַת הַשָּׁמֶשׁ` (קהלת ח:טו)
`וְרָאִיתִי אֶת כָּל מַעֲשֵׂה הָאֱלֹהִים כִּי לֹא יוּכַל הָאָדָם לִמְצוֹא אֶת הַמַּעֲשֶׂה אֲשֶׁר נַעֲשָׂה תַחַת הַשֶּׁמֶשׁ בְּשֶׁל אֲשֶׁר יַעֲמֹל הָאָדָם לְבַקֵּשׁ וְלֹא יִמְצָא וְגַם אִם יֹאמַר הֶחָכָם לָדַעַת לֹא יוּכַל לִמְצֹא` (קהלת ח:יז)
`כִּי אֶת כָּל זֶה נָתַתִּי אֶל לִבִּי וְלָבוּר אֶת כָּל זֶה אֲשֶׁר הַצַּדִּיקִים וְהַחֲכָמִים וַעֲבָדֵיהֶם בְּיַד הָאֱלֹהִים גַּם אַהֲבָה גַם שִׂנְאָה אֵין יוֹדֵעַ הָאָדָם הַכֹּל לִפְנֵיהֶם` (קהלת ט:א)
`זֶה רָע בְּכֹל אֲשֶׁר נַעֲשָׂה תַּחַת הַשֶּׁמֶשׁ כִּי מִקְרֶה אֶחָד לַכֹּל וְגַם לֵב בְּנֵי הָאָדָם מָלֵא רָע וְהוֹלֵלוֹת בִּלְבָבָם בְּחַיֵּיהֶם וְאַחֲרָיו אֶל הַמֵּתִים` (קהלת ט:ג)
`כִּי גַּם לֹא יֵדַע הָאָדָם אֶת עִתּוֹ כַּדָּגִים שֶׁנֶּאֱחָזִים בִּמְצוֹדָה רָעָה וְכַצִּפֳּרִים הָאֲחֻזוֹת בַּפָּח כָּהֵם יוּקָשִׁים בְּנֵי הָאָדָם לְעֵת רָעָה כְּשֶׁתִּפּוֹל עֲלֵיהֶם פִּתְאֹם` (קהלת ט:יב)
`וּמָצָא בָהּ אִישׁ מִסְכֵּן חָכָם וּמִלַּט הוּא אֶת הָעִיר בְּחָכְמָתוֹ וְאָדָם לֹא זָכַר אֶת הָאִישׁ הַמִּסְכֵּן הַהּוּא` (קהלת ט:טו)
`וְהַסָּכָל יַרְבֶּה דְבָרִים לֹא יֵדַע הָאָדָם מַה שֶׁיִּהְיֶה וַאֲשֶׁר יִהְיֶה מֵאַחֲרָיו מִי יַגִּיד לוֹ` (קהלת י:יד)
`כִּי אִם שָׁנִים הַרְבֵּה יִחְיֶה הָאָדָם בְּכֻלָּם יִשְׂמָח וְיִזְכֹּר אֶת יְמֵי הַחֹשֶׁךְ כִּי הַרְבֵּה יִהְיוּ כָּל שֶׁבָּא הָבֶל` (קהלת יא:ח)
`גַּם מִגָּבֹהַּ יִרָאוּ וְחַתְחַתִּים בַּדֶּרֶךְ וְיָנֵאץ הַשָּׁקֵד וְיִסְתַּבֵּל הֶחָגָב וְתָפֵר הָאֲבִיּוֹנָה כִּי הֹלֵךְ הָאָדָם אֶל בֵּית עוֹלָמוֹ וְסָבְבוּ בָשּׁוּק הַסֹּפְדִים` (קהלת יב:ה)
`סוֹף דָּבָר הַכֹּל נִשְׁמָע אֶת הָאֱלֹהִים יְרָא וְאֶת מִצְוֺתָיו שְׁמוֹר כִּי זֶה כָּל הָאָדָם` (קהלת יב:יג)
`לְעַוֵּת אָדָם בְּרִיבוֹ אֲדֹנָי לֹא רָאָה` (איכה ג:לו)
`מַה יִּתְאוֹנֵן אָדָם חָי גֶּבֶר עַל חֶטְאוֹ` (איכה ג:לט)
`וָאֶשְׁמַע קוֹל אָדָם בֵּין אוּלָי וַיִּקְרָא וַיֹּאמַר גַּבְרִיאֵל הָבֵן לְהַלָּז אֶת הַמַּרְאֶה` (דניאל ח:טז)
`וַיָּבֹא אֵצֶל עָמְדִי וּבְבֹאוֹ נִבְעַתִּי וָאֶפְּלָה עַל פָּנָי וַיֹּאמֶר אֵלַי הָבֵן בֶּן אָדָם כִּי לְעֶת קֵץ הֶחָזוֹן` (דניאל ח:יז)
`וְהִנֵּה כִּדְמוּת בְּנֵי אָדָם נֹגֵעַ עַל שְׂפָתָי וָאֶפְתַּח פִּי וָאֲדַבְּרָה וָאֹמְרָה אֶל הָעֹמֵד לְנֶגְדִּי אֲדֹנִי בַּמַּרְאָה נֶהֶפְכוּ צִירַי עָלַי וְלֹא עָצַרְתִּי כֹּחַ` (דניאל י:טז)
`וַיֹּסֶף וַיִּגַּע בִּי כְּמַרְאֵה אָדָם וַיְחַזְּקֵנִי` (דניאל י:יח)
`וַיִּשְׁמַע סַנְבַלַּט הַחֹרֹנִי וְטוֹבִיָּה הָעֶבֶד הָעַמֹּנִי וַיֵּרַע לָהֶם רָעָה גְדֹלָה אֲשֶׁר בָּא אָדָם לְבַקֵּשׁ טוֹבָה לִבְנֵי יִשְׂרָאֵל` (נחמיה ב:י)
`וָאָקוּם לַיְלָה אֲנִי וַאֲנָשִׁים מְעַט עִמִּי וְלֹא הִגַּדְתִּי לְאָדָם מָה אֱלֹהַי נֹתֵן אֶל לִבִּי לַעֲשׂוֹת לִירוּשָׁלִָם וּבְהֵמָה אֵין עִמִּי כִּי אִם הַבְּהֵמָה אֲשֶׁר אֲנִי רֹכֵב בָּהּ` (נחמיה ב:יב)
`וַתָּעַד בָּהֶם לַהֲשִׁיבָם אֶל תּוֹרָתֶךָ וְהֵמָּה הֵזִידוּ וְלֹא שָׁמְעוּ לְמִצְוֺתֶיךָ וּבְמִשְׁפָּטֶיךָ חָטְאוּ בָם אֲשֶׁר יַעֲשֶׂה אָדָם וְחָיָה בָהֶם וַיִּתְּנוּ כָתֵף סוֹרֶרֶת וְעָרְפָּם הִקְשׁוּ וְלֹא שָׁמֵעוּ` (נחמיה ט:כט)
